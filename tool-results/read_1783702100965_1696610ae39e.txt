   105→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   106→  const visible = users.slice(0, max)
   107→  const extra = users.length - max
   108→  return (
   109→    <div className="flex -space-x-2">
   110→      {visible.map((u, i) =>
   111→        u ? (
   112→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   113→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   114→              {getInitials(u.name)}
   115→            </AvatarFallback>
   116→          </Avatar>
   117→        ) : null
   118→      )}
   119→      {extra > 0 && (
   120→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   121→          +{extra}
   122→        </div>
   123→      )}
   124→    </div>
   125→  )
   126→}
   127→
   128→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   129→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   130→  if (!user) return null
   131→  return (
   132→    <Avatar className={sizeClasses}>
   133→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   134→        {getInitials(user.name)}
   135→      </AvatarFallback>
   136→    </Avatar>
   137→  )
   138→}

   157→
   158→export function TeamProjects() {
   159→  const { user, setSelectedProjectId } = useAuthStore()
   160→  const [projects, setProjects] = useState<Project[]>([])
   161→  const [loading, setLoading] = useState(true)
   162→  const [expandedId, setExpandedId] = useState<string | null>(null)
   163→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   164→  const [expandedLoading, setExpandedLoading] = useState(false)
   165→  const [taskFilter, setTaskFilter] = useState<string>('all')
   166→  const [createOpen, setCreateOpen] = useState(false)
   167→  const [createLoading, setCreateLoading] = useState(false)
   168→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   169→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   170→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   171→
   172→  // Create project form
   173→  const [projName, setProjName] = useState('')
   174→  const [projDesc, setProjDesc] = useState('')
   175→  const [projPriority, setProjPriority] = useState('medium')
   176→  const [projDueDate, setProjDueDate] = useState<Date>()
   177→  const [projColor, setProjColor] = useState(PROJECT_COLORS[0])
   178→  const [projCalendarOpen, setProjCalendarOpen] = useState(false)
   179→
   180→  // Create task form
   181→  const [taskTitle, setTaskTitle] = useState('')
   182→  const [taskDesc, setTaskDesc] = useState('')
   183→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   184→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   185→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   186→  const [taskTags, setTaskTags] = useState('')
   187→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   188→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   189→
   190→  const fetchProjects = useCallback(async () => {
   191→    try {
   192→      const res = await fetch('/api/projects')
   193→      const data = await res.json()
   194→      const filtered = (data.projects || []).filter((p: Project) =>
   195→        p.members?.some((m) => m.userId === user?.id)
   196→      )
   197→      setProjects(filtered)
   198→    } catch {
   199→      toast.error('Failed to load projects')
   200→    } finally {
   201→      setLoading(false)
   202→    }
   203→  }, [user])
   204→
   205→  useEffect(() => { fetchProjects() }, [fetchProjects])
   206→
   207→  const handleExpand = async (projectId: string) => {
   208→    if (expandedId === projectId) {
   209→      setExpandedId(null)
   210→      setExpandedTasks([])
   211→      return
   212→    }
   213→    setExpandedId(projectId)
   214→    setTaskFilter('all')
   215→    setExpandedLoading(true)
   216→    try {
   217→      const res = await fetch(`/api/projects/${projectId}`)
   218→      const data = await res.json()
   219→      setExpandedTasks(data.project?.tasks || [])
   220→    } catch {
   221→      toast.error('Failed to load tasks')
   222→    } finally {
   223→      setExpandedLoading(false)
   224→    }
   225→  }
   226→
   227→  const handleCreateProject = async () => {
   228→    if (!projName.trim()) { toast.error('Project name is required'); return }
   229→    setCreateLoading(true)
   230→    try {
   231→      await fetch('/api/projects', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({
   235→          name: projName, description: projDesc || null, priority: projPriority,
   236→          dueDate: projDueDate ? projDueDate.toISOString() : null, color: projColor,
   237→        }),
   238→      })
   239→      toast.success('Project created')
   240→      setCreateOpen(false)
   241→      setProjName(''); setProjDesc(''); setProjPriority('medium')
   242→      setProjDueDate(undefined); setProjColor(PROJECT_COLORS[0])
   243→      fetchProjects()
   244→    } catch {
   245→      toast.error('Failed to create project')
   246→    } finally {
   247→      setCreateLoading(false)
   248→    }
   249→  }
   250→
   251→  const handleCreateTask = async () => {
   252→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   253→    if (!user?.id) { toast.error('Not authenticated'); return }
   254→    setCreateTaskLoading(true)
   255→    try {
   256→      const res = await fetch('/api/tasks', {
   257→        method: 'POST',
   258→        headers: { 'Content-Type': 'application/json' },
   259→        body: JSON.stringify({
   260→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   261→          priority: taskPriority, projectId: createTaskProjectId,
   262→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   263→          tags: taskTags,
   264→          createdById: user.id,
   265→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   266→        }),
   267→      })
   268→      if (!res.ok) {
   269→        const d = await res.json()
   270→        throw new Error(d.error || 'Failed to create task')
   271→      }
   272→      toast.success('Task created')
   273→      setCreateTaskOpen(false)
   274→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   275→      setTaskPriority('medium'); setTaskDueDate(undefined)
   276→      setTaskTags(''); setTaskAssignee('')
   277→      // Always refresh the expanded project view
   278→      handleExpand(createTaskProjectId)
   279→    } catch (e: unknown) {
   280→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   281→    } finally {
   282→      setCreateTaskLoading(false)
   283→    }
   284→  }
   285→
   286→  const openCreateTaskForProject = (projectId: string) => {
   287→    setCreateTaskProjectId(projectId)
   288→    setCreateTaskOpen(true)
   289→  }
   290→
   291→  const getProjectTaskStats = (projectId: string) => {
   292→    if (expandedId !== projectId) return null
   293→    const total = expandedTasks.length
   294→    const done = expandedTasks.filter((t) => t.status === 'done').length
   295→    return { total, done }
   296→  }
   297→
   298→  const filteredTasks = taskFilter === 'all'
   299→    ? expandedTasks
   300→    : expandedTasks.filter((t) => t.status === taskFilter)
   301→
   302→  if (loading) {
   303→    return (
   304→      <div>
   305→        <div className="flex items-center justify-between mb-6">
   306→          <Skeleton className="h-8 w-48" />
   307→          <Skeleton className="h-10 w-36" />
   308→        </div>
   309→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   310→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   311→        </div>
   312→      </div>
   313→    )
   314→  }
   315→
   316→  return (
   317→    <div className="space-y-6">
   318→      {/* Header */}
   319→      <div className="flex items-center justify-between">
   320→        <div>
   321→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   322→          <p className="text-sm text-muted-foreground mt-1">
   323→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   324→          </p>
   325→        </div>
   326→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   327→          <DialogTrigger asChild>
   328→            <Button className="gap-2">
   329→              <Plus className="h-4 w-4" /> Create Project
   330→            </Button>
   331→          </DialogTrigger>
   332→          <DialogContent className="sm:max-w-lg">
   333→            <DialogHeader>
   334→              <DialogTitle>Create New Project</DialogTitle>
   335→              <DialogDescription>Add a new project to your workspace.</DialogDescription>
   336→            </DialogHeader>
   337→            <div className="space-y-4 py-2">
   338→              <div className="space-y-2">
   339→                <Label>Project Name</Label>
   340→                <Input placeholder="Enter project name" value={projName} onChange={(e) => setProjName(e.target.value)} />
   341→              </div>
   342→              <div className="space-y-2">
   343→                <Label>Description</Label>
   344→                <Textarea placeholder="Describe the project..." value={projDesc} onChange={(e) => setProjDesc(e.target.value)} rows={3} />
   345→              </div>
   346→              <div className="grid grid-cols-2 gap-4">
   347→                <div className="space-y-2">
   348→                  <Label>Priority</Label>
   349→                  <Select value={projPriority} onValueChange={setProjPriority}>
   350→                    <SelectTrigger><SelectValue /></SelectTrigger>
   351→                    <SelectContent>
   352→                      <SelectItem value="low">Low</SelectItem>
   353→                      <SelectItem value="medium">Medium</SelectItem>
   354→                      <SelectItem value="high">High</SelectItem>
   355→                      <SelectItem value="urgent">Urgent</SelectItem>
   356→                    </SelectContent>
   357→                  </Select>
   358→                </div>
   359→                <div className="space-y-2">
   360→                  <Label>Due Date</Label>
   361→                  <Popover open={projCalendarOpen} onOpenChange={setProjCalendarOpen}>
   362→                    <PopoverTrigger asChild>
   363→                      <Button variant="outline" className="w-full justify-start text-left font-normal gap-2">
   364→                        <CalendarIcon className="h-4 w-4" />
   365→                        {projDueDate ? format(projDueDate, 'PPP') : 'Pick a date'}
   366→                      </Button>
   367→                    </PopoverTrigger>
   368→                    <PopoverContent className="w-auto p-0" align="start">
   369→                      <Calendar mode="single" selected={projDueDate} onSelect={(d) => { setProjDueDate(d); setProjCalendarOpen(false) }} />
   370→                    </PopoverContent>
   371→                  </Popover>
   372→                </div>
   373→              </div>
   374→              <div className="space-y-2">
   375→                <Label>Color</Label>
   376→                <div className="flex flex-wrap gap-2">
   377→                  {PROJECT_COLORS.map((c) => (
   378→                    <button
   379→                      key={c}
   380→                      className={cn(
   381→                        'h-8 w-8 rounded-full transition-all ring-2',
   382→                        projColor === c ? 'ring-offset-2 ring-offset-background scale-110' : 'ring-transparent hover:scale-105',
   383→                      )}
   384→                      style={{ backgroundColor: c, ringColor: c }}
   385→                      onClick={() => setProjColor(c)}
   386→                    />
   387→                  ))}
   388→                </div>
   389→              </div>
   390→            </div>
   391→            <DialogFooter>
   392→              <Button variant="outline" onClick={() => setCreateOpen(false)}>Cancel</Button>
   393→              <Button onClick={handleCreateProject} disabled={createLoading} className="gap-2">
   394→                {createLoading && <Loader2 className="h-4 w-4 animate-spin" />}
   395→                Create Project
   396→              </Button>
   397→            </DialogFooter>
   398→          </DialogContent>
   399→        </Dialog>
   400→      </div>
   401→
   402→      {/* Project Grid */}
   403→      {projects.length === 0 ? (
   404→        <EmptyState
   405→          icon={FolderKanban}
   406→          title="No projects yet"
   407→          description="You haven't been assigned to any projects. Create one or ask an admin to assign you."
   408→        />
   409→      ) : (
   410→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   411→          <AnimatePresence>
   412→            {projects.map((project, idx) => {
   413→              const totalTasks = project._count?.tasks || 0
   414→              const members = project.members || []
   415→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   416→              const pConfig = getPriorityConfig(project.priority)
   417→              const isExpanded = expandedId === project.id
   418→              const stats = getProjectTaskStats(project.id)
   419→
   420→              return (
   421→                <motion.div
   422→                  key={project.id}
   423→                  initial={{ opacity: 0, y: 20 }}
   424→                  animate={{ opacity: 1, y: 0 }}
   425→                  transition={{ delay: idx * 0.05 }}
   426→                >
   427→                  <Card
   428→                    className={cn(
   429→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   430→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   431→                    )}
   432→                    onClick={() => handleExpand(project.id)}
   433→                  >
   434→                    <div className="flex">
   435→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   436→                      <CardContent className="flex-1 p-4">
   437→                        <div className="flex items-start justify-between mb-2">
   438→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   439→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   440→                            {pConfig.label}
   441→                          </Badge>
   442→                        </div>
   443→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   444→                          {project.description || 'No description'}
   445→                        </p>
   446→                        <Progress
   447→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   448→                          className="h-1.5 mb-3"
   449→                        />
   450→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   451→                          <div className="flex items-center gap-3">
   452→                            <span className="flex items-center gap-1">
   453→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   454→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   455→                            </span>
   456→                            <span className="flex items-center gap-1">
   457→                              <Users className="h-3 w-3" />
   458→                              {members.length}
   459→                            </span>
   460→                          </div>
   461→                          {project.dueDate && (
   462→                            <span className={cn(
   463→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   464→                                ? 'text-red-500' : '',
   465→                            )}>
   466→                              {formatDate(project.dueDate)}
   467→                            </span>
   468→                          )}
   469→                        </div>
   470→                        {memberUsers.length > 0 && (
   471→                          <div className="mt-3">
   472→                            <AvatarStack users={memberUsers} max={4} />
   473→                          </div>
   474→                        )}
   475→                      </CardContent>
   476→                    </div>
   477→                  </Card>
   478→
   479→                  {/* Expanded Task List */}
   480→                  <AnimatePresence>
   481→                    {isExpanded && (
   482→                      <motion.div
   483→                        initial={{ opacity: 0, height: 0 }}
   484→                        animate={{ opacity: 1, height: 'auto' }}
   485→                        exit={{ opacity: 0, height: 0 }}
   486→                        transition={{ duration: 0.2 }}
   487→                        className="overflow-hidden"
   488→                        onClick={(e) => e.stopPropagation()}
   489→                      >
   490→                        <Card className="mt-2 p-4 space-y-4">
   491→                          <div className="flex items-center justify-between">
   492→                            <div className="flex items-center gap-2">
   493→                              <h4 className="font-semibold text-sm">Tasks</h4>
   494→                              <div className="flex gap-1">
   495→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   496→                                  <Button
   497→                                    key={s}
   498→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   499→                                    size="sm"
   500→                                    className="h-7 text-xs px-2"
   501→                                    onClick={() => setTaskFilter(s)}
   502→                                  >
   503→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   504→                                  </Button>
   505→                                ))}

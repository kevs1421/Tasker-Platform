   115→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   116→  const visible = users.slice(0, max)
   117→  const extra = users.length - max
   118→  return (
   119→    <div className="flex -space-x-2">
   120→      {visible.map((u, i) =>
   121→        u ? (
   122→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   123→            {u.avatar && <AvatarImage src={u.avatar} alt={u.name} />}
   124→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   125→              {getInitials(u.name)}
   126→            </AvatarFallback>
   127→          </Avatar>
   128→        ) : null
   129→      )}
   130→      {extra > 0 && (
   131→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   132→          +{extra}
   133→        </div>
   134→      )}
   135→    </div>
   136→  )
   137→}

   139→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   140→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   141→  if (!user) return null
   142→  return (
   143→    <Avatar className={sizeClasses}>
   144→      {user.avatar && <AvatarImage src={user.avatar} alt={user.name} />}
   145→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   146→        {getInitials(user.name)}
   147→      </AvatarFallback>
   148→    </Avatar>
   149→  )
   150→}

   170→export function TeamProjects() {
   171→  const { user, setSelectedProjectId } = useAuthStore()
   172→  const isTraining = user?.role === 'training'
   173→  const [projects, setProjects] = useState<Project[]>([])
   174→  const [loading, setLoading] = useState(true)
   175→  const [expandedId, setExpandedId] = useState<string | null>(null)
   176→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   177→  const [expandedLoading, setExpandedLoading] = useState(false)
   178→  const [taskFilter, setTaskFilter] = useState<string>('all')
   179→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   180→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   181→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   182→  const [allTeamUsers, setAllTeamUsers] = useState<User[]>([])
   183→
   184→  // Create task form
   185→  const [taskTitle, setTaskTitle] = useState('')
   186→  const [taskDesc, setTaskDesc] = useState('')
   187→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   188→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   189→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   190→  const [taskTags, setTaskTags] = useState('')
   191→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   192→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   193→
   194→  const fetchProjects = useCallback(async () => {
   195→    try {
   196→      const res = await fetch('/api/projects')
   197→      const data = await res.json()
   198→      const filtered = (data.projects || []).filter((p: Project) =>
   199→        p.members?.some((m) => m.userId === user?.id)
   200→      )
   201→      setProjects(filtered)
   202→    } catch {
   203→      toast.error('Failed to load projects')
   204→    } finally {
   205→      setLoading(false)
   206→    }
   207→  }, [user])
   208→
   209→  useEffect(() => { fetchProjects() }, [fetchProjects])
   210→
   211→  // Fetch all active admin + team users for assignee picker
   212→  useEffect(() => {
   213→    fetch('/api/users')
   214→      .then(r => r.json())
   215→      .then(d => {
   216→        const members = (d.users || []).filter((u: User) => (u.role === 'team' || u.role === 'training' || u.role === 'admin') && u.status === 'active')
   217→        setAllTeamUsers(members)
   218→      })
   219→      .catch(() => {})
   220→  }, [])
   221→
   222→  const handleExpand = async (projectId: string) => {
   223→    if (expandedId === projectId) {
   224→      setExpandedId(null)
   225→      setExpandedTasks([])
   226→      return
   227→    }
   228→    setExpandedId(projectId)
   229→    setTaskFilter('all')
   230→    setExpandedLoading(true)
   231→    try {
   232→      const res = await fetch(`/api/projects/${projectId}`)
   233→      const data = await res.json()
   234→      setExpandedTasks(data.project?.tasks || [])
   235→    } catch {
   236→      toast.error('Failed to load tasks')
   237→    } finally {
   238→      setExpandedLoading(false)
   239→    }
   240→  }
   241→
   242→  const handleCreateTask = async () => {
   243→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   244→    if (!user?.id) { toast.error('Not authenticated'); return }
   245→    setCreateTaskLoading(true)
   246→    try {
   247→      const res = await fetch('/api/tasks', {
   248→        method: 'POST',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify({
   251→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   252→          priority: taskPriority, projectId: createTaskProjectId,
   253→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   254→          tags: taskTags,
   255→          createdById: user.id,
   256→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   257→        }),
   258→      })
   259→      if (!res.ok) {
   260→        const d = await res.json()
   261→        throw new Error(d.error || 'Failed to create task')
   262→      }
   263→      toast.success('Task created')
   264→      setCreateTaskOpen(false)
   265→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   266→      setTaskPriority('medium'); setTaskDueDate(undefined)
   267→      setTaskTags(''); setTaskAssignee('')
   268→      // Always refresh the expanded project view
   269→      handleExpand(createTaskProjectId)
   270→    } catch (e: unknown) {
   271→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   272→    } finally {
   273→      setCreateTaskLoading(false)
   274→    }
   275→  }
   276→
   277→  const openCreateTaskForProject = (projectId: string) => {
   278→    setCreateTaskProjectId(projectId)
   279→    setCreateTaskOpen(true)
   280→  }
   281→
   282→  const getProjectTaskStats = (projectId: string) => {
   283→    if (expandedId !== projectId) return null
   284→    const total = expandedTasks.length
   285→    const done = expandedTasks.filter((t) => t.status === 'done').length
   286→    return { total, done }
   287→  }
   288→
   289→  const filteredTasks = taskFilter === 'all'
   290→    ? expandedTasks
   291→    : expandedTasks.filter((t) => t.status === taskFilter)
   292→
   293→  if (loading) {
   294→    return (
   295→      <div>
   296→        <div className="flex items-center justify-between mb-6">
   297→          <Skeleton className="h-8 w-48" />
   298→          <Skeleton className="h-10 w-36" />
   299→        </div>
   300→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   301→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   302→        </div>
   303→      </div>
   304→    )
   305→  }
   306→
   307→  return (
   308→    <div className="space-y-6">
   309→      {/* Header */}
   310→      <div className="flex items-center justify-between">
   311→        <div>
   312→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   313→          <p className="text-sm text-muted-foreground mt-1">
   314→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   315→          </p>
   316→        </div>
   317→      </div>
   318→
   319→      {/* Project Grid */}
   320→      {projects.length === 0 ? (
   321→        <EmptyState
   322→          icon={FolderKanban}
   323→          title="No projects yet"
   324→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   325→        />
   326→      ) : (
   327→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   328→          <AnimatePresence>
   329→            {projects.map((project, idx) => {
   330→              const totalTasks = project._count?.tasks || 0
   331→              const members = project.members || []
   332→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   333→              const pConfig = getPriorityConfig(project.priority)
   334→              const isExpanded = expandedId === project.id
   335→              const stats = getProjectTaskStats(project.id)
   336→
   337→              return (
   338→                <motion.div
   339→                  key={project.id}
   340→                  initial={{ opacity: 0, y: 20 }}
   341→                  animate={{ opacity: 1, y: 0 }}
   342→                  transition={{ delay: idx * 0.05 }}
   343→                >
   344→                  <Card
   345→                    className={cn(
   346→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   347→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   348→                    )}
   349→                    onClick={() => handleExpand(project.id)}
   350→                  >
   351→                    <div className="flex">
   352→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   353→                      <CardContent className="flex-1 p-4">
   354→                        <div className="flex items-start justify-between mb-2">
   355→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   356→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   357→                            {pConfig.label}
   358→                          </Badge>
   359→                        </div>
   360→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   361→                          {project.description || 'No description'}
   362→                        </p>
   363→                        <Progress
   364→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   365→                          className="h-1.5 mb-3"
   366→                        />
   367→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   368→                          <div className="flex items-center gap-3">
   369→                            <span className="flex items-center gap-1">
   370→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   371→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   372→                            </span>
   373→                            <span className="flex items-center gap-1">
   374→                              <Users className="h-3 w-3" />
   375→                              {members.length}
   376→                            </span>
   377→                          </div>
   378→                          {project.dueDate && (
   379→                            <span className={cn(
   380→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   381→                                ? 'text-red-500' : '',
   382→                            )}>
   383→                              {formatDate(project.dueDate)}
   384→                            </span>
   385→                          )}
   386→                        </div>
   387→                        {memberUsers.length > 0 && (
   388→                          <div className="mt-3">
   389→                            <AvatarStack users={memberUsers} max={4} />
   390→                          </div>
   391→                        )}
   392→                      </CardContent>
   393→                    </div>
   394→                  </Card>
   395→
   396→                  {/* Expanded Task List */}
   397→                  <AnimatePresence>
   398→                    {isExpanded && (
   399→                      <motion.div
   400→                        initial={{ opacity: 0, height: 0 }}
   401→                        animate={{ opacity: 1, height: 'auto' }}
   402→                        exit={{ opacity: 0, height: 0 }}
   403→                        transition={{ duration: 0.2 }}
   404→                        className="overflow-hidden"
   405→                        onClick={(e) => e.stopPropagation()}
   406→                      >
   407→                        <Card className="mt-3 p-4 space-y-4">
   408→                          <div className="flex items-center justify-between">
   409→                            <div className="flex items-center gap-2">
   410→                              <div className="flex gap-1">
   411→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   412→                                  <Button
   413→                                    key={s}
   414→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   415→                                    size="sm"
   416→                                    className="h-7 text-xs px-2"
   417→                                    onClick={() => setTaskFilter(s)}
   418→                                  >
   419→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   420→                                  </Button>
   421→                                ))}
   422→                              </div>
   423→                            </div>
   424→                            {!isTraining && (
   425→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   426→                              <Plus className="h-3 w-3" /> Add Task
   427→                            </Button>
   428→                            )}
   429→                          </div>
   430→                          {expandedLoading ? (
   431→                            <div className="space-y-2">
   432→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   433→                            </div>
   434→                          ) : filteredTasks.length === 0 ? (
   435→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   436→                          ) : (
   437→                            <ScrollArea className="max-h-64">
   438→                              <div className="space-y-2">
   439→                                {filteredTasks.map((task) => {
   440→                                  const sConfig = getStatusConfig(task.status)
   441→                                  const pConf = getPriorityConfig(task.priority)
   442→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   443→                                  return (
   444→                                    <div
   445→                                      key={task.id}
   446→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   447→                                      onClick={() => setSelectedProjectId(project.id)}
   448→                                    >
   449→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   450→                                      <div className="flex-1 min-w-0">
   451→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   452→                                        <div className="flex items-center gap-2 mt-0.5">
   453→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   454→                                          {task.dueDate && (
   455→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   456→                                              {formatDate(task.dueDate)}
   457→                                            </span>
   458→                                          )}
   459→                                        </div>
   460→                                      </div>
   461→                                      <AvatarStack users={assignees} max={2} />
   462→                                    </div>
   463→                                  )
   464→                                })}
   465→                              </div>
   466→                            </ScrollArea>
   467→                          )}
   468→                        </Card>
   469→                      </motion.div>
   470→                    )}
   471→                  </AnimatePresence>
   472→                </motion.div>
   473→              )
   474→            })}
   475→          </AnimatePresence>
   476→        </motion.div>
   477→      )}
   478→
   479→      {/* Create Task Dialog */}
   480→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   481→        <DialogContent className="sm:max-w-lg">
   482→          <DialogHeader>
   483→            <DialogTitle>Create New Task</DialogTitle>
   484→            <DialogDescription>Add a task to the project.</DialogDescription>
   485→          </DialogHeader>
   486→          <div className="space-y-4 py-2">
   487→            <div className="space-y-2">
   488→              <Label>Title</Label>
   489→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   490→            </div>
   491→            <div className="space-y-2">
   492→              <Label>Description</Label>
   493→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   494→            </div>
   495→            <div className="grid grid-cols-2 gap-4">
   496→              <div className="space-y-2">
   497→                <Label>Status</Label>
   498→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   499→                  <SelectTrigger><SelectValue /></SelectTrigger>
   500→                  <SelectContent>
   501→                    <SelectItem value="todo">To Do</SelectItem>
   502→                    <SelectItem value="in_progress">In Progress</SelectItem>
   503→                    <SelectItem value="in_review">In Review</SelectItem>
   504→                    <SelectItem value="done">Done</SelectItem>
   505→                  </SelectContent>
   506→                </Select>
   507→              </div>

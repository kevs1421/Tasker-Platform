    44→function formatElapsed(ms: number): string {
    45→  const totalSeconds = Math.floor(ms / 1000)
    46→  const h = Math.floor(totalSeconds / 3600)
    47→  const m = Math.floor((totalSeconds % 3600) / 60)
    48→  const s = totalSeconds % 60
    49→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    50→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    51→}
    52→
    53→function formatCountdown(ms: number): string {
    54→  if (ms <= 0) return '0:00'
    55→  const totalSeconds = Math.floor(ms / 1000)
    56→  const m = Math.floor(totalSeconds / 60)
    57→  const s = totalSeconds % 60
    58→  return `${m}:${String(s).padStart(2, '0')}`
    59→}

   125→      if (res.ok) {
   126→        const data = await res.json()

   140→                if (config.autoEnabled) {

   151→                      setIsCapturing(false)
   152→                    },
   153→                    onCaptureError: () => setIsCapturing(false),

   155→                }
   156→              })
   157→              .catch(() => {})
   158→          }
   159→        }
   160→      }

   163→    } finally {
   164→      setLoading(false)

   524→export function AppHeader() {
   525→  const { user, logout } = useAuthStore()
   526→  const { theme, setTheme } = useTheme()
   527→  const mounted = useSyncExternalStore(() => () => {}, () => true, () => false)
   528→  const [unreadCount, setUnreadCount] = useState(0)
   529→
   530→  useEffect(() => {
   531→    if (user) {
   532→      fetch(`/api/messages?userId=${user.id}`)
   533→        .then(r => r.json())
   534→        .then(d => setUnreadCount(d.conversations?.reduce((sum: number, c: { unread: number }) => sum + c.unread, 0) || 0))
   535→        .catch(() => {})
   536→    }
   537→  }, [user])
   538→
   539→  // Stop screenshot session on logout
   540→  const handleLogout = () => {
   541→    screenshotSession.stop()
   542→    logout()
   543→  }

   741→function OfflineBanner() {
   742→  const [isOffline, setIsOffline] = useState(() => {
   743→    if (typeof navigator === 'undefined') return false
   744→    return !navigator.onLine
   745→  })
   746→
   747→  useEffect(() => {
   748→    const handleOnline = () => setIsOffline(false)
   749→    const handleOffline = () => setIsOffline(true)
   750→    window.addEventListener('online', handleOnline)
   751→    window.addEventListener('offline', handleOffline)
   752→    return () => {
   753→      window.removeEventListener('online', handleOnline)
   754→      window.removeEventListener('offline', handleOffline)
   755→    }
   756→  }, [])
   757→
   758→  if (!isOffline) return null
   759→
   760→  return (
   761→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   762→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   763→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   764→    </div>
   765→  )
   766→}

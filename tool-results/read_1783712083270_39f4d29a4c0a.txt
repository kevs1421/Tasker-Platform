    64→function ActivityIcon({ action }: { action: string }) {
    65→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    66→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    67→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    68→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    69→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    70→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    71→  return <CircleDot className="h-4 w-4 text-slate-400" />
    72→}

   270→  }

   359→  async function openForm(form: OnboardingForm) {
   360→    try {
   361→      const res = await fetch(`/api/onboarding/${form.id}`)
   362→      if (!res.ok) throw new Error('Failed')

   364→      setSelectedForm(data.form)

   371→  async function handleCreateOnboarding() {
   372→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   373→    setCreating(true)
   374→    try {
   375→      const stepsJson = JSON.stringify(
   376→        createForm.selectedSteps.map(stepId => {
   377→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   378→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   379→        })
   380→      )
   381→      const res = await fetch('/api/onboarding', {
   382→        method: 'POST',
   383→        headers: { 'Content-Type': 'application/json' },
   384→        body: JSON.stringify({
   385→          title: createForm.title.trim(),
   386→          description: createForm.description.trim() || null,
   387→          stepsJson,
   388→          projectId: createForm.projectId,
   389→          clientId: createForm.clientId || null,
   390→          status: 'published',
   391→        }),
   392→      })
   393→      if (!res.ok) {
   394→        const d = await res.json()
   395→        throw new Error(d.error || 'Failed to create')
   396→      }
   397→      toast.success('Onboarding form created successfully')
   398→      setCreateOpen(false)
   399→      setCreateForm({ title: '', description: '', projectId: '', clientId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })
   400→      fetchForms()
   401→    } catch (e: unknown) {
   402→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding form')
   403→    } finally {
   404→      setCreating(false)
   405→    }
   406→  }

   408→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   409→    if (!user) return
   410→    setReviewing(submissionId)
   411→    try {
   412→      const notes = reviewNotes[submissionId] || ''
   413→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   414→        method: 'PATCH',
   415→        headers: { 'Content-Type': 'application/json' },
   416→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   417→      })
   418→      if (!patchRes.ok) {
   419→        const d = await patchRes.json()
   420→        throw new Error(d.error)
   421→      }
   422→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   423→      // Refresh
   424→      if (selectedForm) {
   425→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   426→        if (fresh.ok) {
   427→          const data = await fresh.json()
   428→          setSelectedForm(data.form)
   429→        }
   430→      }
   431→      fetchForms()
   432→    } catch (e: unknown) {
   433→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   434→    } finally {
   435→      setReviewing(null)
   436→    }
   437→  }

    67→    67→function ActivityIcon({ action }: { action: string }) {
    68→    68→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    69→    69→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    70→    70→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    71→    71→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    72→    72→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    73→    73→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    74→    74→  return <CircleDot className="h-4 w-4 text-slate-400" />

   440→   440→      setReviewing(null)
   441→   441→    }

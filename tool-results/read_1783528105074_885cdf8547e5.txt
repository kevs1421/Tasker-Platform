    61→function ActivityIcon({ action }: { action: string }) {
    62→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    63→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    64→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    65→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    66→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    67→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    68→  return <CircleDot className="h-4 w-4 text-slate-400" />
    69→}

   260→function getOnboardingBadge(status: string) {
   261→  switch (status) {
   262→    case 'in_progress': return 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400'
   263→    case 'submitted': return 'bg-purple-100 text-purple-700 dark:bg-purple-900/30 dark:text-purple-400'
   264→    case 'approved': return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   265→    case 'rejected': return 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-400'
   266→    default: return 'bg-slate-100 text-slate-700'
   267→  }
   268→}
   269→
   270→function getOnboardingStatusLabel(status: string) {
   271→  const map: Record<string, string> = { in_progress: 'In Progress', submitted: 'Submitted', approved: 'Approved', rejected: 'Rejected' }
   272→  return map[status] || status
   273→}

   275→export function AdminOnboarding() {
   276→  const { user } = useAuthStore()
   277→  const [loading, setLoading] = useState(true)
   278→  const [forms, setForms] = useState<OnboardingForm[]>([])
   279→  const [selectedForm, setSelectedForm] = useState<(OnboardingForm & { submissions: Onboarding[] }) | null>(null)
   280→  const [dialogOpen, setDialogOpen] = useState(false)
   281→  const [reviewing, setReviewing] = useState<string | null>(null)
   282→  const [reviewNotes, setReviewNotes] = useState<Record<string, string>>({})
   283→
   284→  useEffect(() => {
   285→    async function fetchData() {
   286→      try {
   287→        const res = await fetch('/api/onboarding')
   288→        if (!res.ok) throw new Error('Failed')
   289→        const data = await res.json()
   290→        setForms(data.forms ?? [])
   291→      } catch {
   292→        toast.error('Failed to load onboarding forms')
   293→      } finally {
   294→        setLoading(false)
   295→      }
   296→    }
   297→    fetchData()
   298→  }, [])
   299→
   300→  async function openForm(form: OnboardingForm) {
   301→    try {
   302→      const res = await fetch(`/api/onboarding/${form.id}`)
   303→      if (!res.ok) throw new Error('Failed')
   304→      const data = await res.json()
   305→      setSelectedForm(data.form)
   306→      setDialogOpen(true)
   307→    } catch {
   308→      toast.error('Failed to load form details')
   309→    }
   310→  }
   311→
   312→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   313→    if (!user) return
   314→    setReviewing(submissionId)
   315→    try {
   316→      const notes = reviewNotes[submissionId] || ''
   317→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   318→        method: 'PATCH',
   319→        headers: { 'Content-Type': 'application/json' },
   320→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   321→      })
   322→      if (!patchRes.ok) {
   323→        const d = await patchRes.json()
   324→        throw new Error(d.error)
   325→      }
   326→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   327→      // Refresh
   328→      if (selectedForm) {
   329→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   330→        if (fresh.ok) {
   331→          const data = await fresh.json()
   332→          setSelectedForm(data.form)
   333→        }
   334→      }
   335→    } catch (e: unknown) {
   336→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   337→    } finally {
   338→      setReviewing(null)
   339→    }
   340→  }
   341→
   342→  if (loading) {
   343→    return (
   344→      <div className="space-y-4">
   345→        <Skeleton className="h-10 w-48" />
   346→        <Card><CardContent className="p-4"><Skeleton className="h-64 w-full" /></CardContent></Card>
   347→      </div>
   348→    )
   349→  }
   350→
   351→  return (
   352→    <motion.div {...fadeIn} className="space-y-6">
   353→      <div>
   354→        <h2 className="text-xl font-semibold">Onboarding Management</h2>
   355→        <p className="text-sm text-muted-foreground">Review and manage client onboarding submissions</p>
   356→      </div>
   357→
   358→      <Card>
   359→        <CardContent className="p-0">
   360→          <Table>
   361→            <TableHeader>
   362→              <TableRow>
   363→                <TableHead>Form Title</TableHead>
   364→                <TableHead>Project</TableHead>
   365→                <TableHead className="text-center">Submissions</TableHead>
   366→                <TableHead>Status</TableHead>
   367→                <TableHead className="w-16"></TableHead>
   368→              </TableRow>
   369→            </TableHeader>
   370→            <TableBody>
   371→              {forms.length === 0 && (
   372→                <TableRow><TableCell colSpan={5} className="text-center text-muted-foreground py-8">No onboarding forms found</TableCell></TableRow>
   373→              )}
   374→              {forms.map((form) => (
   375→                <TableRow key={form.id} className="cursor-pointer hover:bg-muted/50" onClick={() => openForm(form)}>
   376→                  <TableCell className="font-medium">{form.title}</TableCell>
   377→                  <TableCell>
   378→                    <div className="flex items-center gap-2">
   379→                      {form.project && <div className="h-3 w-3 rounded-full" style={{ backgroundColor: form.project.color }} />}
   380→                      <span className="text-sm text-muted-foreground">{form.project?.name ?? '—'}</span>
   381→                    </div>
   382→                  </TableCell>
   383→                  <TableCell className="text-center">{form._count?.submissions ?? 0}</TableCell>
   384→                  <TableCell>
   385→                    <Badge variant="secondary" className={getOnboardingBadge(form.status)}>
   386→                      {getOnboardingStatusLabel(form.status)}
   387→                    </Badge>
   388→                  </TableCell>
   389→                  <TableCell>
   390→                    <Button variant="ghost" size="icon" className="h-8 w-8">
   391→                      <Eye className="h-4 w-4" />
   392→                    </Button>
   393→                  </TableCell>
   394→                </TableRow>
   395→              ))}
   396→            </TableBody>
   397→          </Table>
   398→        </CardContent>
   399→      </Card>
   400→
   401→      {/* Submissions Dialog */}
   402→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   403→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-hidden flex flex-col">
   404→          <DialogHeader>
   405→            <DialogTitle>{selectedForm?.title ?? 'Form'}</DialogTitle>
   406→            <DialogDescription>
   407→              {selectedForm?.project?.name ?? 'Project'} — {selectedForm?.submissions?.length ?? 0} submissions
   408→            </DialogDescription>
   409→          </DialogHeader>
   410→          <ScrollArea className="flex-1 -mx-6 px-6">
   411→            <div className="space-y-4 py-2">
   412→              {selectedForm?.submissions?.length === 0 && (
   413→                <p className="text-sm text-muted-foreground text-center py-6">No submissions yet</p>
   414→              )}
   415→              {selectedForm?.submissions?.map((sub) => (
   416→                <Card key={sub.id} className="border">
   417→                  <CardContent className="p-4 space-y-3">
   418→                    <div className="flex items-start justify-between">
   419→                      <div className="flex items-center gap-3">
   420→                        <Avatar className="h-9 w-9">
   421→                          {sub.client?.avatar ? <AvatarImage src={sub.client.avatar} /> : null}
   422→                          <AvatarFallback className="text-xs">{sub.client ? getInitials(sub.client.name) : '?'}</AvatarFallback>
   423→                        </Avatar>
   424→                        <div>
   425→                          <p className="text-sm font-medium">{sub.client?.name ?? 'Unknown'}</p>
   426→                          <p className="text-xs text-muted-foreground">{sub.client?.company ?? 'No company'}</p>
   427→                        </div>
   428→                      </div>
   429→                      <Badge variant="secondary" className={getOnboardingBadge(sub.status)}>
   430→                        {getOnboardingStatusLabel(sub.status)}
   431→                      </Badge>
   432→                    </div>
   433→
   434→                    {sub.submittedAt && (
   435→                      <p className="text-xs text-muted-foreground">Submitted {formatDate(sub.submittedAt)}</p>
   436→                    )}
   437→
   438→                    {sub.reviewNotes && (
   439→                      <div className="bg-muted/50 rounded-md p-2 text-xs">
   440→                        <span className="font-medium">Review notes:</span> {sub.reviewNotes}
   441→                        {sub.reviewer && <span className="text-muted-foreground"> — by {sub.reviewer.name}</span>}
   442→                      </div>
   443→                    )}
   444→
   445→                    {sub.status === 'submitted' && (
   446→                      <div className="space-y-2">
   447→                        <Textarea
   448→                          placeholder="Add review notes..."
   449→                          value={reviewNotes[sub.id] ?? ''}
   450→                          onChange={(e) => setReviewNotes(prev => ({ ...prev, [sub.id]: e.target.value }))}
   451→                          className="text-sm"
   452→                          rows={2}
   453→                        />
   454→                        <div className="flex gap-2">
   455→                          <Button
   456→                            size="sm"
   457→                            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   458→                            onClick={(e) => { e.stopPropagation(); handleReview(sub.id, 'approved') }}
   459→                            disabled={reviewing === sub.id}
   460→                          >
   461→                            {reviewing === sub.id ? <Skeleton className="h-4 w-16" /> : <><ThumbsUp className="h-3 w-3 mr-1" /> Approve</>}
   462→                          </Button>
   463→                          <Button

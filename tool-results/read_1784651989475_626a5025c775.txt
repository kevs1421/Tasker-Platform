    65→    65→function ActivityIcon({ action }: { action: string }) {
    66→    66→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    67→    67→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    68→    68→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    69→    69→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    70→    70→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    71→    71→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    72→    72→  return <CircleDot className="h-4 w-4 text-slate-400" />
    73→    73→}

   411→   411→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   412→   412→    if (!user) return
   413→   413→    setReviewing(submissionId)

   417→   417→        method: 'PATCH',
   418→   418→        headers: { 'Content-Type': 'application/json' },
   419→   419→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   420→   420→      })
   421→   421→      if (!patchRes.ok) {
   422→   422→        const d = await patchRes.json()
   423→   423→        throw new Error(d.error)
   424→   424→      }
   425→   425→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   426→   426→      // Refresh
   427→   427→      if (selectedForm) {
   428→   428→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   429→   429→        if (fresh.ok) {
   430→   430→          const data = await fresh.json()
   431→   431→          setSelectedForm(data.form)
   432→   432→        }
   433→   433→      }
   434→   434→      fetchForms()
   435→   435→    } catch (e: unknown) {
   436→   436→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   437→   437→    } finally {
   438→   438→      setReviewing(null)
   439→   439→    }

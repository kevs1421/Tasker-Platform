    96→    96→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
    97→    97→  const visible = users.slice(0, max)
    98→    98→  const extra = users.length - max
    99→    99→  return (
   100→   100→    <div className="flex -space-x-2">
   101→   101→      {visible.map((u, i) =>
   102→   102→        u ? (
   103→   103→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   104→   104→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   105→   105→              {getInitials(u.name)}
   106→   106→            </AvatarFallback>
   107→   107→          </Avatar>
   108→   108→        ) : null
   109→   109→      )}
   110→   110→      {extra > 0 && (
   111→   111→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   112→   112→          +{extra}
   113→   113→        </div>
   114→   114→      )}
   115→   115→    </div>
   116→   116→  )

   286→   286→  return (
   287→   287→    <div className="space-y-6">
   288→   288→      {/* Header */}
   289→   289→      <div className="flex items-center justify-between">

   409→   409→                          {expandedLoading ? (

   414→   414→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   415→   415→                          ) : (
   416→   416→                            <ScrollArea className="max-h-64">
   417→   417→                              <div className="space-y-2">
   418→   418→                                {filteredTasks.map((task) => {

   421→   421→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   422→   422→                                  return (

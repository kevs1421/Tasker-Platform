    67→function ActivityIcon({ action }: { action: string }) {
    68→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    69→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    70→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    71→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    72→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    73→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    74→  return <CircleDot className="h-4 w-4 text-slate-400" />
    75→}

   364→  async function openForm(form: OnboardingForm) {
   365→    try {
   366→      const res = await fetch(`/api/onboarding/${form.id}`)
   367→      if (!res.ok) throw new Error('Failed')
   368→      const data = await res.json()
   369→      setSelectedForm(data.form)

   376→  async function handleCreateOnboarding() {
   377→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   378→    setCreating(true)
   379→    try {
   380→      const stepsJson = JSON.stringify(
   381→        createForm.selectedSteps.map(stepId => {
   382→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   383→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   384→        })
   385→      )
   386→      const res = await fetch('/api/onboarding', {
   387→        method: 'POST',
   388→        headers: authHeaders({ 'Content-Type': 'application/json' }),
   389→        body: JSON.stringify({
   390→          title: createForm.title.trim(),
   391→          description: createForm.description.trim() || null,
   392→          stepsJson,
   393→          projectId: createForm.projectId,
   394→          clientId: createForm.clientId || null,
   395→          status: 'published',
   396→        }),
   397→      })
   398→      if (!res.ok) {
   399→        const d = await res.json()
   400→        throw new Error(d.error || 'Failed to create')
   401→      }
   402→      toast.success('Onboarding form created successfully')
   403→      setCreateOpen(false)
   404→      setCreateForm({ title: '', description: '', projectId: '', clientId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })
   405→      fetchForms()
   406→    } catch (e: unknown) {
   407→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding form')
   408→    } finally {
   409→      setCreating(false)
   410→    }
   411→  }

   413→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   414→    if (!user) return
   415→    setReviewing(submissionId)
   416→    try {
   417→      const notes = reviewNotes[submissionId] || ''
   418→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   419→        method: 'PATCH',
   420→        headers: authHeaders({ 'Content-Type': 'application/json' }),
   421→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   422→      })
   423→      if (!patchRes.ok) {
   424→        const d = await patchRes.json()
   425→        throw new Error(d.error)
   426→      }
   427→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   428→      // Refresh
   429→      if (selectedForm) {
   430→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   431→        if (fresh.ok) {
   432→          const data = await fresh.json()
   433→          setSelectedForm(data.form)
   434→        }
   435→      }
   436→      fetchForms()
   437→    } catch (e: unknown) {
   438→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   439→    } finally {
   440→      setReviewing(null)
   441→    }
   442→  }

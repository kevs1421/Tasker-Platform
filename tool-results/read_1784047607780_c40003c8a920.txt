    58→function UserAvatar({ user, size = 'md', className }: { user?: User | null; size?: 'sm' | 'md' | 'lg'; className?: string }) {
    59→  const sizeMap = { sm: 'h-8 w-8 text-[10px]', md: 'h-10 w-10 text-xs', lg: 'h-12 w-12 text-sm' }
    60→  if (!user) return <div className={cn(sizeMap[size], 'rounded-full bg-muted flex items-center justify-center', className)}>?</div>
    61→  return (
    62→    <Avatar className={cn(sizeMap[size], className)}>
    63→      {user.avatar && <AvatarImage src={user.avatar} />}
    64→      <AvatarFallback className="bg-slate-100 dark:bg-slate-800 font-medium">
    65→        {getInitials(user.name)}
    66→      </AvatarFallback>
    67→    </Avatar>
    68→  )
    69→}

    82→function CreateGroupDialog({
    83→  open, onOpenChange, onCreated, currentUser,
    84→}: {
    85→  open: boolean; onOpenChange: (open: boolean) => void;
    86→  onCreated: (group: MessageGroup) => void; currentUser: User;
    87→}) {
    88→  const [name, setName] = useState('')
    89→  const [users, setUsers] = useState<User[]>([])
    90→  const [selectedIds, setSelectedIds] = useState<string[]>([])
    91→  const [creating, setCreating] = useState(false)
    92→  const [search, setSearch] = useState('')
    93→  const [loading, setLoading] = useState(false)
    94→  const openCountRef = useRef(0)
    95→  const prevIsOpen = useRef(false)
    96→
    97→  useEffect(() => {
    98→    if (open && !prevIsOpen.current) {
    99→      openCountRef.current++
   100→    }
   101→    prevIsOpen.current = open
   102→  }, [open])
   103→
   104→  useEffect(() => {
   105→    if (!open) return
   106→    setLoading(true)
   107→    setName('')
   108→    setSelectedIds([])
   109→    setSearch('')
   110→    fetch('/api/users')
   111→      .then((r) => r.json())
   112→      .then((d) => {
   113→        setUsers((d.users ?? []).filter((u: User) => u.id !== currentUser.id))
   114→      })
   115→      .catch(() => {})
   116→      .finally(() => setLoading(false))
   117→  }, [open, currentUser.id])
   118→
   119→  const toggleUser = (userId: string) => {
   120→    setSelectedIds((prev) => prev.includes(userId) ? prev.filter((id) => id !== userId) : [...prev, userId])
   121→  }
   122→
   123→  const handleCreate = async () => {
   124→    if (!name.trim() || selectedIds.length === 0) return
   125→    setCreating(true)
   126→    try {
   127→      const res = await fetch('/api/message-groups', {
   128→        method: 'POST',
   129→        headers: { 'Content-Type': 'application/json' },
   130→        body: JSON.stringify({ name: name.trim(), createdBy: currentUser.id, memberIds: selectedIds }),
   131→      })
   132→      if (!res.ok) throw new Error()
   133→      const data = await res.json()
   134→      toast.success('Group created')
   135→      onCreated(data.group)
   136→      onOpenChange(false)
   137→    } catch {
   138→      toast.error('Failed to create group')
   139→    } finally {
   140→      setCreating(false)
   141→    }
   142→  }
   143→
   144→  const filteredUsers = users.filter(
   145→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   146→  )
   147→
   148→  return (
   149→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   150→      <DialogContent className="sm:max-w-md">
   151→        <DialogHeader>
   152→          <DialogTitle className="flex items-center gap-2">
   153→            <Users className="h-5 w-5 text-violet-600" /> Create Group
   154→          </DialogTitle>
   155→          <DialogDescription>Start a group conversation with team members</DialogDescription>
   156→        </DialogHeader>
   157→
   158→        <div className="space-y-4 py-2">
   159→          <div>
   160→            <Input placeholder="Group name..." value={name} onChange={(e) => setName(e.target.value)} autoFocus />
   161→          </div>
   162→
   163→          <div>
   164→            <div className="relative mb-2">
   165→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   166→              <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   167→            </div>
   168→            <ScrollArea className="max-h-48 rounded-md border">
   169→              {loading ? (
   170→                <div className="p-3 space-y-2">
   171→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   172→                </div>
   173→              ) : filteredUsers.length === 0 ? (
   174→                <p className="text-sm text-muted-foreground text-center py-4">No users found</p>
   175→              ) : (
   176→                <div className="p-1">
   177→                  {filteredUsers.map((u) => (
   178→                    <button
   179→                      key={u.id}
   180→                      type="button"
   181→                      className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   182→                      onClick={() => toggleUser(u.id)}
   183→                    >
   184→                      <Checkbox checked={selectedIds.includes(u.id)} />
   185→                      <UserAvatar user={u} size="sm" />
   186→                      <div className="flex-1 min-w-0">
   187→                        <p className="text-sm font-medium truncate">{u.name}</p>
   188→                        <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   189→                      </div>
   190→                      <Badge variant="secondary" className={cn('text-[10px]', getRoleBadgeClasses(u.role))}>{u.role}</Badge>
   191→                    </button>
   192→                  ))}
   193→                </div>
   194→              )}
   195→            </ScrollArea>
   196→            {selectedIds.length > 0 && (
   197→              <p className="text-xs text-muted-foreground mt-2">{selectedIds.length} member{selectedIds.length > 1 ? 's' : ''} selected</p>
   198→            )}
   199→          </div>
   200→        </div>
   201→
   202→        <DialogFooter>
   203→          <Button variant="outline" type="button" onClick={() => onOpenChange(false)}>Cancel</Button>
   204→          <Button
   205→            type="button"
   206→            onClick={handleCreate} disabled={!name.trim() || selectedIds.length === 0 || creating}
   207→            className="bg-violet-600 hover:bg-violet-700 text-white"
   208→          >
   209→            {creating && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   210→            Create Group
   211→          </Button>
   212→        </DialogFooter>
   213→      </DialogContent>
   214→    </Dialog>
   215→  )
   216→}

   307→function GroupInfoDialog({
   308→  open, onOpenChange, group, currentUserId,
   309→}: {
   310→  open: boolean; onOpenChange: (open: boolean) => void
   311→  group: { id: string; name: string; members?: GroupMember[]; _count?: { members: number } } | null
   312→  currentUserId: string
   313→}) {
   314→  const [members, setMembers] = useState<(GroupMember & { user: User })[]>([])
   315→  const [users, setUsers] = useState<User[]>([])
   316→  const [loading, setLoading] = useState(false)
   317→  const [showAddUser, setShowAddUser] = useState(false)
   318→  const [search, setSearch] = useState('')
   319→  const openCountRef = useRef(0)
   320→  const prevIsOpen = useRef(false)
   321→
   322→  useEffect(() => {
   323→    if (open && !prevIsOpen.current) {
   324→      openCountRef.current++
   325→    }
   326→    prevIsOpen.current = open
   327→  }, [open])
   328→
   329→  useEffect(() => {
   330→    if (!open || !group) return
   331→    setLoading(true)
   332→    setSearch('')
   333→    setShowAddUser(false)
   334→    fetch(`/api/message-groups/${group.id}?userId=${currentUserId}`)
   335→      .then((r) => r.json())
   336→      .then((d) => {
   337→        setMembers((d.group?.members ?? []) as (GroupMember & { user: User })[])
   338→      })
   339→      .catch(() => {})
   340→      .finally(() => setLoading(false))
   341→  }, [open, group, currentUserId, openCountRef.current])
   342→
   343→  const handleAddMember = async (userId: string) => {
   344→    if (!group) return
   345→    try {
   346→      const res = await fetch(`/api/message-groups/${group.id}`, {
   347→        method: 'PATCH',
   348→        headers: { 'Content-Type': 'application/json' },
   349→        body: JSON.stringify({ addMemberIds: [userId] }),
   350→      })
   351→      if (!res.ok) throw new Error()
   352→      const data = await res.json()
   353→      setMembers(data.group.members ?? [])
   354→      toast.success('Member added')
   355→    } catch {
   356→      toast.error('Failed to add member')
   357→    }
   358→  }
   359→
   360→  const handleRemoveMember = async (userId: string) => {
   361→    if (!group) return
   362→    try {
   363→      const res = await fetch(`/api/message-groups/${group.id}`, {
   364→        method: 'PATCH',
   365→        headers: { 'Content-Type': 'application/json' },
   366→        body: JSON.stringify({ removeMemberIds: [userId] }),
   367→      })
   368→      if (!res.ok) throw new Error()
   369→      const data = await res.json()
   370→      setMembers(data.group.members ?? [])
   371→      toast.success('Member removed')
   372→    } catch {
   373→      toast.error('Failed to remove member')
   374→    }
   375→  }
   376→
   377→  const handleLoadUsers = () => {
   378→    fetch('/api/users')
   379→      .then((r) => r.json())
   380→      .then((d) => {
   381→        const memberIds = members.map((m) => m.userId)
   382→        setUsers((d.users ?? []).filter((u: User) => !memberIds.includes(u.id)))
   383→      })
   384→      .catch(() => {})
   385→    setShowAddUser(true)
   386→  }
   387→
   388→  const filteredUsers = users.filter(
   389→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   390→  )
   391→
   392→  return (
   393→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   394→      <DialogContent className="sm:max-w-md">
   395→        <DialogHeader>
   396→          <DialogTitle className="flex items-center gap-2">
   397→            <Users className="h-5 w-5 text-violet-600" /> Group Info
   398→          </DialogTitle>
   399→        </DialogHeader>
   400→
   401→        <div className="space-y-4 py-2">
   402→          {group && (
   403→            <div className="flex items-center gap-3">
   404→              <GroupAvatar name={group.name} size="lg" />
   405→              <div>
   406→                <p className="font-semibold">{group.name}</p>
   407→                <p className="text-sm text-muted-foreground">{group._count?.members ?? members.length} members</p>
   408→              </div>
   409→            </div>
   410→          )}
   411→
   412→          <Separator />
   413→
   414→          <div>
   415→            <div className="flex items-center justify-between mb-2">
   416→              <p className="text-sm font-medium">Members</p>
   417→              {!showAddUser && (
   418→                <Button variant="ghost" size="sm" className="h-7 text-xs gap-1" type="button" onClick={handleLoadUsers}>
   419→                  <UserPlus className="h-3.5 w-3.5" /> Add
   420→                </Button>
   421→              )}
   422→            </div>
   423→
   424→            {showAddUser && (
   425→              <div className="mb-3 space-y-2">
   426→                <div className="relative">
   427→                  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   428→                  <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   429→                </div>
   430→                <ScrollArea className="max-h-32 rounded-md border">
   431→                  <div className="p-1">
   432→                    {filteredUsers.slice(0, 8).map((u) => (
   433→                      <button
   434→                        key={u.id}
   435→                        type="button"
   436→                        className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   437→                        onClick={() => handleAddMember(u.id)}
   438→                      >
   439→                        <UserAvatar user={u} size="sm" />
   440→                        <div className="flex-1 min-w-0">
   441→                          <p className="text-sm font-medium truncate">{u.name}</p>
   442→                        </div>
   443→                        <UserPlus className="h-4 w-4 text-muted-foreground" />
   444→                      </button>
   445→                    ))}
   446→                  </div>
   447→                </ScrollArea>
   448→                <Button variant="ghost" size="sm" className="h-7 text-xs" type="button" onClick={() => setShowAddUser(false)}>Done</Button>
   449→              </div>
   450→            )}
   451→
   452→            <ScrollArea className="max-h-48 rounded-md border">
   453→              {loading ? (
   454→                <div className="p-3 space-y-2">
   455→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   456→                </div>
   457→              ) : (
   458→                <div className="p-1">
   459→                  {members.map((m) => (
   460→                    <div key={m.id} className="flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50">
   461→                      <UserAvatar user={m.user} size="sm" />
   462→                      <div className="flex-1 min-w-0">
   463→                        <p className="text-sm font-medium truncate">{m.user?.name ?? 'Unknown'}</p>
   464→                        <p className="text-xs text-muted-foreground">{m.role === 'admin' ? 'Admin' : 'Member'}</p>
   465→                      </div>
   466→                      {m.userId !== currentUserId && (
   467→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-muted-foreground hover:text-destructive" type="button" onClick={() => handleRemoveMember(m.userId)}>
   468→                          <X className="h-3.5 w-3.5" />
   469→                        </Button>
   470→                      )}
   471→                    </div>
   472→                  ))}
   473→                </div>
   474→              )}
   475→            </ScrollArea>
   476→          </div>
   477→        </div>
   478→      </DialogContent>
   479→    </Dialog>
   480→  )
   481→}

   516→  useEffect(() => { fetchConversations() }, [fetchConversations])
   517→
   518→  useEffect(() => {
   519→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   520→  }, [messages])
   521→
   522→  // Determine conversation kind
   523→  const isGroupSelected = selectedConvo?.type === 'group'
   524→  const selectedGroup = isGroupSelected ? (selectedConvo as GroupConversation).group : null
   525→  const selectedDirectUser = !isGroupSelected ? (selectedConvo as DirectConversation | null)?.user : null
   526→
   527→  const selectConversation = async (convo: Conversation) => {
   528→    if (!user?.id) return
   529→    setSelectedConvo(convo)
   530→    setMsgText('')
   531→    setMsgLoading(true)
   532→
   533→    try {
   534→      let url = `/api/messages?userId=${user.id}`
   535→      if (convo.type === 'group') {
   536→        url += `&groupId=${convo.group.id}`
   537→      } else {
   538→        if (!convo.user) return
   539→        url += `&otherId=${convo.user.id}`
   540→      }
   541→      const res = await fetch(url)
   542→      if (!res.ok) throw new Error()
   543→      const data = await res.json()
   544→      setMessages(data.messages ?? [])
   545→    } catch {
   546→      toast.error('Failed to load messages')
   547→    } finally {
   548→      setMsgLoading(false)
   549→    }
   550→
   551→    // Clear unread locally
   552→    setConversations((prev) => prev.map((c) => {
   553→      if (c.type === 'direct' && convo.type === 'direct' && c.user?.id === convo.user?.id) return { ...c, unread: 0 }
   554→      if (c.type === 'group' && convo.type === 'group' && c.group.id === convo.group.id) return { ...c, unread: 0 }
   555→      return c
   556→    }))
   557→  }
   558→
   559→  const sendMessage = async () => {
   560→    if (!user?.id || !msgText.trim() || !selectedConvo) return
   561→
   562→    setSending(true)
   563→    const text = msgText.trim()
   564→    setMsgText('')
   565→
   566→    try {
   567→      const body: Record<string, string> = { content: text, senderId: user.id }
   568→
   569→      if (selectedConvo.type === 'group') {
   570→        body.groupId = selectedConvo.group.id
   571→      } else {
   572→        if (!selectedConvo.user) return
   573→        body.receiverId = selectedConvo.user.id
   574→      }
   575→
   576→      const res = await fetch('/api/messages', {
   577→        method: 'POST',
   578→        headers: { 'Content-Type': 'application/json' },
   579→        body: JSON.stringify(body),
   580→      })
   581→      if (!res.ok) throw new Error()
   582→      const data = await res.json()
   583→      setMessages((prev) => [...prev, data.message])
   584→      fetchConversations()
   585→    } catch {
   586→      toast.error('Failed to send message')
   587→      setMsgText(text)
   588→    } finally {
   589→      setSending(false)
   590→    }
   591→  }
   592→
   593→  const handleKeyDown = (e: React.KeyboardEvent) => {
   594→    if (e.key === 'Enter' && !e.shiftKey) {
   595→      e.preventDefault()
   596→      sendMessage()
   597→    }
   598→  }
   599→
   600→  // Delete a single message
   601→  const deleteMessage = async (messageId: string) => {
   602→    if (!user?.id) return
   603→    try {
   604→      const res = await fetch(`/api/messages?id=${messageId}&userId=${user.id}`, { method: 'DELETE' })
   605→      if (!res.ok) throw new Error()
   606→      setMessages((prev) => prev.filter((m) => m.id !== messageId))
   607→      toast.success('Message deleted')
   608→    } catch {
   609→      toast.error('Failed to delete message')
   610→    }
   611→    setDeleteConfirm(null)
   612→  }
   613→
   614→  // Delete an entire conversation
   615→  const deleteConversation = async () => {
   616→    if (!user?.id || !selectedConvo || !deleteConfirm) return
   617→    try {
   618→      let url = `/api/messages?deleteConversation=true&userId=${user.id}`
   619→      if (selectedConvo.type === 'group') {
   620→        url += `&groupId=${selectedConvo.group.id}`
   621→      } else if (selectedConvo.user) {
   622→        url += `&otherId=${selectedConvo.user.id}`
   623→      }
   624→      const res = await fetch(url, { method: 'DELETE' })
   625→      if (!res.ok) throw new Error()
   626→      setSelectedConvo(null)
   627→      setMessages([])
   628→      fetchConversations()
   629→      toast.success('Conversation deleted')
   630→    } catch {
   631→      toast.error('Failed to delete conversation')
   632→    }
   633→    setDeleteConfirm(null)
   634→  }
   635→
   636→  // Handle selecting a user from New Message dialog
   637→  const handleNewMessageUser = async (selectedUser: User) => {
   638→    // Check if a conversation already exists
   639→    const existing = conversations.find(
   640→      (c) => c.type === 'direct' && c.user?.id === selectedUser.id
   641→    )
   642→    if (existing) {
   643→      await selectConversation(existing)
   644→      return
   645→    }
   646→    // Create a new direct conversation object and select it
   647→    const newConvo: DirectConversation = {
   648→      type: 'direct',
   649→      user: selectedUser,
   650→      lastMessage: null,
   651→      unread: 0,
   652→    }
   653→    setSelectedConvo(newConvo)
   654→    setMessages([])
   655→  }
   656→
   657→  // Filter conversations
   658→  const filteredConversations = conversations.filter((c) => {
   659→    if (!searchQuery.trim()) return true
   660→    const q = searchQuery.toLowerCase()
   661→    if (c.type === 'direct') return c.user?.name.toLowerCase().includes(q) || c.user?.email.toLowerCase().includes(q)
   662→    return c.group.name.toLowerCase().includes(q)
   663→  })
   664→
   665→  // Conversation display helpers
   666→  const getConvoPreview = (c: Conversation) => {
   667→    if (!c.lastMessage) return 'No messages yet'
   668→    const senderName = c.lastMessage.senderId === user?.id ? 'You' : (c.lastMessage.sender?.name ?? 'Someone')
   669→    if (c.type === 'group') return `${senderName}: ${c.lastMessage.content}`
   670→    return `${c.lastMessage.senderId === user?.id ? 'You: ' : ''}${c.lastMessage.content}`
   671→  }
   672→
   673→  // ── Render ───────────────────────────────────────────────
   674→
   675→  const containerHeight = 'h-[calc(100vh-12rem)]'
   676→
   677→  if (loading) {
   678→    return (
   679→      <div className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}>
   680→        <div className="w-80 border-r p-4 space-y-3 flex-shrink-0">
   681→          <Skeleton className="h-10 w-full" />
   682→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-14 w-full" />)}
   683→        </div>
   684→        <div className="flex-1 flex items-center justify-center">
   685→          <p className="text-muted-foreground text-sm">Select a conversation</p>
   686→        </div>
   687→      </div>
   688→    )
   689→  }
   690→
   691→  return (
   692→    <motion.div
   693→      initial={{ opacity: 0, y: 8 }}
   694→      animate={{ opacity: 1, y: 0 }}
   695→      className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}
   696→    >
   697→      {/* ─── Conversation List ──────────────────────────── */}
   698→      <div className="w-80 border-r bg-background flex flex-col flex-shrink-0">
   699→        <div className="p-3 border-b space-y-2">
   700→          <div className="flex items-center justify-between">
   701→            <h2 className="text-lg font-semibold">Messages</h2>
   702→            <div className="flex items-center gap-1">
   703→              <Button
   704→                variant="ghost" size="icon" className="h-8 w-8"
   705→                onClick={() => setShowNewMessage(true)}
   706→                title="New Message"
   707→              >
   708→                <MessageCirclePlus className="h-4 w-4" />
   709→              </Button>
   710→              <Button
   711→                variant="ghost" size="icon" className="h-8 w-8"
   712→                onClick={() => setShowCreateGroup(true)}
   713→                title="Create Group"
   714→              >
   715→                <Plus className="h-4 w-4" />
   716→              </Button>
   717→            </div>
   718→          </div>
   719→          <div className="relative">
   720→            <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   721→            <Input
   722→              placeholder="Search conversations..."
   723→              className="pl-9 h-9 text-sm"
   724→              value={searchQuery}
   725→              onChange={(e) => setSearchQuery(e.target.value)}
   726→            />
   727→          </div>
   728→        </div>
   729→        <ScrollArea className="flex-1">
   730→          <div className="p-1">
   731→            {filteredConversations.length === 0 && (
   732→              <p className="text-sm text-muted-foreground text-center py-8">
   733→                {searchQuery ? 'No matching conversations' : 'No conversations yet'}
   734→              </p>
   735→            )}
   736→            {filteredConversations.map((convo, idx) => {
   737→              const isSelected =
   738→                convo.type === 'direct'
   739→                  ? selectedDirectUser?.id === convo.user?.id
   740→                  : selectedGroup?.id === convo.group.id
   741→              return (
   742→                <div
   743→                  key={convo.type === 'direct' ? convo.user?.id ?? idx : convo.group.id}
   744→                  className={cn(
   745→                    'group flex items-center rounded-md transition-colors',
   746→                    isSelected ? 'bg-muted' : 'hover:bg-muted/50'
   747→                  )}
   748→                >
   749→                  <button
   750→                    className="w-full text-left px-3 py-3 flex items-start gap-3 flex-1 min-w-0"
   751→                    onClick={() => selectConversation(convo)}
   752→                  >
   753→                    {convo.type === 'direct' ? (
   754→                      <UserAvatar user={convo.user} size="md" className="mt-0.5 flex-shrink-0" />
   755→                    ) : (
   756→                      <GroupAvatar name={convo.group.name} size="md" className="mt-0.5 flex-shrink-0" />
   757→                    )}
   758→                    <div className="flex-1 min-w-0">
   759→                      <div className="flex items-center justify-between">
   760→                        <div className="flex items-center gap-1.5 min-w-0">
   761→                          <span className="text-sm font-medium truncate">{convo.type === 'direct' ? (convo.user?.name ?? 'Unknown') : convo.group.name}</span>
   762→                          {convo.type === 'group' && (
   763→                            <Users className="h-3 w-3 text-violet-500 flex-shrink-0" />
   764→                          )}
   765→                        </div>
   766→                        {convo.lastMessage && (
   767→                          <span className="text-[10px] text-muted-foreground flex-shrink-0 ml-2">
   768→                            {formatRelativeTime(convo.lastMessage.createdAt)}
   769→                          </span>
   770→                        )}
   771→                      </div>
   772→                      <p className="text-xs text-muted-foreground truncate mt-0.5">
   773→                        {getConvoPreview(convo)}
   774→                      </p>
   775→                    </div>
   776→                    {convo.unread > 0 && (
   777→                      <div className="h-5 w-5 rounded-full bg-emerald-500 text-white text-[10px] font-bold flex items-center justify-center flex-shrink-0 mt-1">
   778→                        {convo.unread}
   779→                      </div>
   780→                    )}
   781→                  </button>
   782→                  {/* Delete conversation button — shows on hover */}
   783→                  <Button
   784→                    variant="ghost"
   785→                    size="icon"
   786→                    className="h-8 w-8 opacity-0 group-hover:opacity-100 transition-opacity text-muted-foreground hover:text-destructive flex-shrink-0 mr-1 mt-1"
   787→                    onClick={(e) => { e.stopPropagation(); setDeleteConfirm({ type: 'conversation', id: convo.type === 'direct' ? (convo as DirectConversation).user?.id ?? '' : convo.group.id }) }}
   788→                    title="Delete conversation"
   789→                  >
   790→                    <Trash2 className="h-3.5 w-3.5" />
   791→                  </Button>
   792→                </div>
   793→              )
   794→            })}
   795→          </div>
   796→        </ScrollArea>
   797→      </div>
   798→
   799→      {/* ─── Chat Area ─────────────────────────────────── */}
   800→      <div className="flex-1 flex flex-col bg-background">
   801→        {selectedConvo ? (
   802→          <>
   803→            {/* Chat Header */}
   804→            <div className="p-3 border-b flex items-center gap-3">
   805→              {isGroupSelected ? (
   806→                <button onClick={() => setShowGroupInfo(true)} className="flex items-center gap-3 hover:bg-muted/50 rounded-md px-1 -ml-1 py-0.5 transition-colors">
   807→                  <GroupAvatar name={selectedGroup!.name} size="md" />
   808→                  <div className="min-w-0">
   809→                    <p className="text-sm font-medium flex items-center gap-1.5">
   810→                      {selectedGroup!.name}
   811→                      <Users className="h-3.5 w-3.5 text-violet-500" />
   812→                    </p>
   813→                    <p className="text-xs text-muted-foreground">{selectedGroup!._count.members} members</p>
   814→                  </div>
   815→                </button>
   816→              ) : selectedDirectUser ? (
   817→                <>
   818→                  <UserAvatar user={selectedDirectUser} size="md" />
   819→                  <div className="min-w-0">
   820→                    <p className="text-sm font-medium truncate">{selectedDirectUser.name}</p>
   821→                    <p className="text-xs text-muted-foreground truncate">
   822→                      {selectedDirectUser.company ?? selectedDirectUser.email}
   823→                    </p>
   824→                  </div>
   825→                  <Badge variant="secondary" className={cn('ml-auto text-xs flex-shrink-0', getRoleBadgeClasses(selectedDirectUser.role))}>
   826→                    {selectedDirectUser.role}
   827→                  </Badge>
   828→                </>
   829→              ) : null}
   830→
   831→              {/* Delete conversation button in header */}
   832→              <Button
   833→                variant="ghost"
   834→                size="icon"
   835→                className="h-8 w-8 ml-auto text-muted-foreground hover:text-destructive flex-shrink-0"
   836→                onClick={() => setDeleteConfirm({ type: 'conversation', id: '' })}
   837→                title="Delete conversation"
   838→              >
   839→                <Trash2 className="h-4 w-4" />
   840→              </Button>
   841→            </div>
   842→
   843→            {/* Messages */}
   844→            <ScrollArea className="flex-1 p-4">
   845→              <div className="space-y-3 max-w-2xl mx-auto">
   846→                {msgLoading ? (
   847→                  <div className="space-y-3">
   848→                    {Array.from({ length: 4 }).map((_, i) => (
   849→                      <div key={i} className={cn('flex gap-2.5', i % 2 === 0 ? '' : 'justify-end')}>
   850→                        <Skeleton className="h-10 w-48 rounded-2xl" />
   851→                      </div>
   852→                    ))}
   853→                  </div>
   854→                ) : messages.length === 0 ? (
   855→                  <div className="flex flex-col items-center justify-center h-full text-center py-8">
   856→                    <MessageSquare className="h-10 w-10 text-muted-foreground/30 mb-3" />
   857→                    <p className="text-sm text-muted-foreground">
   858→                      {isGroupSelected ? 'No messages yet. Start the conversation!' : 'No messages yet. Say hello!'}
   859→                    </p>
   860→                  </div>
   861→                ) : (
   862→                  messages.map((msg, idx) => {
   863→                    const isSent = msg.senderId === user?.id
   864→                    return (
   865→                      <motion.div
   866→                        key={msg.id}
   867→                        initial={{ opacity: 0, y: 5 }}
   868→                        animate={{ opacity: 1, y: 0 }}
   869→                        transition={{ delay: Math.min(idx * 0.015, 0.2) }}
   870→                        className={cn('group/msg flex gap-2.5', isSent ? 'justify-end' : 'justify-start')}
   871→                      >
   872→                        {!isSent && (
   873→                          <UserAvatar user={msg.sender} size="sm" className="mt-auto" />
   874→                        )}
   875→                        <div className={cn('max-w-[70%]', isSent ? 'text-right' : '')}>
   876→                          {isGroupSelected && !isSent && (
   877→                            <p className="text-xs font-medium text-muted-foreground mb-1 px-1">
   878→                              {msg.sender?.name ?? 'Unknown'}
   879→                            </p>
   880→                          )}
   881→                          <div className="relative">
   882→                            <div className={cn(
   883→                              'inline-block rounded-2xl px-4 py-2.5 text-sm',
   884→                              isSent
   885→                                ? 'bg-emerald-600 text-white rounded-br-md'
   886→                                : 'bg-muted rounded-bl-md',
   887→                            )}>
   888→                              <p className="whitespace-pre-wrap break-words">{msg.content}</p>
   889→                            </div>
   890→                            {/* Delete message button — shows on hover */}
   891→                            <Button
   892→                              variant="ghost"
   893→                              size="icon"
   894→                              className={cn(
   895→                                'h-6 w-6 absolute opacity-0 group-hover/msg:opacity-100 transition-opacity',
   896→                                isSent
   897→                                  ? '-left-8 top-1/2 -translate-y-1/2'
   898→                                  : '-right-8 top-1/2 -translate-y-1/2'
   899→                              )}
   900→                              onClick={() => setDeleteConfirm({ type: 'message', id: msg.id })}
   901→                              title="Delete message"
   902→                            >
   903→                              <Trash2 className="h-3 w-3 text-muted-foreground hover:text-destructive" />
   904→                            </Button>
   905→                          </div>
   906→                          <div className={cn(
   907→                            'flex items-center gap-1 mt-1 text-[10px] text-muted-foreground',
   908→                            isSent ? 'justify-end' : 'justify-start'
   909→                          )}>
   910→                            <span>{formatRelativeTime(msg.createdAt)}</span>
   911→                            {isSent && msg.read && <CheckCheck className="w-3 h-3" />}
   912→                          </div>
   913→                        </div>
   914→                      </motion.div>
   915→                    )
   916→                  })
   917→                )}
   918→                <div ref={messagesEndRef} />
   919→              </div>
   920→            </ScrollArea>
   921→
   922→            {/* Message Input */}
   923→            <div className="p-3 border-t">
   924→              <div className="flex gap-2 max-w-2xl mx-auto">
   925→                <Input
   926→                  placeholder={isGroupSelected ? `Message ${selectedGroup!.name}...` : 'Type a message...'}
   927→                  value={msgText}
   928→                  onChange={(e) => setMsgText(e.target.value)}
   929→                  onKeyDown={handleKeyDown}
   930→                  className="flex-1"
   931→                />
   932→                <Button
   933→                  onClick={sendMessage}
   934→                  disabled={!msgText.trim() || sending}
   935→                  className="bg-emerald-600 hover:bg-emerald-700 text-white flex-shrink-0"
   936→                >
   937→                  {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   938→                </Button>
   939→              </div>
   940→            </div>
   941→          </>
   942→        ) : (
   943→          <div className="flex-1 flex items-center justify-center">
   944→            <div className="text-center space-y-3">
   945→              <div className="h-16 w-16 rounded-full bg-muted flex items-center justify-center mx-auto">
   946→                <MessageSquare className="h-8 w-8 text-muted-foreground" />
   947→              </div>
   948→              <h3 className="font-semibold text-lg">Select a conversation</h3>
   949→              <p className="text-sm text-muted-foreground max-w-xs">
   950→                Choose a conversation from the sidebar, or start a new one.
   951→              </p>
   952→              <div className="flex items-center justify-center gap-2 pt-2">
   953→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setShowNewMessage(true)}>
   954→                  <MessageCirclePlus className="h-4 w-4" /> New Message
   955→                </Button>
   956→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setShowCreateGroup(true)}>
   957→                  <Users className="h-4 w-4" /> Create Group
   958→                </Button>
   959→              </div>
   960→            </div>
   961→          </div>
   962→        )}
   963→      </div>
   964→
   965→      {/* ─── Dialogs ──────────────────────────────────── */}
   966→      <CreateGroupDialog
   967→        open={showCreateGroup}
   968→        onOpenChange={setShowCreateGroup}
   969→        onCreated={() => { fetchConversations(); setShowCreateGroup(false) }}
   970→        currentUser={user!}
   971→      />
   972→      <NewMessageDialog
   973→        open={showNewMessage}
   974→        onOpenChange={setShowNewMessage}
   975→        onUserSelected={handleNewMessageUser}
   976→        currentUser={user!}
   977→      />
   978→      <GroupInfoDialog
   979→        open={showGroupInfo}
   980→        onOpenChange={setShowGroupInfo}
   981→        group={selectedGroup ? { id: selectedGroup.id, name: selectedGroup.name, _count: selectedGroup._count } : null}
   982→        currentUserId={user?.id ?? ''}
   983→      />
   984→
   985→      {/* ─── Delete Confirmation ─────────────────────── */}
   986→      <AlertDialog open={!!deleteConfirm} onOpenChange={(open) => { if (!open) setDeleteConfirm(null) }}>
   987→        <AlertDialogContent>
   988→          <AlertDialogHeader>
   989→            <AlertDialogTitle>
   990→              {deleteConfirm?.type === 'message' ? 'Delete Message' : 'Delete Conversation'}
   991→            </AlertDialogTitle>
   992→            <AlertDialogDescription>
   993→              {deleteConfirm?.type === 'message'
   994→                ? 'Are you sure you want to delete this message? This action cannot be undone.'
   995→                : 'Are you sure you want to delete this entire conversation? All messages will be permanently removed. This action cannot be undone.'
   996→              }
   997→            </AlertDialogDescription>
   998→          </AlertDialogHeader>
   999→          <AlertDialogFooter>
  1000→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1001→            <AlertDialogAction
  1002→              className="bg-destructive text-white hover:bg-destructive/90"
  1003→              onClick={() => {
  1004→                if (!deleteConfirm) return
  1005→                if (deleteConfirm.type === 'message') {
  1006→                  deleteMessage(deleteConfirm.id)
  1007→                } else {
  1008→                  deleteConversation()
  1009→                }
  1010→              }}
  1011→            >
  1012→              Delete
  1013→            </AlertDialogAction>
  1014→          </AlertDialogFooter>
  1015→        </AlertDialogContent>
  1016→      </AlertDialog>
  1017→    </motion.div>
  1018→  )
  1019→}

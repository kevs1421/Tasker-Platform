    38→// ─── Incoming Call Overlay ──────────────────────────────────────────────
    39→
    40→export function IncomingCallOverlay({
    41→  caller,
    42→  callType,
    43→  onAccept,
    44→  onReject,
    45→}: {
    46→  caller: User
    47→  callType: 'audio' | 'video'
    48→  onAccept: () => void
    49→  onReject: () => void
    50→}) {
    51→  return (
    52→    <motion.div
    53→      initial={{ opacity: 0, scale: 0.95 }}
    54→      animate={{ opacity: 1, scale: 1 }}
    55→      exit={{ opacity: 0, scale: 0.95 }}
    56→      className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm"
    57→    >
    58→      <div className="bg-card rounded-2xl shadow-2xl p-8 max-w-sm w-full mx-4 text-center border">
    59→        <div className="mb-6">
    60→          <Avatar className="h-20 w-20 mx-auto mb-4">
    61→            {caller.avatar && <AvatarImage src={caller.avatar} />}
    62→            <AvatarFallback className="bg-slate-200 dark:bg-slate-700 text-2xl font-semibold">
    63→              {getInitials(caller.name)}
    64→            </AvatarFallback>
    65→          </Avatar>
    66→          <h2 className="text-xl font-semibold text-foreground">{caller.name}</h2>
    67→          <p className="text-sm text-muted-foreground mt-1">
    68→            {callType === 'video' ? 'Incoming video call...' : 'Incoming audio call...'}
    69→          </p>
    70→        </div>
    71→        <div className="flex items-center justify-center gap-6">
    72→          <button
    73→            onClick={onReject}
    74→            className="h-14 w-14 rounded-full bg-red-500 hover:bg-red-600 text-white flex items-center justify-center transition-colors shadow-lg"
    75→          >
    76→            <PhoneOff className="h-6 w-6" />
    77→          </button>
    78→          <button
    79→            onClick={onAccept}
    80→            className="h-14 w-14 rounded-full bg-emerald-500 hover:bg-emerald-600 text-white flex items-center justify-center transition-colors shadow-lg animate-pulse"
    81→          >
    82→            <Phone className="h-6 w-6" />
    83→          </button>
    84→        </div>
    85→      </div>
    86→    </motion.div>
    87→  )
    88→}

    92→function ScreenShareBanner({
    93→  isLocal,
    94→  label,
    95→  onStop,
    96→}: {
    97→  isLocal: boolean
    98→  label: string
    99→  onStop: () => void
   100→}) {

   106→      className="flex items-center justify-between px-4 py-2 bg-emerald-600 text-white text-sm"

   109→        <div className="h-2 w-2 rounded-full bg-white animate-pulse" />

   112→          {isLocal ? 'You are sharing your screen' : `${label} is presenting`}

   131→// ─── Active Call Floating Panel ─────────────────────────────────────────
   132→
   133→export function VideoCallPanel({
   134→  currentUser,
   135→  targetUser,
   136→  isInCall,
   137→  callType,
   138→  callStatus,
   139→  onEnd,
   140→  onToggleMic,
   141→  onToggleCamera,
   142→  onToggleScreen,
   143→  micOn,
   144→  cameraOn,
   145→  screenOn,
   146→  remoteScreenOn,
   147→  remoteMicOn,
   148→  remoteCameraOn,
   149→  screenShareLabel,
   150→}: VideoCallProps) {
   151→  const [isMinimized, setIsMinimized] = useState(false)
   152→  const [isFullscreen, setIsFullscreen] = useState(false)
   153→  const [callDuration, setCallDuration] = useState(0)
   154→  const [meetingLink, setMeetingLink] = useState('')
   155→  const [remoteVideoPresent, setRemoteVideoPresent] = useState(false)
   156→  const [remoteScreenPresent, setRemoteScreenPresent] = useState(false)
   157→  const timerRef = useRef<ReturnType<typeof setInterval>>
   158→  const panelRef = useRef<HTMLDivElement>(null)
   159→
   160→  // Timer for connected calls
   161→  useEffect(() => {
   162→    if (callStatus === 'connected') {
   163→      timerRef.current = setInterval(() => setCallDuration((p) => p + 1), 1000)
   164→    } else {
   165→      if (timerRef.current) clearInterval(timerRef.current)
   166→      setCallDuration(0)
   167→    }
   168→    return () => { if (timerRef.current) clearInterval(timerRef.current) }
   169→  }, [callStatus])
   170→
   171→  // Generate meeting link
   172→  useEffect(() => {
   173→    if (isInCall) {
   174→      const id = Math.random().toString(36).substring(2, 10)
   175→      setMeetingLink(`${window.location.origin}/meet/${id}`)
   176→    }
   177→  }, [isInCall])
   178→
   179→  // Monitor remote video tracks
   180→  useEffect(() => {
   181→    if (callStatus !== 'connected') {
   182→      setRemoteVideoPresent(false)
   183→      setRemoteScreenPresent(false)
   184→      return
   185→    }
   186→    const checkTracks = setInterval(() => {
   187→      const remoteVideo = document.querySelector('#remote-video') as HTMLVideoElement
   188→      const hasVideo = remoteVideo?.srcObject instanceof MediaStream &&
   189→        (remoteVideo.srcObject as MediaStream).getVideoTracks().some((t) => t.enabled && t.readyState === 'live')
   190→      setRemoteVideoPresent(!!hasVideo)
   191→
   192→      const remoteScreenVideo = document.querySelector('#remote-screen-video') as HTMLVideoElement
   193→      const hasScreen = remoteScreenVideo?.srcObject instanceof MediaStream &&
   194→        (remoteScreenVideo.srcObject as MediaStream).getVideoTracks().length > 0
   195→      setRemoteScreenPresent(!!hasScreen)
   196→    }, 1000)
   197→    return () => clearInterval(checkTracks)
   198→  }, [callStatus])
   199→
   200→  const copyLink = useCallback(() => {
   201→    if (meetingLink) {
   202→      navigator.clipboard.writeText(meetingLink)
   203→      toast.success('Meeting link copied!')
   204→    }
   205→  }, [meetingLink])
   206→
   207→  const formatTime = (seconds: number) => {
   208→    const h = Math.floor(seconds / 3600)
   209→    const m = Math.floor((seconds % 3600) / 60)
   210→    const s = seconds % 60
   211→    return h > 0
   212→      ? `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   213→      : `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   214→  }
   215→
   216→  const toggleFullscreen = useCallback(() => {
   217→    if (!panelRef.current) return
   218→    if (!document.fullscreenElement) {
   219→      panelRef.current.requestFullscreen().then(() => setIsFullscreen(true)).catch(() => {})
   220→    } else {
   221→      document.exitFullscreen().then(() => setIsFullscreen(false)).catch(() => {})
   222→    }
   223→  }, [])
   224→
   225→  if (!isInCall) return null
   226→
   227→  const isConnected = callStatus === 'connected'
   228→  const isRinging = callStatus === 'ringing'
   229→  const isConnecting = callStatus === 'connecting'
   230→  const isVideoCall = callType === 'video'
   231→  const isGroupCall = targetUser?.role === 'team' && targetUser?.id?.startsWith('group-')
   232→
   233→  // Whether the main view should show screen content
   234→  const showLocalScreen = isConnected && screenOn
   235→  const showRemoteScreen = isConnected && remoteScreenOn && remoteScreenPresent
   236→  const showScreenContent = showLocalScreen || showRemoteScreen
   237→
   238→  // ── Minimized view ──
   239→  if (isMinimized) {
   240→    return (
   241→      <AnimatePresence>
   242→        <motion.div
   243→          initial={{ opacity: 0, y: 40, scale: 0.9 }}
   244→          animate={{ opacity: 1, y: 0, scale: 1 }}
   245→          exit={{ opacity: 0, y: 40, scale: 0.9 }}
   246→          className="fixed bottom-6 right-6 z-50"
   247→        >
   248→          <div className="flex items-center gap-2 bg-card border rounded-2xl shadow-2xl px-3 py-2.5 pr-1.5">
   249→            {targetUser && (
   250→              <div className="flex items-center gap-2 min-w-0">
   251→                <div className={cn(
   252→                  'h-2.5 w-2.5 rounded-full flex-shrink-0',
   253→                  isConnected ? 'bg-emerald-500 animate-pulse' : 'bg-amber-500 animate-pulse'
   254→                )} />
   255→                {showScreenContent && <MonitorUp className="h-3.5 w-3.5 text-emerald-500 flex-shrink-0" />}
   256→                <span className="text-sm font-medium truncate max-w-[120px]">
   257→                  {targetUser.name}
   258→                </span>
   259→                <span className="text-xs text-muted-foreground font-mono">
   260→                  {isConnected ? formatTime(callDuration) : isRinging ? 'Ringing...' : 'Connecting...'}
   261→                </span>
   262→              </div>
   263→            )}
   264→            <Button variant="ghost" size="icon" className="h-8 w-8 flex-shrink-0" onClick={() => setIsMinimized(false)}>
   265→              <Maximize2 className="h-4 w-4" />
   266→            </Button>
   267→            <Button
   268→              variant="ghost" size="icon"
   269→              className="h-8 w-8 text-red-500 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-950/30 flex-shrink-0"
   270→              onClick={onEnd}
   271→            >
   272→              <PhoneOff className="h-4 w-4" />
   273→            </Button>
   274→          </div>
   275→        </motion.div>
   276→      </AnimatePresence>
   277→    )
   278→  }
   279→
   280→  // ── Full call panel ──
   281→  return (
   282→    <AnimatePresence>
   283→      <motion.div
   284→        ref={panelRef}
   285→        initial={{ opacity: 0, scale: 0.95, y: 20 }}
   286→        animate={{ opacity: 1, scale: 1, y: 0 }}
   287→        exit={{ opacity: 0, scale: 0.95, y: 20 }}
   288→        className={cn(
   289→          'fixed z-50 shadow-2xl border-2 border-slate-700 overflow-hidden',
   290→          showScreenContent
   291→            ? 'bottom-4 right-4 left-4 top-4'
   292→            : 'bottom-6 right-6',
   293→          showScreenContent
   294→            ? 'w-auto h-auto'
   295→            : isVideoCall ? 'w-[560px] h-[420px]' : 'w-[380px] h-[280px]',
   296→          'rounded-2xl bg-slate-950 text-white',
   297→          isFullscreen && 'fixed inset-0 !rounded-none !border-0'
   298→        )}
   299→      >
   300→        {/* ── Screen Share Banner ── */}
   301→        <AnimatePresence>
   302→          {showLocalScreen && (
   303→            <ScreenShareBanner
   304→              isLocal
   305→              label={screenShareLabel}
   306→              onStop={onToggleScreen}
   307→            />
   308→          )}
   309→          {showRemoteScreen && !showLocalScreen && (
   310→            <ScreenShareBanner
   311→              isLocal={false}
   312→              label={targetUser?.name ?? 'Participant'}
   313→              onStop={() => {}}
   314→            />
   315→          )}
   316→        </AnimatePresence>
   317→
   318→        {/* ── Call Header ── */}
   319→        <div className="flex items-center justify-between px-4 py-2.5 bg-slate-900/80 border-b border-white/10">
   320→          <div className="flex items-center gap-2.5 min-w-0">
   321→            <div className={cn(
   322→              'h-2.5 w-2.5 rounded-full flex-shrink-0',
   323→              isConnected ? 'bg-emerald-400' : 'bg-amber-400 animate-pulse'
   324→            )} />
   325→            <span className="text-sm font-medium text-white/90 truncate">
   326→              {targetUser ? targetUser.name : 'Meeting'}
   327→            </span>
   328→            {isGroupCall && <Users className="h-3.5 w-3.5 text-white/50" />}
   329→            <span className="text-xs text-white/50 font-mono">
   330→              {isConnected ? formatTime(callDuration) : isRinging ? 'Ringing...' : 'Connecting...'}
   331→            </span>
   332→          </div>
   333→          <div className="flex items-center gap-0.5">
   334→            <Button
   335→              variant="ghost" size="icon"
   336→              className="h-7 w-7 text-white/60 hover:text-white hover:bg-white/10"
   337→              onClick={copyLink}
   338→              title="Copy meeting link"
   339→            >
   340→              <Copy className="h-3.5 w-3.5" />
   341→            </Button>
   342→            <Button
   343→              variant="ghost" size="icon"
   344→              className="h-7 w-7 text-white/60 hover:text-white hover:bg-white/10"
   345→              onClick={toggleFullscreen}
   346→              title={isFullscreen ? 'Exit fullscreen' : 'Fullscreen'}
   347→            >
   348→              <Maximize2 className="h-3.5 w-3.5" />
   349→            </Button>
   350→            <Button
   351→              variant="ghost" size="icon"
   352→              className="h-7 w-7 text-white/60 hover:text-white hover:bg-white/10"
   353→              onClick={() => setIsMinimized(true)}
   354→            >
   355→              <Minimize2 className="h-3.5 w-3.5" />
   356→            </Button>
   357→          </div>
   358→        </div>
   359→
   360→        {/* ── Video Area ── */}
   361→        <div className="relative" style={{ height: 'calc(100% - 110px - 40px)' }}>
   362→          {/* ── SCREEN SHARE MAIN VIEW ── */}
   363→          {showScreenContent ? (
   364→            <div className="absolute inset-0 bg-slate-900">
   365→              {/* Local screen share (presenting) */}
   366→              {showLocalScreen && (
   367→                <div className="absolute inset-0 flex flex-col">
   368→                  <div className="flex-1 flex items-center justify-center p-4">
   369→                    <div className="relative w-full h-full bg-slate-800 rounded-lg overflow-hidden shadow-2xl border border-white/10">
   370→                      <video
   371→                        id="local-screen-video"
   372→                        autoPlay
   373→                        playsInline
   374→                        muted
   375→                        className="w-full h-full object-contain bg-black"
   376→                      />
   377→                      {/* Stop button overlay */}
   378→                      <div className="absolute bottom-4 left-1/2 -translate-x-1/2">
   379→                        <button
   380→                          onClick={onToggleScreen}
   381→                          className="flex items-center gap-2 px-4 py-2 bg-red-500 hover:bg-red-600 text-white rounded-full text-sm font-medium transition-colors shadow-lg"
   382→                        >
   383→                          <MonitorOff className="h-4 w-4" />
   384→                          Stop sharing
   385→                        </button>
   386→                      </div>
   387→                    </div>
   388→                  </div>
   389→                </div>
   390→              )}
   391→
   392→              {/* Remote screen share (viewing) */}
   393→              {showRemoteScreen && !showLocalScreen && (
   394→                <div className="absolute inset-0 bg-black flex items-center justify-center p-4">
   395→                  <div className="relative w-full h-full rounded-lg overflow-hidden">
   396→                    <video
   397→                      id="remote-screen-video"
   398→                      autoPlay
   399→                      playsInline
   400→                      className="w-full h-full object-contain bg-black"
   401→                    />
   402→                    {/* Remote user info when viewing screen */}
   403→                    <div className="absolute top-3 left-3 flex items-center gap-2 px-3 py-1.5 bg-black/60 backdrop-blur-sm rounded-lg">
   404→                      <Avatar className="h-6 w-6">
   405→                        {targetUser?.avatar && <AvatarImage src={targetUser.avatar} />}
   406→                        <AvatarFallback className="bg-slate-600 text-[10px]">
   407→                          {targetUser ? getInitials(targetUser.name) : '?'}
   408→                        </AvatarFallback>
   409→                      </Avatar>
   410→                      <span className="text-xs text-white/80 font-medium">{targetUser?.name ?? 'Participant'}</span>
   411→                      {!remoteCameraOn && <VideoOff className="h-3 w-3 text-white/50" />}
   412→                    </div>
   413→                  </div>
   414→                </div>
   415→              )}
   416→
   417→              {/* Picture-in-picture: local camera when presenting screen */}
   418→              {isVideoCall && showLocalScreen && (
   419→                <div className="absolute bottom-20 right-4 w-[160px] h-[120px] rounded-xl overflow-hidden shadow-xl border-2 border-white/20 bg-slate-800">
   420→                  {cameraOn ? (
   421→                    <video
   422→                      id="local-video"
   423→                      autoPlay
   424→                      playsInline
   425→                      muted
   426→                      className="w-full h-full object-cover transform -scale-x-100"
   427→                    />
   428→                  ) : (
   429→                    <div className="w-full h-full flex items-center justify-center">
   430→                      <Avatar className="h-10 w-10">
   431→                        {currentUser?.avatar && <AvatarImage src={currentUser.avatar} />}
   432→                        <AvatarFallback className="bg-slate-700 text-sm font-semibold">
   433→                          {getInitials(currentUser.name)}
   434→                        </AvatarFallback>
   435→                      </Avatar>
   436→                    </div>
   437→                  )}
   438→                </div>
   439→              )}
   440→            </div>
   441→          ) : (
   442→            /* ── NORMAL VIDEO / AUDIO VIEW ── */
   443→            <div className="absolute inset-0 bg-slate-900 flex items-center justify-center">
   444→              {/* Remote video */}
   445→              {isVideoCall && isConnected && (
   446→                <video
   447→                  id="remote-video"
   448→                  autoPlay
   449→                  playsInline
   450→                  className={cn(
   451→                    'w-full h-full object-cover',
   452→                    !remoteVideoPresent && 'hidden'
   453→                  )}
   454→                />
   455→              )}
   456→
   457→              {/* Waiting/connecting state */}
   458→              {!isConnected && (
   459→                <div className="flex flex-col items-center gap-3">
   460→                  <Avatar className="h-20 w-20">
   461→                    {targetUser?.avatar && <AvatarImage src={targetUser.avatar} />}
   462→                    <AvatarFallback className="bg-slate-700 text-2xl font-semibold">
   463→                      {targetUser ? getInitials(targetUser.name) : '?'}
   464→                    </AvatarFallback>
   465→                  </Avatar>
   466→                  <div className="text-center">
   467→                    <p className="text-white/90 font-medium">
   468→                      {isRinging ? 'Calling...' : 'Connecting...'}
   469→                    </p>
   470→                    {isRinging && (
   471→                      <div className="flex items-center justify-center gap-1.5 mt-2">
   472→                        {[0, 1, 2].map((i) => (
   473→                          <motion.div
   474→                            key={i}
   475→                            className="h-2 w-2 bg-emerald-400 rounded-full"
   476→                            animate={{ opacity: [0.3, 1, 0.3] }}
   477→                            transition={{ duration: 1.5, repeat: Infinity, delay: i * 0.3 }}
   478→                          />
   479→                        ))}
   480→                      </div>
   481→                    )}
   482→                  </div>
   483→                </div>
   484→              )}
   485→
   486→              {/* Remote user avatar (camera off or no video) */}
   487→              {isConnected && isVideoCall && !remoteVideoPresent && !showScreenContent && (
   488→                <div className="absolute inset-0 flex items-center justify-center bg-slate-900">
   489→                  <div className="flex flex-col items-center gap-2">
   490→                    <Avatar className="h-24 w-24">
   491→                      {targetUser?.avatar && <AvatarImage src={targetUser.avatar} />}
   492→                      <AvatarFallback className="bg-slate-700 text-3xl font-semibold">
   493→                        {targetUser ? getInitials(targetUser.name) : '?'}
   494→                      </AvatarFallback>
   495→                    </Avatar>
   496→                    {!remoteCameraOn && (
   497→                      <div className="flex items-center gap-1 text-white/40 text-xs">
   498→                        <VideoOff className="h-3 w-3" />
   499→                        <span>Camera off</span>
   500→                      </div>
   501→                    )}
   502→                    {!remoteMicOn && (
   503→                      <div className="flex items-center gap-1 text-red-400 text-xs">
   504→                        <MicOff className="h-3 w-3" />
   505→                        <span>Muted</span>
   506→                      </div>
   507→                    )}
   508→                  </div>
   509→                </div>
   510→              )}
   511→
   512→              {/* Audio call in progress */}
   513→              {isConnected && !isVideoCall && (
   514→                <div className="flex flex-col items-center gap-3">
   515→                  <div className="h-24 w-24 rounded-full bg-emerald-500/20 flex items-center justify-center">
   516→                    <Phone className="h-10 w-10 text-emerald-400" />
   517→                  </div>
   518→                  <p className="text-white/70 text-sm">Audio call in progress</p>
   519→                  <p className="text-white/50 text-lg font-mono font-medium">
   520→                    {formatTime(callDuration)}
   521→                  </p>
   522→                  {!remoteMicOn && (
   523→                    <div className="flex items-center gap-1 text-red-400 text-xs">
   524→                      <MicOff className="h-3 w-3" />
   525→                      <span>{targetUser?.name} is muted</span>
   526→                    </div>
   527→                  )}
   528→                </div>
   529→              )}
   530→
   531→              {/* Local video picture-in-picture */}
   532→              {isVideoCall && (
   533→                <div className={cn(
   534→                  'absolute bottom-3 right-3 rounded-xl overflow-hidden shadow-lg border-2 border-white/20',
   535→                  cameraOn ? 'w-[140px] h-[105px]' : 'w-[100px] h-[100px]'
   536→                )}>
   537→                  {cameraOn ? (
   538→                    <video
   539→                      id="local-video"
   540→                      autoPlay
   541→                      playsInline
   542→                      muted
   543→                      className="w-full h-full object-cover transform -scale-x-100"
   544→                    />
   545→                  ) : (
   546→                    <div className="w-full h-full bg-slate-800 flex items-center justify-center">
   547→                      <Avatar className="h-10 w-10">
   548→                        {currentUser?.avatar && <AvatarImage src={currentUser.avatar} />}
   549→                        <AvatarFallback className="bg-slate-700 text-sm font-semibold">
   550→                          {getInitials(currentUser.name)}
   551→                        </AvatarFallback>
   552→                      </Avatar>
   553→                    </div>
   554→                  )}
   555→                </div>
   556→              )}
   557→            </div>
   558→          )}
   559→        </div>
   560→
   561→        {/* ── Controls Bar ── */}
   562→        <div className={cn(
   563→          'absolute bottom-0 left-0 right-0 px-4 py-3',
   564→          'bg-gradient-to-t from-slate-950/90 to-transparent'
   565→        )}>
   566→          <div className="flex items-center justify-center gap-2">
   567→            {/* Mic */}
   568→            <ControlButton
   569→              active={micOn}
   570→              activeIcon={<Mic className="h-4 w-4" />}
   571→              inactiveIcon={<MicOff className="h-4 w-4" />}
   572→              onClick={onToggleMic}
   573→              title={micOn ? 'Mute' : 'Unmute'}
   574→            />
   575→
   576→            {/* Camera (video calls only) */}
   577→            {isVideoCall && (
   578→              <ControlButton
   579→                active={cameraOn}
   580→                activeIcon={<Video className="h-4 w-4" />}
   581→                inactiveIcon={<VideoOff className="h-4 w-4" />}
   582→                onClick={onToggleCamera}
   583→                title={cameraOn ? 'Turn off camera' : 'Turn on camera'}
   584→              />
   585→            )}
   586→
   587→            {/* Screen share — with tooltip showing the label */}
   588→            <div className="relative group">
   589→              <ControlButton
   590→                active={screenOn}
   591→                activeIcon={<MonitorOff className="h-4 w-4" />}
   592→                inactiveIcon={<MonitorUp className="h-4 w-4" />}
   593→                onClick={onToggleScreen}
   594→                title={screenOn ? 'Stop sharing' : 'Share screen'}
   595→              />
   596→              {/* Label tooltip */}
   597→              {screenOn && screenShareLabel && (
   598→                <div className="absolute -top-10 left-1/2 -translate-x-1/2 whitespace-nowrap px-2.5 py-1 bg-black text-white text-xs rounded-md opacity-0 group-hover:opacity-100 transition-opacity pointer-events-none">
   599→                  Sharing: {screenShareLabel}
   600→                </div>
   601→              )}
   602→            </div>
   603→
   604→            {/* End Call */}
   605→            <button
   606→              onClick={onEnd}
   607→              className="h-10 w-10 rounded-full bg-red-500 hover:bg-red-600 text-white flex items-center justify-center transition-colors shadow-lg"
   608→              title="End call"
   609→            >
   610→              <PhoneOff className="h-4 w-4" />
   611→            </button>
   612→          </div>
   613→
   614→          {/* Screen share indicator at the bottom */}
   615→          {screenOn && (
   616→            <motion.div
   617→              initial={{ opacity: 0, height: 0 }}
   618→              animate={{ opacity: 1, height: 'auto' }}
   619→              exit={{ opacity: 0, height: 0 }}
   620→              className="flex items-center justify-center gap-2 mt-2 text-xs text-emerald-400"
   621→            >
   622→              <div className="h-1.5 w-1.5 rounded-full bg-emerald-400 animate-pulse" />
   623→              <span>You are presenting</span>
   624→              {screenShareLabel && <span className="text-white/40">({screenShareLabel})</span>}
   625→            </motion.div>
   626→          )}
   627→        </div>
   628→      </div>
   629→      </motion.div>
   630→    </AnimatePresence>
   631→  )
   632→}

   637→  active,
   638→  activeIcon,
   639→  inactiveIcon,
   640→  onClick,
   641→  title,
   642→}: {
   643→  active: boolean
   644→  activeIcon: React.ReactNode
   645→  inactiveIcon: React.ReactNode
   646→  onClick: () => void
   647→  title: string
   648→}) {
   649→  return (
   650→    <button
   651→      onClick={onClick}
   652→      className={cn(
   653→        'h-10 w-10 rounded-full flex items-center justify-center transition-all',
   654→        active ? 'bg-white/15 hover:bg-white/25 text-white' : 'bg-red-500 hover:bg-red-600 text-white'
   655→      )}
   656→      title={title}
   657→    >
   658→      {active ? activeIcon : inactiveIcon}
   659→    </button>
   660→  )
   661→}

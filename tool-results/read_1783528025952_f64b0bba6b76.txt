   138→// ─── Component ────────────────────────────────────────────────────────────────
   139→
   140→export default function AdminDashboard() {
   141→  const { adminView, user: currentUser } = useAuthStore()
   142→
   143→  // ── State ──
   144→  const [loading, setLoading] = useState(true)
   145→  const [stats, setStats] = useState<ExtendedDashboardStats | null>(null)
   146→  const [projects, setProjects] = useState<Project[]>([])
   147→  const [users, setUsers] = useState<ExtendedUser[]>([])
   148→  const [activities, setActivities] = useState<ActivityLog[]>([])
   149→
   150→  // ── Create Project Dialog State ──
   151→  const [createDialogOpen, setCreateDialogOpen] = useState(false)
   152→  const [newProject, setNewProject] = useState({
   153→    name: '',
   154→    description: '',
   155→    color: '#10b981',
   156→    priority: 'medium',
   157→    dueDate: '',
   158→  })
   159→  const [creating, setCreating] = useState(false)
   160→
   161→  // ── Data Fetching ──
   162→
   163→  const fetchStats = useCallback(async () => {
   164→    try {
   165→      const res = await fetch('/api/dashboard/stats')
   166→      if (!res.ok) throw new Error('Failed to fetch stats')
   167→      const data = await res.json()
   168→      setStats(data.stats)
   169→    } catch {
   170→      toast.error('Failed to load dashboard stats')
   171→    }
   172→  }, [])
   173→
   174→  const fetchProjects = useCallback(async () => {
   175→    try {
   176→      const res = await fetch('/api/projects')
   177→      if (!res.ok) throw new Error('Failed to fetch projects')
   178→      const data = await res.json()
   179→      setProjects(data.projects)
   180→    } catch {
   181→      toast.error('Failed to load projects')
   182→    }
   183→  }, [])
   184→
   185→  const fetchUsers = useCallback(async () => {
   186→    try {
   187→      const res = await fetch('/api/users')
   188→      if (!res.ok) throw new Error('Failed to fetch users')
   189→      const data = await res.json()
   190→      setUsers(data.users)
   191→    } catch {
   192→      toast.error('Failed to load users')
   193→    }
   194→  }, [])
   195→
   196→  const fetchActivities = useCallback(async () => {
   197→    try {
   198→      const res = await fetch('/api/dashboard/activity')
   199→      if (!res.ok) throw new Error('Failed to fetch activities')
   200→      const data = await res.json()
   201→      setActivities(data.activities)
   202→    } catch {
   203→      toast.error('Failed to load activities')
   204→    }
   205→  }, [])
   206→
   207→  useEffect(() => {
   208→    const loadAll = async () => {
   209→      setLoading(true)
   210→      await Promise.all([fetchStats(), fetchProjects(), fetchUsers(), fetchActivities()])
   211→      setLoading(false)
   212→    }
   213→    loadAll()
   214→  }, [fetchStats, fetchProjects, fetchUsers, fetchActivities])
   215→
   216→  // ── Handlers ──
   217→
   218→  const handleCreateProject = useCallback(async () => {
   219→    if (!newProject.name.trim()) {
   220→      toast.error('Project name is required')
   221→      return
   222→    }
   223→    if (!currentUser) return
   224→
   225→    setCreating(true)
   226→    try {
   227→      const res = await fetch('/api/projects', {
   228→        method: 'POST',
   229→        headers: { 'Content-Type': 'application/json' },
   230→        body: JSON.stringify({
   231→          ...newProject,
   232→          createdBy: currentUser.id,
   233→          dueDate: newProject.dueDate || null,
   234→        }),
   235→      })
   236→      if (!res.ok) throw new Error('Failed to create project')
   237→      toast.success('Project created successfully')
   238→      setCreateDialogOpen(false)
   239→      setNewProject({ name: '', description: '', color: '#10b981', priority: 'medium', dueDate: '' })
   240→      fetchProjects()
   241→      fetchStats()
   242→    } catch {
   243→      toast.error('Failed to create project')
   244→    } finally {
   245→      setCreating(false)
   246→    }
   247→  }, [newProject, currentUser, fetchProjects, fetchStats])
   248→
   249→  const handleUpdateUserRole = useCallback(async (userId: string, role: string) => {
   250→    try {
   251→      const res = await fetch('/api/users', {
   252→        method: 'PATCH',
   253→        headers: { 'Content-Type': 'application/json' },
   254→        body: JSON.stringify({ id: userId, role }),
   255→      })
   256→      if (!res.ok) throw new Error('Failed to update user role')
   257→      toast.success('User role updated')
   258→      fetchUsers()
   259→    } catch {
   260→      toast.error('Failed to update user role')
   261→    }
   262→  }, [fetchUsers])
   263→
   264→  const handleToggleUserStatus = useCallback(async (userId: string, currentStatus: string) => {
   265→    const newStatus = currentStatus === 'active' ? 'inactive' : 'active'
   266→    try {
   267→      const res = await fetch('/api/users', {
   268→        method: 'PATCH',
   269→        headers: { 'Content-Type': 'application/json' },
   270→        body: JSON.stringify({ id: userId, status: newStatus }),
   271→      })
   272→      if (!res.ok) throw new Error('Failed to update user status')
   273→      toast.success(`User ${newStatus === 'active' ? 'activated' : 'deactivated'}`)
   274→      fetchUsers()
   275→    } catch {
   276→      toast.error('Failed to update user status')
   277→    }
   278→  }, [fetchUsers])
   279→
   280→  const handleUpdateProjectStatus = useCallback(async (projectId: string, status: string) => {
   281→    try {
   282→      const res = await fetch(`/api/projects/${projectId}`, {
   283→        method: 'PATCH',
   284→        headers: { 'Content-Type': 'application/json' },
   285→        body: JSON.stringify({ status }),
   286→      })
   287→      if (!res.ok) throw new Error('Failed to update project status')
   288→      toast.success('Project status updated')
   289→      fetchProjects()
   290→      fetchStats()
   291→    } catch {
   292→      toast.error('Failed to update project status')
   293→    }
   294→  }, [fetchProjects, fetchStats])
   295→
   296→  // ── Chart Data ──
   297→
   298→  const chartData = stats
   299→    ? [
   300→        { name: 'Todo', count: stats.todoTasks, fill: '#6366f1' },
   301→        { name: 'In Progress', count: stats.inProgressTasks, fill: '#f59e0b' },
   302→        { name: 'In Review', count: stats.inReviewTasks, fill: '#06b6d4' },
   303→        { name: 'Done', count: stats.completedTasks, fill: '#10b981' },
   304→      ]
   305→    : []
   306→
   307→  // ── Render Helpers ──
   308→
   309→  const getActivityIcon = (action: string) => {
   310→    const lower = action.toLowerCase()
   311→    if (lower.includes('comment')) return ACTIVITY_ICONS.commented
   312→    if (lower.includes('complet') || lower.includes('done')) return ACTIVITY_ICONS.completed
   313→    if (lower.includes('delet')) return ACTIVITY_ICONS.deleted
   314→    if (lower.includes('assign')) return ACTIVITY_ICONS.assigned
   315→    if (lower.includes('status') || lower.includes('move')) return ACTIVITY_ICONS.status_changed
   316→    if (lower.includes('updat') || lower.includes('edit')) return ACTIVITY_ICONS.updated
   317→    return ACTIVITY_ICONS.created
   318→  }
   319→
   320→  // ── Loading State ──
   321→
   322→  if (loading) {
   323→    return (
   324→      <div className="space-y-6 p-4 md:p-6">
   325→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-4">
   326→          {Array.from({ length: 7 }).map((_, i) => (
   327→            <Skeleton key={i} className="h-28 rounded-xl" />
   328→          ))}
   329→        </div>
   330→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   331→          <Skeleton className="h-80 rounded-xl" />
   332→          <Skeleton className="h-80 rounded-xl" />
   333→        </div>
   334→      </div>
   335→    )
   336→  }
   337→
   338→  // ── Tab Rendering ──
   339→
   340→  return (
   341→    <div className="space-y-6 p-4 md:p-6">
   342→      {adminView === 'overview' && (
   343→        <OverviewTab
   344→          stats={stats!}
   345→          chartData={chartData}
   346→          activities={activities}
   347→          getActivityIcon={getActivityIcon}
   348→        />
   349→      )}
   350→      {adminView === 'projects' && (
   351→        <ProjectsTab
   352→          projects={projects}
   353→          onCreateClick={() => setCreateDialogOpen(true)}
   354→          onUpdateStatus={handleUpdateProjectStatus}
   355→        />
   356→      )}
   357→      {adminView === 'users' && (
   358→        <UsersTab
   359→          users={users}
   360→          onUpdateRole={handleUpdateUserRole}
   361→          onToggleStatus={handleToggleUserStatus}
   362→        />
   363→      )}
   364→      {adminView === 'activity' && (
   365→        <ActivityTab
   366→          activities={activities}
   367→          getActivityIcon={getActivityIcon}
   368→        />
   369→      )}
   370→
   371→      {/* ── Create Project Dialog ── */}
   372→      <Dialog open={createDialogOpen} onOpenChange={setCreateDialogOpen}>
   373→        <DialogContent className="sm:max-w-[480px]">
   374→          <DialogHeader>
   375→            <DialogTitle>Create New Project</DialogTitle>
   376→            <DialogDescription>
   377→              Add a new project to your workspace. Fill in the details below.
   378→            </DialogDescription>
   379→          </DialogHeader>
   380→          <div className="space-y-4 py-2">
   381→            <div className="space-y-2">
   382→              <Label htmlFor="project-name">Name *</Label>
   383→              <Input
   384→                id="project-name"
   385→                placeholder="Enter project name"
   386→                value={newProject.name}
   387→                onChange={(e) => setNewProject((p) => ({ ...p, name: e.target.value }))}
   388→              />
   389→            </div>
   390→            <div className="space-y-2">
   391→              <Label htmlFor="project-desc">Description</Label>
   392→              <Textarea
   393→                id="project-desc"
   394→                placeholder="Brief project description"
   395→                value={newProject.description}
   396→                onChange={(e) => setNewProject((p) => ({ ...p, description: e.target.value }))}
   397→                rows={3}
   398→              />
   399→            </div>
   400→            <div className="space-y-2">
   401→              <Label>Color</Label>
   402→              <div className="flex flex-wrap gap-2">
   403→                {PRESET_COLORS.map((c) => (
   404→                  <button
   405→                    key={c}
   406→                    type="button"
   407→                    className="h-8 w-8 rounded-full border-2 transition-transform hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-emerald-500"
   408→                    style={{
   409→                      backgroundColor: c,
   410→                      borderColor: newProject.color === c ? '#000' : 'transparent',
   411→                    }}
   412→                    onClick={() => setNewProject((p) => ({ ...p, color: c }))}
   413→                  />
   414→                ))}
   415→              </div>
   416→            </div>
   417→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   418→              <div className="space-y-2">
   419→                <Label>Priority</Label>
   420→                <Select
   421→                  value={newProject.priority}
   422→                  onValueChange={(val) => setNewProject((p) => ({ ...p, priority: val }))}
   423→                >
   424→                  <SelectTrigger>
   425→                    <SelectValue />
   426→                  </SelectTrigger>
   427→                  <SelectContent>
   428→                    <SelectItem value="low">Low</SelectItem>
   429→                    <SelectItem value="medium">Medium</SelectItem>
   430→                    <SelectItem value="high">High</SelectItem>
   431→                    <SelectItem value="urgent">Urgent</SelectItem>
   432→                  </SelectContent>
   433→                </Select>
   434→              </div>
   435→              <div className="space-y-2">
   436→                <Label htmlFor="project-due">Due Date</Label>
   437→                <Input
   438→                  id="project-due"
   439→                  type="date"
   440→                  value={newProject.dueDate}
   441→                  onChange={(e) => setNewProject((p) => ({ ...p, dueDate: e.target.value }))}
   442→                />
   443→              </div>
   444→            </div>
   445→          </div>
   446→          <DialogFooter>
   447→            <Button variant="outline" onClick={() => setCreateDialogOpen(false)}>
   448→              Cancel
   449→            </Button>
   450→            <Button onClick={handleCreateProject} disabled={creating || !newProject.name.trim()}>
   451→              {creating ? 'Creating...' : 'Create Project'}
   452→            </Button>
   453→          </DialogFooter>
   454→        </DialogContent>
   455→      </Dialog>
   456→    </div>
   457→  )
   458→}

   460→// ─── Overview Tab ─────────────────────────────────────────────────────────────
   461→
   462→function OverviewTab({
   463→  stats,
   464→  chartData,
   465→  activities,
   466→  getActivityIcon,
   467→}: {
   468→  stats: ExtendedDashboardStats
   469→  chartData: { name: string; count: number; fill: string }[]
   470→  activities: ActivityLog[]
   471→  getActivityIcon: (action: string) => React.ReactNode
   472→}) {
   473→  const statCards = [
   474→    {
   475→      label: 'Total Tasks',
   476→      value: stats.totalTasks,
   477→      icon: <ListTodo className="h-5 w-5" />,
   478→      color: 'text-emerald-600',
   479→      bg: 'bg-emerald-50',
   480→    },
   481→    {
   482→      label: 'Completed',
   483→      value: stats.completedTasks,
   484→      icon: <CheckCircle2 className="h-5 w-5" />,
   485→      color: 'text-emerald-600',
   486→      bg: 'bg-emerald-50',
   487→    },
   488→    {
   489→      label: 'In Progress',
   490→      value: stats.inProgressTasks,
   491→      icon: <Clock className="h-5 w-5" />,
   492→      color: 'text-amber-600',
   493→      bg: 'bg-amber-50',
   494→    },
   495→    {
   496→      label: 'Overdue',
   497→      value: stats.overdueTasks,
   498→      icon: <AlertTriangle className="h-5 w-5" />,
   499→      color: 'text-rose-600',
   500→      bg: 'bg-rose-50',
   501→    },
   502→    {
   503→      label: 'Projects',
   504→      value: stats.totalProjects,
   505→      icon: <FolderKanban className="h-5 w-5" />,
   506→      color: 'text-violet-600',
   507→      bg: 'bg-violet-50',
   508→    },
   509→    {
   510→      label: 'Members',
   511→      value: stats.totalMembers,
   512→      icon: <Users className="h-5 w-5" />,
   513→      color: 'text-cyan-600',
   514→      bg: 'bg-cyan-50',
   515→    },
   516→    {
   517→      label: 'Clients',
   518→      value: stats.totalClients,
   519→      icon: <UserCheck className="h-5 w-5" />,
   520→      color: 'text-pink-600',
   521→      bg: 'bg-pink-50',
   522→    },
   523→  ]
   524→
   525→  const completionRate = stats.totalTasks > 0
   526→    ? Math.round((stats.completedTasks / stats.totalTasks) * 100)
   527→    : 0
   528→
   529→  return (
   530→    <>
   531→      {/* Stat Cards */}
   532→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-4">
   533→        {statCards.map((card) => (
   534→          <motion.div
   535→            key={card.label}
   536→            initial={{ opacity: 0, y: 10 }}
   537→            animate={{ opacity: 1, y: 0 }}
   538→            transition={{ duration: 0.3 }}
   539→          >
   540→            <Card className="hover:shadow-md transition-shadow">
   541→              <CardContent className="p-4">
   542→                <div className="flex items-center justify-between">
   543→                  <div>
   544→                    <p className="text-sm text-muted-foreground">{card.label}</p>
   545→                    <p className="text-2xl font-bold mt-1">{card.value}</p>
   546→                  </div>
   547→                  <div className={`p-2.5 rounded-lg ${card.bg}`}>
   548→                    <span className={card.color}>{card.icon}</span>
   549→                  </div>
   550→                </div>
   551→              </CardContent>
   552→            </Card>
   553→          </motion.div>
   554→        ))}
   555→      </div>
   556→
   557→      {/* Completion Rate Banner */}
   558→      <motion.div
   559→        initial={{ opacity: 0, y: 10 }}
   560→        animate={{ opacity: 1, y: 0 }}
   561→        transition={{ duration: 0.3, delay: 0.1 }}
   562→      >
   563→        <Card>
   564→          <CardContent className="p-4">
   565→            <div className="flex items-center justify-between mb-2">
   566→              <div className="flex items-center gap-2">
   567→                <TrendingUp className="h-5 w-5 text-emerald-500" />
   568→                <span className="font-medium text-sm">Overall Completion Rate</span>
   569→              </div>
   570→              <span className="text-2xl font-bold text-emerald-600">{completionRate}%</span>
   571→            </div>
   572→            <div className="w-full bg-gray-100 rounded-full h-2.5">
   573→              <div
   574→                className="bg-emerald-500 h-2.5 rounded-full transition-all duration-500"
   575→                style={{ width: `${completionRate}%` }}
   576→              />
   577→            </div>
   578→            <div className="flex justify-between mt-2 text-xs text-muted-foreground">
   579→              <span>{stats.thisWeekCompleted} completed this week</span>
   580→              <span>{stats.totalTasks - stats.completedTasks} remaining</span>
   581→            </div>
   582→          </CardContent>
   583→        </Card>
   584→      </motion.div>
   585→
   586→      {/* Chart + Recent Activity */}
   587→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   588→        <motion.div
   589→          initial={{ opacity: 0, y: 10 }}
   590→          animate={{ opacity: 1, y: 0 }}
   591→          transition={{ duration: 0.3, delay: 0.2 }}
   592→        >

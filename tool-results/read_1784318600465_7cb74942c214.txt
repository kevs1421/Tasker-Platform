    58→    58→function UserAvatar({ user, size = 'md', className }: { user?: User | null; size?: 'sm' | 'md' | 'lg'; className?: string }) {
    59→    59→  const sizeMap = { sm: 'h-8 w-8 text-[10px]', md: 'h-10 w-10 text-xs', lg: 'h-12 w-12 text-sm' }
    60→    60→  if (!user) return <div className={cn(sizeMap[size], 'rounded-full bg-muted flex items-center justify-center', className)}>?</div>
    61→    61→  return (
    62→    62→    <Avatar className={cn(sizeMap[size], className)}>
    63→    63→      {user.avatar && <AvatarImage src={user.avatar} />}
    64→    64→      <AvatarFallback className="bg-slate-100 dark:bg-slate-800 font-medium">
    65→    65→        {getInitials(user.name)}
    66→    66→      </AvatarFallback>
    67→    67→    </Avatar>
    68→    68→  )
    69→    69→}

    82→    82→function CreateGroupDialog({
    83→    83→  open, onOpenChange, onCreated, currentUser,
    84→    84→}: {
    85→    85→  open: boolean; onOpenChange: (open: boolean) => void;
    86→    86→  onCreated: (group: MessageGroup) => void; currentUser: User;
    87→    87→}) {
    88→    88→  const [name, setName] = useState('')
    89→    89→  const [users, setUsers] = useState<User[]>([])
    90→    90→  const [selectedIds, setSelectedIds] = useState<string[]>([])

    92→    92→  const [search, setSearch] = useState('')
    93→    93→  const [loading, setLoading] = useState(false)
    94→    94→
    95→    95→  useEffect(() => {
    96→    96→    if (!open) return
    97→    97→    setLoading(true)
    98→    98→    setName('')
    99→    99→    setSelectedIds([])
   100→   100→    setSearch('')
   101→   101→    fetch('/api/users')
   102→   102→      .then((r) => r.json())
   103→   103→      .then((d) => {
   104→   104→        setUsers((d.users ?? []).filter((u: User) => u.id !== currentUser.id))
   105→   105→      })
   106→   106→      .catch(() => {})
   107→   107→      .finally(() => setLoading(false))
   108→   108→  }, [open, currentUser.id])
   109→   109→
   110→   110→  const toggleUser = (userId: string) => {
   111→   111→    setSelectedIds((prev) => prev.includes(userId) ? prev.filter((id) => id !== userId) : [...prev, userId])
   112→   112→  }
   113→   113→
   114→   114→  const handleCreate = async () => {
   115→   115→    if (!name.trim() || selectedIds.length === 0) return
   116→   116→    setCreating(true)
   117→   117→    try {
   118→   118→      const res = await fetch('/api/message-groups', {
   119→   119→        method: 'POST',
   120→   120→        headers: { 'Content-Type': 'application/json' },
   121→   121→        body: JSON.stringify({ name: name.trim(), createdBy: currentUser.id, memberIds: selectedIds }),
   122→   122→      })
   123→   123→      if (!res.ok) throw new Error()
   124→   124→      const data = await res.json()
   125→   125→      toast.success('Group created')
   126→   126→      onCreated(data.group)
   127→   127→      onOpenChange(false)
   128→   128→    } catch {
   129→   129→      toast.error('Failed to create group')
   130→   130→    } finally {
   131→   131→      setCreating(false)
   132→   132→    }
   133→   133→  }
   134→   134→
   135→   135→  const filteredUsers = users.filter(
   136→   136→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   137→   137→  )
   138→   138→
   139→   139→  return (
   140→   140→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   141→   141→      <DialogContent className="sm:max-w-md">
   142→   142→        <DialogHeader>
   143→   143→          <DialogTitle className="flex items-center gap-2">
   144→   144→            <Users className="h-5 w-5 text-violet-600" /> Create Group
   145→   145→          </DialogTitle>
   146→   146→          <DialogDescription>Start a group conversation with team members</DialogDescription>
   147→   147→        </DialogHeader>
   148→   148→
   149→   149→        <div className="space-y-4 py-2">
   150→   150→          <div>
   151→   151→            <Input placeholder="Group name..." value={name} onChange={(e) => setName(e.target.value)} autoFocus />
   152→   152→          </div>
   153→   153→
   154→   154→          <div>
   155→   155→            <div className="relative mb-2">
   156→   156→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   157→   157→              <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   158→   158→            </div>
   159→   159→            <ScrollArea className="max-h-48 rounded-md border">
   160→   160→              {loading ? (
   161→   161→                <div className="p-3 space-y-2">
   162→   162→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   163→   163→                </div>
   164→   164→              ) : filteredUsers.length === 0 ? (
   165→   165→                <p className="text-sm text-muted-foreground text-center py-4">No users found</p>
   166→   166→              ) : (
   167→   167→                <div className="p-1">
   168→   168→                  {filteredUsers.map((u) => (
   169→   169→                    <button
   170→   170→                      key={u.id}
   171→   171→                      type="button"
   172→   172→                      className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   173→   173→                      onClick={() => toggleUser(u.id)}
   174→   174→                    >
   175→   175→                      <Checkbox checked={selectedIds.includes(u.id)} />
   176→   176→                      <UserAvatar user={u} size="sm" />
   177→   177→                      <div className="flex-1 min-w-0">
   178→   178→                        <p className="text-sm font-medium truncate">{u.name}</p>
   179→   179→                        <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   180→   180→                      </div>
   181→   181→                      <Badge variant="secondary" className={cn('text-[10px]', getRoleBadgeClasses(u.role))}>{u.role}</Badge>
   182→   182→                    </button>
   183→   183→                  ))}
   184→   184→                </div>
   185→   185→              )}
   186→   186→            </ScrollArea>
   187→   187→            {selectedIds.length > 0 && (
   188→   188→              <p className="text-xs text-muted-foreground mt-2">{selectedIds.length} member{selectedIds.length > 1 ? 's' : ''} selected</p>
   189→   189→            )}
   190→   190→          </div>
   191→   191→        </div>
   192→   192→
   193→   193→        <DialogFooter>
   194→   194→          <Button variant="outline" type="button" onClick={() => onOpenChange(false)}>Cancel</Button>
   195→   195→          <Button
   196→   196→            type="button"
   197→   197→            onClick={handleCreate} disabled={!name.trim() || selectedIds.length === 0 || creating}
   198→   198→            className="bg-violet-600 hover:bg-violet-700 text-white"
   199→   199→          >
   200→   200→            {creating && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   201→   201→            Create Group
   202→   202→          </Button>
   203→   203→        </DialogFooter>
   204→   204→      </DialogContent>
   205→   205→    </Dialog>
   206→   206→  )
   207→   207→}

   289→   289→function GroupInfoDialog({
   290→   290→  open, onOpenChange, group, currentUserId,
   291→   291→}: {
   292→   292→  open: boolean; onOpenChange: (open: boolean) => void
   293→   293→  group: { id: string; name: string; members?: GroupMember[]; _count?: { members: number } } | null
   294→   294→  currentUserId: string
   295→   295→}) {
   296→   296→  const [members, setMembers] = useState<(GroupMember & { user: User })[]>([])
   297→   297→  const [users, setUsers] = useState<User[]>([])
   298→   298→  const [loading, setLoading] = useState(false)
   299→   299→  const [showAddUser, setShowAddUser] = useState(false)
   300→   300→  const [search, setSearch] = useState('')
   301→   301→  useEffect(() => {
   302→   302→    if (!open || !group) return
   303→   303→    setLoading(true)
   304→   304→    setSearch('')
   305→   305→    setShowAddUser(false)
   306→   306→    fetch(`/api/message-groups/${group.id}?userId=${currentUserId}`)
   307→   307→      .then((r) => r.json())
   308→   308→      .then((d) => {
   309→   309→        setMembers((d.group?.members ?? []) as (GroupMember & { user: User })[])
   310→   310→      })
   311→   311→      .catch(() => {})
   312→   312→      .finally(() => setLoading(false))
   313→   313→  }, [open, group, currentUserId])
   314→   314→
   315→   315→  const handleAddMember = async (userId: string) => {
   316→   316→    if (!group) return
   317→   317→    try {
   318→   318→      const res = await fetch(`/api/message-groups/${group.id}`, {
   319→   319→        method: 'PATCH',
   320→   320→        headers: { 'Content-Type': 'application/json' },
   321→   321→        body: JSON.stringify({ addMemberIds: [userId] }),
   322→   322→      })
   323→   323→      if (!res.ok) throw new Error()

   336→   336→        method: 'PATCH',
   337→   337→        headers: { 'Content-Type': 'application/json' },
   338→   338→        body: JSON.stringify({ removeMemberIds: [userId] }),
   339→   339→      })

   354→   354→        setUsers((d.users ?? []).filter((u: User) => !memberIds.includes(u.id)))

   356→   356→      .catch(() => {})

   359→   359→
   360→   360→  const filteredUsers = users.filter(
   361→   361→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())

   368→   368→          <DialogTitle className="flex items-center gap-2">
   369→   369→            <Users className="h-5 w-5 text-violet-600" /> Group Info

   399→   399→                  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />

   412→   412→                        <div className="flex-1 min-w-0">
   413→   413→                          <p className="text-sm font-medium truncate">{u.name}</p>

   424→   424→            <ScrollArea className="max-h-48 rounded-md border">
   425→   425→              {loading ? (
   426→   426→                <div className="p-3 space-y-2">

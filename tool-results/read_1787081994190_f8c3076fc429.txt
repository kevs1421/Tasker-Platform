    66→    66→function ActivityIcon({ action }: { action: string }) {
    67→    67→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    68→    68→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    69→    69→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    70→    70→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    71→    71→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    72→    72→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    73→    73→  return <CircleDot className="h-4 w-4 text-slate-400" />

   412→   412→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   413→   413→    if (!user) return
   414→   414→    setReviewing(submissionId)

   416→   416→      const notes = reviewNotes[submissionId] || ''
   417→   417→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   418→   418→        method: 'PATCH',
   419→   419→        headers: { 'Content-Type': 'application/json' },
   420→   420→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   421→   421→      })
   422→   422→      if (!patchRes.ok) {
   423→   423→        const d = await patchRes.json()
   424→   424→        throw new Error(d.error)
   425→   425→      }
   426→   426→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)

   428→   428→      if (selectedForm) {
   429→   429→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   430→   430→        if (fresh.ok) {
   431→   431→          const data = await fresh.json()
   432→   432→          setSelectedForm(data.form)
   433→   433→        }
   434→   434→      }
   435→   435→      fetchForms()
   436→   436→    } catch (e: unknown) {
   437→   437→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   438→   438→    } finally {
   439→   439→      setReviewing(null)
   440→   440→    }
   441→   441→  }

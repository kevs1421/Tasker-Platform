   104→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   105→  const visible = users.slice(0, max)
   106→  const extra = users.length - max
   107→  return (
   108→    <div className="flex -space-x-2">
   109→      {visible.map((u, i) =>
   110→        u ? (
   111→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   112→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   113→              {getInitials(u.name)}
   114→            </AvatarFallback>
   115→          </Avatar>
   116→        ) : null
   117→      )}
   118→      {extra > 0 && (
   119→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   120→          +{extra}
   121→        </div>
   122→      )}
   123→    </div>
   124→  )
   125→}
   126→
   127→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   128→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   129→  if (!user) return null
   130→  return (
   131→    <Avatar className={sizeClasses}>
   132→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   133→        {getInitials(user.name)}
   134→      </AvatarFallback>
   135→    </Avatar>
   136→  )
   137→}

   155→// ─── TeamProjects ───────────────────────────────────────────
   156→
   157→export function TeamProjects() {
   158→  const { user, setSelectedProjectId } = useAuthStore()
   159→  const [projects, setProjects] = useState<Project[]>([])
   160→  const [loading, setLoading] = useState(true)
   161→  const [expandedId, setExpandedId] = useState<string | null>(null)
   162→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   163→  const [expandedLoading, setExpandedLoading] = useState(false)
   164→  const [taskFilter, setTaskFilter] = useState<string>('all')
   165→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   166→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   167→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   168→  const [allTeamUsers, setAllTeamUsers] = useState<User[]>([])
   169→
   170→  // Create task form
   171→  const [taskTitle, setTaskTitle] = useState('')
   172→  const [taskDesc, setTaskDesc] = useState('')
   173→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   174→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   175→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   176→  const [taskTags, setTaskTags] = useState('')
   177→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   178→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   179→
   180→  const fetchProjects = useCallback(async () => {
   181→    try {
   182→      const res = await fetch('/api/projects')
   183→      const data = await res.json()
   184→      const filtered = (data.projects || []).filter((p: Project) =>
   185→        p.members?.some((m) => m.userId === user?.id)
   186→      )
   187→      setProjects(filtered)
   188→    } catch {
   189→      toast.error('Failed to load projects')
   190→    } finally {
   191→      setLoading(false)
   192→    }
   193→  }, [user])
   194→
   195→  useEffect(() => { fetchProjects() }, [fetchProjects])
   196→
   197→  // Fetch all active admin + team users for assignee picker
   198→  useEffect(() => {
   199→    fetch('/api/users')
   200→      .then(r => r.json())
   201→      .then(d => {
   202→        const members = (d.users || []).filter((u: User) => (u.role === 'team' || u.role === 'admin') && u.status === 'active')
   203→        setAllTeamUsers(members)
   204→      })
   205→      .catch(() => {})
   206→  }, [])
   207→
   208→  const handleExpand = async (projectId: string) => {
   209→    if (expandedId === projectId) {
   210→      setExpandedId(null)
   211→      setExpandedTasks([])
   212→      return
   213→    }
   214→    setExpandedId(projectId)
   215→    setTaskFilter('all')
   216→    setExpandedLoading(true)
   217→    try {
   218→      const res = await fetch(`/api/projects/${projectId}`)
   219→      const data = await res.json()
   220→      setExpandedTasks(data.project?.tasks || [])
   221→    } catch {
   222→      toast.error('Failed to load tasks')
   223→    } finally {
   224→      setExpandedLoading(false)
   225→    }
   226→  }
   227→
   228→  const handleCreateTask = async () => {
   229→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   230→    if (!user?.id) { toast.error('Not authenticated'); return }
   231→    setCreateTaskLoading(true)
   232→    try {
   233→      const res = await fetch('/api/tasks', {
   234→        method: 'POST',
   235→        headers: { 'Content-Type': 'application/json' },
   236→        body: JSON.stringify({
   237→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   238→          priority: taskPriority, projectId: createTaskProjectId,
   239→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   240→          tags: taskTags,
   241→          createdById: user.id,
   242→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   243→        }),
   244→      })
   245→      if (!res.ok) {
   246→        const d = await res.json()
   247→        throw new Error(d.error || 'Failed to create task')
   248→      }
   249→      toast.success('Task created')
   250→      setCreateTaskOpen(false)
   251→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   252→      setTaskPriority('medium'); setTaskDueDate(undefined)
   253→      setTaskTags(''); setTaskAssignee('')
   254→      // Always refresh the expanded project view
   255→      handleExpand(createTaskProjectId)
   256→    } catch (e: unknown) {
   257→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   258→    } finally {
   259→      setCreateTaskLoading(false)
   260→    }
   261→  }
   262→
   263→  const openCreateTaskForProject = (projectId: string) => {
   264→    setCreateTaskProjectId(projectId)
   265→    setCreateTaskOpen(true)
   266→  }
   267→
   268→  const getProjectTaskStats = (projectId: string) => {
   269→    if (expandedId !== projectId) return null
   270→    const total = expandedTasks.length
   271→    const done = expandedTasks.filter((t) => t.status === 'done').length
   272→    return { total, done }
   273→  }
   274→
   275→  const filteredTasks = taskFilter === 'all'
   276→    ? expandedTasks
   277→    : expandedTasks.filter((t) => t.status === taskFilter)
   278→
   279→  if (loading) {
   280→    return (
   281→      <div>
   282→        <div className="flex items-center justify-between mb-6">
   283→          <Skeleton className="h-8 w-48" />
   284→          <Skeleton className="h-10 w-36" />
   285→        </div>
   286→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   287→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   288→        </div>
   289→      </div>
   290→    )
   291→  }
   292→
   293→  return (
   294→    <div className="space-y-6">
   295→      {/* Header */}
   296→      <div className="flex items-center justify-between">
   297→        <div>
   298→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   299→          <p className="text-sm text-muted-foreground mt-1">
   300→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   301→          </p>
   302→        </div>
   303→      </div>
   304→
   305→      {/* Project Grid */}
   306→      {projects.length === 0 ? (
   307→        <EmptyState
   308→          icon={FolderKanban}
   309→          title="No projects yet"
   310→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   311→        />
   312→      ) : (
   313→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   314→          <AnimatePresence>
   315→            {projects.map((project, idx) => {
   316→              const totalTasks = project._count?.tasks || 0
   317→              const members = project.members || []
   318→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   319→              const pConfig = getPriorityConfig(project.priority)
   320→              const isExpanded = expandedId === project.id
   321→              const stats = getProjectTaskStats(project.id)
   322→
   323→              return (
   324→                <motion.div
   325→                  key={project.id}
   326→                  initial={{ opacity: 0, y: 20 }}
   327→                  animate={{ opacity: 1, y: 0 }}
   328→                  transition={{ delay: idx * 0.05 }}
   329→                >
   330→                  <Card
   331→                    className={cn(
   332→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   333→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   334→                    )}
   335→                    onClick={() => handleExpand(project.id)}
   336→                  >
   337→                    <div className="flex">
   338→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   339→                      <CardContent className="flex-1 p-4">
   340→                        <div className="flex items-start justify-between mb-2">
   341→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   342→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   343→                            {pConfig.label}
   344→                          </Badge>
   345→                        </div>
   346→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   347→                          {project.description || 'No description'}
   348→                        </p>
   349→                        <Progress
   350→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   351→                          className="h-1.5 mb-3"
   352→                        />
   353→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   354→                          <div className="flex items-center gap-3">
   355→                            <span className="flex items-center gap-1">
   356→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   357→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   358→                            </span>
   359→                            <span className="flex items-center gap-1">
   360→                              <Users className="h-3 w-3" />
   361→                              {members.length}
   362→                            </span>
   363→                          </div>
   364→                          {project.dueDate && (
   365→                            <span className={cn(
   366→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   367→                                ? 'text-red-500' : '',
   368→                            )}>
   369→                              {formatDate(project.dueDate)}
   370→                            </span>
   371→                          )}
   372→                        </div>
   373→                        {memberUsers.length > 0 && (
   374→                          <div className="mt-3">
   375→                            <AvatarStack users={memberUsers} max={4} />
   376→                          </div>
   377→                        )}
   378→                      </CardContent>
   379→                    </div>
   380→                  </Card>
   381→
   382→                  {/* Expanded Task List */}
   383→                  <AnimatePresence>
   384→                    {isExpanded && (
   385→                      <motion.div
   386→                        initial={{ opacity: 0, height: 0 }}
   387→                        animate={{ opacity: 1, height: 'auto' }}
   388→                        exit={{ opacity: 0, height: 0 }}
   389→                        transition={{ duration: 0.2 }}
   390→                        className="overflow-hidden"
   391→                        onClick={(e) => e.stopPropagation()}
   392→                      >
   393→                        <Card className="mt-3 p-4 space-y-4">
   394→                          <div className="flex items-center justify-between">
   395→                            <div className="flex items-center gap-2">
   396→                              <div className="flex gap-1">
   397→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   398→                                  <Button
   399→                                    key={s}
   400→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   401→                                    size="sm"
   402→                                    className="h-7 text-xs px-2"
   403→                                    onClick={() => setTaskFilter(s)}
   404→                                  >
   405→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   406→                                  </Button>
   407→                                ))}
   408→                              </div>
   409→                            </div>
   410→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   411→                              <Plus className="h-3 w-3" /> Add Task
   412→                            </Button>
   413→                          </div>
   414→                          {expandedLoading ? (
   415→                            <div className="space-y-2">
   416→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   417→                            </div>
   418→                          ) : filteredTasks.length === 0 ? (
   419→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   420→                          ) : (
   421→                            <ScrollArea className="max-h-64">
   422→                              <div className="space-y-2">
   423→                                {filteredTasks.map((task) => {
   424→                                  const sConfig = getStatusConfig(task.status)
   425→                                  const pConf = getPriorityConfig(task.priority)
   426→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   427→                                  return (
   428→                                    <div
   429→                                      key={task.id}
   430→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   431→                                      onClick={() => setSelectedProjectId(project.id)}
   432→                                    >
   433→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   434→                                      <div className="flex-1 min-w-0">
   435→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   436→                                        <div className="flex items-center gap-2 mt-0.5">
   437→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   438→                                          {task.dueDate && (
   439→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   440→                                              {formatDate(task.dueDate)}
   441→                                            </span>
   442→                                          )}
   443→                                        </div>
   444→                                      </div>
   445→                                      <AvatarStack users={assignees} max={2} />
   446→                                    </div>
   447→                                  )
   448→                                })}
   449→                              </div>
   450→                            </ScrollArea>
   451→                          )}
   452→                        </Card>
   453→                      </motion.div>
   454→                    )}
   455→                  </AnimatePresence>
   456→                </motion.div>
   457→              )
   458→            })}
   459→          </AnimatePresence>
   460→        </motion.div>
   461→      )}
   462→
   463→      {/* Create Task Dialog */}
   464→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   465→        <DialogContent className="sm:max-w-lg">
   466→          <DialogHeader>
   467→            <DialogTitle>Create New Task</DialogTitle>
   468→            <DialogDescription>Add a task to the project.</DialogDescription>
   469→          </DialogHeader>
   470→          <div className="space-y-4 py-2">
   471→            <div className="space-y-2">
   472→              <Label>Title</Label>
   473→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   474→            </div>
   475→            <div className="space-y-2">
   476→              <Label>Description</Label>
   477→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   478→            </div>
   479→            <div className="grid grid-cols-2 gap-4">
   480→              <div className="space-y-2">
   481→                <Label>Status</Label>
   482→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   483→                  <SelectTrigger><SelectValue /></SelectTrigger>
   484→                  <SelectContent>
   485→                    <SelectItem value="todo">To Do</SelectItem>
   486→                    <SelectItem value="in_progress">In Progress</SelectItem>
   487→                    <SelectItem value="in_review">In Review</SelectItem>
   488→                    <SelectItem value="done">Done</SelectItem>
   489→                  </SelectContent>
   490→                </Select>
   491→              </div>
   492→              <div className="space-y-2">
   493→                <Label>Priority</Label>
   494→                <Select value={taskPriority} onValueChange={(v) => setTaskPriority(v as typeof taskPriority)}>
   495→                  <SelectTrigger><SelectValue /></SelectTrigger>
   496→                  <SelectContent>
   497→                    <SelectItem value="low">Low</SelectItem>
   498→                    <SelectItem value="medium">Medium</SelectItem>
   499→                    <SelectItem value="high">High</SelectItem>
   500→                    <SelectItem value="urgent">Urgent</SelectItem>
   501→                  </SelectContent>
   502→                </Select>
   503→              </div>

  1846→function formatDuration(minutes: number): string {
  1847→  const h = Math.floor(minutes / 60)
  1848→  const m = minutes % 60
  1849→  if (h === 0) return `${m}m`
  1850→  if (m === 0) return `${h}h`
  1851→  return `${h}h ${m}m`
  1852→}

  1854→function formatElapsed(startDate: string): string {
  1855→  const ms = Date.now() - new Date(startDate).getTime()
  1856→  const totalSec = Math.floor(ms / 1000)
  1857→  const h = Math.floor(totalSec / 3600)
  1858→  const m = Math.floor((totalSec % 3600) / 60)
  1859→  const s = totalSec % 60
  1860→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  1861→}

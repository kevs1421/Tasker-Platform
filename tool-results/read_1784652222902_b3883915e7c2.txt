    65→function ActivityIcon({ action }: { action: string }) {
    66→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    67→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    68→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    69→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    70→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    71→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    72→  return <CircleDot className="h-4 w-4 text-slate-400" />
    73→}

   292→  const [loading, setLoading] = useState(true)
   293→  const [forms, setForms] = useState<OnboardingForm[]>([])
   294→  const [selectedForm, setSelectedForm] = useState<(OnboardingForm & { submissions: Onboarding[] }) | null>(null)
   295→  const [dialogOpen, setDialogOpen] = useState(false)
   296→  const [reviewing, setReviewing] = useState<string | null>(null)

   298→
   299→  // Create onboarding form state
   300→  const [createOpen, setCreateOpen] = useState(false)
   301→  const [creating, setCreating] = useState(false)

   304→  const [createForm, setCreateForm] = useState({
   305→    title: '',
   306→    description: '',
   307→    projectId: '',
   308→    clientId: '',
   309→    selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'],

   362→  async function openForm(form: OnboardingForm) {
   363→    try {
   364→      const res = await fetch(`/api/onboarding/${form.id}`)
   365→      if (!res.ok) throw new Error('Failed')
   366→      const data = await res.json()
   367→      setSelectedForm(data.form)
   368→      setDialogOpen(true)
   369→    } catch {
   370→      toast.error('Failed to load form details')
   371→    }
   372→  }
   373→
   374→  async function handleCreateOnboarding() {
   375→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   376→    setCreating(true)
   377→    try {
   378→      const stepsJson = JSON.stringify(
   379→        createForm.selectedSteps.map(stepId => {
   380→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   381→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   382→        })
   383→      )
   384→      const res = await fetch('/api/onboarding', {
   385→        method: 'POST',
   386→        headers: { 'Content-Type': 'application/json' },
   387→        body: JSON.stringify({
   388→          title: createForm.title.trim(),
   389→          description: createForm.description.trim() || null,
   390→          stepsJson,
   391→          projectId: createForm.projectId,
   392→          clientId: createForm.clientId || null,
   393→          status: 'published',
   394→        }),
   395→      })
   396→      if (!res.ok) {
   397→        const d = await res.json()
   398→        throw new Error(d.error || 'Failed to create')
   399→      }
   400→      toast.success('Onboarding form created successfully')
   401→      setCreateOpen(false)
   402→      setCreateForm({ title: '', description: '', projectId: '', clientId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })
   403→      fetchForms()
   404→    } catch (e: unknown) {
   405→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding form')
   406→    } finally {
   407→      setCreating(false)
   408→    }
   409→  }

   411→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   412→    if (!user) return
   413→    setReviewing(submissionId)
   414→    try {
   415→      const notes = reviewNotes[submissionId] || ''
   416→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   417→        method: 'PATCH',
   418→        headers: { 'Content-Type': 'application/json' },
   419→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   420→      })
   421→      if (!patchRes.ok) {
   422→        const d = await patchRes.json()
   423→        throw new Error(d.error)
   424→      }
   425→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   426→      // Refresh
   427→      if (selectedForm) {
   428→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   429→        if (fresh.ok) {
   430→          const data = await fresh.json()
   431→          setSelectedForm(data.form)
   432→        }
   433→      }
   434→      fetchForms()
   435→    } catch (e: unknown) {
   436→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   437→    } finally {
   438→      setReviewing(null)
   439→    }
   440→  }

   442→  async function handleDeleteForm(formId: string, formTitle: string) {
   443→    try {

   448→      toast.success(`"${formTitle}" deleted`)
   449→      setForms(prev => prev.filter(f => f.id !== formId))
   450→    } catch {
   451→      toast.error('Failed to delete onboarding form')
   452→    }
   453→  }

   462→      clientId: form.clientId ?? '',
   463→      selectedSteps: steps.length > 0 ? steps : ['business_info', 'brand_assets', 'project_kickoff', 'documents'],
   464→    })

   105→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   106→  const visible = users.slice(0, max)
   107→  const extra = users.length - max
   108→  return (
   109→    <div className="flex -space-x-2">
   110→      {visible.map((u, i) =>
   111→        u ? (
   112→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   113→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   114→              {getInitials(u.name)}
   115→            </AvatarFallback>
   116→          </Avatar>
   117→        ) : null
   118→      )}
   119→      {extra > 0 && (
   120→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   121→          +{extra}
   122→        </div>
   123→      )}
   124→    </div>
   125→  )
   126→}
   127→
   128→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   129→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   130→  if (!user) return null
   131→  return (
   132→    <Avatar className={sizeClasses}>
   133→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   134→        {getInitials(user.name)}
   135→      </AvatarFallback>
   136→    </Avatar>
   137→  )
   138→}

   156→// ─── TeamProjects ───────────────────────────────────────────
   157→
   158→export function TeamProjects() {
   159→  const { user, setSelectedProjectId } = useAuthStore()
   160→  const [projects, setProjects] = useState<Project[]>([])
   161→  const [loading, setLoading] = useState(true)
   162→  const [expandedId, setExpandedId] = useState<string | null>(null)
   163→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   164→  const [expandedLoading, setExpandedLoading] = useState(false)
   165→  const [taskFilter, setTaskFilter] = useState<string>('all')
   166→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   167→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   168→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   169→  const [allTeamUsers, setAllTeamUsers] = useState<User[]>([])
   170→
   171→  // Create task form
   172→  const [taskTitle, setTaskTitle] = useState('')
   173→  const [taskDesc, setTaskDesc] = useState('')
   174→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   175→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   176→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   177→  const [taskTags, setTaskTags] = useState('')
   178→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   179→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   180→
   181→  const fetchProjects = useCallback(async () => {
   182→    try {
   183→      const res = await fetch('/api/projects')
   184→      const data = await res.json()
   185→      const filtered = (data.projects || []).filter((p: Project) =>
   186→        p.members?.some((m) => m.userId === user?.id)
   187→      )
   188→      setProjects(filtered)
   189→    } catch {
   190→      toast.error('Failed to load projects')
   191→    } finally {
   192→      setLoading(false)
   193→    }
   194→  }, [user])
   195→
   196→  useEffect(() => { fetchProjects() }, [fetchProjects])
   197→
   198→  // Fetch all active admin + team users for assignee picker
   199→  useEffect(() => {
   200→    fetch('/api/users')
   201→      .then(r => r.json())
   202→      .then(d => {
   203→        const members = (d.users || []).filter((u: User) => (u.role === 'team' || u.role === 'admin') && u.status === 'active')
   204→        setAllTeamUsers(members)
   205→      })
   206→      .catch(() => {})
   207→  }, [])
   208→
   209→  const handleExpand = async (projectId: string) => {
   210→    if (expandedId === projectId) {
   211→      setExpandedId(null)
   212→      setExpandedTasks([])
   213→      return
   214→    }
   215→    setExpandedId(projectId)
   216→    setTaskFilter('all')
   217→    setExpandedLoading(true)
   218→    try {
   219→      const res = await fetch(`/api/projects/${projectId}`)
   220→      const data = await res.json()
   221→      setExpandedTasks(data.project?.tasks || [])
   222→    } catch {
   223→      toast.error('Failed to load tasks')
   224→    } finally {
   225→      setExpandedLoading(false)
   226→    }
   227→  }
   228→
   229→  const handleCreateTask = async () => {
   230→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   231→    if (!user?.id) { toast.error('Not authenticated'); return }
   232→    setCreateTaskLoading(true)
   233→    try {
   234→      const res = await fetch('/api/tasks', {
   235→        method: 'POST',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify({
   238→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   239→          priority: taskPriority, projectId: createTaskProjectId,
   240→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   241→          tags: taskTags,
   242→          createdById: user.id,
   243→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   244→        }),
   245→      })
   246→      if (!res.ok) {
   247→        const d = await res.json()
   248→        throw new Error(d.error || 'Failed to create task')
   249→      }
   250→      toast.success('Task created')
   251→      setCreateTaskOpen(false)
   252→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   253→      setTaskPriority('medium'); setTaskDueDate(undefined)
   254→      setTaskTags(''); setTaskAssignee('')
   255→      // Always refresh the expanded project view
   256→      handleExpand(createTaskProjectId)
   257→    } catch (e: unknown) {
   258→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   259→    } finally {
   260→      setCreateTaskLoading(false)
   261→    }
   262→  }
   263→
   264→  const openCreateTaskForProject = (projectId: string) => {
   265→    setCreateTaskProjectId(projectId)
   266→    setCreateTaskOpen(true)
   267→  }
   268→
   269→  const getProjectTaskStats = (projectId: string) => {
   270→    if (expandedId !== projectId) return null
   271→    const total = expandedTasks.length
   272→    const done = expandedTasks.filter((t) => t.status === 'done').length
   273→    return { total, done }
   274→  }
   275→
   276→  const filteredTasks = taskFilter === 'all'
   277→    ? expandedTasks
   278→    : expandedTasks.filter((t) => t.status === taskFilter)
   279→
   280→  if (loading) {
   281→    return (
   282→      <div>
   283→        <div className="flex items-center justify-between mb-6">
   284→          <Skeleton className="h-8 w-48" />
   285→          <Skeleton className="h-10 w-36" />
   286→        </div>
   287→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   288→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   289→        </div>
   290→      </div>
   291→    )
   292→  }
   293→
   294→  return (
   295→    <div className="space-y-6">
   296→      {/* Header */}
   297→      <div className="flex items-center justify-between">
   298→        <div>
   299→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   300→          <p className="text-sm text-muted-foreground mt-1">
   301→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   302→          </p>
   303→        </div>
   304→      </div>
   305→
   306→      {/* Project Grid */}
   307→      {projects.length === 0 ? (
   308→        <EmptyState
   309→          icon={FolderKanban}
   310→          title="No projects yet"
   311→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   312→        />
   313→      ) : (
   314→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   315→          <AnimatePresence>
   316→            {projects.map((project, idx) => {
   317→              const totalTasks = project._count?.tasks || 0
   318→              const members = project.members || []
   319→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   320→              const pConfig = getPriorityConfig(project.priority)
   321→              const isExpanded = expandedId === project.id
   322→              const stats = getProjectTaskStats(project.id)
   323→
   324→              return (
   325→                <motion.div
   326→                  key={project.id}
   327→                  initial={{ opacity: 0, y: 20 }}
   328→                  animate={{ opacity: 1, y: 0 }}
   329→                  transition={{ delay: idx * 0.05 }}
   330→                >
   331→                  <Card
   332→                    className={cn(
   333→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   334→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   335→                    )}
   336→                    onClick={() => handleExpand(project.id)}
   337→                  >
   338→                    <div className="flex">
   339→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   340→                      <CardContent className="flex-1 p-4">
   341→                        <div className="flex items-start justify-between mb-2">
   342→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   343→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   344→                            {pConfig.label}
   345→                          </Badge>
   346→                        </div>
   347→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   348→                          {project.description || 'No description'}
   349→                        </p>
   350→                        <Progress
   351→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   352→                          className="h-1.5 mb-3"
   353→                        />
   354→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   355→                          <div className="flex items-center gap-3">
   356→                            <span className="flex items-center gap-1">
   357→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   358→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   359→                            </span>
   360→                            <span className="flex items-center gap-1">
   361→                              <Users className="h-3 w-3" />
   362→                              {members.length}
   363→                            </span>
   364→                          </div>
   365→                          {project.dueDate && (
   366→                            <span className={cn(
   367→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   368→                                ? 'text-red-500' : '',
   369→                            )}>
   370→                              {formatDate(project.dueDate)}
   371→                            </span>
   372→                          )}
   373→                        </div>
   374→                        {memberUsers.length > 0 && (
   375→                          <div className="mt-3">
   376→                            <AvatarStack users={memberUsers} max={4} />
   377→                          </div>
   378→                        )}
   379→                      </CardContent>
   380→                    </div>
   381→                  </Card>
   382→
   383→                  {/* Expanded Task List */}
   384→                  <AnimatePresence>
   385→                    {isExpanded && (
   386→                      <motion.div
   387→                        initial={{ opacity: 0, height: 0 }}
   388→                        animate={{ opacity: 1, height: 'auto' }}
   389→                        exit={{ opacity: 0, height: 0 }}
   390→                        transition={{ duration: 0.2 }}
   391→                        className="overflow-hidden"
   392→                        onClick={(e) => e.stopPropagation()}
   393→                      >
   394→                        <Card className="mt-3 p-4 space-y-4">
   395→                          <div className="flex items-center justify-between">
   396→                            <div className="flex items-center gap-2">
   397→                              <div className="flex gap-1">
   398→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   399→                                  <Button
   400→                                    key={s}
   401→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   402→                                    size="sm"
   403→                                    className="h-7 text-xs px-2"
   404→                                    onClick={() => setTaskFilter(s)}
   405→                                  >
   406→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   407→                                  </Button>
   408→                                ))}
   409→                              </div>
   410→                            </div>
   411→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   412→                              <Plus className="h-3 w-3" /> Add Task
   413→                            </Button>
   414→                          </div>
   415→                          {expandedLoading ? (
   416→                            <div className="space-y-2">
   417→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   418→                            </div>
   419→                          ) : filteredTasks.length === 0 ? (
   420→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   421→                          ) : (
   422→                            <ScrollArea className="max-h-64">
   423→                              <div className="space-y-2">
   424→                                {filteredTasks.map((task) => {
   425→                                  const sConfig = getStatusConfig(task.status)
   426→                                  const pConf = getPriorityConfig(task.priority)
   427→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   428→                                  return (
   429→                                    <div
   430→                                      key={task.id}
   431→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   432→                                      onClick={() => setSelectedProjectId(project.id)}
   433→                                    >
   434→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   435→                                      <div className="flex-1 min-w-0">
   436→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   437→                                        <div className="flex items-center gap-2 mt-0.5">
   438→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   439→                                          {task.dueDate && (
   440→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   441→                                              {formatDate(task.dueDate)}
   442→                                            </span>
   443→                                          )}
   444→                                        </div>
   445→                                      </div>
   446→                                      <AvatarStack users={assignees} max={2} />
   447→                                    </div>
   448→                                  )
   449→                                })}
   450→                              </div>
   451→                            </ScrollArea>
   452→                          )}
   453→                        </Card>
   454→                      </motion.div>
   455→                    )}
   456→                  </AnimatePresence>
   457→                </motion.div>
   458→              )
   459→            })}
   460→          </AnimatePresence>
   461→        </motion.div>
   462→      )}
   463→
   464→      {/* Create Task Dialog */}
   465→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   466→        <DialogContent className="sm:max-w-lg">
   467→          <DialogHeader>
   468→            <DialogTitle>Create New Task</DialogTitle>
   469→            <DialogDescription>Add a task to the project.</DialogDescription>
   470→          </DialogHeader>
   471→          <div className="space-y-4 py-2">
   472→            <div className="space-y-2">
   473→              <Label>Title</Label>
   474→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   475→            </div>
   476→            <div className="space-y-2">
   477→              <Label>Description</Label>
   478→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   479→            </div>
   480→            <div className="grid grid-cols-2 gap-4">
   481→              <div className="space-y-2">
   482→                <Label>Status</Label>
   483→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   484→                  <SelectTrigger><SelectValue /></SelectTrigger>
   485→                  <SelectContent>
   486→                    <SelectItem value="todo">To Do</SelectItem>
   487→                    <SelectItem value="in_progress">In Progress</SelectItem>
   488→                    <SelectItem value="in_review">In Review</SelectItem>
   489→                    <SelectItem value="done">Done</SelectItem>
   490→                  </SelectContent>
   491→                </Select>
   492→              </div>
   493→              <div className="space-y-2">
   494→                <Label>Priority</Label>
   495→                <Select value={taskPriority} onValueChange={(v) => setTaskPriority(v as typeof taskPriority)}>
   496→                  <SelectTrigger><SelectValue /></SelectTrigger>
   497→                  <SelectContent>
   498→                    <SelectItem value="low">Low</SelectItem>
   499→                    <SelectItem value="medium">Medium</SelectItem>
   500→                    <SelectItem value="high">High</SelectItem>
   501→                    <SelectItem value="urgent">Urgent</SelectItem>
   502→                  </SelectContent>

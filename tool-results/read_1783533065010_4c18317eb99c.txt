   130→// ═══════════════════════════════════════════════════════════════════
   131→export function ClientOnboarding() {
   132→  const { user } = useAuthStore()
   133→  const [forms, setForms] = useState<(OnboardingForm & { _count?: { submissions: number } })[]>([])
   134→  const [loading, setLoading] = useState(true)
   135→  const [selectedForm, setSelectedForm] = useState<(OnboardingForm & { submissions: Onboarding[]; _count?: { submissions: number } }) | null>(null)
   136→  const [formLoading, setFormLoading] = useState(false)
   137→
   138→  const fetchForms = useCallback(async () => {
   139→    try {
   140→      const res = await fetch('/api/onboarding')
   141→      const data = await res.json()
   142→      setForms(data.forms || [])
   143→    } catch {
   144→      toast.error('Failed to load onboarding forms')
   145→    } finally {
   146→      setLoading(false)
   147→    }
   148→  }, [])
   149→
   150→  useEffect(() => { fetchForms() }, [fetchForms])
   151→
   152→  const openForm = async (formId: string) => {
   153→    setFormLoading(true)
   154→    try {
   155→      const res = await fetch(`/api/onboarding/${formId}`)
   156→      const data = await res.json()
   157→      setSelectedForm(data.form)
   158→    } catch {
   159→      toast.error('Failed to load form')
   160→    } finally {
   161→      setFormLoading(false)
   162→    }
   163→  }
   164→
   165→  // ─── Form List View ─────────────────────────────────────
   166→  if (selectedForm) {
   167→    return (
   168→      <OnboardingFormView
   169→        form={selectedForm}
   170→        userId={user?.id || ''}
   171→        onBack={() => { setSelectedForm(null); fetchForms() }}
   172→      />
   173→    )
   174→  }
   175→
   176→  return (
   177→    <div className="p-4 md:p-6 space-y-6">
   178→      <div className="flex items-center gap-3">
   179→        <div className="w-10 h-10 rounded-lg bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center">
   180→          <ClipboardList className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   181→        </div>
   182→        <div>
   183→          <h1 className="text-2xl font-bold">Onboarding</h1>
   184→          <p className="text-sm text-muted-foreground">Complete your onboarding forms to get started</p>
   185→        </div>
   186→      </div>
   187→
   188→      {loading ? <PageSkeleton /> : forms.length === 0 ? (
   189→        <EmptyState
   190→          icon={ClipboardList}
   191→          title="No onboarding forms"
   192→          description="There are no onboarding forms assigned to you yet."
   193→        />
   194→      ) : (
   195→        <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
   196→          {forms.map((form, i) => (
   197→            <motion.div
   198→              key={form.id}
   199→              initial={{ opacity: 0, y: 16 }}
   200→              animate={{ opacity: 1, y: 0 }}
   201→              transition={{ delay: i * 0.05 }}
   202→            >
   203→              <Card
   204→                className="cursor-pointer hover:shadow-md transition-shadow border-l-4 border-l-amber-500"
   205→                onClick={() => openForm(form.id)}
   206→              >
   207→                <CardHeader className="pb-3">
   208→                  <div className="flex items-start justify-between">
   209→                    <CardTitle className="text-base font-semibold leading-tight">{form.title}</CardTitle>
   210→                    {form.project && (
   211→                      <Badge variant="outline" className="ml-2 shrink-0 text-xs" style={{ borderColor: form.project.color, color: form.project.color }}>
   212→                        {form.project.name}
   213→                      </Badge>
   214→                    )}
   215→                  </div>
   216→                </CardHeader>
   217→                <CardContent>
   218→                  <p className="text-sm text-muted-foreground mb-3 line-clamp-2">
   219→                    {form.description || 'Complete this onboarding form to proceed with your project.'}
   220→                  </p>
   221→                  <div className="flex items-center gap-2 text-xs text-muted-foreground">
   222→                    <FileText className="w-3.5 h-3.5" />
   223→                    <span>{form._count?.submissions || 0} submission{((form._count?.submissions || 0) !== 1) ? 's' : ''}</span>
   224→                  </div>
   225→                </CardContent>
   226→              </Card>
   227→            </motion.div>
   228→          ))}
   229→        </div>
   230→      )}
   231→      {formLoading && <div className="fixed inset-0 bg-background/60 flex items-center justify-center z-50">
   232→        <Loader2 className="w-8 h-8 animate-spin text-amber-500" />
   233→      </div>}
   234→    </div>
   235→  )
   236→}

   238→// ─── Onboarding Form Wizard / Read-only View ─────────────────
   239→function OnboardingFormView({
   240→  form, userId, onBack
   241→}: {
   242→  form: OnboardingForm & { submissions: Onboarding[]; _count?: { submissions: number } }
   243→  userId: string
   244→  onBack: () => void
   245→}) {
   246→  const [steps, setSteps] = useState<StepDef[]>([])
   247→  const [currentStep, setCurrentStep] = useState(0)
   248→  const [responses, setResponses] = useState<Record<string, string>>({})
   249→  const [submitting, setSubmitting] = useState(false)
   250→
   251→  const mySubmission = form.submissions?.find(s => s.clientId === userId)
   252→
   253→  // Parse steps
   254→  useEffect(() => {
   255→    try {
   256→      const parsed = JSON.parse(form.stepsJson)
   257→      if (Array.isArray(parsed)) setSteps(parsed)
   258→    } catch {
   259→      toast.error('Failed to parse form steps')
   260→    }
   261→  }, [form.stepsJson])
   262→
   263→  // Load existing responses for editing
   264→  useEffect(() => {
   265→    if (mySubmission?.responsesJson) {
   266→      try {
   267→        setResponses(JSON.parse(mySubmission.responsesJson))
   268→      } catch { /* ignore */ }
   269→    }
   270→  }, [mySubmission])
   271→
   272→  const updateResponse = (name: string, value: string) => {
   273→    setResponses(prev => ({ ...prev, [name]: value }))
   274→  }
   275→
   276→  const currentFields = steps[currentStep]?.fields || []
   277→  const isStepValid = currentFields.every(f => !f.required || responses[f.name]?.trim())
   278→
   279→  const handleSubmit = async () => {
   280→    if (!isStepValid) return
   281→    setSubmitting(true)
   282→    try {
   283→      const submissionId = mySubmission?.id
   284→      const res = await fetch(`/api/onboarding/${submissionId || form.id}`, {
   285→        method: 'PATCH',
   286→        headers: { 'Content-Type': 'application/json' },
   287→        body: JSON.stringify({
   288→          action: 'submit',
   289→          responsesJson: JSON.stringify(responses),
   290→          ...(submissionId ? {} : { formId: form.id, clientId: userId })
   291→        })
   292→      })
   293→      if (!res.ok) throw new Error()
   294→      toast.success('Onboarding submitted successfully!')
   295→      onBack()
   296→    } catch {
   297→      toast.error('Failed to submit onboarding')
   298→    } finally {
   299→      setSubmitting(false)
   300→    }
   301→  }
   302→
   303→  const renderReviewResult = () => {
   304→    if (!mySubmission) return null
   305→    if (mySubmission.status === 'approved') return (
   306→      <div className="flex items-start gap-3 p-4 rounded-lg bg-emerald-50 dark:bg-emerald-900/20 border border-emerald-200 dark:border-emerald-800">
   307→        <CheckCircle2 className="w-5 h-5 text-emerald-600 mt-0.5 shrink-0" />
   308→        <div>
   309→          <p className="font-medium text-emerald-800 dark:text-emerald-300">Approved</p>
   310→          {mySubmission.reviewNotes && <p className="text-sm text-emerald-700 dark:text-emerald-400 mt-1">{mySubmission.reviewNotes}</p>}
   311→          {mySubmission.reviewedAt && <p className="text-xs text-emerald-600 dark:text-emerald-500 mt-1">Reviewed {formatRelativeTime(mySubmission.reviewedAt)}</p>}
   312→        </div>
   313→      </div>
   314→    )
   315→    if (mySubmission.status === 'rejected') return (
   316→      <div className="flex items-start gap-3 p-4 rounded-lg bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800">
   317→        <XCircle className="w-5 h-5 text-red-600 mt-0.5 shrink-0" />
   318→        <div>
   319→          <p className="font-medium text-red-800 dark:text-red-300">Needs Revision</p>
   320→          {mySubmission.reviewNotes && <p className="text-sm text-red-700 dark:text-red-400 mt-1">{mySubmission.reviewNotes}</p>}
   321→          {mySubmission.reviewedAt && <p className="text-xs text-red-600 dark:text-red-500 mt-1">Reviewed {formatRelativeTime(mySubmission.reviewedAt)}</p>}
   322→        </div>
   323→      </div>
   324→    )
   325→    if (mySubmission.status === 'submitted') return (
   326→      <div className="flex items-start gap-3 p-4 rounded-lg bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800">
   327→        <Clock className="w-5 h-5 text-amber-600 mt-0.5 shrink-0" />
   328→        <div>
   329→          <p className="font-medium text-amber-800 dark:text-amber-300">Under Review</p>
   330→          <p className="text-sm text-amber-700 dark:text-amber-400 mt-1">Your submission is being reviewed by the team.</p>
   331→        </div>
   332→      </div>
   333→    )
   334→    return null
   335→  }
   336→
   337→  const renderReadOnlyResponses = () => {
   338→    const allFields = steps.flatMap(s => s.fields)
   339→    return (
   340→      <div className="space-y-4">
   341→        {allFields.map(field => (
   342→          <div key={field.name} className="space-y-1">
   343→            <Label className="text-sm font-medium">{field.label}</Label>
   344→            <div className="p-3 rounded-lg bg-muted/50 border text-sm min-h-[40px]">
   345→              {responses[field.name] || (
   346→                <span className="text-muted-foreground italic">Not provided</span>
   347→              )}
   348→            </div>
   349→          </div>
   350→        ))}
   351→      </div>
   352→    )
   353→  }
   354→
   355→  const renderField = (field: StepField) => {
   356→    const value = responses[field.name] || ''
   357→    if (field.type === 'textarea') return (
   358→      <div key={field.name} className="space-y-2">
   359→        <Label htmlFor={field.name}>
   360→          {field.label} {field.required && <span className="text-red-500">*</span>}
   361→        </Label>
   362→        <Textarea
   363→          id={field.name}
   364→          value={value}
   365→          onChange={e => updateResponse(field.name, e.target.value)}
   366→          placeholder={field.placeholder || `Enter ${field.label.toLowerCase()}`}
   367→          rows={3}
   368→        />
   369→      </div>
   370→    )
   371→    if (field.type === 'select') return (
   372→      <div key={field.name} className="space-y-2">
   373→        <Label htmlFor={field.name}>
   374→          {field.label} {field.required && <span className="text-red-500">*</span>}
   375→        </Label>
   376→        <select
   377→          id={field.name}
   378→          value={value}
   379→          onChange={e => updateResponse(field.name, e.target.value)}
   380→          className="flex h-10 w-full rounded-md border border-input bg-background px-3 py-2 text-sm ring-offset-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   381→        >
   382→          <option value="">Select...</option>
   383→          {field.options?.map(opt => (
   384→            <option key={opt} value={opt}>{opt}</option>
   385→          ))}
   386→        </select>
   387→      </div>
   388→    )
   389→    return (
   390→      <div key={field.name} className="space-y-2">
   391→        <Label htmlFor={field.name}>
   392→          {field.label} {field.required && <span className="text-red-500">*</span>}
   393→        </Label>
   394→        <Input
   395→          id={field.name}
   396→          type={field.type === 'email' ? 'email' : field.type === 'url' ? 'url' : 'text'}
   397→          value={value}
   398→          onChange={e => updateResponse(field.name, e.target.value)}
   399→          placeholder={field.placeholder || `Enter ${field.label.toLowerCase()}`}
   400→        />
   401→      </div>
   402→    )
   403→  }
   404→
   405→  const isLocked = mySubmission?.status === 'approved' || mySubmission?.status === 'submitted'
   406→
   407→  return (
   408→    <div className="p-4 md:p-6 space-y-6">
   409→      <div className="flex items-center gap-3">
   410→        <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0">
   411→          <ArrowLeft className="w-5 h-5" />
   412→        </Button>
   413→        <div className="min-w-0">
   414→          <h1 className="text-2xl font-bold truncate">{form.title}</h1>
   415→          {form.description && (
   416→            <p className="text-sm text-muted-foreground truncate">{form.description}</p>
   417→          )}
   418→        </div>
   419→      </div>
   420→
   421→      {form.project && (
   422→        <Badge variant="outline" style={{ borderColor: form.project.color, color: form.project.color }}>
   423→          {form.project.name}
   424→        </Badge>
   425→      )}
   426→
   427→      {/* Review result banner */}
   428→      {renderReviewResult()}
   429→
   430→      {/* Stepper / Progress */}
   431→      {steps.length > 0 && (
   432→        <>
   433→          <div className="flex items-center gap-2 overflow-x-auto pb-2">
   434→            {steps.map((step, i) => {
   435→              const isCompleted = i < currentStep
   436→              const isCurrent = i === currentStep
   437→              return (
   438→                <button
   439→                  key={i}
   440→                  onClick={() => !isLocked && i <= currentStep && setCurrentStep(i)}
   441→                  disabled={isLocked || i > currentStep}
   442→                  className={`flex items-center gap-2 px-3 py-1.5 rounded-full text-sm font-medium whitespace-nowrap transition-colors ${
   443→                    isCompleted ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400' :
   444→                    isCurrent ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400' :
   445→                    'bg-muted text-muted-foreground'
   446→                  }`}
   447→                >
   448→                  {isCompleted ? <Check className="w-3.5 h-3.5" /> : <span className="w-5 h-5 rounded-full border-2 flex items-center justify-center text-xs">{i + 1}</span>}
   449→                  {step.title}
   450→                  {i < steps.length - 1 && <ChevronRight className="w-3.5 h-3.5 ml-1 text-muted-foreground" />}
   451→                </button>
   452→              )
   453→            })}
   454→          </div>
   455→
   456→          <Progress value={((currentStep + 1) / steps.length) * 100} className="h-2" />
   457→        </>
   458→      )}
   459→
   460→      {/* Content */}
   461→      <AnimatePresence mode="wait">
   462→        <motion.div
   463→          key={currentStep}
   464→          initial={{ opacity: 0, x: 20 }}
   465→          animate={{ opacity: 1, x: 0 }}
   466→          exit={{ opacity: 0, x: -20 }}
   467→          transition={{ duration: 0.2 }}
   468→        >
   469→          {steps[currentStep] && (
   470→            <Card>
   471→              <CardHeader>
   472→                <CardTitle className="text-lg">{steps[currentStep].title}</CardTitle>
   473→                {steps[currentStep].description && (
   474→                  <p className="text-sm text-muted-foreground">{steps[currentStep].description}</p>
   475→                )}
   476→              </CardHeader>
   477→              <CardContent className="space-y-4">
   478→                {isLocked ? renderReadOnlyResponses() : currentFields.map(renderField)}
   479→              </CardContent>
   480→            </Card>
   481→          )}
   482→        </motion.div>
   483→      </AnimatePresence>
   484→
   485→      {/* Navigation buttons */}
   486→      {!isLocked && steps.length > 0 && (
   487→        <div className="flex items-center justify-between">
   488→          <Button
   489→            variant="outline"
   490→            onClick={() => setCurrentStep(Math.max(0, currentStep - 1))}
   491→            disabled={currentStep === 0}
   492→          >
   493→            <ChevronLeft className="w-4 h-4 mr-1" /> Previous
   494→          </Button>
   495→          <div className="text-sm text-muted-foreground">
   496→            Step {currentStep + 1} of {steps.length}
   497→          </div>
   498→          {currentStep < steps.length - 1 ? (
   499→            <Button onClick={() => setCurrentStep(currentStep + 1)} disabled={!isStepValid}>
   500→              Next <ChevronRight className="w-4 h-4 ml-1" />
   501→            </Button>
   502→          ) : (
   503→            <Button onClick={handleSubmit} disabled={!isStepValid || submitting}>
   504→              {submitting ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Check className="w-4 h-4 mr-2" />}
   505→              Submit
   506→            </Button>
   507→          )}
   508→        </div>
   509→      )}
   510→    </div>
   511→  )
   512→}

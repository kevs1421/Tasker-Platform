    98→const ONBOARDING_STEPS = [
    99→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   100→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   101→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   102→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   103→]
   104→
   105→export function ClientProjects() {
   106→  const { user } = useAuthStore()
   107→  const [projects, setProjects] = useState<ClientProject[]>([])
   108→  const [loading, setLoading] = useState(true)
   109→
   110→  // Upload files state
   111→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   112→  const [uploading, setUploading] = useState(false)
   113→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   114→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   115→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   116→  const fileInputRef = useRef<HTMLInputElement>(null)
   117→
   118→  // Onboarding creation state
   119→
   120→
   121→  const fetchProjects = useCallback(async () => {
   122→    if (!user?.id) return
   123→    try {
   124→      // Use the client-only endpoint that queries through ClientProject table
   125→      // This ensures only projects explicitly assigned to this client are returned
   126→      const res = await fetch(`/api/client/projects?clientId=${user.id}`)
   127→      const data = await res.json()
   128→      setProjects(data.projects || [])
   129→    } catch {
   130→      toast.error('Failed to load projects')
   131→    } finally {
   132→      setLoading(false)
   133→    }
   134→  }, [user?.id])
   135→
   136→  // Fetch uploaded files from localStorage
   137→  const fetchUploadedFiles = useCallback(() => {
   138→    if (!user) return
   139→    try {
   140→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   141→      if (stored) setUploadedFiles(JSON.parse(stored))
   142→    } catch { /* ignore */ }
   143→  }, [user])
   144→
   145→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   146→    if (!user) return
   147→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   148→  }, [user])
   149→
   150→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   151→
   152→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   153→  const handleFileUpload = async (files: FileList | null) => {
   154→    if (!files || files.length === 0) return
   155→    if (!uploadDialogProjectId) {
   156→      toast.error('No project selected')
   157→      return
   158→    }
   159→    setUploading(true)
   160→    const newFiles: typeof uploadedFiles = []
   161→    for (let i = 0; i < files.length; i++) {
   162→      const file = files[i]
   163→      try {
   164→        const formData = new FormData()
   165→        formData.append('file', file)
   166→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   167→        if (!res.ok) {
   168→          const d = await res.json()
   169→          throw new Error(d.error || 'Upload failed')
   170→        }
   171→        const data = await res.json()
   172→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   173→      } catch (e: unknown) {
   174→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   175→      }
   176→    }
   177→    if (newFiles.length > 0) {
   178→      const updated = [...uploadedFiles, ...newFiles]
   179→      setUploadedFiles(updated)
   180→      saveUploadedFiles(updated)
   181→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   182→    }
   183→    setUploading(false)
   184→    if (fileInputRef.current) fileInputRef.current.value = ''
   185→  }
   186→
   187→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   188→    e.preventDefault()
   189→    setUploadDragOver(null)
   190→    setUploadDialogProjectId(projectId)
   191→    handleFileUpload(e.dataTransfer.files)
   192→  }
   193→
   194→  const openUploadDialog = (projectId: string) => {
   195→    setUploadDialogProjectId(projectId)
   196→    setUploadDialogOpen(true)
   197→  }
   198→
   199→  const deleteUploadedFile = (idx: number) => {
   200→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   201→    setUploadedFiles(updated)
   202→    saveUploadedFiles(updated)
   203→    toast.success('File removed')
   204→  }
   205→
   206→  const getFileIcon = (type: string) => {
   207→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   208→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   209→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   210→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   211→    return <File className="w-4 h-4 text-muted-foreground" />
   212→  }
   213→
   214→  const formatFileSize = (bytes: number) => {
   215→    if (bytes < 1024) return `${bytes} B`
   216→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   217→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   218→  }
   219→
   220→  if (loading) return <PageSkeleton />
   221→
   222→  return (
   223→    <div className="p-4 md:p-6 space-y-6">
   224→      {/* Page Header */}
   225→      <div className="flex items-center gap-3">
   226→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   227→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   228→        </div>
   229→        <div>
   230→          <h1 className="text-2xl font-bold">My Projects</h1>
   231→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   232→        </div>
   233→      </div>
   234→
   235→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   236→      <input
   237→        ref={fileInputRef}
   238→        type="file"
   239→        multiple
   240→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   241→        className="hidden"
   242→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   243→      />
   244→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   245→        <DialogContent className="sm:max-w-md">
   246→          <DialogHeader>
   247→            <DialogTitle className="flex items-center gap-2">
   248→              <Upload className="w-5 h-5 text-primary" />
   249→              Upload Files
   250→            </DialogTitle>
   251→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   252→          </DialogHeader>
   253→          <div
   254→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   255→            onDragLeave={() => setUploadDragOver(null)}
   256→            onDrop={(e) => {
   257→              e.preventDefault()
   258→              setUploadDragOver(null)
   259→              // Set the project context before handling files
   260→              const files = e.dataTransfer.files
   261→              if (files && files.length > 0) {
   262→                // Use a separate handler for dialog drops
   263→                void (async () => {
   264→                  setUploading(true)
   265→                  const newFiles: typeof uploadedFiles = []
   266→                  for (let i = 0; i < files.length; i++) {
   267→                    const file = files[i]
   268→                    try {
   269→                      const formData = new FormData()
   270→                      formData.append('file', file)
   271→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   272→                      if (!res.ok) {
   273→                        const d = await res.json()
   274→                        throw new Error(d.error || 'Upload failed')
   275→                      }
   276→                      const data = await res.json()
   277→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   278→                    } catch (err: unknown) {
   279→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   280→                    }
   281→                  }
   282→                  if (newFiles.length > 0) {
   283→                    const updated = [...uploadedFiles, ...newFiles]
   284→                    setUploadedFiles(updated)
   285→                    saveUploadedFiles(updated)
   286→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   287→                  }
   288→                  setUploading(false)
   289→                })()
   290→              }
   291→            }}
   292→            onClick={() => fileInputRef.current?.click()}
   293→            className={cn(
   294→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   295→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   296→            )}
   297→          >
   298→            {uploading ? (
   299→              <div className="flex flex-col items-center gap-2">
   300→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   301→                <p className="text-sm text-muted-foreground">Uploading...</p>
   302→              </div>
   303→            ) : (
   304→              <div className="flex flex-col items-center gap-2">
   305→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   306→                  <Upload className="w-5 h-5 text-muted-foreground" />
   307→                </div>
   308→                <div>
   309→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   310→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   311→                </div>
   312→              </div>
   313→            )}
   314→          </div>
   315→          <DialogFooter>
   316→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   317→          </DialogFooter>
   318→        </DialogContent>
   319→      </Dialog>
   320→
   321→      {/* ─── Project Progress Overview ──────────────────────────── */}
   322→      {projects.length > 0 && (
   323→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   324→          <Card>
   325→            <CardHeader className="pb-3">
   326→              <CardTitle className="text-base flex items-center gap-2">
   327→                <BarChart3 className="w-4 h-4 text-primary" />
   328→                Project Progress
   329→              </CardTitle>
   330→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   331→            </CardHeader>
   332→            <CardContent>
   333→              <div className="space-y-5">
   334→                {projects.map((project) => {
   335→                  const todo = project.taskStats?.todo || 0
   336→                  const inProgress = project.taskStats?.in_progress || 0
   337→                  const inReview = project.taskStats?.in_review || 0
   338→                  const completed = project.taskStats?.done || 0
   339→                  const total = todo + inProgress + inReview + completed
   340→                  if (total === 0) return null
   341→
   342→                  const todoPct = Math.round((todo / total) * 100)
   343→                  const inProgressPct = Math.round((inProgress / total) * 100)
   344→                  const inReviewPct = Math.round((inReview / total) * 100)
   345→                  const completedPct = Math.round((completed / total) * 100)
   346→
   347→                  return (
   348→                    <div key={project.id} className="space-y-2.5">
   349→                      <div className="flex items-center justify-between">
   350→                        <div className="flex items-center gap-2">
   351→                          <span className="w-3 h-3 rounded-full" style={{ backgroundColor: project.color }} />
   352→                          <span className="text-sm font-medium">{project.name}</span>
   353→                        </div>
   354→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   355→                      </div>
   356→                      {/* Stacked progress bar */}
   357→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   358→                        {todo > 0 && (
   359→                          <div
   360→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   361→                            style={{ width: `${todoPct}%` }}
   362→                            title={`Not Started: ${todo}`}
   363→                          />
   364→                        )}
   365→                        {inProgress > 0 && (
   366→                          <div
   367→                            className="bg-amber-500 transition-all duration-500"
   368→                            style={{ width: `${inProgressPct}%` }}
   369→                            title={`In Progress: ${inProgress}`}
   370→                          />
   371→                        )}
   372→                        {inReview > 0 && (
   373→                          <div
   374→                            className="bg-blue-500 transition-all duration-500"
   375→                            style={{ width: `${inReviewPct}%` }}
   376→                            title={`In Review: ${inReview}`}
   377→                          />
   378→                        )}
   379→                        {completed > 0 && (
   380→                          <div
   381→                            className="bg-emerald-500 transition-all duration-500"
   382→                            style={{ width: `${completedPct}%` }}
   383→                            title={`Completed: ${completed}`}
   384→                          />
   385→                        )}
   386→                      </div>
   387→                      {/* Legend */}
   388→                      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   389→                        <span className="flex items-center gap-1.5">
   390→                          <span className="w-2.5 h-2.5 rounded-sm bg-slate-400 dark:bg-slate-500" />
   391→                          Not Started ({todo})
   392→                        </span>
   393→                        <span className="flex items-center gap-1.5">
   394→                          <span className="w-2.5 h-2.5 rounded-sm bg-amber-500" />
   395→                          In Progress ({inProgress})
   396→                        </span>
   397→                        <span className="flex items-center gap-1.5">
   398→                          <span className="w-2.5 h-2.5 rounded-sm bg-blue-500" />
   399→                          In Review ({inReview})
   400→                        </span>
   401→                        <span className="flex items-center gap-1.5">
   402→                          <span className="w-2.5 h-2.5 rounded-sm bg-emerald-500" />
   403→                          Completed ({completed})
   404→                        </span>
   405→                      </div>
   406→                    </div>
   407→                  )
   408→                })}
   409→              </div>
   410→            </CardContent>
   411→          </Card>
   412→        </motion.div>
   413→      )}
   414→
   415→      {projects.length === 0 ? (
   416→        <EmptyState
   417→          icon={FolderKanban}
   418→          title="No active projects"
   419→          description="You don't have any active projects yet."
   420→        />
   421→      ) : (
   422→        <>
   423→          {/* ─── Project Cards ──────────────────────────────────── */}
   424→          <div className="space-y-4">
   425→            {projects.map((project, i) => {
   426→              const doneTasks = project.taskStats?.done || 0
   427→              const totalT = project._count?.tasks || 0
   428→              const pct = project.progress ?? 0
   429→              const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   430→
   431→              // Compute progress bar gradient colors
   432→              const barColor = pct >= 75 ? '#10b981' : pct >= 40 ? '#f59e0b' : '#ef4444'
   433→
   434→              // Files for this project only
   435→              const projectFiles = uploadedFiles.filter(f => f.projectId === project.id)
   436→
   437→              return (
   438→                <motion.div
   439→                  key={project.id}
   440→                  initial={{ opacity: 0, y: 12 }}
   441→                  animate={{ opacity: 1, y: 0 }}
   442→                  transition={{ delay: i * 0.05 }}
   443→                >
   444→                  <Card
   445→                    className="hover:shadow-md transition-all border-l-4"
   446→                    style={{ borderLeftColor: project.color }}
   447→                  >
   448→                    <CardContent className="p-4 md:p-5">
   449→                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   450→                        <div className="min-w-0 flex-1">
   451→                          <div className="flex items-center gap-2 mb-1">
   452→                            <h3 className="font-semibold text-base truncate">{project.name}</h3>
   453→                            <PriorityBadge priority={project.priority} />
   454→                          </div>
   455→                          <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   456→                        </div>
   457→                        <div className="flex items-center gap-2 shrink-0 text-sm flex-wrap">
   458→                          {project.dueDate && (
   459→                            <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   460→                              <Calendar className="w-3.5 h-3.5" />
   461→                              {formatDate(project.dueDate)}
   462→                            </span>
   463→                          )}
   464→                          <span className="text-muted-foreground flex items-center gap-1">
   465→                            <Users className="w-3.5 h-3.5" />
   466→                            {project._count?.members || 0}
   467→                          </span>
   468→                          <Button
   469→                            size="sm"
   470→                            variant="outline"
   471→                            className="h-7 gap-1.5 text-xs"
   472→                            onClick={(e) => { e.stopPropagation(); openUploadDialog(project.id) }}
   473→                          >
   474→                            <Upload className="w-3.5 h-3.5" />
   475→                            Upload Files
   476→                          </Button>
   477→                          <Button
   478→                            size="sm"
   479→                            variant="outline"
   480→                            className="h-7 gap-1.5 text-xs"
   481→                            onClick={(e) => { e.stopPropagation(); useAuthStore.getState().setRoute({ page: 'client-dashboard', tab: 'onboarding' }) }}
   482→                          >
   483→                            <ClipboardCheck className="w-3.5 h-3.5" />
   484→                            Onboarding
   485→                          </Button>
   486→                        </div>
   487→                      </div>
   488→
   489→                      {/* Progress bar with percentage */}
   490→                      <div className="mt-3 space-y-1">
   491→                        <div className="flex items-center justify-between text-xs">
   492→                          <span className="text-muted-foreground">{doneTasks}/{totalT} tasks completed</span>
   493→                          <span className="font-semibold" style={{ color: barColor }}>{pct}%</span>
   494→                        </div>
   495→                        <div className="relative h-2.5 w-full overflow-hidden rounded-full bg-primary/20">
   496→                          <div
   497→                            className="absolute top-0 left-0 h-2.5 rounded-full transition-all duration-500"
   498→                            style={{
   499→                              width: `${pct}%`,
   500→                              background: `linear-gradient(90deg, ${barColor}88, ${barColor})`,
   501→                            }}
   502→                          />
   503→                        </div>
   504→                      </div>
   505→
   506→                      {/* Per-project uploaded files */}
   507→                      {projectFiles.length > 0 && (
   508→                        <div className="mt-4 pt-3 border-t">

   510→                            Files ({projectFiles.length})
   511→                          </p>
   512→                          <div className="space-y-1.5 max-h-48 overflow-y-auto">
   513→                            {projectFiles.map((file, idx) => {

   516→                                <motion.div
   517→                                  key={file.filename}
   518→                                  initial={{ opacity: 0, x: -8 }}

   549→                    </CardContent>

   551→                </motion.div>

   554→          </div>

   557→
   558→    </div>

   568→// CLIENT PROFILE
   569→// ═══════════════════════════════════════════════════════════════════
   570→export function ClientProfile() {
   571→  const { user, updateUser } = useAuthStore()
   572→  const [loading, setLoading] = useState(true)
   573→  const [saving, setSaving] = useState(false)
   574→  const [savingPassword, setSavingPassword] = useState(false)
   575→
   576→  const [name, setName] = useState('')
   577→  const [email, setEmail] = useState('')
   578→  const [phone, setPhone] = useState('')
   579→  const [company, setCompany] = useState('')
   580→
   581→  const [currentPassword, setCurrentPassword] = useState('')
   582→  const [newPassword, setNewPassword] = useState('')
   583→  const [confirmPassword, setConfirmPassword] = useState('')
   584→
   585→  useEffect(() => {
   586→    if (!user?.id) return
   587→    const fetchProfile = async () => {
   588→      try {
   589→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   590→        const data = await res.json()
   591→        if (data.user) {
   592→          setName(data.user.name || '')
   593→          setEmail(data.user.email || '')
   594→          setPhone(data.user.phone || '')
   595→          setCompany(data.user.company || '')
   596→        }
   597→      } catch {
   598→        toast.error('Failed to load profile')
   599→      } finally {
   600→        setLoading(false)
   601→      }
   602→    }
   603→    fetchProfile()
   604→  }, [user?.id])
   605→
   606→  const saveProfile = async () => {
   607→    if (!user?.id) return
   608→    setSaving(true)
   609→    try {
   610→      const res = await fetch('/api/user/profile', {
   611→        method: 'PATCH',
   612→        headers: { 'Content-Type': 'application/json' },
   613→        body: JSON.stringify({ userId: user.id, name, email, phone, company })
   614→      })
   615→      if (!res.ok) {
   616→        const data = await res.json()
   617→        throw new Error(data.error || 'Failed')
   618→      }
   619→      const data = await res.json()
   620→      toast.success('Profile updated successfully')
   621→      // Update user in store
   622→      if (data.user && user) {
   623→        updateUser({ name: data.user.name, email: data.user.email, phone: data.user.phone, company: data.user.company })
   624→      }
   625→    } catch (e: unknown) {
   626→      const msg = e instanceof Error ? e.message : 'Failed to update profile'
   627→      toast.error(msg)
   628→    } finally {
   629→      setSaving(false)
   630→    }
   631→  }
   632→
   633→  const changePassword = async () => {
   634→    if (!user?.id) return
   635→    if (!currentPassword || !newPassword || !confirmPassword) {
   636→      toast.error('Please fill in all password fields')
   637→      return
   638→    }
   639→    if (newPassword !== confirmPassword) {
   640→      toast.error('New passwords do not match')
   641→      return
   642→    }
   643→    if (newPassword.length < 6) {
   644→      toast.error('Password must be at least 6 characters')
   645→      return
   646→    }
   647→    setSavingPassword(true)
   648→    try {
   649→      const res = await fetch('/api/user/profile', {
   650→        method: 'PATCH',
   651→        headers: { 'Content-Type': 'application/json' },
   652→        body: JSON.stringify({
   653→          userId: user.id,
   654→          currentPassword,
   655→          newPassword
   656→        })
   657→      })
   658→      if (!res.ok) {
   659→        const data = await res.json()
   660→        throw new Error(data.error || 'Failed')
   661→      }
   662→      toast.success('Password changed successfully')
   663→      setCurrentPassword('')
   664→      setNewPassword('')
   665→      setConfirmPassword('')
   666→    } catch (e: unknown) {
   667→      const msg = e instanceof Error ? e.message : 'Failed to change password'
   668→      toast.error(msg)
   669→    } finally {
   670→      setSavingPassword(false)
   671→    }
   672→  }
   673→
   674→  if (loading) return <PageSkeleton />
   675→
   676→  return (
   677→    <div className="p-4 md:p-6 space-y-6 max-w-2xl">
   678→      <div className="flex items-center gap-3">
   679→        <div className="w-10 h-10 rounded-lg bg-rose-100 dark:bg-rose-900/30 flex items-center justify-center">
   680→          <UserCircle className="w-5 h-5 text-rose-600 dark:text-rose-400" />
   681→        </div>
   682→        <div>
   683→          <h1 className="text-2xl font-bold">Profile Settings</h1>
   684→          <p className="text-sm text-muted-foreground">Manage your account information</p>
   685→        </div>
   686→      </div>
   687→
   688→      {/* Profile Card */}
   689→      <motion.div {...fadeSlide}>
   690→        <Card>
   691→          <CardHeader>
   692→            <div className="flex items-center justify-between">
   693→              <CardTitle className="text-lg">Personal Information</CardTitle>
   694→              <span className={`text-xs px-2.5 py-1 rounded-full font-medium ${getRoleBadgeClasses(user?.role || 'client')}`}>
   695→                Client
   696→              </span>
   697→            </div>
   698→            {user?.createdAt && (
   699→              <p className="text-sm text-muted-foreground">
   700→                Member since {formatDate(user.createdAt)}
   701→              </p>
   702→            )}
   703→          </CardHeader>
   704→          <CardContent className="space-y-4">
   705→            <div className="space-y-2">
   706→              <Label htmlFor="profile-name">Full Name</Label>
   707→              <Input
   708→                id="profile-name"
   709→                value={name}
   710→                onChange={e => setName(e.target.value)}
   711→                placeholder="Your full name"
   712→              />
   713→            </div>
   714→            <div className="space-y-2">
   715→              <Label htmlFor="profile-email">Email Address</Label>
   716→              <Input
   717→                id="profile-email"
   718→                type="email"
   719→                value={email}
   720→                onChange={e => setEmail(e.target.value)}
   721→                placeholder="your@email.com"
   722→              />
   723→            </div>
   724→            <div className="space-y-2">
   725→              <Label htmlFor="profile-phone">Phone Number</Label>
   726→              <Input
   727→                id="profile-phone"
   728→                type="tel"
   729→                value={phone}
   730→                onChange={e => setPhone(e.target.value)}
   731→                placeholder="+1 (555) 000-0000"
   732→              />
   733→            </div>
   734→            <div className="space-y-2">
   735→              <Label htmlFor="profile-company">Company</Label>
   736→              <Input
   737→                id="profile-company"
   738→                value={company}
   739→                onChange={e => setCompany(e.target.value)}
   740→                placeholder="Your company name"
   741→              />
   742→            </div>
   743→            <div className="pt-2">
   744→              <Button onClick={saveProfile} disabled={saving} className="w-full sm:w-auto">
   745→                {saving ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   746→                Save Changes
   747→              </Button>
   748→            </div>
   749→          </CardContent>
   750→        </Card>
   751→      </motion.div>
   752→
   753→      {/* Change Password */}
   754→      <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   755→        <Card>
   756→          <CardHeader>
   757→            <CardTitle className="text-lg flex items-center gap-2">
   758→              <Shield className="w-5 h-5 text-muted-foreground" />
   759→              Change Password
   760→            </CardTitle>
   761→            <p className="text-sm text-muted-foreground">
   762→              Update your password to keep your account secure
   763→            </p>
   764→          </CardHeader>
   765→          <CardContent className="space-y-4">
   766→            <div className="space-y-2">
   767→              <Label htmlFor="current-password">Current Password</Label>
   768→              <div className="relative">
   769→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   770→                <Input
   771→                  id="current-password"
   772→                  type="password"
   773→                  value={currentPassword}
   774→                  onChange={e => setCurrentPassword(e.target.value)}
   775→                  placeholder="Enter current password"
   776→                  className="pl-9"
   777→                />
   778→              </div>
   779→            </div>
   780→            <div className="space-y-2">
   781→              <Label htmlFor="new-password">New Password</Label>
   782→              <div className="relative">
   783→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   784→                <Input
   785→                  id="new-password"
   786→                  type="password"
   787→                  value={newPassword}
   788→                  onChange={e => setNewPassword(e.target.value)}
   789→                  placeholder="Enter new password"
   790→                  className="pl-9"
   791→                />
   792→              </div>
   793→            </div>
   794→            <div className="space-y-2">
   795→              <Label htmlFor="confirm-password">Confirm New Password</Label>
   796→              <div className="relative">
   797→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   798→                <Input
   799→                  id="confirm-password"
   800→                  type="password"
   801→                  value={confirmPassword}
   802→                  onChange={e => setConfirmPassword(e.target.value)}
   803→                  placeholder="Confirm new password"
   804→                  className="pl-9"
   805→                />
   806→              </div>
   807→            </div>
   808→            <div className="pt-2">
   809→              <Button onClick={changePassword} disabled={savingPassword} variant="outline" className="w-full sm:w-auto">
   810→                {savingPassword ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Shield className="w-4 h-4 mr-2" />}
   811→                Change Password
   812→              </Button>
   813→            </div>
   814→          </CardContent>
   815→        </Card>
   816→      </motion.div>
   817→    </div>
   818→  )

    62→function ActivityIcon({ action }: { action: string }) {
    63→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    64→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    65→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    66→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    67→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    68→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    69→  return <CircleDot className="h-4 w-4 text-slate-400" />
    70→}

   258→function getOnboardingBadge(status: string) {
   259→  switch (status) {
   260→    case 'in_progress': return 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400'
   261→    case 'submitted': return 'bg-purple-100 text-purple-700 dark:bg-purple-900/30 dark:text-purple-400'
   262→    case 'approved': return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   263→    case 'rejected': return 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-400'
   264→    default: return 'bg-slate-100 text-slate-700'
   265→  }
   266→}
   267→
   268→function getOnboardingStatusLabel(status: string) {
   269→  const map: Record<string, string> = { in_progress: 'In Progress', submitted: 'Submitted', approved: 'Approved', rejected: 'Rejected' }
   270→  return map[status] || status
   271→}

   273→export function AdminOnboarding() {
   274→  const { user } = useAuthStore()
   275→  const [loading, setLoading] = useState(true)
   276→  const [forms, setForms] = useState<OnboardingForm[]>([])
   277→  const [selectedForm, setSelectedForm] = useState<(OnboardingForm & { submissions: Onboarding[] }) | null>(null)
   278→  const [dialogOpen, setDialogOpen] = useState(false)
   279→  const [reviewing, setReviewing] = useState<string | null>(null)
   280→  const [reviewNotes, setReviewNotes] = useState<Record<string, string>>({})
   281→
   282→  useEffect(() => {
   283→    async function fetch() {
   284→      try {
   285→        const res = await fetch('/api/onboarding')
   286→        const data = await res.json()
   287→        setForms(data.forms ?? [])
   288→      } catch { toast.error('Failed to load onboarding forms') }
   289→      finally { setLoading(false) }
   290→    }
   291→    fetch()
   292→  }, [])
   293→
   294→  async function openForm(form: OnboardingForm) {
   295→    try {
   296→      const res = await fetch(`/api/onboarding/${form.id}`)
   297→      const data = await res.json()
   298→      setSelectedForm(data.form)
   299→      setDialogOpen(true)
   300→    } catch { toast.error('Failed to load form details') }
   301→  }
   302→
   303→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   304→    if (!user) return
   305→    setReviewing(submissionId)
   306→    try {
   307→      const notes = reviewNotes[submissionId] || ''
   308→      const res = await fetch(`/api/onboarding/${submissionId}`, {
   309→        method: 'PATCH',
   310→        headers: { 'Content-Type': 'application/json' },
   311→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   312→      })
   313→      if (!res.ok) { const d = await res.json(); throw new Error(d.error) }
   314→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   315→      // Refresh
   316→      if (selectedForm) {
   317→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   318→        const data = await fresh.json()
   319→        setSelectedForm(data.form)
   320→      }
   321→    } catch (e: unknown) {
   322→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   323→    }
   324→    finally { setReviewing(null) }
   325→  }
   326→
   327→  if (loading) {
   328→    return (
   329→      <div className="space-y-4">
   330→        <Skeleton className="h-10 w-48" />
   331→        <Card><CardContent className="p-4"><Skeleton className="h-64 w-full" /></CardContent></Card>
   332→      </div>
   333→    )
   334→  }
   335→
   336→  return (
   337→    <motion.div {...fadeIn} className="space-y-6">
   338→      <div>
   339→        <h2 className="text-xl font-semibold">Onboarding Management</h2>
   340→        <p className="text-sm text-muted-foreground">Review and manage client onboarding submissions</p>
   341→      </div>
   342→
   343→      <Card>
   344→        <CardContent className="p-0">
   345→          <Table>
   346→            <TableHeader>
   347→              <TableRow>
   348→                <TableHead>Form Title</TableHead>
   349→                <TableHead>Project</TableHead>
   350→                <TableHead className="text-center">Submissions</TableHead>
   351→                <TableHead>Status</TableHead>
   352→                <TableHead className="w-16"></TableHead>
   353→              </TableRow>
   354→            </TableHeader>
   355→            <TableBody>
   356→              {forms.length === 0 && (
   357→                <TableRow><TableCell colSpan={5} className="text-center text-muted-foreground py-8">No onboarding forms found</TableCell></TableRow>
   358→              )}
   359→              {forms.map((form) => (
   360→                <TableRow key={form.id} className="cursor-pointer hover:bg-muted/50" onClick={() => openForm(form)}>
   361→                  <TableCell className="font-medium">{form.title}</TableCell>
   362→                  <TableCell>
   363→                    <div className="flex items-center gap-2">
   364→                      {form.project && <div className="h-3 w-3 rounded-full" style={{ backgroundColor: form.project.color }} />}
   365→                      <span className="text-sm text-muted-foreground">{form.project?.name ?? '—'}</span>
   366→                    </div>
   367→                  </TableCell>
   368→                  <TableCell className="text-center">{form._count?.submissions ?? 0}</TableCell>
   369→                  <TableCell>
   370→                    <Badge variant="secondary" className={getOnboardingBadge(form.status)}>
   371→                      {getOnboardingStatusLabel(form.status)}
   372→                    </Badge>
   373→                  </TableCell>
   374→                  <TableCell>
   375→                    <Button variant="ghost" size="icon" className="h-8 w-8">
   376→                      <Eye className="h-4 w-4" />
   377→                    </Button>
   378→                  </TableCell>
   379→                </TableRow>
   380→              ))}
   381→            </TableBody>
   382→          </Table>
   383→        </CardContent>
   384→      </Card>
   385→
   386→      {/* Submissions Dialog */}
   387→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   388→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-hidden flex flex-col">
   389→          <DialogHeader>
   390→            <DialogTitle>{selectedForm?.title ?? 'Form'}</DialogTitle>
   391→            <DialogDescription>
   392→              {selectedForm?.project?.name ?? 'Project'} — {selectedForm?.submissions?.length ?? 0} submissions
   393→            </DialogDescription>
   394→          </DialogHeader>
   395→          <ScrollArea className="flex-1 -mx-6 px-6">
   396→            <div className="space-y-4 py-2">
   397→              {selectedForm?.submissions?.length === 0 && (
   398→                <p className="text-sm text-muted-foreground text-center py-6">No submissions yet</p>
   399→              )}
   400→              {selectedForm?.submissions?.map((sub) => (
   401→                <Card key={sub.id} className="border">
   402→                  <CardContent className="p-4 space-y-3">
   403→                    <div className="flex items-start justify-between">
   404→                      <div className="flex items-center gap-3">
   405→                        <Avatar className="h-9 w-9">
   406→                          {sub.client?.avatar ? <AvatarImage src={sub.client.avatar} /> : null}
   407→                          <AvatarFallback className="text-xs">{sub.client ? getInitials(sub.client.name) : '?'}</AvatarFallback>
   408→                        </Avatar>
   409→                        <div>
   410→                          <p className="text-sm font-medium">{sub.client?.name ?? 'Unknown'}</p>
   411→                          <p className="text-xs text-muted-foreground">{sub.client?.company ?? 'No company'}</p>
   412→                        </div>
   413→                      </div>
   414→                      <Badge variant="secondary" className={getOnboardingBadge(sub.status)}>
   415→                        {getOnboardingStatusLabel(sub.status)}
   416→                      </Badge>
   417→                    </div>
   418→
   419→                    {sub.submittedAt && (
   420→                      <p className="text-xs text-muted-foreground">Submitted {formatDate(sub.submittedAt)}</p>
   421→                    )}
   422→
   423→                    {sub.reviewNotes && (
   424→                      <div className="bg-muted/50 rounded-md p-2 text-xs">
   425→                        <span className="font-medium">Review notes:</span> {sub.reviewNotes}
   426→                        {sub.reviewer && <span className="text-muted-foreground"> — by {sub.reviewer.name}</span>}
   427→                      </div>
   428→                    )}
   429→
   430→                    {sub.status === 'submitted' && (
   431→                      <div className="space-y-2">
   432→                        <Textarea
   433→                          placeholder="Add review notes..."
   434→                          value={reviewNotes[sub.id] ?? ''}
   435→                          onChange={(e) => setReviewNotes(prev => ({ ...prev, [sub.id]: e.target.value }))}
   436→                          className="text-sm"
   437→                          rows={2}
   438→                        />
   439→                        <div className="flex gap-2">
   440→                          <Button
   441→                            size="sm"
   442→                            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   443→                            onClick={(e) => { e.stopPropagation(); handleReview(sub.id, 'approved') }}
   444→                            disabled={reviewing === sub.id}
   445→                          >
   446→                            {reviewing === sub.id ? <Skeleton className="h-4 w-16" /> : <><ThumbsUp className="h-3 w-3 mr-1" /> Approve</>}
   447→                          </Button>
   448→                          <Button

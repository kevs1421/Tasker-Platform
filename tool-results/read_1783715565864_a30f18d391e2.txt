   140→// ─── Custom tooltip for charts ──────────────────────────────────
   141→function CustomBarTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color: string }>; label?: string }) {
   142→  if (!active || !payload?.length) return null
   143→  return (
   144→    <div className="bg-popover border border-border rounded-lg px-3 py-2 shadow-lg text-sm">
   145→      <p className="font-medium mb-1">{label}</p>
   146→      {payload.map((p, i) => (
   147→        <div key={i} className="flex items-center gap-2">
   148→          <span className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: p.color }} />
   149→          <span className="text-muted-foreground">{STATUS_LABELS[p.dataKey] || p.dataKey}:</span>
   150→          <span className="font-medium">{p.value}</span>
   151→        </div>
   152→      ))}
   153→    </div>
   154→  )
   155→}

   157→export function ClientProjects() {
   158→  const { user } = useAuthStore()
   159→  const [projects, setProjects] = useState<ClientProject[]>([])
   160→  const [loading, setLoading] = useState(true)
   161→  const [expandedProjectId, setExpandedProjectId] = useState<string | null>(null)
   162→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   163→  const [taskDetail, setTaskDetail] = useState<(Task & { _count: { comments: number }; comments?: Comment[] }) | null>(null)
   164→  const [taskDetailLoading, setTaskDetailLoading] = useState(false)
   165→
   166→  // Onboarding creation state
   167→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   168→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   169→  const [onboardingTitle, setOnboardingTitle] = useState('')
   170→  const [onboardingDesc, setOnboardingDesc] = useState('')
   171→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   172→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   173→
   174→  const fetchProjects = useCallback(async () => {
   175→    try {
   176→      const res = await fetch('/api/projects')
   177→      const data = await res.json()
   178→      // Filter to active projects and those the client is a member of
   179→      const clientProjects = (data.projects || [])
   180→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   181→      setProjects(clientProjects)
   182→    } catch {
   183→      toast.error('Failed to load projects')
   184→    } finally {
   185→      setLoading(false)
   186→    }
   187→  }, [user?.id])
   188→
   189→  useEffect(() => { fetchProjects() }, [fetchProjects])
   190→
   191→  const expandProject = async (projectId: string) => {
   192→    if (expandedProjectId === projectId) {
   193→      setExpandedProjectId(null)
   194→      setExpandedTasks([])
   195→      return
   196→    }
   197→    setExpandedProjectId(projectId)
   198→    try {
   199→      const res = await fetch(`/api/projects/${projectId}`)
   200→      const data = await res.json()
   201→      setExpandedTasks(data.project?.tasks || [])
   202→    } catch {
   203→      toast.error('Failed to load project tasks')
   204→    }
   205→  }
   206→
   207→  const openTaskDetail = async (taskId: string) => {
   208→    setTaskDetailLoading(true)
   209→    setTaskDetail(null)
   210→    try {
   211→      const res = await fetch(`/api/tasks/${taskId}`)
   212→      const data = await res.json()
   213→      setTaskDetail(data.task)
   214→    } catch {
   215→      toast.error('Failed to load task details')
   216→    } finally {
   217→      setTaskDetailLoading(false)
   218→    }
   219→  }
   220→
   221→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   222→    setOnboardingProjectId(projectId)
   223→    setOnboardingTitle(`Onboarding - ${projectName}`)
   224→    setOnboardingDesc('')
   225→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   226→    setOnboardingDialogOpen(true)
   227→  }
   228→
   229→  const handleCreateOnboarding = async () => {
   230→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   231→    setCreatingOnboarding(true)
   232→    try {
   233→      const stepsJson = JSON.stringify(
   234→        onboardingSteps.map(stepId => {
   235→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   236→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   237→        })
   238→      )
   239→      const res = await fetch('/api/onboarding', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify({
   243→          title: onboardingTitle.trim(),
   244→          description: onboardingDesc.trim() || null,
   245→          stepsJson,
   246→          projectId: onboardingProjectId,
   247→          clientId: user?.id || null,
   248→          status: 'published',
   249→        }),
   250→      })
   251→      if (!res.ok) {
   252→        const d = await res.json()
   253→        throw new Error(d.error || 'Failed to create')
   254→      }
   255→      toast.success('Onboarding form created successfully')
   256→      setOnboardingDialogOpen(false)
   257→    } catch (e: unknown) {
   258→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   259→    } finally {
   260→      setCreatingOnboarding(false)
   261→    }
   262→  }
   263→
   264→  const toggleOnboardingStep = (stepId: string) => {
   265→    setOnboardingSteps(prev =>
   266→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   267→    )
   268→  }
   269→
   270→  // ─── Compute aggregate data for charts ────────────────────────
   271→  const totalTasksAcrossAll = projects.reduce((sum, p) => sum + (p._count?.tasks || 0), 0)
   272→  const totalDoneAcrossAll = projects.reduce((sum, p) => sum + (p.taskStats?.done || 0), 0)
   273→  const overallProgress = totalTasksAcrossAll > 0 ? Math.round((totalDoneAcrossAll / totalTasksAcrossAll) * 100) : 0
   274→
   275→  // Data for the bar chart: stacked tasks per project by status
   276→  const barChartData = projects.map(p => ({
   277→    name: p.name.length > 12 ? p.name.slice(0, 12) + '…' : p.name,
   278→    fullName: p.name,
   279→    todo: p.taskStats?.todo || 0,
   280→    in_progress: p.taskStats?.in_progress || 0,
   281→    in_review: p.taskStats?.in_review || 0,
   282→    done: p.taskStats?.done || 0,
   283→  }))
   284→
   285→  if (loading) return <PageSkeleton />
   286→
   287→  return (
   288→    <div className="p-4 md:p-6 space-y-6">
   289→      {/* Page Header */}
   290→      <div className="flex items-center justify-between flex-wrap gap-4">
   291→        <div className="flex items-center gap-3">
   292→          <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   293→            <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   294→          </div>
   295→          <div>
   296→            <h1 className="text-2xl font-bold">My Projects</h1>
   297→            <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   298→          </div>
   299→        </div>
   300→        {projects.length > 0 && (
   301→          <div className="flex items-center gap-3 bg-card border rounded-xl px-4 py-2.5 shadow-sm">
   302→            <TrendingUp className="w-5 h-5 text-emerald-500" />
   303→            <div>
   304→              <p className="text-xs text-muted-foreground">Overall Progress</p>
   305→              <p className="text-lg font-bold text-emerald-600 dark:text-emerald-400">{overallProgress}%</p>
   306→            </div>
   307→          </div>
   308→        )}
   309→      </div>
   310→
   311→      {projects.length === 0 ? (
   312→        <EmptyState
   313→          icon={FolderKanban}
   314→          title="No active projects"
   315→          description="You don't have any active projects yet."
   316→        />
   317→      ) : (
   318→        <>
   319→          {/* ─── Charts Section ─────────────────────────────────── */}
   320→          <div className="grid grid-cols-1 gap-4">
   321→            {/* Stacked Bar Chart — Tasks per Project by Status */}
   322→            <motion.div {...fadeSlide}>
   323→              <Card className="h-full">
   324→                <CardHeader className="pb-2">
   325→                  <CardTitle className="text-base font-semibold flex items-center gap-2">
   326→                    <BarChart3 className="w-4 h-4 text-muted-foreground" />
   327→                    Project Task Breakdown
   328→                  </CardTitle>
   329→                </CardHeader>
   330→                <CardContent className="pb-4">
   331→                  <div className="h-64">
   332→                    <ResponsiveContainer width="100%" height="100%">
   333→                      <BarChart data={barChartData} margin={{ top: 8, right: 8, left: -8, bottom: 0 }}>
   334→                        <XAxis
   335→                          dataKey="name"
   336→                          tick={{ fontSize: 12 }}
   337→                          axisLine={false}
   338→                          tickLine={false}
   339→                        />
   340→                        <YAxis
   341→                          tick={{ fontSize: 12 }}
   342→                          axisLine={false}
   343→                          tickLine={false}
   344→                          allowDecimals={false}
   345→                        />
   346→                        <Tooltip content={<CustomBarTooltip />} />
   347→                        <Bar dataKey="done" stackId="a" fill={STATUS_COLORS.done} radius={[0, 0, 0, 0]} />
   348→                        <Bar dataKey="in_review" stackId="a" fill={STATUS_COLORS.in_review} />
   349→                        <Bar dataKey="in_progress" stackId="a" fill={STATUS_COLORS.in_progress} />
   350→                        <Bar dataKey="todo" stackId="a" fill={STATUS_COLORS.todo} radius={[4, 4, 0, 0]} />
   351→                      </BarChart>
   352→                    </ResponsiveContainer>
   353→                  </div>
   354→                  {/* Legend */}
   355→                  <div className="flex flex-wrap items-center justify-center gap-4 mt-3">
   356→                    {Object.entries(STATUS_LABELS).map(([key, label]) => (
   357→                      <div key={key} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   358→                        <span className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: STATUS_COLORS[key] }} />
   359→                        {label}
   360→                      </div>
   361→                    ))}
   362→                  </div>
   363→                </CardContent>
   364→              </Card>
   365→            </motion.div>
   366→          </div>
   367→
   368→          {/* ─── Project Cards ──────────────────────────────────── */}
   369→          <div className="space-y-4">
   370→            {projects.map((project, i) => {
   371→              const projectTasks = expandedProjectId === project.id ? expandedTasks : []
   372→              const doneTasks = project.taskStats?.done || 0
   373→              const totalT = project._count?.tasks || 0
   374→              const pct = project.progress ?? 0
   375→              const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   376→              const isExpanded = expandedProjectId === project.id
   377→
   378→              // Compute progress bar gradient colors
   379→              const barColor = pct >= 75 ? '#10b981' : pct >= 40 ? '#f59e0b' : '#ef4444'
   380→
   381→              return (
   382→                <motion.div
   383→                  key={project.id}
   384→                  initial={{ opacity: 0, y: 12 }}
   385→                  animate={{ opacity: 1, y: 0 }}
   386→                  transition={{ delay: i * 0.05 }}
   387→                >
   388→                  <Card
   389→                    className="cursor-pointer hover:shadow-md transition-all border-l-4"
   390→                    style={{ borderLeftColor: project.color }}
   391→                    onClick={() => expandProject(project.id)}
   392→                  >
   393→                    <CardContent className="p-4 md:p-5">
   394→                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   395→                        <div className="min-w-0 flex-1">
   396→                          <div className="flex items-center gap-2 mb-1">
   397→                            <h3 className="font-semibold text-base truncate">{project.name}</h3>
   398→                            <PriorityBadge priority={project.priority} />
   399→                          </div>
   400→                          <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   401→                        </div>
   402→                        <div className="flex items-center gap-3 shrink-0 text-sm">
   403→                          {project.dueDate && (
   404→                            <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   405→                              <Calendar className="w-3.5 h-3.5" />
   406→                              {formatDate(project.dueDate)}
   407→                            </span>
   408→                          )}
   409→                          <span className="text-muted-foreground flex items-center gap-1">
   410→                            <Users className="w-3.5 h-3.5" />
   411→                            {project._count?.members || 0}
   412→                          </span>
   413→                          <Button
   414→                            size="sm"
   415→                            variant="outline"
   416→                            className="h-7 gap-1.5 text-xs"
   417→                            onClick={(e) => { e.stopPropagation(); openOnboardingDialog(project.id, project.name) }}
   418→                          >
   419→                            <ClipboardCheck className="w-3.5 h-3.5" />
   420→                            Onboarding
   421→                          </Button>
   422→                        </div>
   423→                      </div>
   424→
   425→                      {/* Progress bar with percentage */}
   426→                      <div className="mt-3 space-y-1">
   427→                        <div className="flex items-center justify-between text-xs">
   428→                          <span className="text-muted-foreground">{doneTasks}/{totalT} tasks completed</span>
   429→                          <span className="font-semibold" style={{ color: barColor }}>{pct}%</span>
   430→                        </div>
   431→                        <div className="relative">
   432→                          <Progress value={pct} className="h-2.5" />
   433→                          <div
   434→                            className="absolute top-0 left-0 h-2.5 rounded-full transition-all duration-500"
   435→                            style={{
   436→                              width: `${pct}%`,
   437→                              background: `linear-gradient(90deg, ${barColor}88, ${barColor})`,
   438→                            }}
   439→                          />
   440→                        </div>
   441→                      </div>
   442→                    </CardContent>
   443→                  </Card>
   444→
   445→                  {/* Expanded tasks table */}
   446→                  <AnimatePresence>
   447→                    {isExpanded && (
   448→                      <motion.div
   449→                        initial={{ opacity: 0, height: 0 }}
   450→                        animate={{ opacity: 1, height: 'auto' }}
   451→                        exit={{ opacity: 0, height: 0 }}
   452→                        transition={{ duration: 0.2 }}
   453→                        className="overflow-hidden"
   454→                      >
   455→                        <Card className="mt-2 rounded-t-none">
   456→                          <CardContent className="p-0">
   457→                            {projectTasks.length === 0 ? (
   458→                              <div className="py-8 text-center text-sm text-muted-foreground">
   459→                                No tasks in this project yet.
   460→                              </div>
   461→                            ) : (
   462→                              <div className="overflow-x-auto">
   463→                                <table className="w-full text-sm">
   464→                                  <thead>
   465→                                    <tr className="border-b bg-muted/50">
   466→                                      <th className="text-left p-3 font-medium text-muted-foreground">Task</th>
   467→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden md:table-cell">Status</th>
   468→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden sm:table-cell">Priority</th>
   469→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Due Date</th>
   470→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Assignees</th>
   471→                                    </tr>
   472→                                  </thead>
   473→                                  <tbody>
   474→                                    {projectTasks.map(task => (
   475→                                      <tr
   476→                                        key={task.id}
   477→                                        className="border-b last:border-0 hover:bg-muted/30 cursor-pointer transition-colors"
   478→                                        onClick={(e) => { e.stopPropagation(); openTaskDetail(task.id) }}
   479→                                      >
   480→                                        <td className="p-3">
   481→                                          <div className="font-medium">{task.title}</div>
   482→                                          {task.description && (
   483→                                            <div className="text-xs text-muted-foreground line-clamp-1 mt-0.5">{task.description}</div>
   484→                                          )}
   485→                                        </td>
   486→                                        <td className="p-3 hidden md:table-cell"><StatusBadge status={task.status} /></td>
   487→                                        <td className="p-3 hidden sm:table-cell"><PriorityBadge priority={task.priority} /></td>
   488→                                        <td className="p-3 hidden lg:table-cell text-muted-foreground text-xs">
   489→                                          {task.dueDate ? formatDate(task.dueDate) : '—'}
   490→                                        </td>
   491→                                        <td className="p-3 hidden lg:table-cell">

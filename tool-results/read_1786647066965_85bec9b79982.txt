    46→function formatElapsed(ms: number): string {
    47→  const totalSeconds = Math.floor(ms / 1000)
    48→  const h = Math.floor(totalSeconds / 3600)
    49→  const m = Math.floor((totalSeconds % 3600) / 60)
    50→  const s = totalSeconds % 60
    51→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    52→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    53→}
    54→
    55→function formatCountdown(ms: number): string {
    56→  if (ms <= 0) return '0:00'
    57→  const totalSeconds = Math.floor(ms / 1000)
    58→  const m = Math.floor(totalSeconds / 60)
    59→  const s = totalSeconds % 60
    60→  return `${m}:${String(s).padStart(2, '0')}`
    61→}

   893→function OfflineBanner() {
   894→  const [isOffline, setIsOffline] = useState(() => {
   895→    if (typeof navigator === 'undefined') return false
   896→    return !navigator.onLine
   897→  })
   898→
   899→  useEffect(() => {
   900→    const handleOnline = () => setIsOffline(false)
   901→    const handleOffline = () => setIsOffline(true)
   902→    window.addEventListener('online', handleOnline)
   903→    window.addEventListener('offline', handleOffline)
   904→    return () => {
   905→      window.removeEventListener('online', handleOnline)
   906→      window.removeEventListener('offline', handleOffline)
   907→    }
   908→  }, [])
   909→
   910→  if (!isOffline) return null
   911→
   912→  return (
   913→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   914→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   915→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   916→    </div>
   917→  )
   918→}

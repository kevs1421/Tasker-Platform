    65→    65→        {getInitials(user.name)}

   143→   143→  }
   144→   144→
   145→   145→  const filteredUsers = users.filter(
   146→   146→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   147→   147→  )

   163→   163→
   164→   164→          <div>
   165→   165→            <div className="relative mb-2">
   166→   166→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />

   168→   168→            </div>
   169→   169→            <ScrollArea className="max-h-48 rounded-md border">
   170→   170→              {loading ? (
   171→   171→                <div className="p-3 space-y-2">
   172→   172→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   173→   173→                </div>
   174→   174→              ) : filteredUsers.length === 0 ? (
   175→   175→                <p className="text-sm text-muted-foreground text-center py-4">No users found</p>
   176→   176→              ) : (
   177→   177→                <div className="p-1">
   178→   178→                  {filteredUsers.map((u) => (
   179→   179→                    <button
   180→   180→                      key={u.id}
   181→   181→                      type="button"
   182→   182→                      className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   183→   183→                      onClick={() => toggleUser(u.id)}
   184→   184→                    >
   185→   185→                      <Checkbox checked={selectedIds.includes(u.id)} />
   186→   186→                      <UserAvatar user={u} size="sm" />
   187→   187→                      <div className="flex-1 min-w-0">
   188→   188→                        <p className="text-sm font-medium truncate">{u.name}</p>
   189→   189→                        <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   190→   190→                      </div>
   191→   191→                      <Badge variant="secondary" className={cn('text-[10px]', getRoleBadgeClasses(u.role))}>{u.role}</Badge>
   192→   192→                    </button>
   193→   193→                  ))}
   194→   194→                </div>
   195→   195→              )}
   196→   196→            </ScrollArea>
   197→   197→            {selectedIds.length > 0 && (
   198→   198→              <p className="text-xs text-muted-foreground mt-2">{selectedIds.length} member{selectedIds.length > 1 ? 's' : ''} selected</p>
   199→   199→            )}
   200→   200→          </div>
   201→   201→        </div>
   202→   202→
   203→   203→        <DialogFooter>
   204→   204→          <Button variant="outline" type="button" onClick={() => onOpenChange(false)}>Cancel</Button>
   205→   205→          <Button
   206→   206→            type="button"
   207→   207→            onClick={handleCreate} disabled={!name.trim() || selectedIds.length === 0 || creating}
   208→   208→            className="bg-violet-600 hover:bg-violet-700 text-white"
   209→   209→          >
   210→   210→            {creating && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   211→   211→            Create Group
   212→   212→          </Button>
   213→   213→        </DialogFooter>
   214→   214→      </DialogContent>
   215→   215→    </Dialog>
   216→   216→  )
   217→   217→}

   397→   397→          <DialogTitle className="flex items-center gap-2">

   423→   423→            </div>

   425→   425→            {showAddUser && (
   426→   426→              <div className="mb-3 space-y-2">
   427→   427→                <div className="relative">
   428→   428→                  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />

    63→function ActivityIcon({ action }: { action: string }) {
    64→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    65→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    66→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    67→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    68→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    69→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    70→  return <CircleDot className="h-4 w-4 text-slate-400" />
    71→}

   370→  async function handleCreateOnboarding() {
   371→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   372→    setCreating(true)
   373→    try {
   374→      const stepsJson = JSON.stringify(
   375→        createForm.selectedSteps.map(stepId => {
   376→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   377→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   378→        })
   379→      )
   380→      const res = await fetch('/api/onboarding', {
   381→        method: 'POST',
   382→        headers: { 'Content-Type': 'application/json' },
   383→        body: JSON.stringify({
   384→          title: createForm.title.trim(),
   385→          description: createForm.description.trim() || null,
   386→          stepsJson,
   387→          projectId: createForm.projectId,
   388→          clientId: createForm.clientId || null,
   389→          status: 'published',
   390→        }),
   391→      })
   392→      if (!res.ok) {
   393→        const d = await res.json()
   394→        throw new Error(d.error || 'Failed to create')
   395→      }
   396→      toast.success('Onboarding form created successfully')

   398→      setCreateForm({ title: '', description: '', projectId: '', clientId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })

   407→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   408→    if (!user) return
   409→    setReviewing(submissionId)
   410→    try {
   411→      const notes = reviewNotes[submissionId] || ''
   412→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   413→        method: 'PATCH',
   414→        headers: { 'Content-Type': 'application/json' },
   415→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   416→      })
   417→      if (!patchRes.ok) {
   418→        const d = await patchRes.json()
   419→        throw new Error(d.error)
   420→      }
   421→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   422→      // Refresh
   423→      if (selectedForm) {
   424→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   425→        if (fresh.ok) {
   426→          const data = await fresh.json()
   427→          setSelectedForm(data.form)
   428→        }
   429→      }
   430→      fetchForms()
   431→    } catch (e: unknown) {
   432→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   433→    } finally {
   434→      setReviewing(null)
   435→    }
   436→  }

    44→function formatElapsed(ms: number): string {
    45→  const totalSeconds = Math.floor(ms / 1000)
    46→  const h = Math.floor(totalSeconds / 3600)
    47→  const m = Math.floor((totalSeconds % 3600) / 60)
    48→  const s = totalSeconds % 60
    49→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    50→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    51→}

    53→function TimeInOutWidget({ userId }: { userId: string }) {
    54→  const [activeEntry, setActiveEntry] = useState<ActiveTimeEntry | null>(null)
    55→  const [elapsedMs, setElapsedMs] = useState(0)
    56→  const [loading, setLoading] = useState(true)
    57→  const [clockingIn, setClockingIn] = useState(false)
    58→  const [clockingOut, setClockingOut] = useState(false)
    59→  const [dialogOpen, setDialogOpen] = useState(false)
    60→
    61→  // Time In dialog state
    62→  const [projects, setProjects] = useState<Project[]>([])
    63→  const [selectedProjectId, setSelectedProjectId] = useState<string>('')
    64→  const [tasks, setTasks] = useState<Task[]>([])
    65→  const [selectedTaskId, setSelectedTaskId] = useState<string>('')
    66→  const [description, setDescription] = useState('')
    67→  const [loadingProjects, setLoadingProjects] = useState(false)
    68→  const [loadingTasks, setLoadingTasks] = useState(false)
    69→
    70→  // Auto-screenshot state
    71→  const [autoScreenshot, setAutoScreenshot] = useState(true)
    72→  const [screenshotDelay, setScreenshotDelay] = useState('0')
    73→  const [capturingScreenshot, setCapturingScreenshot] = useState(false)
    74→  const screenshotTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    75→
    76→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    77→  const startTimeRef = useRef<number>(0)
    78→
    79→  // Fetch active timer on mount
    80→  const fetchActiveTimer = useCallback(async () => {
    81→    try {
    82→      const res = await fetch(`/api/time-entries?userId=${userId}&active=true`)
    83→      if (res.ok) {
    84→        const data = await res.json()
    85→        const entries = data.timeEntries || []
    86→        const entry: ActiveTimeEntry | null = entries.find((e: ActiveTimeEntry) => e.timeIn) || null
    87→        setActiveEntry(entry)
    88→        if (entry) {
    89→          startTimeRef.current = new Date(entry.timeIn).getTime()
    90→          setElapsedMs(Date.now() - startTimeRef.current)
    91→        }
    92→      }
    93→    } catch {
    94→      // silent
    95→    } finally {
    96→      setLoading(false)
    97→    }
    98→  }, [userId])
    99→
   100→  useEffect(() => {
   101→    fetchActiveTimer()
   102→  }, [fetchActiveTimer])
   103→
   104→  // 1-second interval for elapsed display
   105→  useEffect(() => {
   106→    if (activeEntry) {
   107→      intervalRef.current = setInterval(() => {
   108→        setElapsedMs(Date.now() - startTimeRef.current)
   109→      }, 1000)
   110→    } else {
   111→      if (intervalRef.current) {
   112→        clearInterval(intervalRef.current)
   113→        intervalRef.current = null
   114→      }
   115→    }
   116→    return () => {
   117→      if (intervalRef.current) {
   118→        clearInterval(intervalRef.current)
   119→        intervalRef.current = null
   120→      }
   121→    }
   122→  }, [activeEntry])
   123→
   124→  // Open time-in dialog and fetch projects
   125→  const handleOpenDialog = async () => {
   126→    setDialogOpen(true)
   127→    setSelectedProjectId('')
   128→    setSelectedTaskId('')
   129→    setDescription('')
   130→    setTasks([])
   131→    setLoadingProjects(true)
   132→    try {
   133→      const res = await fetch('/api/projects')
   134→      if (res.ok) {
   135→        const data = await res.json()
   136→        setProjects(data.projects ?? [])
   137→      }
   138→    } catch {
   139→      // silent
   140→    } finally {
   141→      setLoadingProjects(false)
   142→    }
   143→  }
   144→
   145→  // Fetch tasks when project changes
   146→  const handleProjectChange = async (projectId: string) => {
   147→    setSelectedProjectId(projectId)
   148→    setSelectedTaskId('')
   149→    setTasks([])
   150→    if (!projectId) return
   151→    setLoadingTasks(true)
   152→    try {
   153→      const res = await fetch(`/api/projects/${projectId}`)
   154→      if (res.ok) {
   155→        const data = await res.json()
   156→        setTasks(data.project?.tasks ?? [])
   157→      }
   158→    } catch {
   159→      // silent
   160→    } finally {
   161→      setLoadingTasks(false)
   162→    }
   163→  }
   164→
   165→  // Capture auto-screenshot after optional delay
   166→  const captureAutoScreenshot = useCallback(async (projId: string, taskId: string, desc: string, delay: number, userRole: string) => {
   167→    if (delay > 0) {
   168→      setCapturingScreenshot(true)
   169→      // Show countdown toast
   170→      toast.info(`Screenshot will be captured in ${delay} second${delay > 1 ? 's' : ''}...`, { duration: (delay * 1000) - 500 })
   171→      await new Promise(resolve => {
   172→        screenshotTimerRef.current = setTimeout(resolve, delay * 1000)
   173→      })
   174→    }
   175→    try {
   176→      const res = await fetch('/api/screenshots/auto', {
   177→        method: 'POST',
   178→        headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({
   180→          userId,
   181→          userRole,
   182→          projectId: projId || null,
   183→          taskId: taskId || null,
   184→          description: desc || null,
   185→          delaySeconds: delay,
   186→        }),
   187→      })
   188→      if (res.ok) {
   189→        toast.success('Screenshot captured', { description: 'Proof-of-work screenshot saved' })
   190→      } else {
   191→        // Non-blocking — don't fail the whole clock-in
   192→        toast.error('Auto-screenshot failed', { description: 'Time-in was still recorded' })
   193→      }
   194→    } catch {
   195→      // Silent fail — screenshot is optional
   196→    } finally {
   197→      setCapturingScreenshot(false)
   198→    }
   199→  }, [userId])
   200→
   201→  // Clock in
   202→  const handleClockIn = async () => {
   203→    const user = useAuthStore.getState().user
   204→    setClockingIn(true)
   205→    try {
   206→      const body: Record<string, unknown> = { action: 'time_in', userId }
   207→      if (selectedProjectId) body.projectId = selectedProjectId
   208→      if (selectedTaskId) body.taskId = selectedTaskId
   209→      if (description.trim()) body.description = description.trim()
   210→
   211→      const res = await fetch('/api/time-entries', {
   212→        method: 'POST',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify(body),
   215→      })
   216→      if (!res.ok) {
   217→        const err = await res.json().catch(() => ({}))
   218→        throw new Error(err.error || 'Failed to clock in')
   219→      }
   220→      const data = await res.json()
   221→      const entry = data.timeEntry
   222→      if (entry) {
   223→        setActiveEntry(entry)
   224→        startTimeRef.current = new Date(entry.timeIn).getTime()
   225→        setElapsedMs(Date.now() - startTimeRef.current)
   226→      }
   227→      setDialogOpen(false)
   228→      toast.success('Clocked in successfully')
   229→
   230→      // Auto-screenshot (fire-and-forget, after dialog closes)
   231→      if (autoScreenshot && user && user.role !== 'client') {
   232→        const delay = parseInt(screenshotDelay) || 0
   233→        captureAutoScreenshot(selectedProjectId, selectedTaskId, description, delay, user.role)
   234→      }
   235→    } catch (err) {
   236→      toast.error(err instanceof Error ? err.message : 'Failed to clock in')
   237→    } finally {
   238→      setClockingIn(false)
   239→    }
   240→  }
   241→
   242→  // Clock out
   243→  const handleClockOut = async () => {
   244→    if (!activeEntry) return
   245→    setClockingOut(true)
   246→    try {
   247→      const res = await fetch(`/api/time-entries/${activeEntry.id}`, {
   248→        method: 'PATCH',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify({ action: 'time_out' }),
   251→      })
   252→      if (!res.ok) {
   253→        const err = await res.json().catch(() => ({}))
   254→        throw new Error(err.error || 'Failed to clock out')
   255→      }
   256→      setActiveEntry(null)
   257→      setElapsedMs(0)
   258→      toast.success('Clocked out successfully')
   259→    } catch (err) {
   260→      toast.error(err instanceof Error ? err.message : 'Failed to clock out')
   261→    } finally {
   262→      setClockingOut(false)
   263→    }
   264→  }
   265→
   266→  if (loading) {
   267→    return (
   268→      <div className="mb-3 rounded-lg border bg-muted/30 px-3 py-2 flex items-center justify-center gap-2">
   269→        <Loader2 className="w-3.5 h-3.5 animate-spin text-muted-foreground" />
   270→        <span className="text-xs text-muted-foreground">Loading timer…</span>
   271→      </div>
   272→    )
   273→  }
   274→
   275→  if (activeEntry) {
   276→    return (
   277→      <>
   278→        {capturingScreenshot && (
   279→          <div className="mb-2 rounded-lg border border-primary/30 bg-primary/5 dark:bg-primary/10 px-3 py-1.5 flex items-center gap-2 animate-pulse">
   280→            <Camera className="w-3.5 h-3.5 text-primary shrink-0" />
   281→            <span className="text-[11px] text-primary font-medium">Capturing screenshot in {screenshotDelay}s...</span>
   282→            <Loader2 className="w-3 h-3 animate-spin text-primary ml-auto" />
   283→          </div>
   284→        )}
   285→        <div className="mb-3 rounded-lg border border-red-200 bg-red-50 dark:border-red-900/50 dark:bg-red-950/30 px-3 py-2 flex items-center gap-2">
   286→          <Timer className="w-4 h-4 text-red-600 dark:text-red-400 shrink-0" />
   287→          <div className="flex-1 min-w-0">
   288→            <div className="text-[10px] text-red-600 dark:text-red-400 font-medium leading-none mb-0.5">
   289→              {activeEntry.project?.name || 'No Project'}
   290→            </div>
   291→            <div className="text-xs font-mono font-semibold text-red-700 dark:text-red-300 tabular-nums">
   292→              {formatElapsed(elapsedMs)}
   293→            </div>
   294→          </div>
   295→          <Button
   296→            size="sm"
   297→            variant="destructive"
   298→            className="h-7 px-2 text-[11px] shrink-0"
   299→            onClick={handleClockOut}
   300→            disabled={clockingOut}
   301→          >
   302→            {clockingOut ? <Loader2 className="w-3 h-3 animate-spin" /> : <Square className="w-3 h-3 mr-1" />}
   303→            Out
   304→          </Button>
   305→        </div>
   306→      </>
   307→    )
   308→  }
   309→
   310→  return (
   311→    <>
   312→      <button
   313→        onClick={handleOpenDialog}
   314→        className="mb-3 w-full rounded-lg border border-green-200 bg-green-50 dark:border-green-900/50 dark:bg-green-950/30 px-3 py-2 flex items-center justify-center gap-2 hover:bg-green-100 dark:hover:bg-green-950/50 transition-colors cursor-pointer"
   315→      >
   316→        <Play className="w-3.5 h-3.5 text-green-600 dark:text-green-400" />
   317→        <Clock className="w-3.5 h-3.5 text-green-600 dark:text-green-400" />
   318→        <span className="text-xs font-medium text-green-700 dark:text-green-300">Time In</span>
   319→      </button>
   320→
   321→      <Dialog open={dialogOpen} onOpenChange={(open) => {
   322→        setDialogOpen(open)
   323→        // Cleanup timer if dialog is closed
   324→        if (!open && screenshotTimerRef.current) {
   325→          clearTimeout(screenshotTimerRef.current)
   326→          screenshotTimerRef.current = null
   327→          setCapturingScreenshot(false)
   328→        }
   329→      }}>
   330→        <DialogContent className="sm:max-w-md">
   331→          <DialogHeader>
   332→            <DialogTitle>Clock In</DialogTitle>
   333→            <DialogDescription>Optionally select a project, task, and add a description before clocking in.</DialogDescription>
   334→          </DialogHeader>
   335→          <div className="space-y-4 py-2">
   336→            <div className="space-y-2">
   337→              <Label className="text-sm">Project (optional)</Label>
   338→              <Select value={selectedProjectId} onValueChange={handleProjectChange} disabled={loadingProjects}>
   339→                <SelectTrigger className="h-9">
   340→                  <SelectValue placeholder={loadingProjects ? 'Loading…' : 'Select project'} />
   341→                </SelectTrigger>
   342→                <SelectContent>
   343→                  {projects.map((p) => (
   344→                    <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   345→                  ))}
   346→                </SelectContent>
   347→              </Select>
   348→            </div>
   349→            {selectedProjectId && (
   350→              <div className="space-y-2">
   351→                <Label className="text-sm">Task (optional)</Label>
   352→                <Select value={selectedTaskId} onValueChange={setSelectedTaskId} disabled={loadingTasks}>
   353→                  <SelectTrigger className="h-9">
   354→                    <SelectValue placeholder={loadingTasks ? 'Loading…' : 'Select task'} />
   355→                  </SelectTrigger>
   356→                  <SelectContent>
   357→                    {tasks.map((t) => (
   358→                      <SelectItem key={t.id} value={t.id}>{t.title}</SelectItem>
   359→                    ))}
   360→                    </SelectContent>
   361→                </Select>
   362→              </div>
   363→            )}
   364→            <div className="space-y-2">
   365→              <Label className="text-sm">Description (optional)</Label>
   366→              <Input
   367→                placeholder="What are you working on?"
   368→                value={description}
   369→                onChange={(e) => setDescription(e.target.value)}
   370→                className="h-9"
   371→              />
   372→            </div>
   373→
   374→            {/* Auto-screenshot section */}
   375→            <div className="rounded-lg border bg-muted/30 p-3 space-y-3">
   376→              <div className="flex items-center justify-between">
   377→                <div className="flex items-center gap-2">
   378→                  {autoScreenshot
   379→                    ? <Camera className="w-4 h-4 text-primary" />
   380→                    : <CameraOff className="w-4 h-4 text-muted-foreground" />}
   381→                  <Label className="text-sm font-medium cursor-pointer" onClick={() => setAutoScreenshot(!autoScreenshot)}>
   382→                    Auto Screenshot
   383→                  </Label>
   384→                </div>
   385→                <Checkbox
   386→                  checked={autoScreenshot}
   387→                  onCheckedChange={(checked) => setAutoScreenshot(checked === true)}
   388→                  className="data-[state=checked]:bg-primary data-[state=checked]:border-primary"
   389→                />
   390→              </div>
   391→              {autoScreenshot && (
   392→                <div className="flex items-center gap-2 pl-6">
   393→                  <Hourglass className="w-3.5 h-3.5 text-muted-foreground shrink-0" />
   394→                  <Label className="text-xs text-muted-foreground whitespace-nowrap">Capture delay:</Label>
   395→                  <div className="flex items-center gap-1.5">
   396→                    <Input
   397→                      type="number"
   398→                      min="0"
   399→                      max="300"
   400→                      step="5"

   403→                      className="h-8 w-20 text-center text-sm"

   405→                    <span className="text-xs text-muted-foreground">sec</span>
   406→                  </div>
   407→                  <Select
   408→                    value={
   409→                      screenshotDelay === '0' ? 'instant' :
   410→                      screenshotDelay === '5' ? '5s' :
   411→                      screenshotDelay === '10' ? '10s' :
   412→                      screenshotDelay === '30' ? '30s' :
   413→                      screenshotDelay === '60' ? '60s' : 'custom'
   414→                    }
   415→                    onValueChange={(v) => {
   416→                      if (v === 'custom') return
   417→                      setScreenshotDelay(v === 'instant' ? '0' : v.replace('s', ''))
   418→                    }}
   419→                  >
   420→                    <SelectTrigger className="h-8 w-24 text-xs">
   421→                      <SelectValue />
   422→                    </SelectTrigger>
   423→                    <SelectContent>
   424→                      <SelectItem value="instant">Instant</SelectItem>
   425→                      <SelectItem value="5s">5 sec</SelectItem>
   426→                      <SelectItem value="10s">10 sec</SelectItem>
   427→                      <SelectItem value="30s">30 sec</SelectItem>
   428→                      <SelectItem value="60s">1 min</SelectItem>
   429→                      <SelectItem value="custom">Custom</SelectItem>
   430→                    </SelectContent>
   431→                  </Select>
   432→                </div>
   433→              )}
   434→              {autoScreenshot && (
   435→                <p className="text-[11px] text-muted-foreground pl-6">
   436→                  A proof-of-work screenshot will be captured{parseInt(screenshotDelay) > 0 ? ` after ${screenshotDelay}s` : ' instantly'} when you clock in.
   437→                </p>
   438→              )}
   439→            </div>
   440→          </div>
   441→          <DialogFooter>
   442→            <Button variant="outline" onClick={() => setDialogOpen(false)} className="h-9">Cancel</Button>
   443→            <Button onClick={handleClockIn} disabled={clockingIn || capturingScreenshot} className="h-9 bg-green-600 hover:bg-green-700 text-white">
   444→              {(clockingIn || capturingScreenshot) && <Loader2 className="w-3.5 h-3.5 mr-1.5 animate-spin" />}
   445→              <Play className="w-3.5 h-3.5 mr-1.5" />
   446→              {capturingScreenshot ? `Capturing in ${screenshotDelay}s...` : 'Clock In'}
   447→            </Button>
   448→          </DialogFooter>
   449→        </DialogContent>
   450→      </Dialog>
   451→    </>
   452→  )
   453→}

   660→function OfflineBanner() {
   661→  const [isOffline, setIsOffline] = useState(() => {
   662→    if (typeof navigator === 'undefined') return false
   663→    return !navigator.onLine
   664→  })
   665→
   666→  useEffect(() => {
   667→    const handleOnline = () => setIsOffline(false)
   668→    const handleOffline = () => setIsOffline(true)
   669→    window.addEventListener('online', handleOnline)
   670→    window.addEventListener('offline', handleOffline)
   671→    return () => {
   672→      window.removeEventListener('online', handleOnline)
   673→      window.removeEventListener('offline', handleOffline)
   674→    }
   675→  }, [])
   676→
   677→  if (!isOffline) return null
   678→
   679→  return (
   680→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   681→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   682→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   683→    </div>
   684→  )
   685→}

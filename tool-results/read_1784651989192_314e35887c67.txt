    98→    98→const ONBOARDING_STEPS = [
    99→    99→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   100→   100→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },

   102→   102→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   103→   103→]
   104→   104→
   105→   105→export function ClientProjects() {
   106→   106→  const { user } = useAuthStore()
   107→   107→  const [projects, setProjects] = useState<ClientProject[]>([])
   108→   108→  const [loading, setLoading] = useState(true)
   109→   109→
   110→   110→  // Upload files state
   111→   111→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   112→   112→  const [uploading, setUploading] = useState(false)
   113→   113→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   114→   114→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   115→   115→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   116→   116→  const fileInputRef = useRef<HTMLInputElement>(null)
   117→   117→
   118→   118→  // Onboarding creation state
   119→   119→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   120→   120→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   121→   121→  const [onboardingTitle, setOnboardingTitle] = useState('')
   122→   122→  const [onboardingDesc, setOnboardingDesc] = useState('')
   123→   123→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   124→   124→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   125→   125→
   126→   126→  const fetchProjects = useCallback(async () => {
   127→   127→    if (!user?.id) return
   128→   128→    try {
   129→   129→      // Use the client-only endpoint that queries through ClientProject table
   130→   130→      // This ensures only projects explicitly assigned to this client are returned
   131→   131→      const res = await fetch(`/api/client/projects?clientId=${user.id}`)
   132→   132→      const data = await res.json()
   133→   133→      setProjects(data.projects || [])
   134→   134→    } catch {
   135→   135→      toast.error('Failed to load projects')
   136→   136→    } finally {
   137→   137→      setLoading(false)
   138→   138→    }
   139→   139→  }, [user?.id])
   140→   140→
   141→   141→  // Fetch uploaded files from localStorage
   142→   142→  const fetchUploadedFiles = useCallback(() => {
   143→   143→    if (!user) return
   144→   144→    try {
   145→   145→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   146→   146→      if (stored) setUploadedFiles(JSON.parse(stored))
   147→   147→    } catch { /* ignore */ }
   148→   148→  }, [user])
   149→   149→
   150→   150→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   151→   151→    if (!user) return
   152→   152→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   153→   153→  }, [user])
   154→   154→
   155→   155→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   156→   156→
   157→   157→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   158→   158→  const handleFileUpload = async (files: FileList | null) => {
   159→   159→    if (!files || files.length === 0) return
   160→   160→    if (!uploadDialogProjectId) {
   161→   161→      toast.error('No project selected')
   162→   162→      return
   163→   163→    }
   164→   164→    setUploading(true)
   165→   165→    const newFiles: typeof uploadedFiles = []
   166→   166→    for (let i = 0; i < files.length; i++) {
   167→   167→      const file = files[i]
   168→   168→      try {
   169→   169→        const formData = new FormData()
   170→   170→        formData.append('file', file)
   171→   171→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   172→   172→        if (!res.ok) {
   173→   173→          const d = await res.json()
   174→   174→          throw new Error(d.error || 'Upload failed')
   175→   175→        }
   176→   176→        const data = await res.json()
   177→   177→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   178→   178→      } catch (e: unknown) {
   179→   179→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   180→   180→      }
   181→   181→    }
   182→   182→    if (newFiles.length > 0) {
   183→   183→      const updated = [...uploadedFiles, ...newFiles]
   184→   184→      setUploadedFiles(updated)
   185→   185→      saveUploadedFiles(updated)
   186→   186→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   187→   187→    }
   188→   188→    setUploading(false)
   189→   189→    if (fileInputRef.current) fileInputRef.current.value = ''
   190→   190→  }
   191→   191→
   192→   192→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   193→   193→    e.preventDefault()
   194→   194→    setUploadDragOver(null)
   195→   195→    setUploadDialogProjectId(projectId)
   196→   196→    handleFileUpload(e.dataTransfer.files)
   197→   197→  }
   198→   198→
   199→   199→  const openUploadDialog = (projectId: string) => {
   200→   200→    setUploadDialogProjectId(projectId)
   201→   201→    setUploadDialogOpen(true)
   202→   202→  }
   203→   203→
   204→   204→  const deleteUploadedFile = (idx: number) => {
   205→   205→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   206→   206→    setUploadedFiles(updated)
   207→   207→    saveUploadedFiles(updated)
   208→   208→    toast.success('File removed')
   209→   209→  }
   210→   210→
   211→   211→  const getFileIcon = (type: string) => {
   212→   212→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   213→   213→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   214→   214→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   215→   215→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   216→   216→    return <File className="w-4 h-4 text-muted-foreground" />
   217→   217→  }
   218→   218→
   219→   219→  const formatFileSize = (bytes: number) => {
   220→   220→    if (bytes < 1024) return `${bytes} B`
   221→   221→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   222→   222→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   223→   223→  }
   224→   224→
   225→   225→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   226→   226→    setOnboardingProjectId(projectId)
   227→   227→    setOnboardingTitle(`Onboarding - ${projectName}`)
   228→   228→    setOnboardingDesc('')
   229→   229→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   230→   230→    setOnboardingDialogOpen(true)
   231→   231→  }
   232→   232→
   233→   233→  const handleCreateOnboarding = async () => {
   234→   234→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   235→   235→    setCreatingOnboarding(true)
   236→   236→    try {
   237→   237→      const stepsJson = JSON.stringify(
   238→   238→        onboardingSteps.map(stepId => {
   239→   239→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   240→   240→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   241→   241→        })
   242→   242→      )
   243→   243→      const res = await fetch('/api/onboarding', {
   244→   244→        method: 'POST',
   245→   245→        headers: { 'Content-Type': 'application/json' },
   246→   246→        body: JSON.stringify({
   247→   247→          title: onboardingTitle.trim(),
   248→   248→          description: onboardingDesc.trim() || null,
   249→   249→          stepsJson,
   250→   250→          projectId: onboardingProjectId,
   251→   251→          clientId: user?.id || null,
   252→   252→          status: 'published',
   253→   253→        }),
   254→   254→      })
   255→   255→      if (!res.ok) {
   256→   256→        const d = await res.json()
   257→   257→        throw new Error(d.error || 'Failed to create')
   258→   258→      }
   259→   259→      toast.success('Onboarding form created successfully')
   260→   260→      setOnboardingDialogOpen(false)
   261→   261→    } catch (e: unknown) {
   262→   262→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   263→   263→    } finally {
   264→   264→      setCreatingOnboarding(false)
   265→   265→    }
   266→   266→  }
   267→   267→
   268→   268→  const toggleOnboardingStep = (stepId: string) => {
   269→   269→    setOnboardingSteps(prev =>
   270→   270→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   271→   271→    )
   272→   272→  }
   273→   273→
   274→   274→  if (loading) return <PageSkeleton />
   275→   275→
   276→   276→  return (
   277→   277→    <div className="p-4 md:p-6 space-y-6">
   278→   278→      {/* Page Header */}
   279→   279→      <div className="flex items-center gap-3">
   280→   280→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   281→   281→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   282→   282→        </div>
   283→   283→        <div>
   284→   284→          <h1 className="text-2xl font-bold">My Projects</h1>
   285→   285→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   286→   286→        </div>
   287→   287→      </div>
   288→   288→
   289→   289→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   290→   290→      <input
   291→   291→        ref={fileInputRef}
   292→   292→        type="file"
   293→   293→        multiple
   294→   294→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   295→   295→        className="hidden"
   296→   296→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   297→   297→      />
   298→   298→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   299→   299→        <DialogContent className="sm:max-w-md">
   300→   300→          <DialogHeader>
   301→   301→            <DialogTitle className="flex items-center gap-2">
   302→   302→              <Upload className="w-5 h-5 text-primary" />
   303→   303→              Upload Files
   304→   304→            </DialogTitle>
   305→   305→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   306→   306→          </DialogHeader>
   307→   307→          <div
   308→   308→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   309→   309→            onDragLeave={() => setUploadDragOver(null)}
   310→   310→            onDrop={(e) => {
   311→   311→              e.preventDefault()
   312→   312→              setUploadDragOver(null)
   313→   313→              // Set the project context before handling files
   314→   314→              const files = e.dataTransfer.files
   315→   315→              if (files && files.length > 0) {
   316→   316→                // Use a separate handler for dialog drops
   317→   317→                void (async () => {
   318→   318→                  setUploading(true)
   319→   319→                  const newFiles: typeof uploadedFiles = []
   320→   320→                  for (let i = 0; i < files.length; i++) {
   321→   321→                    const file = files[i]
   322→   322→                    try {
   323→   323→                      const formData = new FormData()
   324→   324→                      formData.append('file', file)
   325→   325→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   326→   326→                      if (!res.ok) {
   327→   327→                        const d = await res.json()
   328→   328→                        throw new Error(d.error || 'Upload failed')
   329→   329→                      }
   330→   330→                      const data = await res.json()
   331→   331→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   332→   332→                    } catch (err: unknown) {
   333→   333→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   334→   334→                    }
   335→   335→                  }
   336→   336→                  if (newFiles.length > 0) {
   337→   337→                    const updated = [...uploadedFiles, ...newFiles]
   338→   338→                    setUploadedFiles(updated)
   339→   339→                    saveUploadedFiles(updated)
   340→   340→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   341→   341→                  }
   342→   342→                  setUploading(false)
   343→   343→                })()
   344→   344→              }
   345→   345→            }}
   346→   346→            onClick={() => fileInputRef.current?.click()}
   347→   347→            className={cn(
   348→   348→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   349→   349→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   350→   350→            )}
   351→   351→          >
   352→   352→            {uploading ? (
   353→   353→              <div className="flex flex-col items-center gap-2">
   354→   354→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   355→   355→                <p className="text-sm text-muted-foreground">Uploading...</p>
   356→   356→              </div>
   357→   357→            ) : (
   358→   358→              <div className="flex flex-col items-center gap-2">
   359→   359→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   360→   360→                  <Upload className="w-5 h-5 text-muted-foreground" />
   361→   361→                </div>
   362→   362→                <div>
   363→   363→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   364→   364→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   365→   365→                </div>
   366→   366→              </div>
   367→   367→            )}
   368→   368→          </div>
   369→   369→          <DialogFooter>
   370→   370→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   371→   371→          </DialogFooter>
   372→   372→        </DialogContent>
   373→   373→      </Dialog>
   374→   374→
   375→   375→      {/* ─── Project Progress Overview ──────────────────────────── */}
   376→   376→      {projects.length > 0 && (
   377→   377→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   378→   378→          <Card>
   379→   379→            <CardHeader className="pb-3">
   380→   380→              <CardTitle className="text-base flex items-center gap-2">
   381→   381→                <BarChart3 className="w-4 h-4 text-primary" />
   382→   382→                Project Progress
   383→   383→              </CardTitle>
   384→   384→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   385→   385→            </CardHeader>
   386→   386→            <CardContent>
   387→   387→              <div className="space-y-5">
   388→   388→                {projects.map((project) => {
   389→   389→                  const todo = project.taskStats?.todo || 0
   390→   390→                  const inProgress = project.taskStats?.in_progress || 0
   391→   391→                  const inReview = project.taskStats?.in_review || 0
   392→   392→                  const completed = project.taskStats?.done || 0
   393→   393→                  const total = todo + inProgress + inReview + completed
   394→   394→                  if (total === 0) return null
   395→   395→
   396→   396→                  const todoPct = Math.round((todo / total) * 100)
   397→   397→                  const inProgressPct = Math.round((inProgress / total) * 100)
   398→   398→                  const inReviewPct = Math.round((inReview / total) * 100)
   399→   399→                  const completedPct = Math.round((completed / total) * 100)
   400→   400→
   401→   401→                  return (
   402→   402→                    <div key={project.id} className="space-y-2.5">
   403→   403→                      <div className="flex items-center justify-between">
   404→   404→                        <div className="flex items-center gap-2">
   405→   405→                          <span className="w-3 h-3 rounded-full" style={{ backgroundColor: project.color }} />
   406→   406→                          <span className="text-sm font-medium">{project.name}</span>
   407→   407→                        </div>
   408→   408→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   409→   409→                      </div>
   410→   410→                      {/* Stacked progress bar */}
   411→   411→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   412→   412→                        {todo > 0 && (
   413→   413→                          <div
   414→   414→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   415→   415→                            style={{ width: `${todoPct}%` }}
   416→   416→                            title={`Not Started: ${todo}`}
   417→   417→                          />
   418→   418→                        )}
   419→   419→                        {inProgress > 0 && (
   420→   420→                          <div
   421→   421→                            className="bg-amber-500 transition-all duration-500"
   422→   422→                            style={{ width: `${inProgressPct}%` }}
   423→   423→                            title={`In Progress: ${inProgress}`}
   424→   424→                          />
   425→   425→                        )}

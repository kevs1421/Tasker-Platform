    57→function UserAvatar({ user, size = 'md', className }: { user?: User | null; size?: 'sm' | 'md' | 'lg'; className?: string }) {
    58→  const sizeMap = { sm: 'h-8 w-8 text-[10px]', md: 'h-10 w-10 text-xs', lg: 'h-12 w-12 text-sm' }
    59→  if (!user) return <div className={cn(sizeMap[size], 'rounded-full bg-muted flex items-center justify-center', className)}>?</div>
    60→  return (
    61→    <Avatar className={cn(sizeMap[size], className)}>
    62→      {user.avatar && <AvatarImage src={user.avatar} />}
    63→      <AvatarFallback className="bg-slate-100 dark:bg-slate-800 font-medium">
    64→        {getInitials(user.name)}
    65→      </AvatarFallback>
    66→    </Avatar>
    67→  )
    68→}

    81→function CreateGroupDialog({
    82→  open, onOpenChange, onCreated, currentUser,
    83→}: {
    84→  open: boolean; onOpenChange: (open: boolean) => void;
    85→  onCreated: (group: MessageGroup) => void; currentUser: User;
    86→}) {
    87→  const [name, setName] = useState('')
    88→  const [users, setUsers] = useState<User[]>([])
    89→  const [selectedIds, setSelectedIds] = useState<string[]>([])
    90→  const [creating, setCreating] = useState(false)
    91→  const [search, setSearch] = useState('')
    92→  const [loading, setLoading] = useState(false)
    93→
    94→  useEffect(() => {
    95→    if (!open) return
    96→    setLoading(true)
    97→    setName('')
    98→    setSelectedIds([])
    99→    setSearch('')
   100→    fetch('/api/users')
   101→      .then((r) => r.json())
   102→      .then((d) => {
   103→        setUsers((d.users ?? []).filter((u: User) => {
   104→          if (u.id === currentUser.id) return false
   105→          // Training users cannot message clients
   106→          if (currentUser.role === 'training' && u.role === 'client') return false
   107→          return true
   108→        }))
   109→      })
   110→      .catch(() => {})
   111→      .finally(() => setLoading(false))
   112→  }, [open, currentUser.id, currentUser.role])
   113→
   114→  const toggleUser = (userId: string) => {
   115→    setSelectedIds((prev) => prev.includes(userId) ? prev.filter((id) => id !== userId) : [...prev, userId])
   116→  }
   117→
   118→  const handleCreate = async () => {
   119→    if (!name.trim() || selectedIds.length === 0) return
   120→    setCreating(true)
   121→    try {
   122→      const res = await fetch('/api/message-groups', {
   123→        method: 'POST',
   124→        headers: { 'Content-Type': 'application/json' },
   125→        body: JSON.stringify({ name: name.trim(), createdBy: currentUser.id, memberIds: selectedIds }),
   126→      })
   127→      if (!res.ok) throw new Error()
   128→      const data = await res.json()
   129→      toast.success('Group created')
   130→      onCreated(data.group)
   131→      onOpenChange(false)
   132→    } catch {
   133→      toast.error('Failed to create group')
   134→    } finally {
   135→      setCreating(false)
   136→    }
   137→  }
   138→
   139→  const filteredUsers = users.filter(
   140→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   141→  )
   142→
   143→  return (
   144→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   145→      <DialogContent className="sm:max-w-md">
   146→        <DialogHeader>
   147→          <DialogTitle className="flex items-center gap-2">
   148→            <Users className="h-5 w-5 text-violet-600" /> Create Group
   149→          </DialogTitle>
   150→          <DialogDescription>Start a group conversation with team members</DialogDescription>
   151→        </DialogHeader>
   152→
   153→        <div className="space-y-4 py-2">
   154→          <div>
   155→            <Input placeholder="Group name..." value={name} onChange={(e) => setName(e.target.value)} autoFocus />
   156→          </div>
   157→
   158→          <div>
   159→            <div className="relative mb-2">
   160→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   161→              <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   162→            </div>
   163→            <ScrollArea className="max-h-48 rounded-md border">
   164→              {loading ? (
   165→                <div className="p-3 space-y-2">
   166→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   167→                </div>
   168→              ) : filteredUsers.length === 0 ? (
   169→                <p className="text-sm text-muted-foreground text-center py-4">No users found</p>
   170→              ) : (
   171→                <div className="p-1">
   172→                  {filteredUsers.map((u) => (
   173→                    <button
   174→                      key={u.id}
   175→                      type="button"
   176→                      className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   177→                      onClick={() => toggleUser(u.id)}
   178→                    >
   179→                      <Checkbox checked={selectedIds.includes(u.id)} />
   180→                      <UserAvatar user={u} size="sm" />
   181→                      <div className="flex-1 min-w-0">
   182→                        <p className="text-sm font-medium truncate">{u.name}</p>
   183→                        <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   184→                      </div>
   185→                      <Badge variant="secondary" className={cn('text-[10px]', getRoleBadgeClasses(u.role))}>{u.role}</Badge>
   186→                    </button>
   187→                  ))}
   188→                </div>
   189→              )}
   190→            </ScrollArea>
   191→            {selectedIds.length > 0 && (
   192→              <p className="text-xs text-muted-foreground mt-2">{selectedIds.length} member{selectedIds.length > 1 ? 's' : ''} selected</p>
   193→            )}
   194→          </div>
   195→        </div>
   196→
   197→        <DialogFooter>
   198→          <Button variant="outline" type="button" onClick={() => onOpenChange(false)}>Cancel</Button>
   199→          <Button
   200→            type="button"
   201→            onClick={handleCreate} disabled={!name.trim() || selectedIds.length === 0 || creating}
   202→            className="bg-violet-600 hover:bg-violet-700 text-white"
   203→          >
   204→            {creating && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   205→            Create Group
   206→          </Button>
   207→        </DialogFooter>
   208→      </DialogContent>
   209→    </Dialog>
   210→  )
   211→}

   298→function GroupInfoDialog({
   299→  open, onOpenChange, group, currentUserId, currentUserRole,
   300→}: {
   301→  open: boolean; onOpenChange: (open: boolean) => void
   302→  group: { id: string; name: string; members?: GroupMember[]; _count?: { members: number } } | null
   303→  currentUserId: string; currentUserRole: string
   304→}) {
   305→  const [members, setMembers] = useState<(GroupMember & { user: User })[]>([])
   306→  const [users, setUsers] = useState<User[]>([])
   307→  const [loading, setLoading] = useState(false)
   308→  const [showAddUser, setShowAddUser] = useState(false)
   309→  const [search, setSearch] = useState('')
   310→  useEffect(() => {
   311→    if (!open || !group) return
   312→    setLoading(true)
   313→    setSearch('')
   314→    setShowAddUser(false)
   315→    fetch(`/api/message-groups/${group.id}?userId=${currentUserId}`)
   316→      .then((r) => r.json())
   317→      .then((d) => {
   318→        setMembers((d.group?.members ?? []) as (GroupMember & { user: User })[])
   319→      })
   320→      .catch(() => {})
   321→      .finally(() => setLoading(false))
   322→  }, [open, group, currentUserId])
   323→
   324→  const handleAddMember = async (userId: string) => {
   325→    if (!group) return
   326→    try {
   327→      const res = await fetch(`/api/message-groups/${group.id}`, {
   328→        method: 'PATCH',
   329→        headers: { 'Content-Type': 'application/json' },
   330→        body: JSON.stringify({ addMemberIds: [userId] }),
   331→      })
   332→      if (!res.ok) throw new Error()
   333→      const data = await res.json()
   334→      setMembers(data.group.members ?? [])
   335→      toast.success('Member added')
   336→    } catch {
   337→      toast.error('Failed to add member')
   338→    }
   339→  }
   340→
   341→  const handleRemoveMember = async (userId: string) => {
   342→    if (!group) return
   343→    try {
   344→      const res = await fetch(`/api/message-groups/${group.id}`, {
   345→        method: 'PATCH',
   346→        headers: { 'Content-Type': 'application/json' },
   347→        body: JSON.stringify({ removeMemberIds: [userId] }),
   348→      })
   349→      if (!res.ok) throw new Error()
   350→      const data = await res.json()
   351→      setMembers(data.group.members ?? [])
   352→      toast.success('Member removed')
   353→    } catch {
   354→      toast.error('Failed to remove member')
   355→    }
   356→  }
   357→
   358→  const handleLoadUsers = () => {
   359→    fetch('/api/users')
   360→      .then((r) => r.json())
   361→      .then((d) => {
   362→        const memberIds = members.map((m) => m.userId)
   363→        setUsers((d.users ?? []).filter((u: User) => {
   364→          if (memberIds.includes(u.id)) return false
   365→          // Training users cannot add clients to groups
   366→          if (currentUserRole === 'training' && u.role === 'client') return false
   367→          return true
   368→        }))
   369→      })
   370→      .catch(() => {})
   371→    setShowAddUser(true)
   372→  }
   373→
   374→  const filteredUsers = users.filter(
   375→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   376→  )
   377→
   378→  return (
   379→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   380→      <DialogContent className="sm:max-w-md">
   381→        <DialogHeader>
   382→          <DialogTitle className="flex items-center gap-2">
   383→            <Users className="h-5 w-5 text-violet-600" /> Group Info
   384→          </DialogTitle>
   385→        </DialogHeader>
   386→
   387→        <div className="space-y-4 py-2">
   388→          {group && (
   389→            <div className="flex items-center gap-3">
   390→              <GroupAvatar name={group.name} size="lg" />
   391→              <div>
   392→                <p className="font-semibold">{group.name}</p>
   393→                <p className="text-sm text-muted-foreground">{group._count?.members ?? members.length} members</p>
   394→              </div>
   395→            </div>
   396→          )}
   397→
   398→          <Separator />
   399→
   400→          <div>
   401→            <div className="flex items-center justify-between mb-2">
   402→              <p className="text-sm font-medium">Members</p>
   403→              {!showAddUser && (
   404→                <Button variant="ghost" size="sm" className="h-7 text-xs gap-1" type="button" onClick={handleLoadUsers}>
   405→                  <UserPlus className="h-3.5 w-3.5" /> Add
   406→                </Button>
   407→              )}
   408→            </div>
   409→
   410→            {showAddUser && (
   411→              <div className="mb-3 space-y-2">
   412→                <div className="relative">
   413→                  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   414→                  <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   415→                </div>
   416→                <ScrollArea className="max-h-32 rounded-md border">
   417→                  <div className="p-1">
   418→                    {filteredUsers.slice(0, 8).map((u) => (
   419→                      <button
   420→                        key={u.id}
   421→                        type="button"
   422→                        className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   423→                        onClick={() => handleAddMember(u.id)}
   424→                      >
   425→                        <UserAvatar user={u} size="sm" />
   426→                        <div className="flex-1 min-w-0">
   427→                          <p className="text-sm font-medium truncate">{u.name}</p>
   428→                        </div>
   429→                        <UserPlus className="h-4 w-4 text-muted-foreground" />
   430→                      </button>
   431→                    ))}
   432→                  </div>
   433→                </ScrollArea>
   434→                <Button variant="ghost" size="sm" className="h-7 text-xs" type="button" onClick={() => setShowAddUser(false)}>Done</Button>
   435→              </div>
   436→            )}
   437→
   438→            <ScrollArea className="max-h-48 rounded-md border">
   439→              {loading ? (
   440→                <div className="p-3 space-y-2">
   441→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   442→                </div>
   443→              ) : (
   444→                <div className="p-1">
   445→                  {members.map((m) => (
   446→                    <div key={m.id} className="flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50">
   447→                      <UserAvatar user={m.user} size="sm" />
   448→                      <div className="flex-1 min-w-0">
   449→                        <p className="text-sm font-medium truncate">{m.user?.name ?? 'Unknown'}</p>
   450→                        <p className="text-xs text-muted-foreground">{m.role === 'admin' ? 'Admin' : 'Member'}</p>
   451→                      </div>
   452→                      {m.userId !== currentUserId && (
   453→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-muted-foreground hover:text-destructive" type="button" onClick={() => handleRemoveMember(m.userId)}>
   454→                          <X className="h-3.5 w-3.5" />
   455→                        </Button>
   456→                      )}
   457→                    </div>
   458→                  ))}
   459→                </div>
   460→              )}
   461→            </ScrollArea>
   462→          </div>
   463→        </div>
   464→      </DialogContent>
   465→    </Dialog>
   466→  )
   467→}

   511→  const selectedDirectUser = !isGroupSelected ? (selectedConvo as DirectConversation | null)?.user : null
   512→
   513→  const selectConversation = async (convo: Conversation) => {
   514→    if (!user?.id) return
   515→    setSelectedConvo(convo)
   516→    setMsgText('')
   517→    setMsgLoading(true)
   518→
   519→    try {
   520→      let url = `/api/messages?userId=${user.id}`
   521→      if (convo.type === 'group') {
   522→        url += `&groupId=${convo.group.id}`
   523→      } else {
   524→        if (!convo.user) return
   525→        url += `&otherId=${convo.user.id}`
   526→      }
   527→      const res = await fetch(url)
   528→      if (!res.ok) throw new Error()
   529→      const data = await res.json()
   530→      setMessages(data.messages ?? [])
   531→    } catch {
   532→      toast.error('Failed to load messages')
   533→    } finally {
   534→      setMsgLoading(false)
   535→    }
   536→
   537→    // Clear unread locally
   538→    setConversations((prev) => prev.map((c) => {
   539→      if (c.type === 'direct' && convo.type === 'direct' && c.user?.id === convo.user?.id) return { ...c, unread: 0 }
   540→      if (c.type === 'group' && convo.type === 'group' && c.group.id === convo.group.id) return { ...c, unread: 0 }
   541→      return c
   542→    }))
   543→  }
   544→
   545→  const sendMessage = async () => {
   546→    if (!user?.id || !msgText.trim() || !selectedConvo) return
   547→
   548→    setSending(true)
   549→    const text = msgText.trim()
   550→    setMsgText('')
   551→
   552→    try {
   553→      const body: Record<string, string> = { content: text, senderId: user.id }
   554→
   555→      if (selectedConvo.type === 'group') {
   556→        body.groupId = selectedConvo.group.id
   557→      } else {
   558→        if (!selectedConvo.user) return
   559→        body.receiverId = selectedConvo.user.id
   560→      }
   561→
   562→      const res = await fetch('/api/messages', {
   563→        method: 'POST',
   564→        headers: { 'Content-Type': 'application/json' },
   565→        body: JSON.stringify(body),
   566→      })
   567→      if (!res.ok) throw new Error()
   568→      const data = await res.json()
   569→      setMessages((prev) => [...prev, data.message])
   570→      fetchConversations()
   571→    } catch {
   572→      toast.error('Failed to send message')
   573→      setMsgText(text)
   574→    } finally {
   575→      setSending(false)
   576→    }
   577→  }
   578→
   579→  const handleKeyDown = (e: React.KeyboardEvent) => {
   580→    if (e.key === 'Enter' && !e.shiftKey) {
   581→      e.preventDefault()
   582→      sendMessage()
   583→    }
   584→  }
   585→
   586→  // Delete a single message
   587→  const deleteMessage = async (messageId: string) => {
   588→    if (!user?.id) return
   589→    try {
   590→      const res = await fetch(`/api/messages?id=${messageId}&userId=${user.id}`, { method: 'DELETE' })
   591→      if (!res.ok) throw new Error()
   592→      setMessages((prev) => prev.filter((m) => m.id !== messageId))
   593→      toast.success('Message deleted')
   594→    } catch {
   595→      toast.error('Failed to delete message')
   596→    }
   597→    setDeleteConfirm(null)
   598→  }
   599→
   600→  // Delete an entire conversation
   601→  const deleteConversation = async () => {
   602→    if (!user?.id || !selectedConvo || !deleteConfirm) return
   603→    try {
   604→      let url = `/api/messages?deleteConversation=true&userId=${user.id}`
   605→      if (selectedConvo.type === 'group') {
   606→        url += `&groupId=${selectedConvo.group.id}`
   607→      } else if (selectedConvo.user) {
   608→        url += `&otherId=${selectedConvo.user.id}`
   609→      }
   610→      const res = await fetch(url, { method: 'DELETE' })
   611→      if (!res.ok) throw new Error()
   612→      setSelectedConvo(null)
   613→      setMessages([])
   614→      fetchConversations()
   615→      toast.success('Conversation deleted')
   616→    } catch {
   617→      toast.error('Failed to delete conversation')
   618→    }
   619→    setDeleteConfirm(null)
   620→  }
   621→
   622→  // Handle selecting a user from New Message dialog
   623→  const handleNewMessageUser = async (selectedUser: User) => {
   624→    // Check if a conversation already exists
   625→    const existing = conversations.find(
   626→      (c) => c.type === 'direct' && c.user?.id === selectedUser.id
   627→    )
   628→    if (existing) {
   629→      await selectConversation(existing)
   630→      return
   631→    }
   632→    // Create a new direct conversation object and select it
   633→    const newConvo: DirectConversation = {
   634→      type: 'direct',
   635→      user: selectedUser,
   636→      lastMessage: null,
   637→      unread: 0,
   638→    }
   639→    setSelectedConvo(newConvo)
   640→    setMessages([])
   641→  }
   642→
   643→  // Filter conversations
   644→  const filteredConversations = conversations.filter((c) => {
   645→    if (!searchQuery.trim()) return true
   646→    const q = searchQuery.toLowerCase()
   647→    if (c.type === 'direct') return c.user?.name.toLowerCase().includes(q) || c.user?.email.toLowerCase().includes(q)
   648→    return c.group.name.toLowerCase().includes(q)
   649→  })
   650→
   651→  // Conversation display helpers
   652→  const getConvoPreview = (c: Conversation) => {
   653→    if (!c.lastMessage) return 'No messages yet'
   654→    const senderName = c.lastMessage.senderId === user?.id ? 'You' : (c.lastMessage.sender?.name ?? 'Someone')
   655→    if (c.type === 'group') return `${senderName}: ${c.lastMessage.content}`
   656→    return `${c.lastMessage.senderId === user?.id ? 'You: ' : ''}${c.lastMessage.content}`
   657→  }
   658→
   659→  // ── Render ───────────────────────────────────────────────
   660→
   661→  const containerHeight = 'h-[calc(100vh-12rem)]'
   662→
   663→  if (loading) {
   664→    return (
   665→      <div className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}>
   666→        <div className="w-80 border-r p-4 space-y-3 flex-shrink-0">
   667→          <Skeleton className="h-10 w-full" />
   668→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-14 w-full" />)}
   669→        </div>
   670→        <div className="flex-1 flex items-center justify-center">
   671→          <p className="text-muted-foreground text-sm">Select a conversation</p>
   672→        </div>
   673→      </div>
   674→    )
   675→  }
   676→
   677→  return (
   678→    <motion.div
   679→      initial={{ opacity: 0, y: 8 }}
   680→      animate={{ opacity: 1, y: 0 }}
   681→      className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}
   682→    >
   683→      {/* ─── Conversation List ──────────────────────────── */}
   684→      <div className="w-80 border-r bg-background flex flex-col flex-shrink-0">
   685→        <div className="p-3 border-b space-y-2">
   686→          <div className="flex items-center justify-between">
   687→            <h2 className="text-lg font-semibold">Messages</h2>
   688→            <div className="flex items-center gap-1">
   689→              <Button
   690→                variant="ghost" size="icon" className="h-8 w-8"
   691→                onClick={() => setShowNewMessage(true)}
   692→                title="New Message"
   693→              >
   694→                <MessageSquare className="h-4 w-4" />
   695→              </Button>
   696→              <Button
   697→                variant="ghost" size="icon" className="h-8 w-8"
   698→                onClick={() => setShowCreateGroup(true)}
   699→                title="Create Group"
   700→              >
   701→                <Plus className="h-4 w-4" />
   702→              </Button>
   703→            </div>
   704→          </div>
   705→          <div className="relative">
   706→            <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   707→            <Input
   708→              placeholder="Search conversations..."
   709→              className="pl-9 h-9 text-sm"
   710→              value={searchQuery}
   711→              onChange={(e) => setSearchQuery(e.target.value)}
   712→            />
   713→          </div>
   714→        </div>
   715→        <ScrollArea className="flex-1">
   716→          <div className="p-1">
   717→            {filteredConversations.length === 0 && (
   718→              <p className="text-sm text-muted-foreground text-center py-8">
   719→                {searchQuery ? 'No matching conversations' : 'No conversations yet'}
   720→              </p>
   721→            )}
   722→            {filteredConversations.map((convo, idx) => {
   723→              const isSelected =
   724→                convo.type === 'direct'
   725→                  ? selectedDirectUser?.id === convo.user?.id
   726→                  : selectedGroup?.id === convo.group.id
   727→              return (
   728→                <div
   729→                  key={convo.type === 'direct' ? convo.user?.id ?? idx : convo.group.id}
   730→                  className={cn(
   731→                    'group flex items-center rounded-md transition-colors',
   732→                    isSelected ? 'bg-muted' : 'hover:bg-muted/50'
   733→                  )}
   734→                >
   735→                  <button
   736→                    className="w-full text-left px-3 py-3 flex items-start gap-3 flex-1 min-w-0"
   737→                    onClick={() => selectConversation(convo)}
   738→                  >
   739→                    {convo.type === 'direct' ? (
   740→                      <UserAvatar user={convo.user} size="md" className="mt-0.5 flex-shrink-0" />
   741→                    ) : (
   742→                      <GroupAvatar name={convo.group.name} size="md" className="mt-0.5 flex-shrink-0" />
   743→                    )}
   744→                    <div className="flex-1 min-w-0">
   745→                      <div className="flex items-center justify-between">
   746→                        <div className="flex items-center gap-1.5 min-w-0">
   747→                          <span className="text-sm font-medium truncate">{convo.type === 'direct' ? (convo.user?.name ?? 'Unknown') : convo.group.name}</span>
   748→                          {convo.type === 'group' && (
   749→                            <Users className="h-3 w-3 text-violet-500 flex-shrink-0" />
   750→                          )}
   751→                        </div>
   752→                        {convo.lastMessage && (
   753→                          <span className="text-[10px] text-muted-foreground flex-shrink-0 ml-2">
   754→                            {formatRelativeTime(convo.lastMessage.createdAt)}
   755→                          </span>
   756→                        )}
   757→                      </div>
   758→                      <p className="text-xs text-muted-foreground truncate mt-0.5">
   759→                        {getConvoPreview(convo)}
   760→                      </p>
   761→                    </div>
   762→                    {convo.unread > 0 && (
   763→                      <div className="h-5 w-5 rounded-full bg-emerald-500 text-white text-[10px] font-bold flex items-center justify-center flex-shrink-0 mt-1">
   764→                        {convo.unread}
   765→                      </div>
   766→                    )}
   767→                  </button>
   768→                  {/* Delete conversation button — shows on hover */}
   769→                  <Button
   770→                    variant="ghost"
   771→                    size="icon"
   772→                    className="h-8 w-8 opacity-0 group-hover:opacity-100 transition-opacity text-muted-foreground hover:text-destructive flex-shrink-0 mr-1 mt-1"
   773→                    onClick={(e) => { e.stopPropagation(); setDeleteConfirm({ type: 'conversation', id: convo.type === 'direct' ? (convo as DirectConversation).user?.id ?? '' : convo.group.id }) }}
   774→                    title="Delete conversation"
   775→                  >
   776→                    <Trash2 className="h-3.5 w-3.5" />
   777→                  </Button>
   778→                </div>
   779→              )
   780→            })}
   781→          </div>
   782→        </ScrollArea>
   783→      </div>
   784→
   785→      {/* ─── Chat Area ─────────────────────────────────── */}
   786→      <div className="flex-1 flex flex-col bg-background">
   787→        {selectedConvo ? (
   788→          <>
   789→            {/* Chat Header */}
   790→            <div className="p-3 border-b flex items-center gap-3">
   791→              {isGroupSelected ? (
   792→                <button onClick={() => setShowGroupInfo(true)} className="flex items-center gap-3 hover:bg-muted/50 rounded-md px-1 -ml-1 py-0.5 transition-colors">
   793→                  <GroupAvatar name={selectedGroup!.name} size="md" />
   794→                  <div className="min-w-0">
   795→                    <p className="text-sm font-medium flex items-center gap-1.5">
   796→                      {selectedGroup!.name}
   797→                      <Users className="h-3.5 w-3.5 text-violet-500" />
   798→                    </p>
   799→                    <p className="text-xs text-muted-foreground">{selectedGroup!._count.members} members</p>
   800→                  </div>
   801→                </button>
   802→              ) : selectedDirectUser ? (
   803→                <>
   804→                  <UserAvatar user={selectedDirectUser} size="md" />
   805→                  <div className="min-w-0">
   806→                    <p className="text-sm font-medium truncate">{selectedDirectUser.name}</p>
   807→                    <p className="text-xs text-muted-foreground truncate">
   808→                      {selectedDirectUser.company ?? selectedDirectUser.email}
   809→                    </p>
   810→                  </div>
   811→                  <Badge variant="secondary" className={cn('ml-auto text-xs flex-shrink-0', getRoleBadgeClasses(selectedDirectUser.role))}>
   812→                    {selectedDirectUser.role}
   813→                  </Badge>
   814→                </>
   815→              ) : null}
   816→
   817→              {/* Delete conversation button in header */}
   818→              <Button
   819→                variant="ghost"
   820→                size="icon"
   821→                className="h-8 w-8 ml-auto text-muted-foreground hover:text-destructive flex-shrink-0"
   822→                onClick={() => setDeleteConfirm({ type: 'conversation', id: '' })}
   823→                title="Delete conversation"
   824→              >
   825→                <Trash2 className="h-4 w-4" />
   826→              </Button>
   827→            </div>
   828→
   829→            {/* Messages */}
   830→            <ScrollArea className="flex-1 p-4">
   831→              <div className="space-y-3 max-w-2xl mx-auto">
   832→                {msgLoading ? (
   833→                  <div className="space-y-3">
   834→                    {Array.from({ length: 4 }).map((_, i) => (
   835→                      <div key={i} className={cn('flex gap-2.5', i % 2 === 0 ? '' : 'justify-end')}>
   836→                        <Skeleton className="h-10 w-48 rounded-2xl" />
   837→                      </div>
   838→                    ))}
   839→                  </div>
   840→                ) : messages.length === 0 ? (
   841→                  <div className="flex flex-col items-center justify-center h-full text-center py-8">
   842→                    <MessageSquare className="h-10 w-10 text-muted-foreground/30 mb-3" />
   843→                    <p className="text-sm text-muted-foreground">
   844→                      {isGroupSelected ? 'No messages yet. Start the conversation!' : 'No messages yet. Say hello!'}
   845→                    </p>
   846→                  </div>
   847→                ) : (
   848→                  messages.map((msg, idx) => {
   849→                    const isSent = msg.senderId === user?.id
   850→                    return (
   851→                      <motion.div
   852→                        key={msg.id}
   853→                        initial={{ opacity: 0, y: 5 }}
   854→                        animate={{ opacity: 1, y: 0 }}
   855→                        transition={{ delay: Math.min(idx * 0.015, 0.2) }}
   856→                        className={cn('group/msg flex gap-2.5', isSent ? 'justify-end' : 'justify-start')}
   857→                      >
   858→                        {!isSent && (
   859→                          <UserAvatar user={msg.sender} size="sm" className="mt-auto" />
   860→                        )}
   861→                        <div className={cn('max-w-[70%]', isSent ? 'text-right' : '')}>
   862→                          {isGroupSelected && !isSent && (
   863→                            <p className="text-xs font-medium text-muted-foreground mb-1 px-1">
   864→                              {msg.sender?.name ?? 'Unknown'}
   865→                            </p>
   866→                          )}
   867→                          <div className="relative">
   868→                            <div className={cn(
   869→                              'inline-block rounded-2xl px-4 py-2.5 text-sm',
   870→                              isSent
   871→                                ? 'bg-emerald-600 text-white rounded-br-md'
   872→                                : 'bg-muted rounded-bl-md',
   873→                            )}>
   874→                              <p className="whitespace-pre-wrap break-words">{msg.content}</p>
   875→                            </div>
   876→                            {/* Delete message button — shows on hover */}
   877→                            <Button
   878→                              variant="ghost"
   879→                              size="icon"
   880→                              className={cn(
   881→                                'h-6 w-6 absolute opacity-0 group-hover/msg:opacity-100 transition-opacity',
   882→                                isSent
   883→                                  ? '-left-8 top-1/2 -translate-y-1/2'
   884→                                  : '-right-8 top-1/2 -translate-y-1/2'
   885→                              )}
   886→                              onClick={() => setDeleteConfirm({ type: 'message', id: msg.id })}
   887→                              title="Delete message"
   888→                            >
   889→                              <Trash2 className="h-3 w-3 text-muted-foreground hover:text-destructive" />
   890→                            </Button>
   891→                          </div>
   892→                          <div className={cn(
   893→                            'flex items-center gap-1 mt-1 text-[10px] text-muted-foreground',
   894→                            isSent ? 'justify-end' : 'justify-start'
   895→                          )}>
   896→                            <span>{formatRelativeTime(msg.createdAt)}</span>
   897→                            {isSent && msg.read && <CheckCheck className="w-3 h-3" />}
   898→                          </div>
   899→                        </div>
   900→                      </motion.div>
   901→                    )
   902→                  })
   903→                )}
   904→                <div ref={messagesEndRef} />
   905→              </div>
   906→            </ScrollArea>
   907→
   908→            {/* Message Input */}
   909→            <div className="p-3 border-t">
   910→              <div className="flex gap-2 max-w-2xl mx-auto">
   911→                <Input
   912→                  placeholder={isGroupSelected ? `Message ${selectedGroup!.name}...` : 'Type a message...'}
   913→                  value={msgText}
   914→                  onChange={(e) => setMsgText(e.target.value)}
   915→                  onKeyDown={handleKeyDown}
   916→                  className="flex-1"
   917→                />
   918→                <Button
   919→                  onClick={sendMessage}
   920→                  disabled={!msgText.trim() || sending}
   921→                  className="bg-emerald-600 hover:bg-emerald-700 text-white flex-shrink-0"
   922→                >
   923→                  {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   924→                </Button>
   925→              </div>
   926→            </div>
   927→          </>
   928→        ) : (
   929→          <div className="flex-1 flex items-center justify-center">
   930→            <div className="text-center space-y-3">
   931→              <div className="h-16 w-16 rounded-full bg-muted flex items-center justify-center mx-auto">
   932→                <MessageSquare className="h-8 w-8 text-muted-foreground" />
   933→              </div>
   934→              <h3 className="font-semibold text-lg">Select a conversation</h3>
   935→              <p className="text-sm text-muted-foreground max-w-xs">
   936→                Choose a conversation from the sidebar, or start a new one.
   937→              </p>
   938→              <div className="flex items-center justify-center gap-2 pt-2">
   939→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setShowNewMessage(true)}>
   940→                  <MessageSquare className="h-4 w-4" /> New Message
   941→                </Button>
   942→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setShowCreateGroup(true)}>
   943→                  <Users className="h-4 w-4" /> Create Group
   944→                </Button>
   945→              </div>
   946→            </div>
   947→          </div>
   948→        )}
   949→      </div>
   950→
   951→      {/* ─── Dialogs ──────────────────────────────────── */}
   952→      <CreateGroupDialog
   953→        open={showCreateGroup}
   954→        onOpenChange={setShowCreateGroup}
   955→        onCreated={() => { fetchConversations(); setShowCreateGroup(false) }}
   956→        currentUser={user!}
   957→      />
   958→      <NewMessageDialog
   959→        open={showNewMessage}
   960→        onOpenChange={setShowNewMessage}
   961→        onUserSelected={handleNewMessageUser}
   962→        currentUser={user!}
   963→      />
   964→      <GroupInfoDialog
   965→        open={showGroupInfo}
   966→        onOpenChange={setShowGroupInfo}
   967→        group={selectedGroup ? { id: selectedGroup.id, name: selectedGroup.name, _count: selectedGroup._count } : null}
   968→        currentUserId={user?.id ?? ''}
   969→        currentUserRole={user?.role ?? ''}
   970→      />
   971→
   972→      {/* ─── Delete Confirmation ─────────────────────── */}
   973→      <AlertDialog open={!!deleteConfirm} onOpenChange={(open) => { if (!open) setDeleteConfirm(null) }}>
   974→        <AlertDialogContent>
   975→          <AlertDialogHeader>
   976→            <AlertDialogTitle>
   977→              {deleteConfirm?.type === 'message' ? 'Delete Message' : 'Delete Conversation'}
   978→            </AlertDialogTitle>
   979→            <AlertDialogDescription>
   980→              {deleteConfirm?.type === 'message'
   981→                ? 'Are you sure you want to delete this message? This action cannot be undone.'
   982→                : 'Are you sure you want to delete this entire conversation? All messages will be permanently removed. This action cannot be undone.'
   983→              }
   984→            </AlertDialogDescription>
   985→          </AlertDialogHeader>
   986→          <AlertDialogFooter>
   987→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   988→            <AlertDialogAction
   989→              className="bg-destructive text-white hover:bg-destructive/90"
   990→              onClick={() => {
   991→                if (!deleteConfirm) return
   992→                if (deleteConfirm.type === 'message') {
   993→                  deleteMessage(deleteConfirm.id)
   994→                } else {
   995→                  deleteConversation()
   996→                }
   997→              }}
   998→            >
   999→              Delete
  1000→            </AlertDialogAction>
  1001→          </AlertDialogFooter>
  1002→        </AlertDialogContent>
  1003→      </AlertDialog>
  1004→    </motion.div>
  1005→  )

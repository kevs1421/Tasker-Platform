    46→    46→function formatElapsed(ms: number): string {
    47→    47→  const totalSeconds = Math.floor(ms / 1000)
    48→    48→  const h = Math.floor(totalSeconds / 3600)
    49→    49→  const m = Math.floor((totalSeconds % 3600) / 60)
    50→    50→  const s = totalSeconds % 60
    51→    51→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    52→    52→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    53→    53→}
    54→    54→
    55→    55→function formatCountdown(ms: number): string {
    56→    56→  if (ms <= 0) return '0:00'
    57→    57→  const totalSeconds = Math.floor(ms / 1000)
    58→    58→  const m = Math.floor(totalSeconds / 60)
    59→    59→  const s = totalSeconds % 60
    60→    60→  return `${m}:${String(s).padStart(2, '0')}`
    61→    61→}

   874→   874→  return (

   893→   893→function OfflineBanner() {
   894→   894→  const [isOffline, setIsOffline] = useState(() => {
   895→   895→    if (typeof navigator === 'undefined') return false
   896→   896→    return !navigator.onLine
   897→   897→  })
   898→   898→
   899→   899→  useEffect(() => {
   900→   900→    const handleOnline = () => setIsOffline(false)
   901→   901→    const handleOffline = () => setIsOffline(true)
   902→   902→    window.addEventListener('online', handleOnline)
   903→   903→    window.addEventListener('offline', handleOffline)
   904→   904→    return () => {
   905→   905→      window.removeEventListener('online', handleOnline)
   906→   906→      window.removeEventListener('offline', handleOffline)
   907→   907→    }
   908→   908→  }, [])
   909→   909→
   910→   910→  if (!isOffline) return null
   911→   911→
   912→   912→  return (
   913→   913→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   914→   914→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   915→   915→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   916→   916→    </div>
   917→   917→  )
   918→   918→}

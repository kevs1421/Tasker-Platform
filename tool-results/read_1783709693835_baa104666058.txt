   105→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   106→  const visible = users.slice(0, max)
   107→  const extra = users.length - max
   108→  return (
   109→    <div className="flex -space-x-2">
   110→      {visible.map((u, i) =>
   111→        u ? (
   112→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   113→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   114→              {getInitials(u.name)}
   115→            </AvatarFallback>
   116→          </Avatar>
   117→        ) : null
   118→      )}
   119→      {extra > 0 && (
   120→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   121→          +{extra}
   122→        </div>
   123→      )}
   124→    </div>
   125→  )
   126→}
   127→
   128→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   129→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   130→  if (!user) return null
   131→  return (
   132→    <Avatar className={sizeClasses}>
   133→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   134→        {getInitials(user.name)}
   135→      </AvatarFallback>
   136→    </Avatar>
   137→  )
   138→}

   156→// ─── TeamProjects ───────────────────────────────────────────
   157→
   158→export function TeamProjects() {
   159→  const { user, setSelectedProjectId } = useAuthStore()
   160→  const [projects, setProjects] = useState<Project[]>([])
   161→  const [loading, setLoading] = useState(true)
   162→  const [expandedId, setExpandedId] = useState<string | null>(null)
   163→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   164→  const [expandedLoading, setExpandedLoading] = useState(false)
   165→  const [taskFilter, setTaskFilter] = useState<string>('all')
   166→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   167→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   168→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   169→
   170→  // Create task form
   171→  const [taskTitle, setTaskTitle] = useState('')
   172→  const [taskDesc, setTaskDesc] = useState('')
   173→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   174→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   175→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   176→  const [taskTags, setTaskTags] = useState('')
   177→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   178→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   179→
   180→  const fetchProjects = useCallback(async () => {
   181→    try {
   182→      const res = await fetch('/api/projects')
   183→      const data = await res.json()
   184→      const filtered = (data.projects || []).filter((p: Project) =>
   185→        p.members?.some((m) => m.userId === user?.id)
   186→      )
   187→      setProjects(filtered)
   188→    } catch {
   189→      toast.error('Failed to load projects')
   190→    } finally {
   191→      setLoading(false)
   192→    }
   193→  }, [user])
   194→
   195→  useEffect(() => { fetchProjects() }, [fetchProjects])
   196→
   197→  const handleExpand = async (projectId: string) => {
   198→    if (expandedId === projectId) {
   199→      setExpandedId(null)
   200→      setExpandedTasks([])
   201→      return
   202→    }
   203→    setExpandedId(projectId)
   204→    setTaskFilter('all')
   205→    setExpandedLoading(true)
   206→    try {
   207→      const res = await fetch(`/api/projects/${projectId}`)
   208→      const data = await res.json()
   209→      setExpandedTasks(data.project?.tasks || [])
   210→    } catch {
   211→      toast.error('Failed to load tasks')
   212→    } finally {
   213→      setExpandedLoading(false)
   214→    }
   215→  }
   216→
   217→  const handleCreateTask = async () => {
   218→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   219→    if (!user?.id) { toast.error('Not authenticated'); return }
   220→    setCreateTaskLoading(true)
   221→    try {
   222→      const res = await fetch('/api/tasks', {
   223→        method: 'POST',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({
   226→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   227→          priority: taskPriority, projectId: createTaskProjectId,
   228→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   229→          tags: taskTags,
   230→          createdById: user.id,
   231→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   232→        }),
   233→      })
   234→      if (!res.ok) {
   235→        const d = await res.json()
   236→        throw new Error(d.error || 'Failed to create task')
   237→      }
   238→      toast.success('Task created')
   239→      setCreateTaskOpen(false)
   240→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   241→      setTaskPriority('medium'); setTaskDueDate(undefined)
   242→      setTaskTags(''); setTaskAssignee('')
   243→      // Always refresh the expanded project view
   244→      handleExpand(createTaskProjectId)
   245→    } catch (e: unknown) {
   246→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   247→    } finally {
   248→      setCreateTaskLoading(false)
   249→    }
   250→  }
   251→
   252→  const openCreateTaskForProject = (projectId: string) => {
   253→    setCreateTaskProjectId(projectId)
   254→    setCreateTaskOpen(true)
   255→  }
   256→
   257→  const getProjectTaskStats = (projectId: string) => {
   258→    if (expandedId !== projectId) return null
   259→    const total = expandedTasks.length
   260→    const done = expandedTasks.filter((t) => t.status === 'done').length
   261→    return { total, done }
   262→  }
   263→
   264→  const filteredTasks = taskFilter === 'all'
   265→    ? expandedTasks
   266→    : expandedTasks.filter((t) => t.status === taskFilter)
   267→
   268→  if (loading) {
   269→    return (
   270→      <div>
   271→        <div className="flex items-center justify-between mb-6">
   272→          <Skeleton className="h-8 w-48" />
   273→          <Skeleton className="h-10 w-36" />
   274→        </div>
   275→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   276→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   277→        </div>
   278→      </div>
   279→    )
   280→  }
   281→
   282→  return (
   283→    <div className="space-y-6">
   284→      {/* Header */}
   285→      <div className="flex items-center justify-between">
   286→        <div>
   287→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   288→          <p className="text-sm text-muted-foreground mt-1">
   289→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   290→          </p>
   291→        </div>
   292→      </div>
   293→
   294→      {/* Project Grid */}
   295→      {projects.length === 0 ? (
   296→        <EmptyState
   297→          icon={FolderKanban}
   298→          title="No projects yet"
   299→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   300→        />
   301→      ) : (
   302→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   303→          <AnimatePresence>
   304→            {projects.map((project, idx) => {
   305→              const totalTasks = project._count?.tasks || 0
   306→              const members = project.members || []
   307→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   308→              const pConfig = getPriorityConfig(project.priority)
   309→              const isExpanded = expandedId === project.id
   310→              const stats = getProjectTaskStats(project.id)
   311→
   312→              return (
   313→                <motion.div
   314→                  key={project.id}
   315→                  initial={{ opacity: 0, y: 20 }}
   316→                  animate={{ opacity: 1, y: 0 }}
   317→                  transition={{ delay: idx * 0.05 }}
   318→                >
   319→                  <Card
   320→                    className={cn(
   321→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   322→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   323→                    )}
   324→                    onClick={() => handleExpand(project.id)}
   325→                  >
   326→                    <div className="flex">
   327→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   328→                      <CardContent className="flex-1 p-4">
   329→                        <div className="flex items-start justify-between mb-2">
   330→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   331→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   332→                            {pConfig.label}
   333→                          </Badge>
   334→                        </div>
   335→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   336→                          {project.description || 'No description'}
   337→                        </p>
   338→                        <Progress
   339→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   340→                          className="h-1.5 mb-3"
   341→                        />
   342→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   343→                          <div className="flex items-center gap-3">
   344→                            <span className="flex items-center gap-1">
   345→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   346→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   347→                            </span>
   348→                            <span className="flex items-center gap-1">
   349→                              <Users className="h-3 w-3" />
   350→                              {members.length}
   351→                            </span>
   352→                          </div>
   353→                          {project.dueDate && (
   354→                            <span className={cn(
   355→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   356→                                ? 'text-red-500' : '',
   357→                            )}>
   358→                              {formatDate(project.dueDate)}
   359→                            </span>
   360→                          )}
   361→                        </div>
   362→                        {memberUsers.length > 0 && (
   363→                          <div className="mt-3">
   364→                            <AvatarStack users={memberUsers} max={4} />
   365→                          </div>
   366→                        )}
   367→                      </CardContent>
   368→                    </div>
   369→                  </Card>
   370→
   371→                  {/* Expanded Task List */}
   372→                  <AnimatePresence>
   373→                    {isExpanded && (
   374→                      <motion.div
   375→                        initial={{ opacity: 0, height: 0 }}
   376→                        animate={{ opacity: 1, height: 'auto' }}
   377→                        exit={{ opacity: 0, height: 0 }}
   378→                        transition={{ duration: 0.2 }}
   379→                        className="overflow-hidden"
   380→                        onClick={(e) => e.stopPropagation()}
   381→                      >
   382→                        <Card className="mt-3 p-4 space-y-4">
   383→                          <div className="flex items-center justify-between">
   384→                            <div className="flex items-center gap-2">
   385→                              <div className="flex gap-1">
   386→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   387→                                  <Button
   388→                                    key={s}
   389→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   390→                                    size="sm"
   391→                                    className="h-7 text-xs px-2"
   392→                                    onClick={() => setTaskFilter(s)}
   393→                                  >
   394→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   395→                                  </Button>
   396→                                ))}
   397→                              </div>
   398→                            </div>
   399→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   400→                              <Plus className="h-3 w-3" /> Add Task
   401→                            </Button>
   402→                          </div>
   403→                          {expandedLoading ? (
   404→                            <div className="space-y-2">
   405→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   406→                            </div>
   407→                          ) : filteredTasks.length === 0 ? (
   408→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   409→                          ) : (
   410→                            <ScrollArea className="max-h-64">
   411→                              <div className="space-y-2">
   412→                                {filteredTasks.map((task) => {
   413→                                  const sConfig = getStatusConfig(task.status)
   414→                                  const pConf = getPriorityConfig(task.priority)
   415→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   416→                                  return (
   417→                                    <div
   418→                                      key={task.id}
   419→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   420→                                      onClick={() => setSelectedProjectId(project.id)}
   421→                                    >
   422→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   423→                                      <div className="flex-1 min-w-0">
   424→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   425→                                        <div className="flex items-center gap-2 mt-0.5">
   426→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   427→                                          {task.dueDate && (
   428→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   429→                                              {formatDate(task.dueDate)}
   430→                                            </span>
   431→                                          )}
   432→                                        </div>
   433→                                      </div>
   434→                                      <AvatarStack users={assignees} max={2} />
   435→                                    </div>
   436→                                  )
   437→                                })}
   438→                              </div>
   439→                            </ScrollArea>
   440→                          )}
   441→                        </Card>
   442→                      </motion.div>
   443→                    )}
   444→                  </AnimatePresence>
   445→                </motion.div>
   446→              )
   447→            })}
   448→          </AnimatePresence>
   449→        </motion.div>
   450→      )}
   451→
   452→      {/* Create Task Dialog */}
   453→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   454→        <DialogContent className="sm:max-w-lg">
   455→          <DialogHeader>
   456→            <DialogTitle>Create New Task</DialogTitle>
   457→            <DialogDescription>Add a task to the project.</DialogDescription>
   458→          </DialogHeader>
   459→          <div className="space-y-4 py-2">
   460→            <div className="space-y-2">
   461→              <Label>Title</Label>
   462→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   463→            </div>
   464→            <div className="space-y-2">
   465→              <Label>Description</Label>
   466→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   467→            </div>
   468→            <div className="grid grid-cols-2 gap-4">
   469→              <div className="space-y-2">
   470→                <Label>Status</Label>
   471→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   472→                  <SelectTrigger><SelectValue /></SelectTrigger>
   473→                  <SelectContent>
   474→                    <SelectItem value="todo">To Do</SelectItem>
   475→                    <SelectItem value="in_progress">In Progress</SelectItem>
   476→                    <SelectItem value="in_review">In Review</SelectItem>
   477→                    <SelectItem value="done">Done</SelectItem>
   478→                  </SelectContent>
   479→                </Select>
   480→              </div>
   481→              <div className="space-y-2">
   482→                <Label>Priority</Label>
   483→                <Select value={taskPriority} onValueChange={(v) => setTaskPriority(v as typeof taskPriority)}>
   484→                  <SelectTrigger><SelectValue /></SelectTrigger>
   485→                  <SelectContent>
   486→                    <SelectItem value="low">Low</SelectItem>
   487→                    <SelectItem value="medium">Medium</SelectItem>
   488→                    <SelectItem value="high">High</SelectItem>
   489→                    <SelectItem value="urgent">Urgent</SelectItem>
   490→                  </SelectContent>
   491→                </Select>
   492→              </div>
   493→            </div>
   494→            <div className="grid grid-cols-2 gap-4">
   495→              <div className="space-y-2">
   496→                <Label>Due Date</Label>
   497→                <Popover open={taskCalendarOpen} onOpenChange={setTaskCalendarOpen}>
   498→                  <PopoverTrigger asChild>
   499→                    <Button variant="outline" className="w-full justify-start text-left font-normal gap-2">
   500→                      <CalendarIcon className="h-4 w-4" />
   501→                      {taskDueDate ? format(taskDueDate, 'PPP') : 'Pick a date'}
   502→                    </Button>

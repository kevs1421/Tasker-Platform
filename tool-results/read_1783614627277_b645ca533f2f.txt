   126→// ═══════════════════════════════════════════════════════════════════
   127→const ONBOARDING_STEPS = [
   128→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   129→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   130→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   131→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   132→]
   133→
   134→export function ClientProjects() {
   135→  const { user } = useAuthStore()
   136→  const [projects, setProjects] = useState<(Project & { tasks?: (Task & { _count: { comments: number; subtasks: number; completedSubtasks: number } })[] })[]>([])
   137→  const [loading, setLoading] = useState(true)
   138→  const [expandedProjectId, setExpandedProjectId] = useState<string | null>(null)
   139→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   140→  const [taskDetail, setTaskDetail] = useState<(Task & { _count: { comments: number; subtasks: number; completedSubtasks: number }; comments?: Comment[]; subtasks?: Subtask[] }) | null>(null)
   141→  const [taskDetailLoading, setTaskDetailLoading] = useState(false)
   142→
   143→  // Onboarding creation state
   144→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   145→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   146→  const [onboardingTitle, setOnboardingTitle] = useState('')
   147→  const [onboardingDesc, setOnboardingDesc] = useState('')
   148→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   149→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   150→
   151→  const fetchProjects = useCallback(async () => {
   152→    try {
   153→      const res = await fetch('/api/projects')
   154→      const data = await res.json()
   155→      // Filter to active projects and those the client is a member of
   156→      const clientProjects = (data.projects || [])
   157→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   158→      setProjects(clientProjects)
   159→    } catch {
   160→      toast.error('Failed to load projects')
   161→    } finally {
   162→      setLoading(false)
   163→    }
   164→  }, [user?.id])
   165→
   166→  useEffect(() => { fetchProjects() }, [fetchProjects])
   167→
   168→  const expandProject = async (projectId: string) => {
   169→    if (expandedProjectId === projectId) {
   170→      setExpandedProjectId(null)
   171→      setExpandedTasks([])
   172→      return
   173→    }
   174→    setExpandedProjectId(projectId)
   175→    try {
   176→      const res = await fetch(`/api/projects/${projectId}`)
   177→      const data = await res.json()
   178→      setExpandedTasks(data.project?.tasks || [])
   179→    } catch {
   180→      toast.error('Failed to load project tasks')
   181→    }
   182→  }
   183→
   184→  const openTaskDetail = async (taskId: string) => {
   185→    setTaskDetailLoading(true)
   186→    setTaskDetail(null)
   187→    try {
   188→      const res = await fetch(`/api/tasks/${taskId}`)
   189→      const data = await res.json()
   190→      setTaskDetail(data.task)
   191→    } catch {
   192→      toast.error('Failed to load task details')
   193→    } finally {
   194→      setTaskDetailLoading(false)
   195→    }
   196→  }
   197→
   198→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   199→    setOnboardingProjectId(projectId)
   200→    setOnboardingTitle(`Onboarding - ${projectName}`)
   201→    setOnboardingDesc('')
   202→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   203→    setOnboardingDialogOpen(true)
   204→  }
   205→
   206→  const handleCreateOnboarding = async () => {
   207→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   208→    setCreatingOnboarding(true)
   209→    try {
   210→      const stepsJson = JSON.stringify(
   211→        onboardingSteps.map(stepId => {
   212→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   213→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   214→        })
   215→      )
   216→      const res = await fetch('/api/onboarding', {
   217→        method: 'POST',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({
   220→          title: onboardingTitle.trim(),
   221→          description: onboardingDesc.trim() || null,
   222→          stepsJson,
   223→          projectId: onboardingProjectId,
   224→          status: 'published',
   225→        }),
   226→      })
   227→      if (!res.ok) {
   228→        const d = await res.json()
   229→        throw new Error(d.error || 'Failed to create')
   230→      }
   231→      toast.success('Onboarding form created successfully')
   232→      setOnboardingDialogOpen(false)
   233→    } catch (e: unknown) {
   234→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   235→    } finally {
   236→      setCreatingOnboarding(false)
   237→    }
   238→  }
   239→
   240→  const toggleOnboardingStep = (stepId: string) => {
   241→    setOnboardingSteps(prev =>
   242→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   243→    )
   244→  }
   245→
   246→  // Stats
   247→  const allTasks = projects.flatMap(p => p.tasks || [])
   248→  const totalTasks = projects.reduce((sum, p) => sum + (p._count?.tasks || 0), 0)
   249→  const completedTasks = allTasks.filter(t => t.status === 'done').length
   250→  const inProgressTasks = allTasks.filter(t => t.status === 'in_progress').length
   251→
   252→  if (loading) return <PageSkeleton />
   253→
   254→  return (
   255→    <div className="p-4 md:p-6 space-y-6">
   256→      <div className="flex items-center gap-3">
   257→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   258→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   259→        </div>
   260→        <div>
   261→          <h1 className="text-2xl font-bold">My Projects</h1>
   262→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   263→        </div>
   264→      </div>
   265→
   266→      {/* Stats */}
   267→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   268→        <StatCard icon={FolderKanban} label="My Projects" value={projects.length} color="bg-emerald-500" />
   269→        <StatCard icon={ListTodo} label="Total Tasks" value={totalTasks} color="bg-amber-500" />
   270→        <StatCard icon={CheckCircle2} label="Completed" value={completedTasks} color="bg-teal-500" />
   271→        <StatCard icon={Clock} label="In Progress" value={inProgressTasks} color="bg-orange-500" />
   272→      </div>
   273→
   274→      {/* Project Cards */}
   275→      {projects.length === 0 ? (
   276→        <EmptyState
   277→          icon={FolderKanban}
   278→          title="No active projects"
   279→          description="You don't have any active projects yet."
   280→        />
   281→      ) : (
   282→        <div className="space-y-4">
   283→          {projects.map((project, i) => {
   284→            const projectTasks = expandedProjectId === project.id ? expandedTasks : []
   285→            const doneTasks = projectTasks.filter(t => t.status === 'done').length
   286→            const totalT = project._count?.tasks || 0
   287→            const pct = totalT > 0 ? Math.round((doneTasks / totalT) * 100) : 0
   288→            const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   289→
   290→            return (
   291→              <motion.div
   292→                key={project.id}
   293→                initial={{ opacity: 0, y: 12 }}
   294→                animate={{ opacity: 1, y: 0 }}
   295→                transition={{ delay: i * 0.05 }}
   296→              >
   297→                <Card
   298→                  className="cursor-pointer hover:shadow-md transition-shadow border-l-4"
   299→                  style={{ borderLeftColor: project.color }}
   300→                  onClick={() => expandProject(project.id)}
   301→                >
   302→                  <CardContent className="p-4 md:p-5">
   303→                    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   304→                      <div className="min-w-0 flex-1">
   305→                        <div className="flex items-center gap-2 mb-1">
   306→                          <h3 className="font-semibold text-base truncate">{project.name}</h3>
   307→                          <PriorityBadge priority={project.priority} />
   308→                        </div>
   309→                        <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   310→                      </div>
   311→                      <div className="flex items-center gap-3 shrink-0 text-sm">
   312→                        {project.dueDate && (
   313→                          <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   314→                            <Calendar className="w-3.5 h-3.5" />
   315→                            {formatDate(project.dueDate)}
   316→                          </span>
   317→                        )}
   318→                        <span className="text-muted-foreground flex items-center gap-1">
   319→                          <Users className="w-3.5 h-3.5" />
   320→                          {project._count?.members || 0}
   321→                        </span>
   322→                        <Button
   323→                          size="sm"
   324→                          variant="outline"
   325→                          className="h-7 gap-1.5 text-xs"
   326→                          onClick={(e) => { e.stopPropagation(); openOnboardingDialog(project.id, project.name) }}
   327→                        >
   328→                          <ClipboardCheck className="w-3.5 h-3.5" />
   329→                          Onboarding
   330→                        </Button>
   331→                        <span className="text-muted-foreground cursor-pointer">
   332→                          {expandedProjectId === project.id ? '▲' : '▼'}
   333→                        </span>
   334→                      </div>
   335→                    </div>
   336→
   337→                    {/* Progress bar */}
   338→                    <div className="mt-3 space-y-1">
   339→                      <div className="flex items-center justify-between text-xs text-muted-foreground">
   340→                        <span>{doneTasks}/{totalT} tasks completed</span>
   341→                        <span>{pct}%</span>
   342→                      </div>
   343→                      <Progress value={pct} className="h-2" />
   344→                    </div>
   345→                  </CardContent>
   346→                </Card>
   347→
   348→                {/* Expanded tasks table */}
   349→                <AnimatePresence>
   350→                  {expandedProjectId === project.id && (
   351→                    <motion.div
   352→                      initial={{ opacity: 0, height: 0 }}
   353→                      animate={{ opacity: 1, height: 'auto' }}
   354→                      exit={{ opacity: 0, height: 0 }}
   355→                      transition={{ duration: 0.2 }}
   356→                      className="overflow-hidden"
   357→                    >
   358→                      <Card className="mt-2 rounded-t-none">
   359→                        <CardContent className="p-0">
   360→                          {projectTasks.length === 0 ? (
   361→                            <div className="py-8 text-center text-sm text-muted-foreground">
   362→                              No tasks in this project yet.
   363→                            </div>
   364→                          ) : (
   365→                            <div className="overflow-x-auto">
   366→                              <table className="w-full text-sm">
   367→                                <thead>
   368→                                  <tr className="border-b bg-muted/50">
   369→                                    <th className="text-left p-3 font-medium text-muted-foreground">Task</th>
   370→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden md:table-cell">Status</th>
   371→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden sm:table-cell">Priority</th>
   372→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Due Date</th>
   373→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Assignees</th>
   374→                                  </tr>
   375→                                </thead>
   376→                                <tbody>
   377→                                  {projectTasks.map(task => (
   378→                                    <tr
   379→                                      key={task.id}
   380→                                      className="border-b last:border-0 hover:bg-muted/30 cursor-pointer transition-colors"
   381→                                      onClick={(e) => { e.stopPropagation(); openTaskDetail(task.id) }}
   382→                                    >
   383→                                      <td className="p-3">
   384→                                        <div className="font-medium">{task.title}</div>
   385→                                        {task.description && (
   386→                                          <div className="text-xs text-muted-foreground line-clamp-1 mt-0.5">{task.description}</div>
   387→                                        )}
   388→                                      </td>
   389→                                      <td className="p-3 hidden md:table-cell"><StatusBadge status={task.status} /></td>
   390→                                      <td className="p-3 hidden sm:table-cell"><PriorityBadge priority={task.priority} /></td>
   391→                                      <td className="p-3 hidden lg:table-cell text-muted-foreground text-xs">
   392→                                        {task.dueDate ? formatDate(task.dueDate) : '—'}
   393→                                      </td>
   394→                                      <td className="p-3 hidden lg:table-cell">
   395→                                        <div className="flex -space-x-2">
   396→                                          {task.assignees?.slice(0, 3).map(a => (
   397→                                            <Avatar key={a.id} className="w-6 h-6 border-2 border-background">
   398→                                              <AvatarFallback className="text-[10px] bg-muted">
   399→                                                {getInitials(a.user?.name || '?')}
   400→                                              </AvatarFallback>
   401→                                            </Avatar>
   402→                                          ))}
   403→                                          {(task.assignees?.length || 0) > 3 && (
   404→                                            <span className="w-6 h-6 rounded-full bg-muted flex items-center justify-center text-[10px] border-2 border-background">
   405→                                              +{(task.assignees?.length || 0) - 3}
   406→                                            </span>
   407→                                          )}
   408→                                        </div>
   409→                                      </td>
   410→                                    </tr>
   411→                                  ))}
   412→                                </tbody>
   413→                              </table>
   414→                            </div>
   415→                          )}
   416→                        </CardContent>
   417→                      </Card>
   418→                    </motion.div>
   419→                  )}
   420→                </AnimatePresence>
   421→              </motion.div>
   422→            )
   423→          })}
   424→        </div>
   425→      )}
   426→
   427→      {/* Create Onboarding Dialog */}
   428→      <Dialog open={onboardingDialogOpen} onOpenChange={setOnboardingDialogOpen}>
   429→        <DialogContent className="sm:max-w-lg">
   430→          <DialogHeader>
   431→            <DialogTitle className="flex items-center gap-2">
   432→              <ClipboardCheck className="w-5 h-5 text-primary" />
   433→              Create Onboarding
   434→            </DialogTitle>
   435→            <DialogDescription>Create an onboarding form for this project</DialogDescription>
   436→          </DialogHeader>
   437→          <div className="space-y-4 py-2">
   438→            <div className="space-y-2">
   439→              <Label>Title</Label>
   440→              <Input
   441→                placeholder="Onboarding form title"
   442→                value={onboardingTitle}
   443→                onChange={(e) => setOnboardingTitle(e.target.value)}
   444→              />
   445→            </div>
   446→            <div className="space-y-2">
   447→              <Label>Description (optional)</Label>
   448→              <Textarea
   449→                placeholder="Brief description of the onboarding..."
   450→                value={onboardingDesc}
   451→                onChange={(e) => setOnboardingDesc(e.target.value)}
   452→                rows={2}
   453→              />
   454→            </div>
   455→            <div className="space-y-2">
   456→              <div className="flex items-center justify-between">
   457→                <Label>Steps</Label>
   458→                <span className="text-xs text-muted-foreground">{onboardingSteps.length} / {ONBOARDING_STEPS.length} selected</span>
   459→              </div>
   460→              <div className="space-y-2 max-h-48 overflow-y-auto">
   461→                {ONBOARDING_STEPS.map((step) => {
   462→                  const isSelected = onboardingSteps.includes(step.id)
   463→                  return (
   464→                    <button
   465→                      key={step.id}
   466→                      type="button"
   467→                      onClick={() => toggleOnboardingStep(step.id)}
   468→                      className={cn(
   469→                        'w-full flex items-start gap-3 p-3 rounded-lg border text-left transition-colors',
   470→                        isSelected ? 'border-primary bg-primary/5' : 'border-border hover:bg-muted/50'
   471→                      )}
   472→                    >
   473→                      <div className={cn(
   474→                        'mt-0.5 h-5 w-5 rounded border-2 flex items-center justify-center shrink-0 transition-colors',
   475→                        isSelected ? 'border-primary bg-primary text-primary-foreground' : 'border-muted-foreground/30'
   476→                      )}>
   477→                        {isSelected && <Check className="w-3 h-3" />}
   478→                      </div>
   479→                      <div className="flex-1 min-w-0">
   480→                        <p className="text-sm font-medium">{step.label}</p>
   481→                        <p className="text-xs text-muted-foreground mt-0.5">{step.description}</p>
   482→                      </div>
   483→                    </button>
   484→                  )
   485→                })}
   486→              </div>
   487→            </div>
   488→          </div>
   489→          <DialogFooter>
   490→            <Button variant="outline" onClick={() => setOnboardingDialogOpen(false)}>Cancel</Button>
   491→            <Button
   492→              onClick={handleCreateOnboarding}
   493→              disabled={creatingOnboarding || !onboardingTitle.trim() || onboardingSteps.length === 0}
   494→              className="gap-2"
   495→            >
   496→              {creatingOnboarding && <Loader2 className="h-4 w-4 animate-spin" />}

   725→
   726→  useEffect(() => {

   729→      try {
   730→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   731→        const data = await res.json()
   732→        if (data.user) {

   735→          setPhone(data.user.phone || '')

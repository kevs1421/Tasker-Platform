    46→function formatElapsed(ms: number): string {
    47→  const totalSeconds = Math.floor(ms / 1000)
    48→  const h = Math.floor(totalSeconds / 3600)
    49→  const m = Math.floor((totalSeconds % 3600) / 60)
    50→  const s = totalSeconds % 60
    51→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    52→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    53→}
    54→
    55→function formatCountdown(ms: number): string {
    56→  if (ms <= 0) return '0:00'
    57→  const totalSeconds = Math.floor(ms / 1000)
    58→  const m = Math.floor(totalSeconds / 60)
    59→  const s = totalSeconds % 60
    60→  return `${m}:${String(s).padStart(2, '0')}`
    61→}

   724→function OfflineBanner() {
   725→  const [isOffline, setIsOffline] = useState(() => {
   726→    if (typeof navigator === 'undefined') return false
   727→    return !navigator.onLine
   728→  })
   729→
   730→  useEffect(() => {
   731→    const handleOnline = () => setIsOffline(false)
   732→    const handleOffline = () => setIsOffline(true)
   733→    window.addEventListener('online', handleOnline)
   734→    window.addEventListener('offline', handleOffline)
   735→    return () => {
   736→      window.removeEventListener('online', handleOnline)
   737→      window.removeEventListener('offline', handleOffline)
   738→    }
   739→  }, [])
   740→
   741→  if (!isOffline) return null
   742→
   743→  return (
   744→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   745→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   746→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   747→    </div>
   748→  )
   749→}

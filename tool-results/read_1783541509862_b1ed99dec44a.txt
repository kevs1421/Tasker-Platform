   101→   101→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   102→   102→  const visible = users.slice(0, max)
   103→   103→  const extra = users.length - max
   104→   104→  return (
   105→   105→    <div className="flex -space-x-2">
   106→   106→      {visible.map((u, i) =>
   107→   107→        u ? (
   108→   108→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   109→   109→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   110→   110→              {getInitials(u.name)}
   111→   111→            </AvatarFallback>
   112→   112→          </Avatar>
   113→   113→        ) : null
   114→   114→      )}
   115→   115→      {extra > 0 && (
   116→   116→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   117→   117→          +{extra}
   118→   118→        </div>
   119→   119→      )}
   120→   120→    </div>
   121→   121→  )

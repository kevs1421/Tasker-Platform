    98→const ONBOARDING_STEPS = [
    99→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   100→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   101→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   102→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   103→]
   104→
   105→export function ClientProjects() {
   106→  const { user } = useAuthStore()
   107→  const [projects, setProjects] = useState<ClientProject[]>([])
   108→  const [loading, setLoading] = useState(true)
   109→
   110→  // Upload files state
   111→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   112→  const [uploading, setUploading] = useState(false)
   113→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   114→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   115→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   116→  const fileInputRef = useRef<HTMLInputElement>(null)
   117→
   118→  // Onboarding creation state
   119→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   120→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   121→  const [onboardingTitle, setOnboardingTitle] = useState('')
   122→  const [onboardingDesc, setOnboardingDesc] = useState('')
   123→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   124→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   125→
   126→  const fetchProjects = useCallback(async () => {
   127→    if (!user?.id) return
   128→    try {
   129→      // Use the client-only endpoint that queries through ClientProject table
   130→      // This ensures only projects explicitly assigned to this client are returned
   131→      const res = await fetch(`/api/client/projects?clientId=${user.id}`)
   132→      const data = await res.json()
   133→      setProjects(data.projects || [])
   134→    } catch {
   135→      toast.error('Failed to load projects')
   136→    } finally {
   137→      setLoading(false)
   138→    }
   139→  }, [user?.id])
   140→
   141→  // Fetch uploaded files from localStorage
   142→  const fetchUploadedFiles = useCallback(() => {
   143→    if (!user) return
   144→    try {
   145→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   146→      if (stored) setUploadedFiles(JSON.parse(stored))
   147→    } catch { /* ignore */ }
   148→  }, [user])
   149→
   150→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   151→    if (!user) return
   152→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   153→  }, [user])
   154→
   155→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   156→
   157→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   158→  const handleFileUpload = async (files: FileList | null) => {
   159→    if (!files || files.length === 0) return
   160→    if (!uploadDialogProjectId) {
   161→      toast.error('No project selected')
   162→      return
   163→    }
   164→    setUploading(true)
   165→    const newFiles: typeof uploadedFiles = []
   166→    for (let i = 0; i < files.length; i++) {
   167→      const file = files[i]
   168→      try {
   169→        const formData = new FormData()
   170→        formData.append('file', file)
   171→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   172→        if (!res.ok) {
   173→          const d = await res.json()
   174→          throw new Error(d.error || 'Upload failed')
   175→        }
   176→        const data = await res.json()
   177→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   178→      } catch (e: unknown) {
   179→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   180→      }
   181→    }
   182→    if (newFiles.length > 0) {
   183→      const updated = [...uploadedFiles, ...newFiles]
   184→      setUploadedFiles(updated)
   185→      saveUploadedFiles(updated)
   186→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   187→    }
   188→    setUploading(false)
   189→    if (fileInputRef.current) fileInputRef.current.value = ''
   190→  }
   191→
   192→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   193→    e.preventDefault()
   194→    setUploadDragOver(null)
   195→    setUploadDialogProjectId(projectId)
   196→    handleFileUpload(e.dataTransfer.files)
   197→  }
   198→
   199→  const openUploadDialog = (projectId: string) => {
   200→    setUploadDialogProjectId(projectId)
   201→    setUploadDialogOpen(true)
   202→  }
   203→
   204→  const deleteUploadedFile = (idx: number) => {
   205→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   206→    setUploadedFiles(updated)
   207→    saveUploadedFiles(updated)
   208→    toast.success('File removed')
   209→  }
   210→
   211→  const getFileIcon = (type: string) => {
   212→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   213→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   214→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   215→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   216→    return <File className="w-4 h-4 text-muted-foreground" />
   217→  }
   218→
   219→  const formatFileSize = (bytes: number) => {
   220→    if (bytes < 1024) return `${bytes} B`
   221→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   222→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   223→  }
   224→
   225→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   226→    setOnboardingProjectId(projectId)
   227→    setOnboardingTitle(`Onboarding - ${projectName}`)
   228→    setOnboardingDesc('')
   229→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   230→    setOnboardingDialogOpen(true)
   231→  }
   232→
   233→  const handleCreateOnboarding = async () => {
   234→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   235→    setCreatingOnboarding(true)
   236→    try {
   237→      const stepsJson = JSON.stringify(
   238→        onboardingSteps.map(stepId => {
   239→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   240→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   241→        })
   242→      )
   243→      const res = await fetch('/api/onboarding', {
   244→        method: 'POST',
   245→        headers: { 'Content-Type': 'application/json' },
   246→        body: JSON.stringify({
   247→          title: onboardingTitle.trim(),
   248→          description: onboardingDesc.trim() || null,
   249→          stepsJson,
   250→          projectId: onboardingProjectId,
   251→          clientId: user?.id || null,
   252→          status: 'published',
   253→        }),
   254→      })
   255→      if (!res.ok) {
   256→        const d = await res.json()
   257→        throw new Error(d.error || 'Failed to create')
   258→      }
   259→      toast.success('Onboarding form created successfully')
   260→      setOnboardingDialogOpen(false)
   261→    } catch (e: unknown) {
   262→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   263→    } finally {
   264→      setCreatingOnboarding(false)
   265→    }
   266→  }
   267→
   268→  const toggleOnboardingStep = (stepId: string) => {
   269→    setOnboardingSteps(prev =>
   270→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   271→    )
   272→  }
   273→
   274→  if (loading) return <PageSkeleton />
   275→
   276→  return (
   277→    <div className="p-4 md:p-6 space-y-6">
   278→      {/* Page Header */}
   279→      <div className="flex items-center gap-3">
   280→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   281→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   282→        </div>
   283→        <div>
   284→          <h1 className="text-2xl font-bold">My Projects</h1>
   285→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   286→        </div>
   287→      </div>
   288→
   289→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   290→      <input
   291→        ref={fileInputRef}
   292→        type="file"
   293→        multiple
   294→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   295→        className="hidden"
   296→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   297→      />
   298→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   299→        <DialogContent className="sm:max-w-md">
   300→          <DialogHeader>
   301→            <DialogTitle className="flex items-center gap-2">
   302→              <Upload className="w-5 h-5 text-primary" />
   303→              Upload Files
   304→            </DialogTitle>
   305→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   306→          </DialogHeader>
   307→          <div
   308→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   309→            onDragLeave={() => setUploadDragOver(null)}
   310→            onDrop={(e) => {
   311→              e.preventDefault()
   312→              setUploadDragOver(null)
   313→              // Set the project context before handling files
   314→              const files = e.dataTransfer.files
   315→              if (files && files.length > 0) {
   316→                // Use a separate handler for dialog drops
   317→                void (async () => {
   318→                  setUploading(true)
   319→                  const newFiles: typeof uploadedFiles = []
   320→                  for (let i = 0; i < files.length; i++) {
   321→                    const file = files[i]
   322→                    try {
   323→                      const formData = new FormData()
   324→                      formData.append('file', file)
   325→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   326→                      if (!res.ok) {
   327→                        const d = await res.json()
   328→                        throw new Error(d.error || 'Upload failed')
   329→                      }
   330→                      const data = await res.json()
   331→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   332→                    } catch (err: unknown) {
   333→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   334→                    }
   335→                  }
   336→                  if (newFiles.length > 0) {
   337→                    const updated = [...uploadedFiles, ...newFiles]
   338→                    setUploadedFiles(updated)
   339→                    saveUploadedFiles(updated)
   340→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   341→                  }
   342→                  setUploading(false)
   343→                })()
   344→              }
   345→            }}
   346→            onClick={() => fileInputRef.current?.click()}
   347→            className={cn(
   348→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   349→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   350→            )}
   351→          >
   352→            {uploading ? (
   353→              <div className="flex flex-col items-center gap-2">
   354→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   355→                <p className="text-sm text-muted-foreground">Uploading...</p>
   356→              </div>
   357→            ) : (
   358→              <div className="flex flex-col items-center gap-2">
   359→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   360→                  <Upload className="w-5 h-5 text-muted-foreground" />
   361→                </div>
   362→                <div>
   363→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   364→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   365→                </div>
   366→              </div>
   367→            )}
   368→          </div>
   369→          <DialogFooter>
   370→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   371→          </DialogFooter>
   372→        </DialogContent>
   373→      </Dialog>
   374→
   375→      {/* ─── Project Progress Overview ──────────────────────────── */}
   376→      {projects.length > 0 && (
   377→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   378→          <Card>
   379→            <CardHeader className="pb-3">
   380→              <CardTitle className="text-base flex items-center gap-2">
   381→                <BarChart3 className="w-4 h-4 text-primary" />
   382→                Project Progress
   383→              </CardTitle>
   384→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   385→            </CardHeader>
   386→            <CardContent>
   387→              <div className="space-y-5">
   388→                {projects.map((project) => {
   389→                  const todo = project.taskStats?.todo || 0
   390→                  const inProgress = project.taskStats?.in_progress || 0
   391→                  const inReview = project.taskStats?.in_review || 0
   392→                  const completed = project.taskStats?.done || 0
   393→                  const total = todo + inProgress + inReview + completed
   394→                  if (total === 0) return null
   395→
   396→                  const todoPct = Math.round((todo / total) * 100)
   397→                  const inProgressPct = Math.round((inProgress / total) * 100)
   398→                  const inReviewPct = Math.round((inReview / total) * 100)
   399→                  const completedPct = Math.round((completed / total) * 100)
   400→
   401→                  return (
   402→                    <div key={project.id} className="space-y-2.5">
   403→                      <div className="flex items-center justify-between">
   404→                        <div className="flex items-center gap-2">
   405→                          <span className="w-3 h-3 rounded-full" style={{ backgroundColor: project.color }} />
   406→                          <span className="text-sm font-medium">{project.name}</span>
   407→                        </div>
   408→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   409→                      </div>
   410→                      {/* Stacked progress bar */}
   411→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   412→                        {todo > 0 && (
   413→                          <div
   414→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   415→                            style={{ width: `${todoPct}%` }}
   416→                            title={`Not Started: ${todo}`}
   417→                          />
   418→                        )}
   419→                        {inProgress > 0 && (
   420→                          <div
   421→                            className="bg-amber-500 transition-all duration-500"
   422→                            style={{ width: `${inProgressPct}%` }}
   423→                            title={`In Progress: ${inProgress}`}
   424→                          />
   425→                        )}
   426→                        {inReview > 0 && (
   427→                          <div
   428→                            className="bg-blue-500 transition-all duration-500"
   429→                            style={{ width: `${inReviewPct}%` }}
   430→                            title={`In Review: ${inReview}`}
   431→                          />
   432→                        )}
   433→                        {completed > 0 && (
   434→                          <div
   435→                            className="bg-emerald-500 transition-all duration-500"
   436→                            style={{ width: `${completedPct}%` }}
   437→                            title={`Completed: ${completed}`}
   438→                          />
   439→                        )}
   440→                      </div>
   441→                      {/* Legend */}
   442→                      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   443→                        <span className="flex items-center gap-1.5">
   444→                          <span className="w-2.5 h-2.5 rounded-sm bg-slate-400 dark:bg-slate-500" />
   445→                          Not Started ({todo})
   446→                        </span>
   447→                        <span className="flex items-center gap-1.5">
   448→                          <span className="w-2.5 h-2.5 rounded-sm bg-amber-500" />
   449→                          In Progress ({inProgress})
   450→                        </span>
   451→                        <span className="flex items-center gap-1.5">
   452→                          <span className="w-2.5 h-2.5 rounded-sm bg-blue-500" />
   453→                          In Review ({inReview})
   454→                        </span>
   455→                        <span className="flex items-center gap-1.5">
   456→                          <span className="w-2.5 h-2.5 rounded-sm bg-emerald-500" />
   457→                          Completed ({completed})
   458→                        </span>
   459→                      </div>
   460→                    </div>
   461→                  )
   462→                })}
   463→              </div>
   464→            </CardContent>
   465→          </Card>
   466→        </motion.div>
   467→      )}
   468→
   469→      {projects.length === 0 ? (
   470→        <EmptyState
   471→          icon={FolderKanban}
   472→          title="No active projects"
   473→          description="You don't have any active projects yet."
   474→        />
   475→      ) : (
   476→        <>
   477→          {/* ─── Project Cards ──────────────────────────────────── */}
   478→          <div className="space-y-4">
   479→            {projects.map((project, i) => {
   480→              const doneTasks = project.taskStats?.done || 0
   481→              const totalT = project._count?.tasks || 0
   482→              const pct = project.progress ?? 0
   483→              const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   484→
   485→              // Compute progress bar gradient colors
   486→              const barColor = pct >= 75 ? '#10b981' : pct >= 40 ? '#f59e0b' : '#ef4444'
   487→
   488→              // Files for this project only
   489→              const projectFiles = uploadedFiles.filter(f => f.projectId === project.id)
   490→
   491→              return (
   492→                <motion.div
   493→                  key={project.id}
   494→                  initial={{ opacity: 0, y: 12 }}
   495→                  animate={{ opacity: 1, y: 0 }}
   496→                  transition={{ delay: i * 0.05 }}
   497→                >
   498→                  <Card
   499→                    className="hover:shadow-md transition-all border-l-4"
   500→                    style={{ borderLeftColor: project.color }}
   501→                  >
   502→                    <CardContent className="p-4 md:p-5">
   503→                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   504→                        <div className="min-w-0 flex-1">
   505→                          <div className="flex items-center gap-2 mb-1">
   506→                            <h3 className="font-semibold text-base truncate">{project.name}</h3>
   507→                            <PriorityBadge priority={project.priority} />
   508→                          </div>

   699→// CLIENT PROFILE
   700→// ═══════════════════════════════════════════════════════════════════
   701→export function ClientProfile() {
   702→  const { user, login } = useAuthStore()
   703→  const [loading, setLoading] = useState(true)
   704→  const [saving, setSaving] = useState(false)
   705→  const [savingPassword, setSavingPassword] = useState(false)
   706→
   707→  const [name, setName] = useState('')
   708→  const [email, setEmail] = useState('')
   709→  const [phone, setPhone] = useState('')
   710→  const [company, setCompany] = useState('')
   711→
   712→  const [currentPassword, setCurrentPassword] = useState('')
   713→  const [newPassword, setNewPassword] = useState('')
   714→  const [confirmPassword, setConfirmPassword] = useState('')
   715→
   716→  useEffect(() => {
   717→    if (!user?.id) return
   718→    const fetchProfile = async () => {
   719→      try {
   720→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   721→        const data = await res.json()
   722→        if (data.user) {
   723→          setName(data.user.name || '')
   724→          setEmail(data.user.email || '')
   725→          setPhone(data.user.phone || '')
   726→          setCompany(data.user.company || '')
   727→        }
   728→      } catch {
   729→        toast.error('Failed to load profile')
   730→      } finally {
   731→        setLoading(false)
   732→      }
   733→    }
   734→    fetchProfile()
   735→  }, [user?.id])
   736→
   737→  const saveProfile = async () => {
   738→    if (!user?.id) return
   739→    setSaving(true)
   740→    try {
   741→      const res = await fetch('/api/user/profile', {
   742→        method: 'PATCH',
   743→        headers: { 'Content-Type': 'application/json' },
   744→        body: JSON.stringify({ userId: user.id, name, email, phone, company })
   745→      })
   746→      if (!res.ok) {
   747→        const data = await res.json()
   748→        throw new Error(data.error || 'Failed')
   749→      }
   750→      const data = await res.json()
   751→      toast.success('Profile updated successfully')
   752→      // Update user in store
   753→      if (data.user && user) {
   754→        login({ ...user, name: data.user.name, email: data.user.email, phone: data.user.phone, company: data.user.company })
   755→      }
   756→    } catch (e: unknown) {
   757→      const msg = e instanceof Error ? e.message : 'Failed to update profile'
   758→      toast.error(msg)
   759→    } finally {
   760→      setSaving(false)
   761→    }
   762→  }
   763→
   764→  const changePassword = async () => {
   765→    if (!user?.id) return
   766→    if (!currentPassword || !newPassword || !confirmPassword) {
   767→      toast.error('Please fill in all password fields')
   768→      return
   769→    }
   770→    if (newPassword !== confirmPassword) {
   771→      toast.error('New passwords do not match')
   772→      return
   773→    }
   774→    if (newPassword.length < 6) {
   775→      toast.error('Password must be at least 6 characters')
   776→      return
   777→    }
   778→    setSavingPassword(true)
   779→    try {
   780→      const res = await fetch('/api/user/profile', {
   781→        method: 'PATCH',
   782→        headers: { 'Content-Type': 'application/json' },
   783→        body: JSON.stringify({
   784→          userId: user.id,
   785→          currentPassword,
   786→          newPassword
   787→        })
   788→      })
   789→      if (!res.ok) {
   790→        const data = await res.json()
   791→        throw new Error(data.error || 'Failed')
   792→      }
   793→      toast.success('Password changed successfully')
   794→      setCurrentPassword('')
   795→      setNewPassword('')
   796→      setConfirmPassword('')
   797→    } catch (e: unknown) {
   798→      const msg = e instanceof Error ? e.message : 'Failed to change password'
   799→      toast.error(msg)
   800→    } finally {
   801→      setSavingPassword(false)
   802→    }
   803→  }
   804→
   805→  if (loading) return <PageSkeleton />
   806→
   807→  return (
   808→    <div className="p-4 md:p-6 space-y-6 max-w-2xl">
   809→      <div className="flex items-center gap-3">
   810→        <div className="w-10 h-10 rounded-lg bg-rose-100 dark:bg-rose-900/30 flex items-center justify-center">
   811→          <UserCircle className="w-5 h-5 text-rose-600 dark:text-rose-400" />
   812→        </div>
   813→        <div>
   814→          <h1 className="text-2xl font-bold">Profile Settings</h1>
   815→          <p className="text-sm text-muted-foreground">Manage your account information</p>
   816→        </div>
   817→      </div>
   818→
   819→      {/* Profile Card */}
   820→      <motion.div {...fadeSlide}>
   821→        <Card>
   822→          <CardHeader>
   823→            <div className="flex items-center justify-between">
   824→              <CardTitle className="text-lg">Personal Information</CardTitle>
   825→              <span className={`text-xs px-2.5 py-1 rounded-full font-medium ${getRoleBadgeClasses(user?.role || 'client')}`}>
   826→                Client
   827→              </span>
   828→            </div>
   829→            {user?.createdAt && (
   830→              <p className="text-sm text-muted-foreground">
   831→                Member since {formatDate(user.createdAt)}
   832→              </p>
   833→            )}
   834→          </CardHeader>
   835→          <CardContent className="space-y-4">
   836→            <div className="space-y-2">
   837→              <Label htmlFor="profile-name">Full Name</Label>
   838→              <Input
   839→                id="profile-name"
   840→                value={name}
   841→                onChange={e => setName(e.target.value)}
   842→                placeholder="Your full name"
   843→              />
   844→            </div>
   845→            <div className="space-y-2">
   846→              <Label htmlFor="profile-email">Email Address</Label>
   847→              <Input
   848→                id="profile-email"
   849→                type="email"
   850→                value={email}
   851→                onChange={e => setEmail(e.target.value)}
   852→                placeholder="your@email.com"
   853→              />
   854→            </div>
   855→            <div className="space-y-2">
   856→              <Label htmlFor="profile-phone">Phone Number</Label>
   857→              <Input
   858→                id="profile-phone"
   859→                type="tel"
   860→                value={phone}
   861→                onChange={e => setPhone(e.target.value)}
   862→                placeholder="+1 (555) 000-0000"
   863→              />
   864→            </div>
   865→            <div className="space-y-2">
   866→              <Label htmlFor="profile-company">Company</Label>
   867→              <Input
   868→                id="profile-company"
   869→                value={company}
   870→                onChange={e => setCompany(e.target.value)}
   871→                placeholder="Your company name"
   872→              />
   873→            </div>
   874→            <div className="pt-2">
   875→              <Button onClick={saveProfile} disabled={saving} className="w-full sm:w-auto">
   876→                {saving ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   877→                Save Changes
   878→              </Button>
   879→            </div>
   880→          </CardContent>
   881→        </Card>
   882→      </motion.div>
   883→
   884→      {/* Change Password */}
   885→      <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   886→        <Card>
   887→          <CardHeader>
   888→            <CardTitle className="text-lg flex items-center gap-2">
   889→              <Shield className="w-5 h-5 text-muted-foreground" />
   890→              Change Password
   891→            </CardTitle>
   892→            <p className="text-sm text-muted-foreground">
   893→              Update your password to keep your account secure
   894→            </p>
   895→          </CardHeader>
   896→          <CardContent className="space-y-4">
   897→            <div className="space-y-2">
   898→              <Label htmlFor="current-password">Current Password</Label>
   899→              <div className="relative">
   900→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   901→                <Input
   902→                  id="current-password"
   903→                  type="password"
   904→                  value={currentPassword}
   905→                  onChange={e => setCurrentPassword(e.target.value)}
   906→                  placeholder="Enter current password"
   907→                  className="pl-9"
   908→                />
   909→              </div>
   910→            </div>
   911→            <div className="space-y-2">
   912→              <Label htmlFor="new-password">New Password</Label>
   913→              <div className="relative">
   914→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   915→                <Input
   916→                  id="new-password"
   917→                  type="password"
   918→                  value={newPassword}
   919→                  onChange={e => setNewPassword(e.target.value)}
   920→                  placeholder="Enter new password"
   921→                  className="pl-9"
   922→                />
   923→              </div>
   924→            </div>
   925→            <div className="space-y-2">
   926→              <Label htmlFor="confirm-password">Confirm New Password</Label>

   928→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   929→                <Input
   930→                  id="confirm-password"
   931→                  type="password"
   932→                  value={confirmPassword}
   933→                  onChange={e => setConfirmPassword(e.target.value)}
   934→                  placeholder="Confirm new password"
   935→                  className="pl-9"
   936→                />
   937→              </div>
   938→            </div>
   939→            <div className="pt-2">
   940→              <Button onClick={changePassword} disabled={savingPassword} variant="outline" className="w-full sm:w-auto">
   941→                {savingPassword ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Shield className="w-4 h-4 mr-2" />}
   942→                Change Password
   943→              </Button>
   944→            </div>

   946→        </Card>
   947→      </motion.div>
   948→    </div>

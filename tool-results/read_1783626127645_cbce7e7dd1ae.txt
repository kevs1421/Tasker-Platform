   104→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   105→  const visible = users.slice(0, max)
   106→  const extra = users.length - max
   107→  return (
   108→    <div className="flex -space-x-2">
   109→      {visible.map((u, i) =>
   110→        u ? (
   111→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   112→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   113→              {getInitials(u.name)}
   114→            </AvatarFallback>
   115→          </Avatar>
   116→        ) : null
   117→      )}
   118→      {extra > 0 && (
   119→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   120→          +{extra}
   121→        </div>
   122→      )}
   123→    </div>
   124→  )
   125→}
   126→
   127→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   128→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   129→  if (!user) return null
   130→  return (
   131→    <Avatar className={sizeClasses}>
   132→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   133→        {getInitials(user.name)}
   134→      </AvatarFallback>
   135→    </Avatar>
   136→  )
   137→}

   156→
   157→export function TeamProjects() {
   158→  const { user, setSelectedProjectId } = useAuthStore()
   159→  const [projects, setProjects] = useState<Project[]>([])
   160→  const [loading, setLoading] = useState(true)
   161→  const [expandedId, setExpandedId] = useState<string | null>(null)
   162→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   163→  const [expandedLoading, setExpandedLoading] = useState(false)
   164→  const [taskFilter, setTaskFilter] = useState<string>('all')
   165→  const [createOpen, setCreateOpen] = useState(false)
   166→  const [createLoading, setCreateLoading] = useState(false)
   167→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   168→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   169→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   170→
   171→  // Create project form
   172→  const [projName, setProjName] = useState('')
   173→  const [projDesc, setProjDesc] = useState('')
   174→  const [projPriority, setProjPriority] = useState('medium')
   175→  const [projDueDate, setProjDueDate] = useState<Date>()
   176→  const [projColor, setProjColor] = useState(PROJECT_COLORS[0])
   177→  const [projCalendarOpen, setProjCalendarOpen] = useState(false)
   178→
   179→  // Create task form
   180→  const [taskTitle, setTaskTitle] = useState('')
   181→  const [taskDesc, setTaskDesc] = useState('')
   182→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   183→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   184→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   185→  const [taskTags, setTaskTags] = useState('')
   186→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   187→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   188→
   189→  const fetchProjects = useCallback(async () => {
   190→    try {
   191→      const res = await fetch('/api/projects')
   192→      const data = await res.json()
   193→      const filtered = (data.projects || []).filter((p: Project) =>
   194→        p.members?.some((m) => m.userId === user?.id)
   195→      )
   196→      setProjects(filtered)
   197→    } catch {
   198→      toast.error('Failed to load projects')
   199→    } finally {
   200→      setLoading(false)
   201→    }
   202→  }, [user])
   203→
   204→  useEffect(() => { fetchProjects() }, [fetchProjects])
   205→
   206→  const handleExpand = async (projectId: string) => {
   207→    if (expandedId === projectId) {
   208→      setExpandedId(null)
   209→      setExpandedTasks([])
   210→      return
   211→    }
   212→    setExpandedId(projectId)
   213→    setTaskFilter('all')
   214→    setExpandedLoading(true)
   215→    try {
   216→      const res = await fetch(`/api/projects/${projectId}`)
   217→      const data = await res.json()
   218→      setExpandedTasks(data.project?.tasks || [])
   219→    } catch {
   220→      toast.error('Failed to load tasks')
   221→    } finally {
   222→      setExpandedLoading(false)
   223→    }
   224→  }
   225→
   226→  const handleCreateProject = async () => {
   227→    if (!projName.trim()) { toast.error('Project name is required'); return }
   228→    setCreateLoading(true)
   229→    try {
   230→      await fetch('/api/projects', {
   231→        method: 'POST',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify({
   234→          name: projName, description: projDesc || null, priority: projPriority,
   235→          dueDate: projDueDate ? projDueDate.toISOString() : null, color: projColor,
   236→        }),
   237→      })
   238→      toast.success('Project created')
   239→      setCreateOpen(false)
   240→      setProjName(''); setProjDesc(''); setProjPriority('medium')
   241→      setProjDueDate(undefined); setProjColor(PROJECT_COLORS[0])
   242→      fetchProjects()
   243→    } catch {
   244→      toast.error('Failed to create project')
   245→    } finally {
   246→      setCreateLoading(false)
   247→    }
   248→  }
   249→
   250→  const handleCreateTask = async () => {
   251→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   252→    setCreateTaskLoading(true)
   253→    try {
   254→      await fetch('/api/tasks', {
   255→        method: 'POST',
   256→        headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({
   258→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   259→          priority: taskPriority, projectId: createTaskProjectId,
   260→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   261→          tags: taskTags, assigneeId: taskAssignee || null,
   262→        }),
   263→      })
   264→      toast.success('Task created')
   265→      setCreateTaskOpen(false)
   266→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   267→      setTaskPriority('medium'); setTaskDueDate(undefined)
   268→      setTaskTags(''); setTaskAssignee('')
   269→      if (expandedId === createTaskProjectId) handleExpand(createTaskProjectId)
   270→      else handleExpand(createTaskProjectId)
   271→    } catch {
   272→      toast.error('Failed to create task')
   273→    } finally {
   274→      setCreateTaskLoading(false)
   275→    }
   276→  }
   277→
   278→  const openCreateTaskForProject = (projectId: string) => {
   279→    setCreateTaskProjectId(projectId)
   280→    setCreateTaskOpen(true)
   281→  }
   282→
   283→  const getProjectTaskStats = (projectId: string) => {
   284→    if (expandedId !== projectId) return null
   285→    const total = expandedTasks.length
   286→    const done = expandedTasks.filter((t) => t.status === 'done').length
   287→    return { total, done }
   288→  }
   289→
   290→  const filteredTasks = taskFilter === 'all'
   291→    ? expandedTasks
   292→    : expandedTasks.filter((t) => t.status === taskFilter)
   293→
   294→  if (loading) {
   295→    return (
   296→      <div>
   297→        <div className="flex items-center justify-between mb-6">
   298→          <Skeleton className="h-8 w-48" />
   299→          <Skeleton className="h-10 w-36" />
   300→        </div>
   301→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   302→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   303→        </div>
   304→      </div>
   305→    )
   306→  }
   307→
   308→  return (
   309→    <div className="space-y-6">
   310→      {/* Header */}
   311→      <div className="flex items-center justify-between">
   312→        <div>
   313→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   314→          <p className="text-sm text-muted-foreground mt-1">
   315→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   316→          </p>
   317→        </div>
   318→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   319→          <DialogTrigger asChild>
   320→            <Button className="gap-2">
   321→              <Plus className="h-4 w-4" /> Create Project
   322→            </Button>
   323→          </DialogTrigger>
   324→          <DialogContent className="sm:max-w-lg">
   325→            <DialogHeader>
   326→              <DialogTitle>Create New Project</DialogTitle>
   327→              <DialogDescription>Add a new project to your workspace.</DialogDescription>
   328→            </DialogHeader>
   329→            <div className="space-y-4 py-2">
   330→              <div className="space-y-2">
   331→                <Label>Project Name</Label>
   332→                <Input placeholder="Enter project name" value={projName} onChange={(e) => setProjName(e.target.value)} />
   333→              </div>
   334→              <div className="space-y-2">
   335→                <Label>Description</Label>
   336→                <Textarea placeholder="Describe the project..." value={projDesc} onChange={(e) => setProjDesc(e.target.value)} rows={3} />
   337→              </div>
   338→              <div className="grid grid-cols-2 gap-4">
   339→                <div className="space-y-2">
   340→                  <Label>Priority</Label>
   341→                  <Select value={projPriority} onValueChange={setProjPriority}>
   342→                    <SelectTrigger><SelectValue /></SelectTrigger>
   343→                    <SelectContent>
   344→                      <SelectItem value="low">Low</SelectItem>
   345→                      <SelectItem value="medium">Medium</SelectItem>
   346→                      <SelectItem value="high">High</SelectItem>
   347→                      <SelectItem value="urgent">Urgent</SelectItem>
   348→                    </SelectContent>
   349→                  </Select>
   350→                </div>
   351→                <div className="space-y-2">
   352→                  <Label>Due Date</Label>
   353→                  <Popover open={projCalendarOpen} onOpenChange={setProjCalendarOpen}>
   354→                    <PopoverTrigger asChild>
   355→                      <Button variant="outline" className="w-full justify-start text-left font-normal gap-2">
   356→                        <CalendarIcon className="h-4 w-4" />
   357→                        {projDueDate ? format(projDueDate, 'PPP') : 'Pick a date'}
   358→                      </Button>
   359→                    </PopoverTrigger>
   360→                    <PopoverContent className="w-auto p-0" align="start">
   361→                      <Calendar mode="single" selected={projDueDate} onSelect={(d) => { setProjDueDate(d); setProjCalendarOpen(false) }} />
   362→                    </PopoverContent>
   363→                  </Popover>
   364→                </div>
   365→              </div>
   366→              <div className="space-y-2">
   367→                <Label>Color</Label>
   368→                <div className="flex flex-wrap gap-2">
   369→                  {PROJECT_COLORS.map((c) => (
   370→                    <button
   371→                      key={c}
   372→                      className={cn(
   373→                        'h-8 w-8 rounded-full transition-all ring-2',
   374→                        projColor === c ? 'ring-offset-2 ring-offset-background scale-110' : 'ring-transparent hover:scale-105',
   375→                      )}
   376→                      style={{ backgroundColor: c, ringColor: c }}
   377→                      onClick={() => setProjColor(c)}
   378→                    />
   379→                  ))}
   380→                </div>
   381→              </div>
   382→            </div>
   383→            <DialogFooter>
   384→              <Button variant="outline" onClick={() => setCreateOpen(false)}>Cancel</Button>
   385→              <Button onClick={handleCreateProject} disabled={createLoading} className="gap-2">
   386→                {createLoading && <Loader2 className="h-4 w-4 animate-spin" />}
   387→                Create Project
   388→              </Button>
   389→            </DialogFooter>
   390→          </DialogContent>
   391→        </Dialog>
   392→      </div>
   393→
   394→      {/* Project Grid */}
   395→      {projects.length === 0 ? (
   396→        <EmptyState
   397→          icon={FolderKanban}
   398→          title="No projects yet"
   399→          description="You haven't been assigned to any projects. Create one or ask an admin to assign you."
   400→        />
   401→      ) : (
   402→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   403→          <AnimatePresence>
   404→            {projects.map((project, idx) => {
   405→              const totalTasks = project._count?.tasks || 0
   406→              const members = project.members || []
   407→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   408→              const pConfig = getPriorityConfig(project.priority)
   409→              const isExpanded = expandedId === project.id
   410→              const stats = getProjectTaskStats(project.id)
   411→
   412→              return (
   413→                <motion.div
   414→                  key={project.id}
   415→                  initial={{ opacity: 0, y: 20 }}
   416→                  animate={{ opacity: 1, y: 0 }}
   417→                  transition={{ delay: idx * 0.05 }}
   418→                >
   419→                  <Card
   420→                    className={cn(
   421→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   422→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   423→                    )}
   424→                    onClick={() => handleExpand(project.id)}
   425→                  >
   426→                    <div className="flex">
   427→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   428→                      <CardContent className="flex-1 p-4">
   429→                        <div className="flex items-start justify-between mb-2">
   430→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   431→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   432→                            {pConfig.label}
   433→                          </Badge>
   434→                        </div>
   435→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   436→                          {project.description || 'No description'}
   437→                        </p>
   438→                        <Progress
   439→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   440→                          className="h-1.5 mb-3"
   441→                        />
   442→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   443→                          <div className="flex items-center gap-3">
   444→                            <span className="flex items-center gap-1">
   445→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   446→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   447→                            </span>
   448→                            <span className="flex items-center gap-1">
   449→                              <Users className="h-3 w-3" />
   450→                              {members.length}
   451→                            </span>
   452→                          </div>
   453→                          {project.dueDate && (
   454→                            <span className={cn(
   455→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   456→                                ? 'text-red-500' : '',
   457→                            )}>
   458→                              {formatDate(project.dueDate)}
   459→                            </span>
   460→                          )}
   461→                        </div>
   462→                        {memberUsers.length > 0 && (
   463→                          <div className="mt-3">
   464→                            <AvatarStack users={memberUsers} max={4} />
   465→                          </div>
   466→                        )}
   467→                      </CardContent>
   468→                    </div>
   469→                  </Card>
   470→
   471→                  {/* Expanded Task List */}
   472→                  <AnimatePresence>
   473→                    {isExpanded && (
   474→                      <motion.div
   475→                        initial={{ opacity: 0, height: 0 }}
   476→                        animate={{ opacity: 1, height: 'auto' }}
   477→                        exit={{ opacity: 0, height: 0 }}
   478→                        transition={{ duration: 0.2 }}
   479→                        className="overflow-hidden"
   480→                        onClick={(e) => e.stopPropagation()}
   481→                      >
   482→                        <Card className="mt-2 p-4 space-y-4">
   483→                          <div className="flex items-center justify-between">
   484→                            <div className="flex items-center gap-2">
   485→                              <h4 className="font-semibold text-sm">Tasks</h4>
   486→                              <div className="flex gap-1">
   487→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   488→                                  <Button
   489→                                    key={s}
   490→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   491→                                    size="sm"
   492→                                    className="h-7 text-xs px-2"
   493→                                    onClick={() => setTaskFilter(s)}
   494→                                  >
   495→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   496→                                  </Button>
   497→                                ))}
   498→                              </div>
   499→                            </div>
   500→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   501→                              <Plus className="h-3 w-3" /> Add Task
   502→                            </Button>
   503→                          </div>

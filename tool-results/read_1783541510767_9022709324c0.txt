    66→    66→function FileUploadZone({
    67→    67→  label,
    68→    68→  required,
    69→    69→  accept,
    70→    70→  file,
    71→    71→  onUpload,
    72→    72→  onRemove,
    73→    73→  icon: Icon = Upload,
    74→    74→  description,
    75→    75→}: {
    76→    76→  label: string; required?: boolean; accept?: string
    77→    77→  file: UploadedFile | null; onUpload: (file: UploadedFile) => void; onRemove: () => void
    78→    78→  icon?: React.ElementType; description?: string
    79→    79→}) {
    80→    80→  const [uploading, setUploading] = useState(false)
    81→    81→  const inputRef = useRef<HTMLInputElement>(null)
    82→    82→
    83→    83→  const handleFile = async (fileObj: File) => {
    84→    84→    setUploading(true)
    85→    85→    try {
    86→    86→      const formData = new FormData()
    87→    87→      formData.append('file', fileObj)
    88→    88→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
    89→    89→      if (!res.ok) {
    90→    90→        const d = await res.json()
    91→    91→        throw new Error(d.error || 'Upload failed')
    92→    92→      }
    93→    93→      const data = await res.json()
    94→    94→      onUpload({
    95→    95→        url: data.url,
    96→    96→        originalName: data.originalName,
    97→    97→        filename: data.filename,
    98→    98→        size: data.size,
    99→    99→        type: data.type,
   100→   100→      })
   101→   101→      toast.success('File uploaded')
   102→   102→    } catch (e: unknown) {
   103→   103→      toast.error(e instanceof Error ? e.message : 'Upload failed')
   104→   104→    } finally {
   105→   105→      setUploading(false)
   106→   106→    }
   107→   107→  }
   108→   108→
   109→   109→  const handleDrop = (e: React.DragEvent) => {
   110→   110→    e.preventDefault()
   111→   111→    const f = e.dataTransfer.files[0]
   112→   112→    if (f) handleFile(f)
   113→   113→  }
   114→   114→
   115→   115→  const formatSize = (bytes: number) => {
   116→   116→    if (bytes < 1024) return `${bytes} B`
   117→   117→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   118→   118→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   119→   119→  }
   120→   120→
   121→   121→  return (
   122→   122→    <div className="space-y-2">
   123→   123→      <Label className="text-sm font-medium">
   124→   124→        {label} {required && <span className="text-red-500">*</span>}
   125→   125→      </Label>
   126→   126→      {file ? (
   127→   127→        <div className="flex items-center gap-3 p-3 rounded-lg border bg-muted/30">
   128→   128→          <div className="w-10 h-10 rounded-lg bg-primary/10 flex items-center justify-center flex-shrink-0">
   129→   129→            <FileText className="w-5 h-5 text-primary" />
   130→   130→          </div>
   131→   131→          <div className="flex-1 min-w-0">
   132→   132→            <p className="text-sm font-medium truncate">{file.originalName}</p>
   133→   133→            <p className="text-xs text-muted-foreground">{formatSize(file.size)}</p>
   134→   134→          </div>
   135→   135→          <Button
   136→   136→            type="button" variant="ghost" size="icon" className="h-8 w-8 text-muted-foreground hover:text-destructive"
   137→   137→            onClick={onRemove}
   138→   138→          >
   139→   139→            <X className="h-4 w-4" />
   140→   140→          </Button>
   141→   141→        </div>
   142→   142→      ) : (
   143→   143→        <div
   144→   144→          className="border-2 border-dashed rounded-lg p-6 flex flex-col items-center justify-center gap-2 cursor-pointer hover:border-primary/50 hover:bg-muted/20 transition-colors"
   145→   145→          onClick={() => inputRef.current?.click()}
   146→   146→          onDragOver={(e) => e.preventDefault()}
   147→   147→          onDrop={handleDrop}
   148→   148→        >
   149→   149→          {uploading ? (
   150→   150→            <Loader2 className="w-8 h-8 text-primary animate-spin" />
   151→   151→          ) : (
   152→   152→            <>
   153→   153→              <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center">
   154→   154→                <Icon className="w-6 h-6 text-muted-foreground" />
   155→   155→              </div>
   156→   156→              <div className="text-center">
   157→   157→                <p className="text-sm font-medium text-muted-foreground">Click to upload</p>
   158→   158→                {description && <p className="text-xs text-muted-foreground mt-1">{description}</p>}
   159→   159→              </div>
   160→   160→            </>
   161→   161→          )}
   162→   162→          <input
   163→   163→            ref={inputRef}
   164→   164→            type="file"
   165→   165→            className="hidden"
   166→   166→            accept={accept}
   167→   167→            onChange={(e) => {
   168→   168→              const f = e.target.files?.[0]
   169→   169→              if (f) handleFile(f)
   170→   170→              e.target.value = ''
   171→   171→            }}
   172→   172→          />
   173→   173→        </div>
   174→   174→      )}
   175→   175→    </div>
   176→   176→  )
   177→   177→}

   181→   181→function StepCard({
   182→   182→  step, index, currentStep, isCompleted, onClick,
   183→   183→}: {
   184→   184→  step: string; index: number; currentStep: number; isCompleted: boolean; onClick: () => void
   185→   185→}) {
   186→   186→  const Icon = STEP_ICONS[index]
   187→   187→  const isCurrent = index === currentStep
   188→   188→  const isClickable = isCompleted || index === currentStep
   189→   189→
   190→   190→  return (
   191→   191→    <button
   192→   192→      type="button"
   193→   193→      onClick={isClickable ? onClick : undefined}
   194→   194→      disabled={!isClickable}
   195→   195→      className={cn(
   196→   196→        'flex-1 min-w-0 flex flex-col items-center gap-1.5 py-3 px-2 rounded-xl transition-all duration-200 border',
   197→   197→        isCompleted && 'bg-emerald-50 border-emerald-200 dark:bg-emerald-900/20 dark:border-emerald-800',
   198→   198→        isCurrent && !isCompleted && 'bg-primary/10 border-primary/30',
   199→   199→        !isCurrent && !isCompleted && 'bg-card border-transparent opacity-50',
   200→   200→        isClickable && 'cursor-pointer hover:shadow-sm',
   201→   201→        !isClickable && 'cursor-not-allowed',
   202→   202→      )}
   203→   203→    >
   204→   204→      <div className={cn(
   205→   205→        'w-10 h-10 rounded-full flex items-center justify-center transition-colors',
   206→   206→        isCompleted && 'bg-emerald-500 text-white',
   207→   207→        isCurrent && !isCompleted && 'bg-primary text-primary-foreground',
   208→   208→        !isCurrent && !isCompleted && 'bg-muted text-muted-foreground',
   209→   209→      )}>
   210→   210→        {isCompleted ? <Check className="w-5 h-5" /> : <Icon className="w-5 h-5" />}
   211→   211→      </div>
   212→   212→      <span className={cn(
   213→   213→        'text-[11px] md:text-xs font-medium text-center leading-tight',
   214→   214→        isCompleted && 'text-emerald-700 dark:text-emerald-400',
   215→   215→        isCurrent && !isCompleted && 'text-primary',
   216→   216→        !isCurrent && !isCompleted && 'text-muted-foreground',
   217→   217→      )}>
   218→   218→        {step}
   219→   219→      </span>
   220→   220→    </button>
   221→   221→  )
   222→   222→}

    99→const ONBOARDING_STEPS = [
   100→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   101→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   102→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   103→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   104→]
   105→
   106→export function ClientProjects() {
   107→  const { user } = useAuthStore()
   108→  const [projects, setProjects] = useState<ClientProject[]>([])
   109→  const [loading, setLoading] = useState(true)
   110→
   111→  // Upload files state
   112→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   113→  const [uploading, setUploading] = useState(false)
   114→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   115→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   116→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   117→  const fileInputRef = useRef<HTMLInputElement>(null)
   118→
   119→  // Onboarding creation state
   120→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   121→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   122→  const [onboardingTitle, setOnboardingTitle] = useState('')
   123→  const [onboardingDesc, setOnboardingDesc] = useState('')
   124→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   125→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   126→
   127→  const fetchProjects = useCallback(async () => {
   128→    if (!user?.id) return
   129→    try {
   130→      // Use the client-only endpoint that queries through ClientProject table
   131→      // This ensures only projects explicitly assigned to this client are returned
   132→      const res = await fetch(`/api/client/projects?clientId=${user.id}`)
   133→      const data = await res.json()
   134→      setProjects(data.projects || [])
   135→    } catch {
   136→      toast.error('Failed to load projects')
   137→    } finally {
   138→      setLoading(false)
   139→    }
   140→  }, [user?.id])
   141→
   142→  // Fetch uploaded files from localStorage
   143→  const fetchUploadedFiles = useCallback(() => {
   144→    if (!user) return
   145→    try {
   146→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   147→      if (stored) setUploadedFiles(JSON.parse(stored))
   148→    } catch { /* ignore */ }
   149→  }, [user])
   150→
   151→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   152→    if (!user) return
   153→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   154→  }, [user])
   155→
   156→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   157→
   158→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   159→  const handleFileUpload = async (files: FileList | null) => {
   160→    if (!files || files.length === 0) return
   161→    if (!uploadDialogProjectId) {
   162→      toast.error('No project selected')
   163→      return
   164→    }
   165→    setUploading(true)
   166→    const newFiles: typeof uploadedFiles = []
   167→    for (let i = 0; i < files.length; i++) {
   168→      const file = files[i]
   169→      try {
   170→        const formData = new FormData()
   171→        formData.append('file', file)
   172→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   173→        if (!res.ok) {
   174→          const d = await res.json()
   175→          throw new Error(d.error || 'Upload failed')
   176→        }
   177→        const data = await res.json()
   178→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   179→      } catch (e: unknown) {
   180→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   181→      }
   182→    }
   183→    if (newFiles.length > 0) {
   184→      const updated = [...uploadedFiles, ...newFiles]
   185→      setUploadedFiles(updated)
   186→      saveUploadedFiles(updated)
   187→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   188→    }
   189→    setUploading(false)
   190→    if (fileInputRef.current) fileInputRef.current.value = ''
   191→  }
   192→
   193→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   194→    e.preventDefault()
   195→    setUploadDragOver(null)
   196→    setUploadDialogProjectId(projectId)
   197→    handleFileUpload(e.dataTransfer.files)
   198→  }
   199→
   200→  const openUploadDialog = (projectId: string) => {
   201→    setUploadDialogProjectId(projectId)
   202→    setUploadDialogOpen(true)
   203→  }
   204→
   205→  const deleteUploadedFile = (idx: number) => {
   206→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   207→    setUploadedFiles(updated)
   208→    saveUploadedFiles(updated)
   209→    toast.success('File removed')
   210→  }
   211→
   212→  const getFileIcon = (type: string) => {
   213→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   214→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   215→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   216→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   217→    return <File className="w-4 h-4 text-muted-foreground" />
   218→  }
   219→
   220→  const formatFileSize = (bytes: number) => {
   221→    if (bytes < 1024) return `${bytes} B`
   222→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   223→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   224→  }
   225→
   226→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   227→    setOnboardingProjectId(projectId)
   228→    setOnboardingTitle(`Onboarding - ${projectName}`)
   229→    setOnboardingDesc('')
   230→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   231→    setOnboardingDialogOpen(true)
   232→  }
   233→
   234→  const handleCreateOnboarding = async () => {
   235→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   236→    setCreatingOnboarding(true)
   237→    try {
   238→      const stepsJson = JSON.stringify(
   239→        onboardingSteps.map(stepId => {
   240→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   241→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   242→        })
   243→      )
   244→      const res = await fetch('/api/onboarding', {
   245→        method: 'POST',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({
   248→          title: onboardingTitle.trim(),
   249→          description: onboardingDesc.trim() || null,
   250→          stepsJson,
   251→          projectId: onboardingProjectId,
   252→          clientId: user?.id || null,
   253→          status: 'published',
   254→        }),
   255→      })
   256→      if (!res.ok) {
   257→        const d = await res.json()
   258→        throw new Error(d.error || 'Failed to create')
   259→      }
   260→      toast.success('Onboarding form created successfully')
   261→      setOnboardingDialogOpen(false)
   262→    } catch (e: unknown) {
   263→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   264→    } finally {
   265→      setCreatingOnboarding(false)
   266→    }
   267→  }
   268→
   269→  const toggleOnboardingStep = (stepId: string) => {
   270→    setOnboardingSteps(prev =>
   271→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   272→    )
   273→  }
   274→
   275→  if (loading) return <PageSkeleton />
   276→
   277→  return (
   278→    <div className="p-4 md:p-6 space-y-6">
   279→      {/* Page Header */}
   280→      <div className="flex items-center gap-3">
   281→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   282→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   283→        </div>
   284→        <div>
   285→          <h1 className="text-2xl font-bold">My Projects</h1>
   286→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   287→        </div>
   288→      </div>
   289→
   290→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   291→      <input
   292→        ref={fileInputRef}
   293→        type="file"
   294→        multiple
   295→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   296→        className="hidden"
   297→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   298→      />
   299→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   300→        <DialogContent className="sm:max-w-md">
   301→          <DialogHeader>
   302→            <DialogTitle className="flex items-center gap-2">
   303→              <Upload className="w-5 h-5 text-primary" />
   304→              Upload Files
   305→            </DialogTitle>
   306→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   307→          </DialogHeader>
   308→          <div
   309→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   310→            onDragLeave={() => setUploadDragOver(null)}
   311→            onDrop={(e) => {
   312→              e.preventDefault()
   313→              setUploadDragOver(null)
   314→              // Set the project context before handling files
   315→              const files = e.dataTransfer.files
   316→              if (files && files.length > 0) {
   317→                // Use a separate handler for dialog drops
   318→                void (async () => {
   319→                  setUploading(true)
   320→                  const newFiles: typeof uploadedFiles = []
   321→                  for (let i = 0; i < files.length; i++) {
   322→                    const file = files[i]
   323→                    try {
   324→                      const formData = new FormData()
   325→                      formData.append('file', file)
   326→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   327→                      if (!res.ok) {
   328→                        const d = await res.json()
   329→                        throw new Error(d.error || 'Upload failed')
   330→                      }
   331→                      const data = await res.json()
   332→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   333→                    } catch (err: unknown) {
   334→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   335→                    }
   336→                  }
   337→                  if (newFiles.length > 0) {
   338→                    const updated = [...uploadedFiles, ...newFiles]
   339→                    setUploadedFiles(updated)
   340→                    saveUploadedFiles(updated)
   341→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   342→                  }
   343→                  setUploading(false)
   344→                })()
   345→              }
   346→            }}
   347→            onClick={() => fileInputRef.current?.click()}
   348→            className={cn(
   349→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   350→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   351→            )}
   352→          >
   353→            {uploading ? (
   354→              <div className="flex flex-col items-center gap-2">
   355→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   356→                <p className="text-sm text-muted-foreground">Uploading...</p>
   357→              </div>
   358→            ) : (
   359→              <div className="flex flex-col items-center gap-2">
   360→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   361→                  <Upload className="w-5 h-5 text-muted-foreground" />
   362→                </div>
   363→                <div>
   364→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   365→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   366→                </div>
   367→              </div>
   368→            )}
   369→          </div>
   370→          <DialogFooter>
   371→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   372→          </DialogFooter>
   373→        </DialogContent>
   374→      </Dialog>
   375→
   376→      {/* ─── Project Progress Overview ──────────────────────────── */}
   377→      {projects.length > 0 && (
   378→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   379→          <Card>
   380→            <CardHeader className="pb-3">
   381→              <CardTitle className="text-base flex items-center gap-2">
   382→                <BarChart3 className="w-4 h-4 text-primary" />
   383→                Project Progress
   384→              </CardTitle>
   385→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   386→            </CardHeader>
   387→            <CardContent>
   388→              <div className="space-y-5">
   389→                {projects.map((project) => {
   390→                  const todo = project.taskStats?.todo || 0
   391→                  const inProgress = project.taskStats?.in_progress || 0
   392→                  const completed = project.taskStats?.done || 0
   393→                  const total = todo + inProgress + completed
   394→                  if (total === 0) return null
   395→
   396→                  const todoPct = Math.round((todo / total) * 100)
   397→                  const inProgressPct = Math.round((inProgress / total) * 100)
   398→                  const completedPct = Math.round((completed / total) * 100)
   399→
   400→                  return (
   401→                    <div key={project.id} className="space-y-2.5">
   402→                      <div className="flex items-center justify-between">
   403→                        <div className="flex items-center gap-2">
   404→                          <span className="w-3 h-3 rounded-full" style={{ backgroundColor: project.color }} />
   405→                          <span className="text-sm font-medium">{project.name}</span>
   406→                        </div>
   407→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   408→                      </div>
   409→                      {/* Stacked progress bar */}
   410→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   411→                        {todo > 0 && (
   412→                          <div
   413→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   414→                            style={{ width: `${todoPct}%` }}
   415→                            title={`Not Started: ${todo}`}
   416→                          />
   417→                        )}
   418→                        {inProgress > 0 && (
   419→                          <div
   420→                            className="bg-amber-500 transition-all duration-500"
   421→                            style={{ width: `${inProgressPct}%` }}
   422→                            title={`In Progress: ${inProgress}`}
   423→                          />
   424→                        )}
   425→                        {completed > 0 && (
   426→                          <div
   427→                            className="bg-emerald-500 transition-all duration-500"
   428→                            style={{ width: `${completedPct}%` }}
   429→                            title={`Completed: ${completed}`}
   430→                          />
   431→                        )}
   432→                      </div>
   433→                      {/* Legend */}
   434→                      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   435→                        <span className="flex items-center gap-1.5">
   436→                          <span className="w-2.5 h-2.5 rounded-sm bg-slate-400 dark:bg-slate-500" />
   437→                          Not Started ({todo})
   438→                        </span>
   439→                        <span className="flex items-center gap-1.5">
   440→                          <span className="w-2.5 h-2.5 rounded-sm bg-amber-500" />
   441→                          In Progress ({inProgress})
   442→                        </span>
   443→                        <span className="flex items-center gap-1.5">
   444→                          <span className="w-2.5 h-2.5 rounded-sm bg-emerald-500" />
   445→                          Completed ({completed})
   446→                        </span>
   447→                      </div>
   448→                    </div>
   449→                  )
   450→                })}
   451→              </div>
   452→            </CardContent>
   453→          </Card>
   454→        </motion.div>
   455→      )}
   456→
   457→      {projects.length === 0 ? (
   458→        <EmptyState
   459→          icon={FolderKanban}
   460→          title="No active projects"
   461→          description="You don't have any active projects yet."
   462→        />
   463→      ) : (
   464→        <>
   465→          {/* ─── Project Cards ──────────────────────────────────── */}
   466→          <div className="space-y-4">
   467→            {projects.map((project, i) => {
   468→              const doneTasks = project.taskStats?.done || 0
   469→              const totalT = project._count?.tasks || 0
   470→              const pct = project.progress ?? 0
   471→              const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   472→
   473→              // Compute progress bar gradient colors
   474→              const barColor = pct >= 75 ? '#10b981' : pct >= 40 ? '#f59e0b' : '#ef4444'
   475→
   476→              // Files for this project only
   477→              const projectFiles = uploadedFiles.filter(f => f.projectId === project.id)
   478→
   479→              return (
   480→                <motion.div
   481→                  key={project.id}
   482→                  initial={{ opacity: 0, y: 12 }}
   483→                  animate={{ opacity: 1, y: 0 }}
   484→                  transition={{ delay: i * 0.05 }}
   485→                >
   486→                  <Card
   487→                    className="hover:shadow-md transition-all border-l-4"
   488→                    style={{ borderLeftColor: project.color }}
   489→                  >
   490→                    <CardContent className="p-4 md:p-5">
   491→                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   492→                        <div className="min-w-0 flex-1">
   493→                          <div className="flex items-center gap-2 mb-1">
   494→                            <h3 className="font-semibold text-base truncate">{project.name}</h3>
   495→                            <PriorityBadge priority={project.priority} />
   496→                          </div>
   497→                          <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   498→                        </div>
   499→                        <div className="flex items-center gap-2 shrink-0 text-sm flex-wrap">
   500→                          {project.dueDate && (
   501→                            <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   502→                              <Calendar className="w-3.5 h-3.5" />
   503→                              {formatDate(project.dueDate)}
   504→                            </span>
   505→                          )}
   506→                          <span className="text-muted-foreground flex items-center gap-1">
   507→                            <Users className="w-3.5 h-3.5" />
   508→                            {project._count?.members || 0}

   688→// CLIENT PROFILE
   689→// ═══════════════════════════════════════════════════════════════════
   690→export function ClientProfile() {
   691→  const { user, login } = useAuthStore()
   692→  const [loading, setLoading] = useState(true)
   693→  const [saving, setSaving] = useState(false)
   694→  const [savingPassword, setSavingPassword] = useState(false)
   695→
   696→  const [name, setName] = useState('')
   697→  const [email, setEmail] = useState('')
   698→  const [phone, setPhone] = useState('')
   699→  const [company, setCompany] = useState('')
   700→
   701→  const [currentPassword, setCurrentPassword] = useState('')
   702→  const [newPassword, setNewPassword] = useState('')
   703→  const [confirmPassword, setConfirmPassword] = useState('')
   704→
   705→  useEffect(() => {
   706→    if (!user?.id) return
   707→    const fetchProfile = async () => {
   708→      try {
   709→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   710→        const data = await res.json()
   711→        if (data.user) {
   712→          setName(data.user.name || '')
   713→          setEmail(data.user.email || '')
   714→          setPhone(data.user.phone || '')
   715→          setCompany(data.user.company || '')
   716→        }
   717→      } catch {
   718→        toast.error('Failed to load profile')
   719→      } finally {
   720→        setLoading(false)
   721→      }
   722→    }
   723→    fetchProfile()
   724→  }, [user?.id])
   725→
   726→  const saveProfile = async () => {
   727→    if (!user?.id) return
   728→    setSaving(true)
   729→    try {
   730→      const res = await fetch('/api/user/profile', {
   731→        method: 'PATCH',
   732→        headers: { 'Content-Type': 'application/json' },
   733→        body: JSON.stringify({ userId: user.id, name, email, phone, company })
   734→      })
   735→      if (!res.ok) {
   736→        const data = await res.json()
   737→        throw new Error(data.error || 'Failed')
   738→      }
   739→      const data = await res.json()
   740→      toast.success('Profile updated successfully')
   741→      // Update user in store
   742→      if (data.user && user) {
   743→        login({ ...user, name: data.user.name, email: data.user.email, phone: data.user.phone, company: data.user.company })
   744→      }
   745→    } catch (e: unknown) {
   746→      const msg = e instanceof Error ? e.message : 'Failed to update profile'
   747→      toast.error(msg)
   748→    } finally {
   749→      setSaving(false)
   750→    }
   751→  }
   752→
   753→  const changePassword = async () => {
   754→    if (!user?.id) return
   755→    if (!currentPassword || !newPassword || !confirmPassword) {
   756→      toast.error('Please fill in all password fields')
   757→      return
   758→    }
   759→    if (newPassword !== confirmPassword) {
   760→      toast.error('New passwords do not match')
   761→      return
   762→    }
   763→    if (newPassword.length < 6) {
   764→      toast.error('Password must be at least 6 characters')
   765→      return
   766→    }
   767→    setSavingPassword(true)
   768→    try {
   769→      const res = await fetch('/api/user/profile', {
   770→        method: 'PATCH',
   771→        headers: { 'Content-Type': 'application/json' },
   772→        body: JSON.stringify({
   773→          userId: user.id,
   774→          currentPassword,
   775→          newPassword
   776→        })
   777→      })
   778→      if (!res.ok) {
   779→        const data = await res.json()
   780→        throw new Error(data.error || 'Failed')
   781→      }
   782→      toast.success('Password changed successfully')
   783→      setCurrentPassword('')
   784→      setNewPassword('')
   785→      setConfirmPassword('')
   786→    } catch (e: unknown) {
   787→      const msg = e instanceof Error ? e.message : 'Failed to change password'
   788→      toast.error(msg)
   789→    } finally {
   790→      setSavingPassword(false)
   791→    }
   792→  }
   793→
   794→  if (loading) return <PageSkeleton />
   795→
   796→  return (
   797→    <div className="p-4 md:p-6 space-y-6 max-w-2xl">
   798→      <div className="flex items-center gap-3">
   799→        <div className="w-10 h-10 rounded-lg bg-rose-100 dark:bg-rose-900/30 flex items-center justify-center">
   800→          <UserCircle className="w-5 h-5 text-rose-600 dark:text-rose-400" />
   801→        </div>
   802→        <div>
   803→          <h1 className="text-2xl font-bold">Profile Settings</h1>
   804→          <p className="text-sm text-muted-foreground">Manage your account information</p>
   805→        </div>
   806→      </div>
   807→
   808→      {/* Profile Card */}
   809→      <motion.div {...fadeSlide}>
   810→        <Card>
   811→          <CardHeader>
   812→            <div className="flex items-center justify-between">
   813→              <CardTitle className="text-lg">Personal Information</CardTitle>
   814→              <span className={`text-xs px-2.5 py-1 rounded-full font-medium ${getRoleBadgeClasses(user?.role || 'client')}`}>
   815→                Client
   816→              </span>
   817→            </div>
   818→            {user?.createdAt && (
   819→              <p className="text-sm text-muted-foreground">
   820→                Member since {formatDate(user.createdAt)}
   821→              </p>
   822→            )}
   823→          </CardHeader>
   824→          <CardContent className="space-y-4">
   825→            <div className="space-y-2">
   826→              <Label htmlFor="profile-name">Full Name</Label>
   827→              <Input
   828→                id="profile-name"
   829→                value={name}
   830→                onChange={e => setName(e.target.value)}
   831→                placeholder="Your full name"
   832→              />
   833→            </div>
   834→            <div className="space-y-2">
   835→              <Label htmlFor="profile-email">Email Address</Label>
   836→              <Input
   837→                id="profile-email"
   838→                type="email"
   839→                value={email}
   840→                onChange={e => setEmail(e.target.value)}
   841→                placeholder="your@email.com"
   842→              />
   843→            </div>
   844→            <div className="space-y-2">
   845→              <Label htmlFor="profile-phone">Phone Number</Label>
   846→              <Input
   847→                id="profile-phone"
   848→                type="tel"
   849→                value={phone}
   850→                onChange={e => setPhone(e.target.value)}
   851→                placeholder="+1 (555) 000-0000"
   852→              />
   853→            </div>
   854→            <div className="space-y-2">
   855→              <Label htmlFor="profile-company">Company</Label>
   856→              <Input
   857→                id="profile-company"
   858→                value={company}
   859→                onChange={e => setCompany(e.target.value)}
   860→                placeholder="Your company name"
   861→              />
   862→            </div>
   863→            <div className="pt-2">
   864→              <Button onClick={saveProfile} disabled={saving} className="w-full sm:w-auto">
   865→                {saving ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   866→                Save Changes
   867→              </Button>
   868→            </div>
   869→          </CardContent>
   870→        </Card>
   871→      </motion.div>
   872→
   873→      {/* Change Password */}
   874→      <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   875→        <Card>
   876→          <CardHeader>
   877→            <CardTitle className="text-lg flex items-center gap-2">
   878→              <Shield className="w-5 h-5 text-muted-foreground" />
   879→              Change Password
   880→            </CardTitle>
   881→            <p className="text-sm text-muted-foreground">
   882→              Update your password to keep your account secure
   883→            </p>
   884→          </CardHeader>
   885→          <CardContent className="space-y-4">
   886→            <div className="space-y-2">
   887→              <Label htmlFor="current-password">Current Password</Label>
   888→              <div className="relative">
   889→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   890→                <Input
   891→                  id="current-password"
   892→                  type="password"
   893→                  value={currentPassword}
   894→                  onChange={e => setCurrentPassword(e.target.value)}
   895→                  placeholder="Enter current password"
   896→                  className="pl-9"
   897→                />
   898→              </div>
   899→            </div>
   900→            <div className="space-y-2">
   901→              <Label htmlFor="new-password">New Password</Label>
   902→              <div className="relative">
   903→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   904→                <Input
   905→                  id="new-password"
   906→                  type="password"
   907→                  value={newPassword}
   908→                  onChange={e => setNewPassword(e.target.value)}
   909→                  placeholder="Enter new password"

   911→                />

   918→                <Input

   923→                  placeholder="Confirm new password"
   924→                  className="pl-9"
   925→                />
   926→              </div>
   927→            </div>
   928→            <div className="pt-2">
   929→              <Button onClick={changePassword} disabled={savingPassword} variant="outline" className="w-full sm:w-auto">
   930→                {savingPassword ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Shield className="w-4 h-4 mr-2" />}
   931→                Change Password

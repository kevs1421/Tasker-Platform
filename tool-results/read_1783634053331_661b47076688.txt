   105→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   106→  const visible = users.slice(0, max)
   107→  const extra = users.length - max
   108→  return (
   109→    <div className="flex -space-x-2">
   110→      {visible.map((u, i) =>
   111→        u ? (
   112→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   113→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   114→              {getInitials(u.name)}
   115→            </AvatarFallback>
   116→          </Avatar>
   117→        ) : null
   118→      )}
   119→      {extra > 0 && (
   120→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   121→          +{extra}
   122→        </div>
   123→      )}
   124→    </div>
   125→  )
   126→}
   127→
   128→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   129→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   130→  if (!user) return null
   131→  return (
   132→    <Avatar className={sizeClasses}>
   133→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   134→        {getInitials(user.name)}
   135→      </AvatarFallback>
   136→    </Avatar>
   137→  )
   138→}

   157→
   158→export function TeamProjects() {
   159→  const { user, setSelectedProjectId } = useAuthStore()
   160→  const [projects, setProjects] = useState<Project[]>([])
   161→  const [loading, setLoading] = useState(true)
   162→  const [expandedId, setExpandedId] = useState<string | null>(null)
   163→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   164→  const [expandedLoading, setExpandedLoading] = useState(false)
   165→  const [taskFilter, setTaskFilter] = useState<string>('all')
   166→  const [createOpen, setCreateOpen] = useState(false)
   167→  const [createLoading, setCreateLoading] = useState(false)
   168→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   169→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   170→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   171→
   172→  // Create project form
   173→  const [projName, setProjName] = useState('')
   174→  const [projDesc, setProjDesc] = useState('')
   175→  const [projPriority, setProjPriority] = useState('medium')
   176→  const [projDueDate, setProjDueDate] = useState<Date>()
   177→  const [projColor, setProjColor] = useState(PROJECT_COLORS[0])
   178→  const [projCalendarOpen, setProjCalendarOpen] = useState(false)
   179→
   180→  // Create task form
   181→  const [taskTitle, setTaskTitle] = useState('')
   182→  const [taskDesc, setTaskDesc] = useState('')
   183→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   184→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   185→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   186→  const [taskTags, setTaskTags] = useState('')
   187→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   188→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   189→
   190→  const fetchProjects = useCallback(async () => {
   191→    try {
   192→      const res = await fetch('/api/projects')
   193→      const data = await res.json()
   194→      const filtered = (data.projects || []).filter((p: Project) =>
   195→        p.members?.some((m) => m.userId === user?.id)
   196→      )
   197→      setProjects(filtered)
   198→    } catch {
   199→      toast.error('Failed to load projects')
   200→    } finally {
   201→      setLoading(false)
   202→    }
   203→  }, [user])
   204→
   205→  useEffect(() => { fetchProjects() }, [fetchProjects])
   206→
   207→  const handleExpand = async (projectId: string) => {
   208→    if (expandedId === projectId) {
   209→      setExpandedId(null)
   210→      setExpandedTasks([])
   211→      return
   212→    }
   213→    setExpandedId(projectId)
   214→    setTaskFilter('all')
   215→    setExpandedLoading(true)
   216→    try {
   217→      const res = await fetch(`/api/projects/${projectId}`)
   218→      const data = await res.json()
   219→      setExpandedTasks(data.project?.tasks || [])
   220→    } catch {
   221→      toast.error('Failed to load tasks')
   222→    } finally {
   223→      setExpandedLoading(false)
   224→    }
   225→  }
   226→
   227→  const handleCreateProject = async () => {
   228→    if (!projName.trim()) { toast.error('Project name is required'); return }
   229→    setCreateLoading(true)
   230→    try {
   231→      await fetch('/api/projects', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({
   235→          name: projName, description: projDesc || null, priority: projPriority,
   236→          dueDate: projDueDate ? projDueDate.toISOString() : null, color: projColor,
   237→        }),
   238→      })
   239→      toast.success('Project created')
   240→      setCreateOpen(false)
   241→      setProjName(''); setProjDesc(''); setProjPriority('medium')
   242→      setProjDueDate(undefined); setProjColor(PROJECT_COLORS[0])
   243→      fetchProjects()
   244→    } catch {
   245→      toast.error('Failed to create project')
   246→    } finally {
   247→      setCreateLoading(false)
   248→    }
   249→  }
   250→
   251→  const handleCreateTask = async () => {
   252→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   253→    setCreateTaskLoading(true)
   254→    try {
   255→      await fetch('/api/tasks', {
   256→        method: 'POST',
   257→        headers: { 'Content-Type': 'application/json' },
   258→        body: JSON.stringify({
   259→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   260→          priority: taskPriority, projectId: createTaskProjectId,
   261→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   262→          tags: taskTags, assigneeId: taskAssignee || null,
   263→        }),
   264→      })
   265→      toast.success('Task created')
   266→      setCreateTaskOpen(false)
   267→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   268→      setTaskPriority('medium'); setTaskDueDate(undefined)
   269→      setTaskTags(''); setTaskAssignee('')
   270→      if (expandedId === createTaskProjectId) handleExpand(createTaskProjectId)
   271→      else handleExpand(createTaskProjectId)
   272→    } catch {
   273→      toast.error('Failed to create task')
   274→    } finally {
   275→      setCreateTaskLoading(false)
   276→    }
   277→  }
   278→
   279→  const openCreateTaskForProject = (projectId: string) => {
   280→    setCreateTaskProjectId(projectId)
   281→    setCreateTaskOpen(true)
   282→  }
   283→
   284→  const getProjectTaskStats = (projectId: string) => {
   285→    if (expandedId !== projectId) return null
   286→    const total = expandedTasks.length
   287→    const done = expandedTasks.filter((t) => t.status === 'done').length
   288→    return { total, done }
   289→  }
   290→
   291→  const filteredTasks = taskFilter === 'all'
   292→    ? expandedTasks
   293→    : expandedTasks.filter((t) => t.status === taskFilter)
   294→
   295→  if (loading) {
   296→    return (
   297→      <div>
   298→        <div className="flex items-center justify-between mb-6">
   299→          <Skeleton className="h-8 w-48" />
   300→          <Skeleton className="h-10 w-36" />
   301→        </div>
   302→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   303→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   304→        </div>
   305→      </div>
   306→    )
   307→  }
   308→
   309→  return (
   310→    <div className="space-y-6">
   311→      {/* Header */}
   312→      <div className="flex items-center justify-between">
   313→        <div>
   314→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   315→          <p className="text-sm text-muted-foreground mt-1">
   316→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   317→          </p>
   318→        </div>
   319→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   320→          <DialogTrigger asChild>
   321→            <Button className="gap-2">
   322→              <Plus className="h-4 w-4" /> Create Project
   323→            </Button>
   324→          </DialogTrigger>
   325→          <DialogContent className="sm:max-w-lg">
   326→            <DialogHeader>
   327→              <DialogTitle>Create New Project</DialogTitle>
   328→              <DialogDescription>Add a new project to your workspace.</DialogDescription>
   329→            </DialogHeader>
   330→            <div className="space-y-4 py-2">
   331→              <div className="space-y-2">
   332→                <Label>Project Name</Label>
   333→                <Input placeholder="Enter project name" value={projName} onChange={(e) => setProjName(e.target.value)} />
   334→              </div>
   335→              <div className="space-y-2">
   336→                <Label>Description</Label>
   337→                <Textarea placeholder="Describe the project..." value={projDesc} onChange={(e) => setProjDesc(e.target.value)} rows={3} />
   338→              </div>
   339→              <div className="grid grid-cols-2 gap-4">
   340→                <div className="space-y-2">
   341→                  <Label>Priority</Label>
   342→                  <Select value={projPriority} onValueChange={setProjPriority}>
   343→                    <SelectTrigger><SelectValue /></SelectTrigger>
   344→                    <SelectContent>
   345→                      <SelectItem value="low">Low</SelectItem>
   346→                      <SelectItem value="medium">Medium</SelectItem>
   347→                      <SelectItem value="high">High</SelectItem>
   348→                      <SelectItem value="urgent">Urgent</SelectItem>
   349→                    </SelectContent>
   350→                  </Select>
   351→                </div>
   352→                <div className="space-y-2">
   353→                  <Label>Due Date</Label>
   354→                  <Popover open={projCalendarOpen} onOpenChange={setProjCalendarOpen}>
   355→                    <PopoverTrigger asChild>
   356→                      <Button variant="outline" className="w-full justify-start text-left font-normal gap-2">
   357→                        <CalendarIcon className="h-4 w-4" />
   358→                        {projDueDate ? format(projDueDate, 'PPP') : 'Pick a date'}
   359→                      </Button>
   360→                    </PopoverTrigger>
   361→                    <PopoverContent className="w-auto p-0" align="start">
   362→                      <Calendar mode="single" selected={projDueDate} onSelect={(d) => { setProjDueDate(d); setProjCalendarOpen(false) }} />
   363→                    </PopoverContent>
   364→                  </Popover>
   365→                </div>
   366→              </div>
   367→              <div className="space-y-2">
   368→                <Label>Color</Label>
   369→                <div className="flex flex-wrap gap-2">
   370→                  {PROJECT_COLORS.map((c) => (
   371→                    <button
   372→                      key={c}
   373→                      className={cn(
   374→                        'h-8 w-8 rounded-full transition-all ring-2',
   375→                        projColor === c ? 'ring-offset-2 ring-offset-background scale-110' : 'ring-transparent hover:scale-105',
   376→                      )}
   377→                      style={{ backgroundColor: c, ringColor: c }}
   378→                      onClick={() => setProjColor(c)}
   379→                    />
   380→                  ))}
   381→                </div>
   382→              </div>
   383→            </div>
   384→            <DialogFooter>
   385→              <Button variant="outline" onClick={() => setCreateOpen(false)}>Cancel</Button>
   386→              <Button onClick={handleCreateProject} disabled={createLoading} className="gap-2">
   387→                {createLoading && <Loader2 className="h-4 w-4 animate-spin" />}
   388→                Create Project
   389→              </Button>
   390→            </DialogFooter>
   391→          </DialogContent>
   392→        </Dialog>
   393→      </div>
   394→
   395→      {/* Project Grid */}
   396→      {projects.length === 0 ? (
   397→        <EmptyState
   398→          icon={FolderKanban}
   399→          title="No projects yet"
   400→          description="You haven't been assigned to any projects. Create one or ask an admin to assign you."
   401→        />
   402→      ) : (
   403→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   404→          <AnimatePresence>
   405→            {projects.map((project, idx) => {
   406→              const totalTasks = project._count?.tasks || 0
   407→              const members = project.members || []
   408→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   409→              const pConfig = getPriorityConfig(project.priority)
   410→              const isExpanded = expandedId === project.id
   411→              const stats = getProjectTaskStats(project.id)
   412→
   413→              return (
   414→                <motion.div
   415→                  key={project.id}
   416→                  initial={{ opacity: 0, y: 20 }}
   417→                  animate={{ opacity: 1, y: 0 }}
   418→                  transition={{ delay: idx * 0.05 }}
   419→                >
   420→                  <Card
   421→                    className={cn(
   422→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   423→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   424→                    )}
   425→                    onClick={() => handleExpand(project.id)}
   426→                  >
   427→                    <div className="flex">
   428→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   429→                      <CardContent className="flex-1 p-4">
   430→                        <div className="flex items-start justify-between mb-2">
   431→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   432→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   433→                            {pConfig.label}
   434→                          </Badge>
   435→                        </div>
   436→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   437→                          {project.description || 'No description'}
   438→                        </p>
   439→                        <Progress
   440→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   441→                          className="h-1.5 mb-3"
   442→                        />
   443→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   444→                          <div className="flex items-center gap-3">
   445→                            <span className="flex items-center gap-1">
   446→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   447→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   448→                            </span>
   449→                            <span className="flex items-center gap-1">
   450→                              <Users className="h-3 w-3" />
   451→                              {members.length}
   452→                            </span>
   453→                          </div>
   454→                          {project.dueDate && (
   455→                            <span className={cn(
   456→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   457→                                ? 'text-red-500' : '',
   458→                            )}>
   459→                              {formatDate(project.dueDate)}
   460→                            </span>
   461→                          )}
   462→                        </div>
   463→                        {memberUsers.length > 0 && (
   464→                          <div className="mt-3">
   465→                            <AvatarStack users={memberUsers} max={4} />
   466→                          </div>
   467→                        )}
   468→                      </CardContent>
   469→                    </div>
   470→                  </Card>
   471→
   472→                  {/* Expanded Task List */}
   473→                  <AnimatePresence>
   474→                    {isExpanded && (
   475→                      <motion.div
   476→                        initial={{ opacity: 0, height: 0 }}
   477→                        animate={{ opacity: 1, height: 'auto' }}
   478→                        exit={{ opacity: 0, height: 0 }}
   479→                        transition={{ duration: 0.2 }}
   480→                        className="overflow-hidden"
   481→                        onClick={(e) => e.stopPropagation()}
   482→                      >
   483→                        <Card className="mt-2 p-4 space-y-4">
   484→                          <div className="flex items-center justify-between">
   485→                            <div className="flex items-center gap-2">
   486→                              <h4 className="font-semibold text-sm">Tasks</h4>
   487→                              <div className="flex gap-1">
   488→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   489→                                  <Button
   490→                                    key={s}
   491→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   492→                                    size="sm"
   493→                                    className="h-7 text-xs px-2"
   494→                                    onClick={() => setTaskFilter(s)}
   495→                                  >
   496→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   497→                                  </Button>
   498→                                ))}
   499→                              </div>
   500→                            </div>
   501→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   502→                              <Plus className="h-3 w-3" /> Add Task
   503→                            </Button>
   504→                          </div>

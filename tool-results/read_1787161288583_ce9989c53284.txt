    98→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
    99→  const visible = users.slice(0, max)
   100→  const extra = users.length - max
   101→  return (
   102→    <div className="flex -space-x-2">
   103→      {visible.map((u, i) =>
   104→        u ? (
   105→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   106→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   107→              {getInitials(u.name)}
   108→            </AvatarFallback>
   109→          </Avatar>
   110→        ) : null
   111→      )}
   112→      {extra > 0 && (
   113→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   114→          +{extra}
   115→        </div>
   116→      )}
   117→    </div>
   118→  )
   119→}

   121→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   122→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   123→  if (!user) return null
   124→  return (
   125→    <Avatar className={sizeClasses}>
   126→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   127→        {getInitials(user.name)}
   128→      </AvatarFallback>
   129→    </Avatar>
   130→  )
   131→}

   149→// ─── TeamProjects ───────────────────────────────────────────
   150→
   151→export function TeamProjects() {
   152→  const { user, setSelectedProjectId } = useAuthStore()
   153→  const isTraining = user?.role === 'training'
   154→  const [projects, setProjects] = useState<Project[]>([])
   155→  const [loading, setLoading] = useState(true)
   156→  const [expandedId, setExpandedId] = useState<string | null>(null)
   157→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   158→  const [expandedLoading, setExpandedLoading] = useState(false)
   159→  const [taskFilter, setTaskFilter] = useState<string>('all')
   160→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   161→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   162→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   163→  const [allTeamUsers, setAllTeamUsers] = useState<User[]>([])
   164→
   165→  // Create task form
   166→  const [taskTitle, setTaskTitle] = useState('')
   167→  const [taskDesc, setTaskDesc] = useState('')
   168→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   169→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   170→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   171→  const [taskTags, setTaskTags] = useState('')
   172→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   173→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   174→
   175→  const fetchProjects = useCallback(async () => {
   176→    try {
   177→      const res = await fetch('/api/projects')
   178→      const data = await res.json()
   179→      const filtered = (data.projects || []).filter((p: Project) =>
   180→        p.members?.some((m) => m.userId === user?.id)
   181→      )
   182→      setProjects(filtered)
   183→    } catch {
   184→      toast.error('Failed to load projects')
   185→    } finally {
   186→      setLoading(false)
   187→    }
   188→  }, [user])
   189→
   190→  useEffect(() => { fetchProjects() }, [fetchProjects])
   191→
   192→  // Fetch all active admin + team users for assignee picker
   193→  useEffect(() => {
   194→    fetch('/api/users')
   195→      .then(r => r.json())
   196→      .then(d => {
   197→        const members = (d.users || []).filter((u: User) => (u.role === 'team' || u.role === 'training' || u.role === 'admin') && u.status === 'active')
   198→        setAllTeamUsers(members)
   199→      })
   200→      .catch(() => {})
   201→  }, [])
   202→
   203→  const handleExpand = async (projectId: string) => {
   204→    if (expandedId === projectId) {
   205→      setExpandedId(null)
   206→      setExpandedTasks([])
   207→      return
   208→    }
   209→    setExpandedId(projectId)
   210→    setTaskFilter('all')
   211→    setExpandedLoading(true)
   212→    try {
   213→      const res = await fetch(`/api/projects/${projectId}`)
   214→      const data = await res.json()
   215→      setExpandedTasks(data.project?.tasks || [])
   216→    } catch {
   217→      toast.error('Failed to load tasks')
   218→    } finally {
   219→      setExpandedLoading(false)
   220→    }
   221→  }
   222→
   223→  const handleCreateTask = async () => {
   224→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   225→    if (!user?.id) { toast.error('Not authenticated'); return }
   226→    setCreateTaskLoading(true)
   227→    try {
   228→      const res = await fetch('/api/tasks', {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify({
   232→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   233→          priority: taskPriority, projectId: createTaskProjectId,
   234→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   235→          tags: taskTags,
   236→          createdById: user.id,
   237→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   238→        }),
   239→      })
   240→      if (!res.ok) {
   241→        const d = await res.json()
   242→        throw new Error(d.error || 'Failed to create task')
   243→      }
   244→      toast.success('Task created')
   245→      setCreateTaskOpen(false)
   246→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   247→      setTaskPriority('medium'); setTaskDueDate(undefined)
   248→      setTaskTags(''); setTaskAssignee('')
   249→      // Always refresh the expanded project view
   250→      handleExpand(createTaskProjectId)
   251→    } catch (e: unknown) {
   252→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   253→    } finally {
   254→      setCreateTaskLoading(false)
   255→    }
   256→  }
   257→
   258→  const openCreateTaskForProject = (projectId: string) => {
   259→    setCreateTaskProjectId(projectId)
   260→    setCreateTaskOpen(true)
   261→  }
   262→
   263→  const getProjectTaskStats = (projectId: string) => {
   264→    if (expandedId !== projectId) return null
   265→    const total = expandedTasks.length
   266→    const done = expandedTasks.filter((t) => t.status === 'done').length
   267→    return { total, done }
   268→  }
   269→
   270→  const filteredTasks = taskFilter === 'all'
   271→    ? expandedTasks
   272→    : expandedTasks.filter((t) => t.status === taskFilter)
   273→
   274→  if (loading) {
   275→    return (
   276→      <div>
   277→        <div className="flex items-center justify-between mb-6">
   278→          <Skeleton className="h-8 w-48" />
   279→          <Skeleton className="h-10 w-36" />
   280→        </div>
   281→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   282→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   283→        </div>
   284→      </div>
   285→    )
   286→  }
   287→
   288→  return (
   289→    <div className="space-y-6">
   290→      {/* Header */}
   291→      <div className="flex items-center justify-between">
   292→        <div>
   293→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   294→          <p className="text-sm text-muted-foreground mt-1">
   295→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   296→          </p>
   297→        </div>
   298→      </div>
   299→
   300→      {/* Project Grid */}
   301→      {projects.length === 0 ? (
   302→        <EmptyState
   303→          icon={FolderKanban}
   304→          title="No projects yet"
   305→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   306→        />
   307→      ) : (
   308→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   309→          <AnimatePresence>
   310→            {projects.map((project, idx) => {
   311→              const totalTasks = project._count?.tasks || 0
   312→              const members = project.members || []
   313→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   314→              const pConfig = getPriorityConfig(project.priority)
   315→              const isExpanded = expandedId === project.id
   316→              const stats = getProjectTaskStats(project.id)
   317→
   318→              return (
   319→                <motion.div
   320→                  key={project.id}
   321→                  initial={{ opacity: 0, y: 20 }}
   322→                  animate={{ opacity: 1, y: 0 }}
   323→                  transition={{ delay: idx * 0.05 }}
   324→                >
   325→                  <Card
   326→                    className={cn(
   327→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   328→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   329→                    )}
   330→                    onClick={() => handleExpand(project.id)}
   331→                  >
   332→                    <div className="flex">
   333→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   334→                      <CardContent className="flex-1 p-4">
   335→                        <div className="flex items-start justify-between mb-2">
   336→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   337→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   338→                            {pConfig.label}
   339→                          </Badge>
   340→                        </div>
   341→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   342→                          {project.description || 'No description'}
   343→                        </p>
   344→                        <Progress
   345→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   346→                          className="h-1.5 mb-3"
   347→                        />
   348→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   349→                          <div className="flex items-center gap-3">
   350→                            <span className="flex items-center gap-1">
   351→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   352→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   353→                            </span>
   354→                            <span className="flex items-center gap-1">
   355→                              <Users className="h-3 w-3" />
   356→                              {members.length}
   357→                            </span>
   358→                          </div>
   359→                          {project.dueDate && (
   360→                            <span className={cn(
   361→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   362→                                ? 'text-red-500' : '',
   363→                            )}>
   364→                              {formatDate(project.dueDate)}
   365→                            </span>
   366→                          )}
   367→                        </div>
   368→                        {memberUsers.length > 0 && (
   369→                          <div className="mt-3">
   370→                            <AvatarStack users={memberUsers} max={4} />
   371→                          </div>
   372→                        )}
   373→                      </CardContent>
   374→                    </div>
   375→                  </Card>
   376→
   377→                  {/* Expanded Task List */}
   378→                  <AnimatePresence>
   379→                    {isExpanded && (
   380→                      <motion.div
   381→                        initial={{ opacity: 0, height: 0 }}
   382→                        animate={{ opacity: 1, height: 'auto' }}
   383→                        exit={{ opacity: 0, height: 0 }}
   384→                        transition={{ duration: 0.2 }}
   385→                        className="overflow-hidden"
   386→                        onClick={(e) => e.stopPropagation()}
   387→                      >
   388→                        <Card className="mt-3 p-4 space-y-4">
   389→                          <div className="flex items-center justify-between">
   390→                            <div className="flex items-center gap-2">
   391→                              <div className="flex gap-1">
   392→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   393→                                  <Button
   394→                                    key={s}
   395→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   396→                                    size="sm"
   397→                                    className="h-7 text-xs px-2"
   398→                                    onClick={() => setTaskFilter(s)}
   399→                                  >
   400→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   401→                                  </Button>
   402→                                ))}
   403→                              </div>
   404→                            </div>
   405→                            {!isTraining && (
   406→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   407→                              <Plus className="h-3 w-3" /> Add Task
   408→                            </Button>
   409→                            )}
   410→                          </div>
   411→                          {expandedLoading ? (
   412→                            <div className="space-y-2">
   413→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   414→                            </div>
   415→                          ) : filteredTasks.length === 0 ? (
   416→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   417→                          ) : (
   418→                            <ScrollArea className="max-h-64">
   419→                              <div className="space-y-2">
   420→                                {filteredTasks.map((task) => {
   421→                                  const sConfig = getStatusConfig(task.status)
   422→                                  const pConf = getPriorityConfig(task.priority)
   423→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   424→                                  return (
   425→                                    <div
   426→                                      key={task.id}
   427→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   428→                                      onClick={() => setSelectedProjectId(project.id)}
   429→                                    >
   430→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   431→                                      <div className="flex-1 min-w-0">
   432→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   433→                                        <div className="flex items-center gap-2 mt-0.5">
   434→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   435→                                          {task.dueDate && (
   436→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   437→                                              {formatDate(task.dueDate)}
   438→                                            </span>
   439→                                          )}
   440→                                        </div>
   441→                                      </div>
   442→                                      <AvatarStack users={assignees} max={2} />
   443→                                    </div>
   444→                                  )
   445→                                })}
   446→                              </div>
   447→                            </ScrollArea>
   448→                          )}
   449→                        </Card>
   450→                      </motion.div>
   451→                    )}
   452→                  </AnimatePresence>
   453→                </motion.div>
   454→              )
   455→            })}
   456→          </AnimatePresence>
   457→        </motion.div>
   458→      )}
   459→
   460→      {/* Create Task Dialog */}
   461→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   462→        <DialogContent className="sm:max-w-lg">
   463→          <DialogHeader>
   464→            <DialogTitle>Create New Task</DialogTitle>
   465→            <DialogDescription>Add a task to the project.</DialogDescription>
   466→          </DialogHeader>
   467→          <div className="space-y-4 py-2">
   468→            <div className="space-y-2">
   469→              <Label>Title</Label>
   470→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   471→            </div>
   472→            <div className="space-y-2">
   473→              <Label>Description</Label>
   474→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   475→            </div>
   476→            <div className="grid grid-cols-2 gap-4">
   477→              <div className="space-y-2">
   478→                <Label>Status</Label>
   479→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   480→                  <SelectTrigger><SelectValue /></SelectTrigger>
   481→                  <SelectContent>
   482→                    <SelectItem value="todo">To Do</SelectItem>
   483→                    <SelectItem value="in_progress">In Progress</SelectItem>
   484→                    <SelectItem value="in_review">In Review</SelectItem>
   485→                    <SelectItem value="done">Done</SelectItem>
   486→                  </SelectContent>
   487→                </Select>
   488→              </div>
   489→              <div className="space-y-2">
   490→                <Label>Priority</Label>
   491→                <Select value={taskPriority} onValueChange={(v) => setTaskPriority(v as typeof taskPriority)}>
   492→                  <SelectTrigger><SelectValue /></SelectTrigger>
   493→                  <SelectContent>
   494→                    <SelectItem value="low">Low</SelectItem>
   495→                    <SelectItem value="medium">Medium</SelectItem>
   496→                    <SelectItem value="high">High</SelectItem>
   497→                    <SelectItem value="urgent">Urgent</SelectItem>
   498→                  </SelectContent>
   499→                </Select>
   500→              </div>
   501→            </div>
   502→            <div className="grid grid-cols-2 gap-4">
   503→              <div className="space-y-2">
   504→                <Label>Due Date</Label>
   505→                <Popover open={taskCalendarOpen} onOpenChange={setTaskCalendarOpen}>
   506→                  <PopoverTrigger asChild>

  1862→function formatDuration(minutes: number): string {
  1863→  const h = Math.floor(minutes / 60)
  1864→  const m = minutes % 60
  1865→  if (h === 0) return `${m}m`
  1866→  if (m === 0) return `${h}h`
  1867→  return `${h}h ${m}m`
  1868→}

  1870→function formatElapsed(startDate: string): string {
  1871→  const ms = Date.now() - new Date(startDate).getTime()
  1872→  const totalSec = Math.floor(ms / 1000)
  1873→  const h = Math.floor(totalSec / 3600)
  1874→  const m = Math.floor((totalSec % 3600) / 60)
  1875→  const s = totalSec % 60
  1876→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  1877→}

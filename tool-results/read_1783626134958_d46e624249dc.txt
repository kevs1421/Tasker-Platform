   104→   104→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   105→   105→  const visible = users.slice(0, max)
   106→   106→  const extra = users.length - max
   107→   107→  return (
   108→   108→    <div className="flex -space-x-2">
   109→   109→      {visible.map((u, i) =>
   110→   110→        u ? (
   111→   111→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   112→   112→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   113→   113→              {getInitials(u.name)}
   114→   114→            </AvatarFallback>
   115→   115→          </Avatar>
   116→   116→        ) : null
   117→   117→      )}
   118→   118→      {extra > 0 && (
   119→   119→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   120→   120→          +{extra}
   121→   121→        </div>
   122→   122→      )}
   123→   123→    </div>
   124→   124→  )

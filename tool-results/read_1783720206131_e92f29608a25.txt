   116→const ONBOARDING_STEPS = [
   117→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   118→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   119→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   120→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   121→]
   122→
   123→export function ClientProjects() {
   124→  const { user } = useAuthStore()
   125→  const [projects, setProjects] = useState<ClientProject[]>([])
   126→  const [loading, setLoading] = useState(true)
   127→  const [expandedProjectId, setExpandedProjectId] = useState<string | null>(null)
   128→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   129→  const [taskDetail, setTaskDetail] = useState<(Task & { _count: { comments: number }; comments?: Comment[] }) | null>(null)
   130→  const [taskDetailLoading, setTaskDetailLoading] = useState(false)
   131→
   132→  // Onboarding creation state
   133→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   134→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   135→  const [onboardingTitle, setOnboardingTitle] = useState('')
   136→  const [onboardingDesc, setOnboardingDesc] = useState('')
   137→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   138→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   139→
   140→  const fetchProjects = useCallback(async () => {
   141→    try {
   142→      const res = await fetch('/api/projects')
   143→      const data = await res.json()
   144→      // Filter to active projects and those the client is a member of
   145→      const clientProjects = (data.projects || [])
   146→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   147→      setProjects(clientProjects)
   148→    } catch {
   149→      toast.error('Failed to load projects')
   150→    } finally {
   151→      setLoading(false)
   152→    }
   153→  }, [user?.id])
   154→
   155→  useEffect(() => { fetchProjects() }, [fetchProjects])
   156→
   157→  const expandProject = async (projectId: string) => {
   158→    if (expandedProjectId === projectId) {
   159→      setExpandedProjectId(null)
   160→      setExpandedTasks([])
   161→      return
   162→    }
   163→    setExpandedProjectId(projectId)
   164→    try {
   165→      const res = await fetch(`/api/projects/${projectId}`)
   166→      const data = await res.json()
   167→      setExpandedTasks(data.project?.tasks || [])
   168→    } catch {
   169→      toast.error('Failed to load project tasks')
   170→    }
   171→  }
   172→
   173→  const openTaskDetail = async (taskId: string) => {
   174→    setTaskDetailLoading(true)
   175→    setTaskDetail(null)
   176→    try {
   177→      const res = await fetch(`/api/tasks/${taskId}`)
   178→      const data = await res.json()
   179→      setTaskDetail(data.task)
   180→    } catch {
   181→      toast.error('Failed to load task details')
   182→    } finally {
   183→      setTaskDetailLoading(false)
   184→    }
   185→  }
   186→
   187→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   188→    setOnboardingProjectId(projectId)
   189→    setOnboardingTitle(`Onboarding - ${projectName}`)
   190→    setOnboardingDesc('')
   191→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   192→    setOnboardingDialogOpen(true)
   193→  }
   194→
   195→  const handleCreateOnboarding = async () => {
   196→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   197→    setCreatingOnboarding(true)
   198→    try {
   199→      const stepsJson = JSON.stringify(
   200→        onboardingSteps.map(stepId => {
   201→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   202→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   203→        })
   204→      )
   205→      const res = await fetch('/api/onboarding', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({
   209→          title: onboardingTitle.trim(),
   210→          description: onboardingDesc.trim() || null,
   211→          stepsJson,
   212→          projectId: onboardingProjectId,
   213→          clientId: user?.id || null,
   214→          status: 'published',
   215→        }),
   216→      })
   217→      if (!res.ok) {
   218→        const d = await res.json()
   219→        throw new Error(d.error || 'Failed to create')
   220→      }
   221→      toast.success('Onboarding form created successfully')
   222→      setOnboardingDialogOpen(false)
   223→    } catch (e: unknown) {
   224→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   225→    } finally {
   226→      setCreatingOnboarding(false)
   227→    }
   228→  }
   229→
   230→  const toggleOnboardingStep = (stepId: string) => {
   231→    setOnboardingSteps(prev =>
   232→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   233→    )
   234→  }
   235→
   236→  if (loading) return <PageSkeleton />
   237→
   238→  return (
   239→    <div className="p-4 md:p-6 space-y-6">
   240→      {/* Page Header */}
   241→      <div className="flex items-center gap-3">
   242→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   243→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   244→        </div>
   245→        <div>
   246→          <h1 className="text-2xl font-bold">My Projects</h1>
   247→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   248→        </div>
   249→      </div>
   250→
   251→      {projects.length === 0 ? (
   252→        <EmptyState
   253→          icon={FolderKanban}
   254→          title="No active projects"
   255→          description="You don't have any active projects yet."
   256→        />
   257→      ) : (
   258→        <>
   259→          {/* ─── Project Cards ──────────────────────────────────── */}
   260→          <div className="space-y-4">
   261→            {projects.map((project, i) => {
   262→              const projectTasks = expandedProjectId === project.id ? expandedTasks : []
   263→              const doneTasks = project.taskStats?.done || 0
   264→              const totalT = project._count?.tasks || 0
   265→              const pct = project.progress ?? 0
   266→              const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   267→              const isExpanded = expandedProjectId === project.id
   268→
   269→              // Compute progress bar gradient colors
   270→              const barColor = pct >= 75 ? '#10b981' : pct >= 40 ? '#f59e0b' : '#ef4444'
   271→
   272→              return (
   273→                <motion.div
   274→                  key={project.id}
   275→                  initial={{ opacity: 0, y: 12 }}
   276→                  animate={{ opacity: 1, y: 0 }}
   277→                  transition={{ delay: i * 0.05 }}
   278→                >
   279→                  <Card
   280→                    className="cursor-pointer hover:shadow-md transition-all border-l-4"
   281→                    style={{ borderLeftColor: project.color }}
   282→                    onClick={() => expandProject(project.id)}
   283→                  >
   284→                    <CardContent className="p-4 md:p-5">
   285→                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   286→                        <div className="min-w-0 flex-1">
   287→                          <div className="flex items-center gap-2 mb-1">
   288→                            <h3 className="font-semibold text-base truncate">{project.name}</h3>
   289→                            <PriorityBadge priority={project.priority} />
   290→                          </div>
   291→                          <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   292→                        </div>
   293→                        <div className="flex items-center gap-3 shrink-0 text-sm">
   294→                          {project.dueDate && (
   295→                            <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   296→                              <Calendar className="w-3.5 h-3.5" />
   297→                              {formatDate(project.dueDate)}
   298→                            </span>
   299→                          )}
   300→                          <span className="text-muted-foreground flex items-center gap-1">
   301→                            <Users className="w-3.5 h-3.5" />
   302→                            {project._count?.members || 0}
   303→                          </span>
   304→                          <Button
   305→                            size="sm"
   306→                            variant="outline"
   307→                            className="h-7 gap-1.5 text-xs"
   308→                            onClick={(e) => { e.stopPropagation(); openOnboardingDialog(project.id, project.name) }}
   309→                          >
   310→                            <ClipboardCheck className="w-3.5 h-3.5" />
   311→                            Onboarding
   312→                          </Button>
   313→                        </div>
   314→                      </div>
   315→
   316→                      {/* Progress bar with percentage */}
   317→                      <div className="mt-3 space-y-1">
   318→                        <div className="flex items-center justify-between text-xs">
   319→                          <span className="text-muted-foreground">{doneTasks}/{totalT} tasks completed</span>
   320→                          <span className="font-semibold" style={{ color: barColor }}>{pct}%</span>
   321→                        </div>
   322→                        <div className="relative">
   323→                          <Progress value={pct} className="h-2.5" />
   324→                          <div
   325→                            className="absolute top-0 left-0 h-2.5 rounded-full transition-all duration-500"
   326→                            style={{
   327→                              width: `${pct}%`,
   328→                              background: `linear-gradient(90deg, ${barColor}88, ${barColor})`,
   329→                            }}
   330→                          />
   331→                        </div>
   332→                      </div>
   333→                    </CardContent>
   334→                  </Card>
   335→
   336→                  {/* Expanded tasks table */}
   337→                  <AnimatePresence>
   338→                    {isExpanded && (
   339→                      <motion.div
   340→                        initial={{ opacity: 0, height: 0 }}
   341→                        animate={{ opacity: 1, height: 'auto' }}
   342→                        exit={{ opacity: 0, height: 0 }}
   343→                        transition={{ duration: 0.2 }}
   344→                        className="overflow-hidden"
   345→                      >
   346→                        <Card className="mt-2 rounded-t-none">
   347→                          <CardContent className="p-0">
   348→                            {projectTasks.length === 0 ? (
   349→                              <div className="py-8 text-center text-sm text-muted-foreground">
   350→                                No tasks in this project yet.
   351→                              </div>
   352→                            ) : (
   353→                              <div className="overflow-x-auto">
   354→                                <table className="w-full text-sm">
   355→                                  <thead>
   356→                                    <tr className="border-b bg-muted/50">
   357→                                      <th className="text-left p-3 font-medium text-muted-foreground">Task</th>
   358→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden md:table-cell">Status</th>
   359→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden sm:table-cell">Priority</th>
   360→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Due Date</th>
   361→                                      <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Assignees</th>
   362→                                    </tr>
   363→                                  </thead>
   364→                                  <tbody>
   365→                                    {projectTasks.map(task => (
   366→                                      <tr
   367→                                        key={task.id}
   368→                                        className="border-b last:border-0 hover:bg-muted/30 cursor-pointer transition-colors"
   369→                                        onClick={(e) => { e.stopPropagation(); openTaskDetail(task.id) }}
   370→                                      >
   371→                                        <td className="p-3">
   372→                                          <div className="font-medium">{task.title}</div>
   373→                                          {task.description && (
   374→                                            <div className="text-xs text-muted-foreground line-clamp-1 mt-0.5">{task.description}</div>
   375→                                          )}
   376→                                        </td>
   377→                                        <td className="p-3 hidden md:table-cell"><StatusBadge status={task.status} /></td>
   378→                                        <td className="p-3 hidden sm:table-cell"><PriorityBadge priority={task.priority} /></td>
   379→                                        <td className="p-3 hidden lg:table-cell text-muted-foreground text-xs">
   380→                                          {task.dueDate ? formatDate(task.dueDate) : '—'}
   381→                                        </td>
   382→                                        <td className="p-3 hidden lg:table-cell">
   383→                                          <div className="flex -space-x-2">
   384→                                            {task.assignees?.slice(0, 3).map(a => (
   385→                                              <Avatar key={a.id} className="w-6 h-6 border-2 border-background">
   386→                                                <AvatarFallback className="text-[10px] bg-muted">
   387→                                                  {getInitials(a.user?.name || '?')}
   388→                                                </AvatarFallback>
   389→                                              </Avatar>
   390→                                            ))}
   391→                                            {(task.assignees?.length || 0) > 3 && (
   392→                                              <span className="w-6 h-6 rounded-full bg-muted flex items-center justify-center text-[10px] border-2 border-background">
   393→                                                +{(task.assignees?.length || 0) - 3}
   394→                                              </span>
   395→                                            )}
   396→                                          </div>
   397→                                        </td>
   398→                                      </tr>
   399→                                    ))}
   400→                                  </tbody>
   401→                                </table>
   402→                              </div>
   403→                            )}
   404→                          </CardContent>
   405→                        </Card>
   406→                      </motion.div>
   407→                    )}
   408→                  </AnimatePresence>
   409→                </motion.div>
   410→              )
   411→            })}
   412→          </div>
   413→        </>
   414→      )}
   415→
   416→      {/* Create Onboarding Dialog */}
   417→      <Dialog open={onboardingDialogOpen} onOpenChange={setOnboardingDialogOpen}>
   418→        <DialogContent className="sm:max-w-lg">
   419→          <DialogHeader>
   420→            <DialogTitle className="flex items-center gap-2">
   421→              <ClipboardCheck className="w-5 h-5 text-primary" />
   422→              Create Onboarding
   423→            </DialogTitle>
   424→            <DialogDescription>Create an onboarding form for this project</DialogDescription>
   425→          </DialogHeader>
   426→          <div className="space-y-4 py-2">
   427→            <div className="space-y-2">
   428→              <Label>Title</Label>
   429→              <Input
   430→                placeholder="Onboarding form title"
   431→                value={onboardingTitle}
   432→                onChange={(e) => setOnboardingTitle(e.target.value)}
   433→              />
   434→            </div>
   435→            <div className="space-y-2">
   436→              <Label>Description (optional)</Label>
   437→              <Textarea
   438→                placeholder="Brief description of the onboarding..."
   439→                value={onboardingDesc}
   440→                onChange={(e) => setOnboardingDesc(e.target.value)}
   441→                rows={2}
   442→              />
   443→            </div>
   444→            <div className="space-y-2">
   445→              <div className="flex items-center justify-between">
   446→                <Label>Steps</Label>
   447→                <span className="text-xs text-muted-foreground">{onboardingSteps.length} / {ONBOARDING_STEPS.length} selected</span>
   448→              </div>
   449→              <div className="space-y-2 max-h-48 overflow-y-auto">
   450→                {ONBOARDING_STEPS.map((step) => {
   451→                  const isSelected = onboardingSteps.includes(step.id)
   452→                  return (
   453→                    <button
   454→                      key={step.id}
   455→                      type="button"
   456→                      onClick={() => toggleOnboardingStep(step.id)}
   457→                      className={cn(
   458→                        'w-full flex items-start gap-3 p-3 rounded-lg border text-left transition-colors',
   459→                        isSelected ? 'border-primary bg-primary/5' : 'border-border hover:bg-muted/50'
   460→                      )}
   461→                    >
   462→                      <div className={cn(
   463→                        'mt-0.5 h-5 w-5 rounded border-2 flex items-center justify-center shrink-0 transition-colors',
   464→                        isSelected ? 'border-primary bg-primary text-primary-foreground' : 'border-muted-foreground/30'
   465→                      )}>
   466→                        {isSelected && <Check className="w-3 h-3" />}
   467→                      </div>
   468→                      <div className="flex-1 min-w-0">
   469→                        <p className="text-sm font-medium">{step.label}</p>
   470→                        <p className="text-xs text-muted-foreground mt-0.5">{step.description}</p>
   471→                      </div>
   472→                    </button>
   473→                  )
   474→                })}
   475→              </div>
   476→            </div>
   477→          </div>
   478→          <DialogFooter>
   479→            <Button variant="outline" onClick={() => setOnboardingDialogOpen(false)}>Cancel</Button>
   480→            <Button
   481→              onClick={handleCreateOnboarding}
   482→              disabled={creatingOnboarding || !onboardingTitle.trim() || onboardingSteps.length === 0}
   483→              className="gap-2"
   484→            >
   485→              {creatingOnboarding && <Loader2 className="h-4 w-4 animate-spin" />}
   486→              <Plus className="h-4 w-4" />
   487→              Create Onboarding
   488→            </Button>
   489→          </DialogFooter>
   490→        </DialogContent>
   491→      </Dialog>
   492→
   493→      {/* Task Detail Dialog */}
   494→      <Dialog open={!!taskDetail} onOpenChange={(open) => { if (!open) setTaskDetail(null) }}>
   495→        <DialogContent className="max-w-2xl max-h-[85vh] overflow-hidden flex flex-col">
   496→          {taskDetailLoading ? (
   497→            <div className="p-6 space-y-4">
   498→              <Skeleton className="h-6 w-3/4" />
   499→              <Skeleton className="h-4 w-1/2" />
   500→              <Skeleton className="h-20 w-full" />
   501→            </div>
   502→          ) : taskDetail ? (
   503→            <TaskDetailDialog task={taskDetail} userId={user?.id || ''} onUpdate={() => openTaskDetail(taskDetail.id)} />
   504→          ) : null}
   505→        </DialogContent>
   506→      </Dialog>
   507→    </div>
   508→  )
   509→}

   511→// ─── Task Detail Dialog ─────────────────────────────────────
   512→function TaskDetailDialog({
   513→  task, userId, onUpdate
   514→}: {

   516→  userId: string
   517→  onUpdate: () => void

   536→        headers: { 'Content-Type': 'application/json' },
   537→        body: JSON.stringify({ content: newComment.trim(), taskId: task.id, userId })
   538→      })
   539→      if (!res.ok) throw new Error()
   540→      const data = await res.json()
   541→      setComments(prev => [...prev, data.comment])
   542→      setNewComment('')
   543→      toast.success('Comment added')

   545→      toast.error('Failed to add comment')
   546→    } finally {
   547→      setSendingComment(false)

   566→          </div>
   567→        )}
   568→      </DialogHeader>
   569→
   570→      <ScrollArea className="flex-1 -mx-6 px-6" ref={scrollRef}>
   571→        <div className="space-y-5 py-4">
   572→          {/* Description */}
   573→          {task.description && (
   574→            <div>
   575→              <h4 className="text-sm font-medium text-muted-foreground mb-1">Description</h4>

   580→          {/* Details Grid */}

   589→              </div>

   591→            <div className="space-y-1">
   592→              <span className="text-xs text-muted-foreground">Assignees</span>

   595→                  <Avatar key={a.id} className="w-6 h-6">
   596→                    <AvatarFallback className="text-[10px] bg-muted">

   598→                    </AvatarFallback>
   599→                  </Avatar>

   602→                  <span className="text-sm text-muted-foreground">Unassigned</span>
   603→                )}
   604→              </div>
   605→            </div>
   606→            <div className="space-y-1">
   607→              <span className="text-xs text-muted-foreground">Created</span>

   610→          </div>

   612→          <Separator />

   614→          {/* Comments */}

   617→              <MessageCircle className="w-3.5 h-3.5" />
   618→              Comments ({comments.length})
   619→            </h4>
   620→            <div className="space-y-3">
   621→              {comments.map(comment => (
   622→                <div key={comment.id} className="flex gap-3">
   623→                  <Avatar className="w-8 h-8 shrink-0">
   624→                    <AvatarFallback className="text-xs bg-muted">
   625→                      {getInitials(comment.user?.name || '?')}
   626→                    </AvatarFallback>
   627→                  </Avatar>
   628→                  <div className="min-w-0 flex-1">
   629→                    <div className="flex items-center gap-2">
   630→                      <span className="text-sm font-medium">{comment.user?.name || 'Unknown'}</span>

   633→                        <span className={`text-[10px] px-1.5 py-0.5 rounded-full ${getRoleBadgeClasses(comment.user.role)}`}>

   684→  const [phone, setPhone] = useState('')
   685→  const [company, setCompany] = useState('')
   686→
   687→  const [currentPassword, setCurrentPassword] = useState('')
   688→  const [newPassword, setNewPassword] = useState('')
   689→  const [confirmPassword, setConfirmPassword] = useState('')
   690→
   691→  useEffect(() => {
   692→    if (!user?.id) return
   693→    const fetchProfile = async () => {
   694→      try {
   695→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   696→        const data = await res.json()
   697→        if (data.user) {
   698→          setName(data.user.name || '')

   701→          setCompany(data.user.company || '')

   714→    setSaving(true)

   716→      const res = await fetch('/api/user/profile', {
   717→        method: 'PATCH',

   788→        <div>
   789→          <h1 className="text-2xl font-bold">Profile Settings</h1>

   801→                Client
   802→              </span>
   803→            </div>

   806→                Member since {formatDate(user.createdAt)}

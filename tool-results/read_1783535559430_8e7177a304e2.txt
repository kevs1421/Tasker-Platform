    80→    80→}) {

   130→   130→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())

   204→   204→  open, onOpenChange, group, currentUserId,

   206→   206→  open: boolean; onOpenChange: (open: boolean) => void
   207→   207→  group: { id: string; name: string; members?: GroupMember[]; _count?: { members: number } } | null
   208→   208→  currentUserId: string
   209→   209→}) {
   210→   210→  const [members, setMembers] = useState<(GroupMember & { user: User })[]>([])
   211→   211→  const [users, setUsers] = useState<User[]>([])
   212→   212→  const [loading, setLoading] = useState(false)
   213→   213→  const [showAddUser, setShowAddUser] = useState(false)
   214→   214→  const [search, setSearch] = useState('')
   215→   215→
   216→   216→  useEffect(() => {
   217→   217→    if (!open || !group) return
   218→   218→    setLoading(true)
   219→   219→    fetch(`/api/message-groups/${group.id}?userId=${currentUserId}`)
   220→   220→      .then((r) => r.json())

   222→   222→        setMembers((d.group?.members ?? []) as (GroupMember & { user: User })[])

   225→   225→      .finally(() => setLoading(false))

   230→   230→  const handleAddMember = async (userId: string) => {

   233→   233→      const res = await fetch(`/api/message-groups/${group.id}`, {
   234→   234→        method: 'PATCH',
   235→   235→        headers: { 'Content-Type': 'application/json' },
   236→   236→        body: JSON.stringify({ addMemberIds: [userId] }),
   237→   237→      })

   243→   243→      toast.error('Failed to add member')

   248→   248→    if (!group) return
   249→   249→    try {
   250→   250→      const res = await fetch(`/api/message-groups/${group.id}`, {
   251→   251→        method: 'PATCH',
   252→   252→        headers: { 'Content-Type': 'application/json' },
   253→   253→        body: JSON.stringify({ removeMemberIds: [userId] }),
   254→   254→      })
   255→   255→      if (!res.ok) throw new Error()
   256→   256→      const data = await res.json()
   257→   257→      setMembers(data.group.members ?? [])
   258→   258→      toast.success('Member removed')

   269→   269→        setUsers((d.users ?? []).filter((u: User) => !memberIds.includes(u.id)))
   270→   270→      })
   271→   271→      .catch(() => {})
   272→   272→    setShowAddUser(true)
   273→   273→  }
   274→   274→
   275→   275→  const filteredUsers = users.filter(
   276→   276→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   277→   277→  )
   278→   278→
   279→   279→  return (
   280→   280→    <Dialog open={open} onOpenChange={onOpenChange}>
   281→   281→      <DialogContent className="sm:max-w-md">
   282→   282→        <DialogHeader>
   283→   283→          <DialogTitle className="flex items-center gap-2">
   284→   284→            <Users className="h-5 w-5 text-violet-600" /> Group Info
   285→   285→          </DialogTitle>
   286→   286→        </DialogHeader>
   287→   287→
   288→   288→        <div className="space-y-4 py-2">
   289→   289→          {group && (
   290→   290→            <div className="flex items-center gap-3">
   291→   291→              <GroupAvatar name={group.name} size="lg" />
   292→   292→              <div>
   293→   293→                <p className="font-semibold">{group.name}</p>

   295→   295→              </div>
   296→   296→            </div>
   297→   297→          )}
   298→   298→
   299→   299→          <Separator />
   300→   300→
   301→   301→          <div>
   302→   302→            <div className="flex items-center justify-between mb-2">
   303→   303→              <p className="text-sm font-medium">Members</p>
   304→   304→              {!showAddUser && (

   371→   371→export function MessagesPanel() {

   375→   375→  const [selectedConvo, setSelectedConvo] = useState<Conversation | null>(null)

   377→   377→  const [msgText, setMsgText] = useState('')

   381→   381→  const [showCreateGroup, setShowCreateGroup] = useState(false)

   383→   383→  const messagesEndRef = useRef<HTMLDivElement>(null)

   385→   385→  const fetchConversations = useCallback(async () => {
   386→   386→    if (!user?.id) return
   387→   387→    setLoading(true)
   388→   388→    try {
   389→   389→      const res = await fetch(`/api/messages?userId=${user.id}`)

   391→   391→      const data = await res.json()

   394→   394→      toast.error('Failed to load conversations')
   395→   395→    } finally {
   396→   396→      setLoading(false)
   397→   397→    }
   398→   398→  }, [user?.id])
   399→   399→
   400→   400→  useEffect(() => { fetchConversations() }, [fetchConversations])
   401→   401→
   402→   402→  useEffect(() => {
   403→   403→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   404→   404→  }, [messages])

   406→   406→  // Determine conversation kind
   407→   407→  const isGroupSelected = selectedConvo?.type === 'group'
   408→   408→  const selectedGroup = isGroupSelected ? (selectedConvo as GroupConversation).group : null
   409→   409→  const selectedDirectUser = !isGroupSelected ? (selectedConvo as DirectConversation | null)?.user : null
   410→   410→
   411→   411→  const selectConversation = async (convo: Conversation) => {
   412→   412→    if (!user?.id) return
   413→   413→    setSelectedConvo(convo)
   414→   414→    setMsgText('')
   415→   415→    setMsgLoading(true)

   417→   417→    try {

   419→   419→      if (convo.type === 'group') {
   420→   420→        url += `&groupId=${convo.group.id}`
   421→   421→      } else {
   422→   422→        if (!convo.user) return
   423→   423→        url += `&otherId=${convo.user.id}`
   424→   424→      }
   425→   425→      const res = await fetch(url)
   426→   426→      if (!res.ok) throw new Error()
   427→   427→      const data = await res.json()
   428→   428→      setMessages(data.messages ?? [])
   429→   429→    } catch {
   430→   430→      toast.error('Failed to load messages')
   431→   431→    } finally {
   432→   432→      setMsgLoading(false)
   433→   433→    }
   434→   434→
   435→   435→    // Clear unread locally
   436→   436→    setConversations((prev) => prev.map((c) => {
   437→   437→      if (c.type === 'direct' && convo.type === 'direct' && c.user?.id === convo.user?.id) return { ...c, unread: 0 }
   438→   438→      if (c.type === 'group' && convo.type === 'group' && c.group.id === convo.group.id) return { ...c, unread: 0 }
   439→   439→      return c
   440→   440→    }))
   441→   441→  }
   442→   442→
   443→   443→  const sendMessage = async () => {
   444→   444→    if (!user?.id || !msgText.trim()) return
   445→   445→    if (!selectedConvo) return
   446→   446→
   447→   447→    setSending(true)
   448→   448→    const text = msgText.trim()
   449→   449→    setMsgText('')
   450→   450→
   451→   451→    try {
   452→   452→      const body: Record<string, string> = {
   453→   453→        content: text,
   454→   454→        senderId: user.id,
   455→   455→      }
   456→   456→
   457→   457→      if (selectedConvo.type === 'group') {
   458→   458→        body.groupId = selectedConvo.group.id
   459→   459→      } else {
   460→   460→        if (!selectedConvo.user) return
   461→   461→        body.receiverId = selectedConvo.user.id
   462→   462→      }
   463→   463→
   464→   464→      const res = await fetch('/api/messages', {
   465→   465→        method: 'POST',
   466→   466→        headers: { 'Content-Type': 'application/json' },
   467→   467→        body: JSON.stringify(body),
   468→   468→      })
   469→   469→      if (!res.ok) throw new Error()
   470→   470→      const data = await res.json()
   471→   471→      setMessages((prev) => [...prev, data.message])
   472→   472→      fetchConversations()
   473→   473→    } catch {
   474→   474→      toast.error('Failed to send message')
   475→   475→      setMsgText(text)
   476→   476→    } finally {
   477→   477→      setSending(false)
   478→   478→    }
   479→   479→  }
   480→   480→
   481→   481→  const handleKeyDown = (e: React.KeyboardEvent) => {
   482→   482→    if (e.key === 'Enter' && !e.shiftKey) {
   483→   483→      e.preventDefault()
   484→   484→      sendMessage()
   485→   485→    }
   486→   486→  }
   487→   487→
   488→   488→  // Filter conversations
   489→   489→  const filteredConversations = conversations.filter((c) => {
   490→   490→    if (!searchQuery.trim()) return true
   491→   491→    const q = searchQuery.toLowerCase()
   492→   492→    if (c.type === 'direct') return c.user?.name.toLowerCase().includes(q) || c.user?.email.toLowerCase().includes(q)
   493→   493→    return c.group.name.toLowerCase().includes(q)
   494→   494→  })
   495→   495→
   496→   496→  // Conversation display helpers
   497→   497→  const getConvoTitle = (c: Conversation) => c.type === 'direct' ? (c.user?.name ?? 'Unknown') : c.group.name
   498→   498→  const getConvoSubtitle = (c: Conversation) => {
   499→   499→    if (c.type === 'group') {
   500→   500→      return `${c.group._count.members} members`
   501→   501→    }
   502→   502→    return c.user?.company ?? c.user?.email ?? ''
   503→   503→  }
   504→   504→  const getConvoPreview = (c: Conversation) => {
   505→   505→    if (!c.lastMessage) return 'No messages yet'
   506→   506→    const senderName = c.lastMessage.senderId === user?.id ? 'You' : (c.lastMessage.sender?.name ?? 'Someone')
   507→   507→    if (c.type === 'group') {
   508→   508→      return `${senderName}: ${c.lastMessage.content}`
   509→   509→    }
   510→   510→    return `${c.lastMessage.senderId === user?.id ? 'You: ' : ''}${c.lastMessage.content}`
   511→   511→  }
   512→   512→
   513→   513→  // ── Render ───────────────────────────────────────────────
   514→   514→
   515→   515→  const containerHeight = 'h-[calc(100vh-12rem)]'
   516→   516→
   517→   517→  if (loading) {
   518→   518→    return (
   519→   519→      <div className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}>
   520→   520→        <div className="w-80 border-r p-4 space-y-3 flex-shrink-0">
   521→   521→          <Skeleton className="h-10 w-full" />
   522→   522→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-14 w-full" />)}
   523→   523→        </div>
   524→   524→        <div className="flex-1 flex items-center justify-center">
   525→   525→          <p className="text-muted-foreground text-sm">Select a conversation</p>
   526→   526→        </div>
   527→   527→      </div>
   528→   528→    )
   529→   529→  }
   530→   530→
   531→   531→  return (
   532→   532→    <motion.div
   533→   533→      initial={{ opacity: 0, y: 8 }}
   534→   534→      animate={{ opacity: 1, y: 0 }}
   535→   535→      className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}
   536→   536→    >
   537→   537→      {/* ─── Conversation List ──────────────────────────── */}
   538→   538→      <div className="w-80 border-r bg-background flex flex-col flex-shrink-0">
   539→   539→        <div className="p-3 border-b space-y-2">
   540→   540→          <div className="flex items-center justify-between">
   541→   541→            <h2 className="text-lg font-semibold">Messages</h2>
   542→   542→            <Button
   543→   543→              variant="ghost" size="icon" className="h-8 w-8"
   544→   544→              onClick={() => setShowCreateGroup(true)}
   545→   545→              title="Create Group"
   546→   546→            >
   547→   547→              <Plus className="h-4 w-4" />
   548→   548→            </Button>
   549→   549→          </div>
   550→   550→          <div className="relative">
   551→   551→            <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   552→   552→            <Input
   553→   553→              placeholder="Search conversations..."
   554→   554→              className="pl-9 h-9 text-sm"
   555→   555→              value={searchQuery}
   556→   556→              onChange={(e) => setSearchQuery(e.target.value)}
   557→   557→            />
   558→   558→          </div>
   559→   559→        </div>
   560→   560→        <ScrollArea className="flex-1">
   561→   561→          <div className="p-1">
   562→   562→            {filteredConversations.length === 0 && (
   563→   563→              <p className="text-sm text-muted-foreground text-center py-8">
   564→   564→                {searchQuery ? 'No matching conversations' : 'No conversations yet'}
   565→   565→              </p>
   566→   566→            )}
   567→   567→            {filteredConversations.map((convo, idx) => {
   568→   568→              const isSelected =
   569→   569→                convo.type === 'direct'
   570→   570→                  ? selectedDirectUser?.id === convo.user?.id
   571→   571→                  : selectedGroup?.id === convo.group.id
   572→   572→              return (
   573→   573→                <button
   574→   574→                  key={convo.type === 'direct' ? convo.user?.id ?? idx : convo.group.id}
   575→   575→                  className={cn(
   576→   576→                    'w-full text-left px-3 py-3 rounded-md flex items-start gap-3 transition-colors',
   577→   577→                    isSelected ? 'bg-muted' : 'hover:bg-muted/50'
   578→   578→                  )}
   579→   579→                  onClick={() => selectConversation(convo)}
   580→   580→                >
   581→   581→                  {convo.type === 'direct' ? (
   582→   582→                    <UserAvatar user={convo.user} size="md" className="mt-0.5 flex-shrink-0" />
   583→   583→                  ) : (
   584→   584→                    <GroupAvatar name={convo.group.name} size="md" className="mt-0.5 flex-shrink-0" />
   585→   585→                  )}
   586→   586→                  <div className="flex-1 min-w-0">
   587→   587→                    <div className="flex items-center justify-between">
   588→   588→                      <div className="flex items-center gap-1.5 min-w-0">
   589→   589→                        <span className="text-sm font-medium truncate">{getConvoTitle(convo)}</span>
   590→   590→                        {convo.type === 'group' && (
   591→   591→                          <Users className="h-3 w-3 text-violet-500 flex-shrink-0" />
   592→   592→                        )}
   593→   593→                      </div>
   594→   594→                      {convo.lastMessage && (
   595→   595→                        <span className="text-[10px] text-muted-foreground flex-shrink-0 ml-2">
   596→   596→                          {formatRelativeTime(convo.lastMessage.createdAt)}
   597→   597→                        </span>
   598→   598→                      )}
   599→   599→                    </div>
   600→   600→                    <p className="text-xs text-muted-foreground truncate mt-0.5">
   601→   601→                      {getConvoPreview(convo)}
   602→   602→                    </p>
   603→   603→                  </div>
   604→   604→                  {convo.unread > 0 && (
   605→   605→                    <div className="h-5 w-5 rounded-full bg-emerald-500 text-white text-[10px] font-bold flex items-center justify-center flex-shrink-0 mt-1">
   606→   606→                      {convo.unread}
   607→   607→                    </div>
   608→   608→                  )}
   609→   609→                </button>
   610→   610→              )
   611→   611→            })}
   612→   612→          </div>
   613→   613→        </ScrollArea>
   614→   614→      </div>
   615→   615→
   616→   616→      {/* ─── Chat Area ─────────────────────────────────── */}
   617→   617→      <div className="flex-1 flex flex-col bg-background">
   618→   618→        {selectedConvo ? (
   619→   619→          <>
   620→   620→            {/* Chat Header */}
   621→   621→            <div className="p-3 border-b flex items-center gap-3">
   622→   622→              {isGroupSelected ? (
   623→   623→                <>
   624→   624→                  <button onClick={() => setShowGroupInfo(true)} className="flex items-center gap-3 hover:bg-muted/50 rounded-md px-1 -ml-1 py-0.5 transition-colors">
   625→   625→                    <GroupAvatar name={selectedGroup!.name} size="md" />
   626→   626→                    <div>
   627→   627→                      <p className="text-sm font-medium flex items-center gap-1.5">
   628→   628→                        {selectedGroup!.name}
   629→   629→                        <Users className="h-3.5 w-3.5 text-violet-500" />
   630→   630→                      </p>
   631→   631→                      <p className="text-xs text-muted-foreground">
   632→   632→                        {selectedGroup!._count.members} members
   633→   633→                      </p>
   634→   634→                    </div>
   635→   635→                  </button>
   636→   636→                </>
   637→   637→              ) : selectedDirectUser ? (
   638→   638→                <>
   639→   639→                  <UserAvatar user={selectedDirectUser} size="md" />
   640→   640→                  <div>
   641→   641→                    <p className="text-sm font-medium">{selectedDirectUser.name}</p>
   642→   642→                    <p className="text-xs text-muted-foreground">
   643→   643→                      {selectedDirectUser.company ?? selectedDirectUser.email}
   644→   644→                    </p>
   645→   645→                  </div>
   646→   646→                  <Badge variant="secondary" className={cn('ml-auto text-xs', getRoleBadgeClasses(selectedDirectUser.role))}>
   647→   647→                    {selectedDirectUser.role}
   648→   648→                  </Badge>
   649→   649→                </>
   650→   650→              ) : null}
   651→   651→            </div>
   652→   652→
   653→   653→            {/* Messages */}
   654→   654→            <ScrollArea className="flex-1 p-4">
   655→   655→              <div className="space-y-3 max-w-2xl mx-auto">
   656→   656→                {msgLoading ? (
   657→   657→                  <div className="space-y-3">
   658→   658→                    {Array.from({ length: 4 }).map((_, i) => (
   659→   659→                      <div key={i} className={cn('flex gap-2.5', i % 2 === 0 ? '' : 'justify-end')}>
   660→   660→                        <Skeleton className="h-10 w-48 rounded-2xl" />
   661→   661→                      </div>
   662→   662→                    ))}
   663→   663→                  </div>
   664→   664→                ) : messages.length === 0 ? (
   665→   665→                  <div className="flex flex-col items-center justify-center h-full text-center py-8">
   666→   666→                    <MessageSquare className="h-10 w-10 text-muted-foreground/30 mb-3" />
   667→   667→                    <p className="text-sm text-muted-foreground">
   668→   668→                      {isGroupSelected ? 'No messages yet. Start the conversation!' : 'No messages yet. Say hello!'}
   669→   669→                    </p>
   670→   670→                  </div>
   671→   671→                ) : (
   672→   672→                  messages.map((msg, idx) => {
   673→   673→                    const isSent = msg.senderId === user?.id
   674→   674→                    return (
   675→   675→                      <motion.div
   676→   676→                        key={msg.id}
   677→   677→                        initial={{ opacity: 0, y: 5 }}
   678→   678→                        animate={{ opacity: 1, y: 0 }}
   679→   679→                        transition={{ delay: Math.min(idx * 0.015, 0.2) }}
   680→   680→                        className={cn('flex gap-2.5', isSent ? 'justify-end' : 'justify-start')}
   681→   681→                      >
   682→   682→                        {!isSent && (
   683→   683→                          <UserAvatar user={msg.sender} size="sm" className="mt-auto" />
   684→   684→                        )}
   685→   685→                        <div className={cn('max-w-[70%]', isSent ? 'text-right' : '')}>
   686→   686→                          {/* Show sender name in group chats */}
   687→   687→                          {isGroupSelected && !isSent && (
   688→   688→                            <p className="text-xs font-medium text-muted-foreground mb-1 px-1">
   689→   689→                              {msg.sender?.name ?? 'Unknown'}
   690→   690→                            </p>
   691→   691→                          )}
   692→   692→                          <div className={cn(
   693→   693→                            'inline-block rounded-2xl px-4 py-2.5 text-sm',
   694→   694→                            isSent
   695→   695→                              ? 'bg-emerald-600 text-white rounded-br-md'
   696→   696→                              : 'bg-muted rounded-bl-md',
   697→   697→                          )}>
   698→   698→                            <p className="whitespace-pre-wrap break-words">{msg.content}</p>
   699→   699→                          </div>
   700→   700→                          <div className={cn(
   701→   701→                            'flex items-center gap-1 mt-1 text-[10px] text-muted-foreground',
   702→   702→                            isSent ? 'justify-end' : 'justify-start'
   703→   703→                          )}>
   704→   704→                            <span>{formatRelativeTime(msg.createdAt)}</span>
   705→   705→                            {isSent && msg.read && <CheckCheck className="w-3 h-3" />}
   706→   706→                          </div>
   707→   707→                        </div>
   708→   708→                      </motion.div>
   709→   709→                    )
   710→   710→                  })
   711→   711→                )}
   712→   712→                <div ref={messagesEndRef} />
   713→   713→              </div>
   714→   714→            </ScrollArea>
   715→   715→
   716→   716→            {/* Message Input */}
   717→   717→            <div className="p-3 border-t">
   718→   718→              <div className="flex gap-2 max-w-2xl mx-auto">
   719→   719→                <Input
   720→   720→                  placeholder={isGroupSelected ? `Message ${selectedGroup!.name}...` : 'Type a message...'}
   721→   721→                  value={msgText}
   722→   722→                  onChange={(e) => setMsgText(e.target.value)}
   723→   723→                  onKeyDown={handleKeyDown}
   724→   724→                  className="flex-1"
   725→   725→                />
   726→   726→                <Button
   727→   727→                  onClick={sendMessage}
   728→   728→                  disabled={!msgText.trim() || sending}
   729→   729→                  className="bg-emerald-600 hover:bg-emerald-700 text-white flex-shrink-0"
   730→   730→                >
   731→   731→                  {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   732→   732→                </Button>
   733→   733→              </div>
   734→   734→            </div>
   735→   735→          </>
   736→   736→        ) : (
   737→   737→          <div className="flex-1 flex items-center justify-center">
   738→   738→            <div className="text-center space-y-3">
   739→   739→              <div className="h-16 w-16 rounded-full bg-muted flex items-center justify-center mx-auto">
   740→   740→                <MessageSquare className="h-8 w-8 text-muted-foreground" />
   741→   741→              </div>
   742→   742→              <h3 className="font-semibold text-lg">Select a conversation</h3>
   743→   743→              <p className="text-sm text-muted-foreground max-w-xs">
   744→   744→                Choose a conversation from the sidebar or create a new group.
   745→   745→              </p>
   746→   746→            </div>
   747→   747→          </div>
   748→   748→        )}
   749→   749→      </div>
   750→   750→
   751→   751→      {/* Dialogs */}
   752→   752→      <CreateGroupDialog
   753→   753→        open={showCreateGroup}
   754→   754→        onOpenChange={setShowCreateGroup}
   755→   755→        onCreated={() => { fetchConversations(); setShowCreateGroup(false) }}
   756→   756→        currentUser={user!}
   757→   757→      />
   758→   758→      <GroupInfoDialog
   759→   759→        open={showGroupInfo}
   760→   760→        onOpenChange={setShowGroupInfo}
   761→   761→        group={selectedGroup ? { id: selectedGroup.id, name: selectedGroup.name, _count: selectedGroup._count } : null}
   762→   762→        currentUserId={user?.id ?? ''}
   763→   763→      />
   764→   764→    </motion.div>
   765→   765→  )

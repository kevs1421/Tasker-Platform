    99→    99→const ONBOARDING_STEPS = [
   100→   100→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   101→   101→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },

   103→   103→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   104→   104→]
   105→   105→
   106→   106→export function ClientProjects() {
   107→   107→  const { user } = useAuthStore()
   108→   108→  const [projects, setProjects] = useState<ClientProject[]>([])
   109→   109→  const [loading, setLoading] = useState(true)
   110→   110→
   111→   111→  // Upload files state
   112→   112→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   113→   113→  const [uploading, setUploading] = useState(false)
   114→   114→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   115→   115→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   116→   116→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   117→   117→  const fileInputRef = useRef<HTMLInputElement>(null)
   118→   118→
   119→   119→  // Onboarding creation state
   120→   120→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   121→   121→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   122→   122→  const [onboardingTitle, setOnboardingTitle] = useState('')
   123→   123→  const [onboardingDesc, setOnboardingDesc] = useState('')
   124→   124→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   125→   125→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   126→   126→
   127→   127→  const fetchProjects = useCallback(async () => {
   128→   128→    try {
   129→   129→      const res = await fetch('/api/projects')
   130→   130→      const data = await res.json()
   131→   131→      // Filter to active projects and those the client is a member of
   132→   132→      const clientProjects = (data.projects || [])
   133→   133→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   134→   134→      setProjects(clientProjects)
   135→   135→    } catch {
   136→   136→      toast.error('Failed to load projects')
   137→   137→    } finally {
   138→   138→      setLoading(false)
   139→   139→    }
   140→   140→  }, [user?.id])
   141→   141→
   142→   142→  // Fetch uploaded files from localStorage
   143→   143→  const fetchUploadedFiles = useCallback(() => {
   144→   144→    if (!user) return
   145→   145→    try {
   146→   146→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   147→   147→      if (stored) setUploadedFiles(JSON.parse(stored))
   148→   148→    } catch { /* ignore */ }
   149→   149→  }, [user])
   150→   150→
   151→   151→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   152→   152→    if (!user) return
   153→   153→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   154→   154→  }, [user])
   155→   155→
   156→   156→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   157→   157→
   158→   158→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   159→   159→  const handleFileUpload = async (files: FileList | null) => {
   160→   160→    if (!files || files.length === 0) return
   161→   161→    if (!uploadDialogProjectId) {
   162→   162→      toast.error('No project selected')
   163→   163→      return
   164→   164→    }
   165→   165→    setUploading(true)
   166→   166→    const newFiles: typeof uploadedFiles = []
   167→   167→    for (let i = 0; i < files.length; i++) {
   168→   168→      const file = files[i]
   169→   169→      try {
   170→   170→        const formData = new FormData()
   171→   171→        formData.append('file', file)
   172→   172→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   173→   173→        if (!res.ok) {
   174→   174→          const d = await res.json()
   175→   175→          throw new Error(d.error || 'Upload failed')
   176→   176→        }
   177→   177→        const data = await res.json()
   178→   178→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   179→   179→      } catch (e: unknown) {
   180→   180→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   181→   181→      }
   182→   182→    }
   183→   183→    if (newFiles.length > 0) {
   184→   184→      const updated = [...uploadedFiles, ...newFiles]
   185→   185→      setUploadedFiles(updated)
   186→   186→      saveUploadedFiles(updated)
   187→   187→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   188→   188→    }
   189→   189→    setUploading(false)
   190→   190→    if (fileInputRef.current) fileInputRef.current.value = ''
   191→   191→  }
   192→   192→
   193→   193→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   194→   194→    e.preventDefault()
   195→   195→    setUploadDragOver(null)
   196→   196→    setUploadDialogProjectId(projectId)
   197→   197→    handleFileUpload(e.dataTransfer.files)
   198→   198→  }
   199→   199→
   200→   200→  const openUploadDialog = (projectId: string) => {
   201→   201→    setUploadDialogProjectId(projectId)
   202→   202→    setUploadDialogOpen(true)
   203→   203→  }
   204→   204→
   205→   205→  const deleteUploadedFile = (idx: number) => {
   206→   206→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   207→   207→    setUploadedFiles(updated)
   208→   208→    saveUploadedFiles(updated)
   209→   209→    toast.success('File removed')
   210→   210→  }
   211→   211→
   212→   212→  const getFileIcon = (type: string) => {
   213→   213→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   214→   214→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   215→   215→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   216→   216→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   217→   217→    return <File className="w-4 h-4 text-muted-foreground" />
   218→   218→  }
   219→   219→
   220→   220→  const formatFileSize = (bytes: number) => {
   221→   221→    if (bytes < 1024) return `${bytes} B`
   222→   222→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   223→   223→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   224→   224→  }
   225→   225→
   226→   226→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   227→   227→    setOnboardingProjectId(projectId)
   228→   228→    setOnboardingTitle(`Onboarding - ${projectName}`)
   229→   229→    setOnboardingDesc('')
   230→   230→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   231→   231→    setOnboardingDialogOpen(true)
   232→   232→  }
   233→   233→
   234→   234→  const handleCreateOnboarding = async () => {
   235→   235→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   236→   236→    setCreatingOnboarding(true)
   237→   237→    try {
   238→   238→      const stepsJson = JSON.stringify(
   239→   239→        onboardingSteps.map(stepId => {
   240→   240→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   241→   241→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   242→   242→        })
   243→   243→      )
   244→   244→      const res = await fetch('/api/onboarding', {
   245→   245→        method: 'POST',
   246→   246→        headers: { 'Content-Type': 'application/json' },
   247→   247→        body: JSON.stringify({
   248→   248→          title: onboardingTitle.trim(),
   249→   249→          description: onboardingDesc.trim() || null,
   250→   250→          stepsJson,
   251→   251→          projectId: onboardingProjectId,
   252→   252→          clientId: user?.id || null,
   253→   253→          status: 'published',
   254→   254→        }),
   255→   255→      })
   256→   256→      if (!res.ok) {
   257→   257→        const d = await res.json()
   258→   258→        throw new Error(d.error || 'Failed to create')
   259→   259→      }
   260→   260→      toast.success('Onboarding form created successfully')
   261→   261→      setOnboardingDialogOpen(false)
   262→   262→    } catch (e: unknown) {
   263→   263→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   264→   264→    } finally {
   265→   265→      setCreatingOnboarding(false)
   266→   266→    }
   267→   267→  }
   268→   268→
   269→   269→  const toggleOnboardingStep = (stepId: string) => {
   270→   270→    setOnboardingSteps(prev =>
   271→   271→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   272→   272→    )
   273→   273→  }
   274→   274→
   275→   275→  if (loading) return <PageSkeleton />
   276→   276→
   277→   277→  return (
   278→   278→    <div className="p-4 md:p-6 space-y-6">
   279→   279→      {/* Page Header */}
   280→   280→      <div className="flex items-center gap-3">
   281→   281→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   282→   282→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   283→   283→        </div>
   284→   284→        <div>
   285→   285→          <h1 className="text-2xl font-bold">My Projects</h1>
   286→   286→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   287→   287→        </div>
   288→   288→      </div>
   289→   289→
   290→   290→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   291→   291→      <input
   292→   292→        ref={fileInputRef}
   293→   293→        type="file"
   294→   294→        multiple
   295→   295→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   296→   296→        className="hidden"
   297→   297→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   298→   298→      />
   299→   299→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   300→   300→        <DialogContent className="sm:max-w-md">
   301→   301→          <DialogHeader>
   302→   302→            <DialogTitle className="flex items-center gap-2">
   303→   303→              <Upload className="w-5 h-5 text-primary" />
   304→   304→              Upload Files
   305→   305→            </DialogTitle>
   306→   306→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   307→   307→          </DialogHeader>
   308→   308→          <div
   309→   309→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   310→   310→            onDragLeave={() => setUploadDragOver(null)}
   311→   311→            onDrop={(e) => {
   312→   312→              e.preventDefault()
   313→   313→              setUploadDragOver(null)
   314→   314→              // Set the project context before handling files
   315→   315→              const files = e.dataTransfer.files
   316→   316→              if (files && files.length > 0) {
   317→   317→                // Use a separate handler for dialog drops
   318→   318→                void (async () => {
   319→   319→                  setUploading(true)
   320→   320→                  const newFiles: typeof uploadedFiles = []
   321→   321→                  for (let i = 0; i < files.length; i++) {
   322→   322→                    const file = files[i]
   323→   323→                    try {
   324→   324→                      const formData = new FormData()
   325→   325→                      formData.append('file', file)
   326→   326→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   327→   327→                      if (!res.ok) {
   328→   328→                        const d = await res.json()
   329→   329→                        throw new Error(d.error || 'Upload failed')
   330→   330→                      }
   331→   331→                      const data = await res.json()
   332→   332→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   333→   333→                    } catch (err: unknown) {
   334→   334→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   335→   335→                    }
   336→   336→                  }
   337→   337→                  if (newFiles.length > 0) {
   338→   338→                    const updated = [...uploadedFiles, ...newFiles]
   339→   339→                    setUploadedFiles(updated)
   340→   340→                    saveUploadedFiles(updated)
   341→   341→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   342→   342→                  }
   343→   343→                  setUploading(false)
   344→   344→                })()
   345→   345→              }
   346→   346→            }}
   347→   347→            onClick={() => fileInputRef.current?.click()}
   348→   348→            className={cn(
   349→   349→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   350→   350→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   351→   351→            )}
   352→   352→          >
   353→   353→            {uploading ? (
   354→   354→              <div className="flex flex-col items-center gap-2">
   355→   355→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   356→   356→                <p className="text-sm text-muted-foreground">Uploading...</p>
   357→   357→              </div>
   358→   358→            ) : (
   359→   359→              <div className="flex flex-col items-center gap-2">
   360→   360→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   361→   361→                  <Upload className="w-5 h-5 text-muted-foreground" />
   362→   362→                </div>
   363→   363→                <div>
   364→   364→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   365→   365→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   366→   366→                </div>
   367→   367→              </div>
   368→   368→            )}
   369→   369→          </div>
   370→   370→          <DialogFooter>
   371→   371→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   372→   372→          </DialogFooter>
   373→   373→        </DialogContent>
   374→   374→      </Dialog>
   375→   375→
   376→   376→      {/* ─── Project Progress Overview ──────────────────────────── */}
   377→   377→      {projects.length > 0 && (
   378→   378→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   379→   379→          <Card>
   380→   380→            <CardHeader className="pb-3">
   381→   381→              <CardTitle className="text-base flex items-center gap-2">
   382→   382→                <BarChart3 className="w-4 h-4 text-primary" />
   383→   383→                Project Progress
   384→   384→              </CardTitle>
   385→   385→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   386→   386→            </CardHeader>
   387→   387→            <CardContent>
   388→   388→              <div className="space-y-5">
   389→   389→                {projects.map((project) => {
   390→   390→                  const todo = project.taskStats?.todo || 0
   391→   391→                  const inProgress = project.taskStats?.in_progress || 0
   392→   392→                  const completed = project.taskStats?.done || 0
   393→   393→                  const total = todo + inProgress + completed
   394→   394→                  if (total === 0) return null
   395→   395→
   396→   396→                  const todoPct = Math.round((todo / total) * 100)
   397→   397→                  const inProgressPct = Math.round((inProgress / total) * 100)
   398→   398→                  const completedPct = Math.round((completed / total) * 100)
   399→   399→
   400→   400→                  return (
   401→   401→                    <div key={project.id} className="space-y-2.5">
   402→   402→                      <div className="flex items-center justify-between">
   403→   403→                        <div className="flex items-center gap-2">
   404→   404→                          <span className="w-3 h-3 rounded-full" style={{ backgroundColor: project.color }} />
   405→   405→                          <span className="text-sm font-medium">{project.name}</span>
   406→   406→                        </div>
   407→   407→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   408→   408→                      </div>
   409→   409→                      {/* Stacked progress bar */}
   410→   410→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   411→   411→                        {todo > 0 && (
   412→   412→                          <div
   413→   413→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   414→   414→                            style={{ width: `${todoPct}%` }}
   415→   415→                            title={`Not Started: ${todo}`}
   416→   416→                          />
   417→   417→                        )}
   418→   418→                        {inProgress > 0 && (
   419→   419→                          <div
   420→   420→                            className="bg-amber-500 transition-all duration-500"
   421→   421→                            style={{ width: `${inProgressPct}%` }}
   422→   422→                            title={`In Progress: ${inProgress}`}
   423→   423→                          />
   424→   424→                        )}
   425→   425→                        {completed > 0 && (

    65→function ActivityIcon({ action }: { action: string }) {
    66→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    67→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    68→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    69→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    70→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    71→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    72→  return <CircleDot className="h-4 w-4 text-slate-400" />
    73→}

   360→  async function openForm(form: OnboardingForm) {

   372→  async function handleCreateOnboarding() {
   373→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   374→    setCreating(true)
   375→    try {
   376→      const stepsJson = JSON.stringify(
   377→        createForm.selectedSteps.map(stepId => {
   378→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   379→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   380→        })
   381→      )
   382→      const res = await fetch('/api/onboarding', {
   383→        method: 'POST',
   384→        headers: { 'Content-Type': 'application/json' },
   385→        body: JSON.stringify({
   386→          title: createForm.title.trim(),
   387→          description: createForm.description.trim() || null,
   388→          stepsJson,
   389→          projectId: createForm.projectId,
   390→          clientId: createForm.clientId || null,
   391→          status: 'published',
   392→        }),
   393→      })
   394→      if (!res.ok) {
   395→        const d = await res.json()
   396→        throw new Error(d.error || 'Failed to create')
   397→      }
   398→      toast.success('Onboarding form created successfully')
   399→      setCreateOpen(false)
   400→      setCreateForm({ title: '', description: '', projectId: '', clientId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })
   401→      fetchForms()
   402→    } catch (e: unknown) {
   403→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding form')
   404→    } finally {
   405→      setCreating(false)
   406→    }
   407→  }

   409→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   410→    if (!user) return
   411→    setReviewing(submissionId)
   412→    try {
   413→      const notes = reviewNotes[submissionId] || ''
   414→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   415→        method: 'PATCH',
   416→        headers: { 'Content-Type': 'application/json' },
   417→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   418→      })
   419→      if (!patchRes.ok) {
   420→        const d = await patchRes.json()
   421→        throw new Error(d.error)
   422→      }
   423→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   424→      // Refresh
   425→      if (selectedForm) {
   426→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   427→        if (fresh.ok) {
   428→          const data = await fresh.json()
   429→          setSelectedForm(data.form)
   430→        }
   431→      }
   432→      fetchForms()
   433→    } catch (e: unknown) {
   434→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   435→    } finally {
   436→      setReviewing(null)
   437→    }
   438→  }

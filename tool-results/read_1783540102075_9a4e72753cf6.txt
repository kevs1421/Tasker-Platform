    63→function ActivityIcon({ action }: { action: string }) {
    64→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    65→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    66→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    67→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    68→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    69→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    70→  return <CircleDot className="h-4 w-4 text-slate-400" />
    71→}

   262→function getOnboardingBadge(status: string) {
   263→  switch (status) {
   264→    case 'in_progress': return 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400'
   265→    case 'submitted': return 'bg-purple-100 text-purple-700 dark:bg-purple-900/30 dark:text-purple-400'
   266→    case 'approved': return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   267→    case 'rejected': return 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-400'
   268→    default: return 'bg-slate-100 text-slate-700'
   269→  }
   270→}

   272→function getOnboardingStatusLabel(status: string) {
   273→  const map: Record<string, string> = { in_progress: 'In Progress', submitted: 'Submitted', approved: 'Approved', rejected: 'Rejected' }
   274→  return map[status] || status

   277→export function AdminOnboarding() {
   278→  const { user } = useAuthStore()
   279→  const [loading, setLoading] = useState(true)
   280→  const [forms, setForms] = useState<OnboardingForm[]>([])
   281→  const [selectedForm, setSelectedForm] = useState<(OnboardingForm & { submissions: Onboarding[] }) | null>(null)
   282→  const [dialogOpen, setDialogOpen] = useState(false)
   283→  const [reviewing, setReviewing] = useState<string | null>(null)
   284→  const [reviewNotes, setReviewNotes] = useState<Record<string, string>>({})
   285→
   286→  useEffect(() => {
   287→    async function fetchData() {
   288→      try {
   289→        const res = await fetch('/api/onboarding')
   290→        if (!res.ok) throw new Error('Failed')
   291→        const data = await res.json()
   292→        setForms(data.forms ?? [])
   293→      } catch {
   294→        toast.error('Failed to load onboarding forms')
   295→      } finally {
   296→        setLoading(false)
   297→      }
   298→    }
   299→    fetchData()
   300→  }, [])
   301→
   302→  async function openForm(form: OnboardingForm) {
   303→    try {
   304→      const res = await fetch(`/api/onboarding/${form.id}`)
   305→      if (!res.ok) throw new Error('Failed')
   306→      const data = await res.json()
   307→      setSelectedForm(data.form)
   308→      setDialogOpen(true)
   309→    } catch {
   310→      toast.error('Failed to load form details')
   311→    }
   312→  }
   313→
   314→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   315→    if (!user) return
   316→    setReviewing(submissionId)
   317→    try {
   318→      const notes = reviewNotes[submissionId] || ''
   319→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   320→        method: 'PATCH',
   321→        headers: { 'Content-Type': 'application/json' },
   322→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   323→      })
   324→      if (!patchRes.ok) {
   325→        const d = await patchRes.json()
   326→        throw new Error(d.error)
   327→      }
   328→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   329→      // Refresh
   330→      if (selectedForm) {
   331→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   332→        if (fresh.ok) {
   333→          const data = await fresh.json()
   334→          setSelectedForm(data.form)
   335→        }
   336→      }
   337→    } catch (e: unknown) {
   338→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   339→    } finally {
   340→      setReviewing(null)
   341→    }
   342→  }
   343→
   344→  if (loading) {
   345→    return (
   346→      <div className="space-y-4">
   347→        <Skeleton className="h-10 w-48" />
   348→        <Card><CardContent className="p-4"><Skeleton className="h-64 w-full" /></CardContent></Card>
   349→      </div>
   350→    )
   351→  }
   352→
   353→  return (
   354→    <motion.div {...fadeIn} className="space-y-6">
   355→      <div>
   356→        <h2 className="text-xl font-semibold">Onboarding Management</h2>
   357→        <p className="text-sm text-muted-foreground">Review and manage client onboarding submissions</p>
   358→      </div>
   359→
   360→      <Card>
   361→        <CardContent className="p-0">
   362→          <Table>
   363→            <TableHeader>
   364→              <TableRow>
   365→                <TableHead>Form Title</TableHead>
   366→                <TableHead>Project</TableHead>
   367→                <TableHead className="text-center">Submissions</TableHead>
   368→                <TableHead>Status</TableHead>
   369→                <TableHead className="w-16"></TableHead>
   370→              </TableRow>
   371→            </TableHeader>
   372→            <TableBody>
   373→              {forms.length === 0 && (
   374→                <TableRow><TableCell colSpan={5} className="text-center text-muted-foreground py-8">No onboarding forms found</TableCell></TableRow>
   375→              )}
   376→              {forms.map((form) => (
   377→                <TableRow key={form.id} className="cursor-pointer hover:bg-muted/50" onClick={() => openForm(form)}>
   378→                  <TableCell className="font-medium">{form.title}</TableCell>
   379→                  <TableCell>
   380→                    <div className="flex items-center gap-2">
   381→                      {form.project && <div className="h-3 w-3 rounded-full" style={{ backgroundColor: form.project.color }} />}
   382→                      <span className="text-sm text-muted-foreground">{form.project?.name ?? '—'}</span>
   383→                    </div>
   384→                  </TableCell>
   385→                  <TableCell className="text-center">{form._count?.submissions ?? 0}</TableCell>
   386→                  <TableCell>
   387→                    <Badge variant="secondary" className={getOnboardingBadge(form.status)}>
   388→                      {getOnboardingStatusLabel(form.status)}
   389→                    </Badge>
   390→                  </TableCell>
   391→                  <TableCell>
   392→                    <Button variant="ghost" size="icon" className="h-8 w-8">
   393→                      <Eye className="h-4 w-4" />
   394→                    </Button>
   395→                  </TableCell>
   396→                </TableRow>
   397→              ))}
   398→            </TableBody>
   399→          </Table>
   400→        </CardContent>
   401→      </Card>
   402→
   403→      {/* Submissions Dialog */}
   404→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   405→        <DialogContent className="max-w-3xl max-h-[85vh] overflow-hidden flex flex-col">
   406→          <DialogHeader>
   407→            <DialogTitle>{selectedForm?.title ?? 'Form'}</DialogTitle>
   408→            <DialogDescription>
   409→              {selectedForm?.project?.name ?? 'Project'} — {selectedForm?.submissions?.length ?? 0} submissions
   410→            </DialogDescription>
   411→          </DialogHeader>
   412→          <ScrollArea className="flex-1 -mx-6 px-6">
   413→            <div className="space-y-4 py-2">
   414→              {selectedForm?.submissions?.length === 0 && (
   415→                <p className="text-sm text-muted-foreground text-center py-6">No submissions yet</p>
   416→              )}
   417→              {selectedForm?.submissions?.map((sub) => {
   418→                let parsedResponses: Record<string, string> = {}
   419→                let parsedFiles: Record<string, { url: string; originalName: string } | null> = {}
   420→                try {
   421→                  const raw = JSON.parse(sub.responsesJson)
   422→                  parsedResponses = raw.responses || {}
   423→                  parsedFiles = raw.files || {}
   424→                } catch { /* old format fallback */ }
   425→
   426→                return (
   427→                  <Card key={sub.id} className="border">
   428→                    <CardContent className="p-4 space-y-4">
   429→                      <div className="flex items-start justify-between">
   430→                        <div className="flex items-center gap-3">
   431→                          <Avatar className="h-9 w-9">
   432→                            {sub.client?.avatar ? <AvatarImage src={sub.client.avatar} /> : null}
   433→                            <AvatarFallback className="text-xs">{sub.client ? getInitials(sub.client.name) : '?'}</AvatarFallback>
   434→                          </Avatar>
   435→                          <div>
   436→                            <p className="text-sm font-medium">{sub.client?.name ?? 'Unknown'}</p>
   437→                            <p className="text-xs text-muted-foreground">{sub.client?.company ?? 'No company'}</p>
   438→                          </div>
   439→                        </div>
   440→                        <Badge variant="secondary" className={getOnboardingBadge(sub.status)}>
   441→                          {getOnboardingStatusLabel(sub.status)}
   442→                        </Badge>
   443→                      </div>
   444→
   445→                      {sub.submittedAt && (
   446→                        <p className="text-xs text-muted-foreground">Submitted {formatDate(sub.submittedAt)}</p>
   447→                      )}
   448→
   449→                      {Object.keys(parsedResponses).length > 0 && (
   450→                        <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   451→                          {Object.entries(parsedResponses).filter(([_, v]) => v?.trim()).map(([key, value]) => (
   452→                            <div key={key} className="p-2.5 rounded-lg bg-muted/50 border">
   453→                              <p className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider mb-0.5">
   454→                                {key.replace(/([A-Z])/g, ' $1').replace(/^./, (s: string) => s.toUpperCase())}
   455→                              </p>
   456→                              <p className="text-sm whitespace-pre-wrap line-clamp-3">{value}</p>
   457→                            </div>
   458→                          ))}
   459→                        </div>
   460→                      )}

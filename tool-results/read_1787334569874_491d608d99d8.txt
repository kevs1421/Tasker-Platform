   113→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   114→  const visible = users.slice(0, max)
   115→  const extra = users.length - max
   116→  return (
   117→    <div className="flex -space-x-2">
   118→      {visible.map((u, i) =>
   119→        u ? (
   120→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   121→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   122→              {getInitials(u.name)}
   123→            </AvatarFallback>
   124→          </Avatar>
   125→        ) : null
   126→      )}
   127→      {extra > 0 && (
   128→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   129→          +{extra}
   130→        </div>
   131→      )}
   132→    </div>
   133→  )
   134→}
   135→
   136→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   137→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   138→  if (!user) return null
   139→  return (
   140→    <Avatar className={sizeClasses}>
   141→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   142→        {getInitials(user.name)}
   143→      </AvatarFallback>
   144→    </Avatar>
   145→  )
   146→}

   164→// ─── TeamProjects ───────────────────────────────────────────
   165→
   166→export function TeamProjects() {
   167→  const { user, setSelectedProjectId } = useAuthStore()
   168→  const isTraining = user?.role === 'training'
   169→  const [projects, setProjects] = useState<Project[]>([])
   170→  const [loading, setLoading] = useState(true)
   171→  const [expandedId, setExpandedId] = useState<string | null>(null)
   172→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   173→  const [expandedLoading, setExpandedLoading] = useState(false)
   174→  const [taskFilter, setTaskFilter] = useState<string>('all')
   175→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   176→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   177→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   178→  const [allTeamUsers, setAllTeamUsers] = useState<User[]>([])
   179→
   180→  // Create task form
   181→  const [taskTitle, setTaskTitle] = useState('')
   182→  const [taskDesc, setTaskDesc] = useState('')
   183→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   184→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   185→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   186→  const [taskTags, setTaskTags] = useState('')
   187→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   188→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   189→
   190→  const fetchProjects = useCallback(async () => {
   191→    try {
   192→      const res = await fetch('/api/projects')
   193→      const data = await res.json()
   194→      const filtered = (data.projects || []).filter((p: Project) =>
   195→        p.members?.some((m) => m.userId === user?.id)
   196→      )
   197→      setProjects(filtered)
   198→    } catch {
   199→      toast.error('Failed to load projects')
   200→    } finally {
   201→      setLoading(false)
   202→    }
   203→  }, [user])
   204→
   205→  useEffect(() => { fetchProjects() }, [fetchProjects])
   206→
   207→  // Fetch all active admin + team users for assignee picker
   208→  useEffect(() => {
   209→    fetch('/api/users')
   210→      .then(r => r.json())
   211→      .then(d => {
   212→        const members = (d.users || []).filter((u: User) => (u.role === 'team' || u.role === 'training' || u.role === 'admin') && u.status === 'active')
   213→        setAllTeamUsers(members)
   214→      })
   215→      .catch(() => {})
   216→  }, [])
   217→
   218→  const handleExpand = async (projectId: string) => {
   219→    if (expandedId === projectId) {
   220→      setExpandedId(null)
   221→      setExpandedTasks([])
   222→      return
   223→    }
   224→    setExpandedId(projectId)
   225→    setTaskFilter('all')
   226→    setExpandedLoading(true)
   227→    try {
   228→      const res = await fetch(`/api/projects/${projectId}`)
   229→      const data = await res.json()
   230→      setExpandedTasks(data.project?.tasks || [])
   231→    } catch {
   232→      toast.error('Failed to load tasks')
   233→    } finally {
   234→      setExpandedLoading(false)
   235→    }
   236→  }
   237→
   238→  const handleCreateTask = async () => {
   239→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   240→    if (!user?.id) { toast.error('Not authenticated'); return }
   241→    setCreateTaskLoading(true)
   242→    try {
   243→      const res = await fetch('/api/tasks', {
   244→        method: 'POST',
   245→        headers: { 'Content-Type': 'application/json' },
   246→        body: JSON.stringify({
   247→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   248→          priority: taskPriority, projectId: createTaskProjectId,
   249→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   250→          tags: taskTags,
   251→          createdById: user.id,
   252→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   253→        }),
   254→      })
   255→      if (!res.ok) {
   256→        const d = await res.json()
   257→        throw new Error(d.error || 'Failed to create task')
   258→      }
   259→      toast.success('Task created')
   260→      setCreateTaskOpen(false)
   261→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   262→      setTaskPriority('medium'); setTaskDueDate(undefined)
   263→      setTaskTags(''); setTaskAssignee('')
   264→      // Always refresh the expanded project view
   265→      handleExpand(createTaskProjectId)
   266→    } catch (e: unknown) {
   267→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   268→    } finally {
   269→      setCreateTaskLoading(false)
   270→    }
   271→  }
   272→
   273→  const openCreateTaskForProject = (projectId: string) => {
   274→    setCreateTaskProjectId(projectId)
   275→    setCreateTaskOpen(true)
   276→  }
   277→
   278→  const getProjectTaskStats = (projectId: string) => {
   279→    if (expandedId !== projectId) return null
   280→    const total = expandedTasks.length
   281→    const done = expandedTasks.filter((t) => t.status === 'done').length
   282→    return { total, done }
   283→  }
   284→
   285→  const filteredTasks = taskFilter === 'all'
   286→    ? expandedTasks
   287→    : expandedTasks.filter((t) => t.status === taskFilter)
   288→
   289→  if (loading) {
   290→    return (
   291→      <div>
   292→        <div className="flex items-center justify-between mb-6">
   293→          <Skeleton className="h-8 w-48" />
   294→          <Skeleton className="h-10 w-36" />
   295→        </div>
   296→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   297→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   298→        </div>
   299→      </div>
   300→    )
   301→  }
   302→
   303→  return (
   304→    <div className="space-y-6">
   305→      {/* Header */}
   306→      <div className="flex items-center justify-between">
   307→        <div>
   308→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   309→          <p className="text-sm text-muted-foreground mt-1">
   310→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   311→          </p>
   312→        </div>
   313→      </div>
   314→
   315→      {/* Project Grid */}
   316→      {projects.length === 0 ? (
   317→        <EmptyState
   318→          icon={FolderKanban}
   319→          title="No projects yet"
   320→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   321→        />
   322→      ) : (
   323→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   324→          <AnimatePresence>
   325→            {projects.map((project, idx) => {
   326→              const totalTasks = project._count?.tasks || 0
   327→              const members = project.members || []
   328→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   329→              const pConfig = getPriorityConfig(project.priority)
   330→              const isExpanded = expandedId === project.id
   331→              const stats = getProjectTaskStats(project.id)
   332→
   333→              return (
   334→                <motion.div
   335→                  key={project.id}
   336→                  initial={{ opacity: 0, y: 20 }}
   337→                  animate={{ opacity: 1, y: 0 }}
   338→                  transition={{ delay: idx * 0.05 }}
   339→                >
   340→                  <Card
   341→                    className={cn(
   342→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   343→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   344→                    )}
   345→                    onClick={() => handleExpand(project.id)}
   346→                  >
   347→                    <div className="flex">
   348→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   349→                      <CardContent className="flex-1 p-4">
   350→                        <div className="flex items-start justify-between mb-2">
   351→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   352→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   353→                            {pConfig.label}
   354→                          </Badge>
   355→                        </div>
   356→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   357→                          {project.description || 'No description'}
   358→                        </p>
   359→                        <Progress
   360→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   361→                          className="h-1.5 mb-3"
   362→                        />
   363→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   364→                          <div className="flex items-center gap-3">
   365→                            <span className="flex items-center gap-1">
   366→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   367→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   368→                            </span>
   369→                            <span className="flex items-center gap-1">
   370→                              <Users className="h-3 w-3" />
   371→                              {members.length}
   372→                            </span>
   373→                          </div>
   374→                          {project.dueDate && (
   375→                            <span className={cn(
   376→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   377→                                ? 'text-red-500' : '',
   378→                            )}>
   379→                              {formatDate(project.dueDate)}
   380→                            </span>
   381→                          )}
   382→                        </div>
   383→                        {memberUsers.length > 0 && (
   384→                          <div className="mt-3">
   385→                            <AvatarStack users={memberUsers} max={4} />
   386→                          </div>
   387→                        )}
   388→                      </CardContent>
   389→                    </div>
   390→                  </Card>
   391→
   392→                  {/* Expanded Task List */}
   393→                  <AnimatePresence>
   394→                    {isExpanded && (
   395→                      <motion.div
   396→                        initial={{ opacity: 0, height: 0 }}
   397→                        animate={{ opacity: 1, height: 'auto' }}
   398→                        exit={{ opacity: 0, height: 0 }}
   399→                        transition={{ duration: 0.2 }}
   400→                        className="overflow-hidden"
   401→                        onClick={(e) => e.stopPropagation()}
   402→                      >
   403→                        <Card className="mt-3 p-4 space-y-4">
   404→                          <div className="flex items-center justify-between">
   405→                            <div className="flex items-center gap-2">
   406→                              <div className="flex gap-1">
   407→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   408→                                  <Button
   409→                                    key={s}
   410→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   411→                                    size="sm"
   412→                                    className="h-7 text-xs px-2"
   413→                                    onClick={() => setTaskFilter(s)}
   414→                                  >
   415→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   416→                                  </Button>
   417→                                ))}
   418→                              </div>
   419→                            </div>
   420→                            {!isTraining && (
   421→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   422→                              <Plus className="h-3 w-3" /> Add Task
   423→                            </Button>
   424→                            )}
   425→                          </div>
   426→                          {expandedLoading ? (
   427→                            <div className="space-y-2">
   428→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   429→                            </div>
   430→                          ) : filteredTasks.length === 0 ? (
   431→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   432→                          ) : (
   433→                            <ScrollArea className="max-h-64">
   434→                              <div className="space-y-2">
   435→                                {filteredTasks.map((task) => {
   436→                                  const sConfig = getStatusConfig(task.status)
   437→                                  const pConf = getPriorityConfig(task.priority)
   438→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   439→                                  return (
   440→                                    <div
   441→                                      key={task.id}
   442→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   443→                                      onClick={() => setSelectedProjectId(project.id)}
   444→                                    >
   445→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   446→                                      <div className="flex-1 min-w-0">
   447→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   448→                                        <div className="flex items-center gap-2 mt-0.5">
   449→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   450→                                          {task.dueDate && (
   451→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   452→                                              {formatDate(task.dueDate)}
   453→                                            </span>
   454→                                          )}
   455→                                        </div>
   456→                                      </div>
   457→                                      <AvatarStack users={assignees} max={2} />
   458→                                    </div>
   459→                                  )
   460→                                })}
   461→                              </div>
   462→                            </ScrollArea>
   463→                          )}
   464→                        </Card>
   465→                      </motion.div>
   466→                    )}
   467→                  </AnimatePresence>
   468→                </motion.div>
   469→              )
   470→            })}
   471→          </AnimatePresence>
   472→        </motion.div>
   473→      )}
   474→
   475→      {/* Create Task Dialog */}
   476→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   477→        <DialogContent className="sm:max-w-lg">
   478→          <DialogHeader>
   479→            <DialogTitle>Create New Task</DialogTitle>
   480→            <DialogDescription>Add a task to the project.</DialogDescription>
   481→          </DialogHeader>
   482→          <div className="space-y-4 py-2">
   483→            <div className="space-y-2">
   484→              <Label>Title</Label>
   485→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   486→            </div>
   487→            <div className="space-y-2">
   488→              <Label>Description</Label>
   489→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   490→            </div>
   491→            <div className="grid grid-cols-2 gap-4">
   492→              <div className="space-y-2">
   493→                <Label>Status</Label>
   494→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   495→                  <SelectTrigger><SelectValue /></SelectTrigger>
   496→                  <SelectContent>
   497→                    <SelectItem value="todo">To Do</SelectItem>
   498→                    <SelectItem value="in_progress">In Progress</SelectItem>
   499→                    <SelectItem value="in_review">In Review</SelectItem>
   500→                    <SelectItem value="done">Done</SelectItem>
   501→                  </SelectContent>
   502→                </Select>
   503→              </div>
   504→              <div className="space-y-2">
   505→                <Label>Priority</Label>
   506→                <Select value={taskPriority} onValueChange={(v) => setTaskPriority(v as typeof taskPriority)}>
   507→                  <SelectTrigger><SelectValue /></SelectTrigger>
   508→                  <SelectContent>

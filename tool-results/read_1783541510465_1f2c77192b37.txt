    66→    66→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    67→    67→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    68→    68→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    69→    69→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    70→    70→  return <CircleDot className="h-4 w-4 text-slate-400" />

   381→   381→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {

   385→   385→      const notes = reviewNotes[submissionId] || ''
   386→   386→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   387→   387→        method: 'PATCH',
   388→   388→        headers: { 'Content-Type': 'application/json' },
   389→   389→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   390→   390→      })
   391→   391→      if (!patchRes.ok) {

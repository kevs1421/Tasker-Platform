   127→   127→      const data = await res.json()
   128→   128→      setProjects(data.projects || [])

   130→   130→      toast.error('Failed to load projects')

   132→   132→      setLoading(false)
   133→   133→    }

   140→   140→      const stored = localStorage.getItem(`client-uploads-${user.id}`)

   156→   156→      toast.error('No project selected')
   157→   157→      return

   190→   190→    setUploadDialogProjectId(projectId)
   191→   191→    handleFileUpload(e.dataTransfer.files)
   192→   192→  }

   195→   195→    setUploadDialogProjectId(projectId)
   196→   196→    setUploadDialogOpen(true)
   197→   197→  }

   200→   200→    const updated = uploadedFiles.filter((_, i) => i !== idx)

   202→   202→    saveUploadedFiles(updated)

   204→   204→  }

   211→   211→    return <File className="w-4 h-4 text-muted-foreground" />
   212→   212→  }
   213→   213→
   214→   214→  const formatFileSize = (bytes: number) => {
   215→   215→    if (bytes < 1024) return `${bytes} B`
   216→   216→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   217→   217→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   218→   218→  }
   219→   219→
   220→   220→  if (loading) return <PageSkeleton />
   221→   221→
   222→   222→  return (
   223→   223→    <div className="p-4 md:p-6 space-y-6">
   224→   224→      {/* Page Header */}
   225→   225→      <div className="flex items-center gap-3">
   226→   226→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">

   228→   228→        </div>

   232→   232→        </div>
   233→   233→      </div>
   234→   234→
   235→   235→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}

   237→   237→        ref={fileInputRef}
   238→   238→        type="file"
   239→   239→        multiple
   240→   240→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   241→   241→        className="hidden"
   242→   242→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   243→   243→      />
   244→   244→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   245→   245→        <DialogContent className="sm:max-w-md">
   246→   246→          <DialogHeader>
   247→   247→            <DialogTitle className="flex items-center gap-2">
   248→   248→              <Upload className="w-5 h-5 text-primary" />
   249→   249→              Upload Files
   250→   250→            </DialogTitle>
   251→   251→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   252→   252→          </DialogHeader>
   253→   253→          <div
   254→   254→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   255→   255→            onDragLeave={() => setUploadDragOver(null)}
   256→   256→            onDrop={(e) => {
   257→   257→              e.preventDefault()
   258→   258→              setUploadDragOver(null)
   259→   259→              // Set the project context before handling files
   260→   260→              const files = e.dataTransfer.files
   261→   261→              if (files && files.length > 0) {
   262→   262→                // Use a separate handler for dialog drops
   263→   263→                void (async () => {
   264→   264→                  setUploading(true)
   265→   265→                  const newFiles: typeof uploadedFiles = []
   266→   266→                  for (let i = 0; i < files.length; i++) {
   267→   267→                    const file = files[i]
   268→   268→                    try {
   269→   269→                      const formData = new FormData()
   270→   270→                      formData.append('file', file)
   271→   271→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   272→   272→                      if (!res.ok) {
   273→   273→                        const d = await res.json()
   274→   274→                        throw new Error(d.error || 'Upload failed')
   275→   275→                      }
   276→   276→                      const data = await res.json()
   277→   277→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   278→   278→                    } catch (err: unknown) {
   279→   279→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   280→   280→                    }
   281→   281→                  }
   282→   282→                  if (newFiles.length > 0) {
   283→   283→                    const updated = [...uploadedFiles, ...newFiles]
   284→   284→                    setUploadedFiles(updated)
   285→   285→                    saveUploadedFiles(updated)
   286→   286→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   287→   287→                  }
   288→   288→                  setUploading(false)
   289→   289→                })()
   290→   290→              }
   291→   291→            }}
   292→   292→            onClick={() => fileInputRef.current?.click()}
   293→   293→            className={cn(
   294→   294→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   295→   295→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   296→   296→            )}
   297→   297→          >
   298→   298→            {uploading ? (
   299→   299→              <div className="flex flex-col items-center gap-2">
   300→   300→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   301→   301→                <p className="text-sm text-muted-foreground">Uploading...</p>

   303→   303→            ) : (
   304→   304→              <div className="flex flex-col items-center gap-2">
   305→   305→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   306→   306→                  <Upload className="w-5 h-5 text-muted-foreground" />
   307→   307→                </div>
   308→   308→                <div>
   309→   309→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   310→   310→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   311→   311→                </div>
   312→   312→              </div>
   313→   313→            )}
   314→   314→          </div>
   315→   315→          <DialogFooter>

   322→   322→      {projects.length > 0 && (

   324→   324→          <Card>
   325→   325→            <CardHeader className="pb-3">
   326→   326→              <CardTitle className="text-base flex items-center gap-2">
   327→   327→                <BarChart3 className="w-4 h-4 text-primary" />
   328→   328→                Project Progress

   330→   330→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>

   332→   332→            <CardContent>
   333→   333→              <div className="space-y-5">
   334→   334→                {projects.map((project) => {
   335→   335→                  const todo = project.taskStats?.todo || 0
   336→   336→                  const inProgress = project.taskStats?.in_progress || 0
   337→   337→                  const inReview = project.taskStats?.in_review || 0

   362→   362→                            title={`Not Started: ${todo}`}
   363→   363→                          />
   364→   364→                        )}
   365→   365→                        {inProgress > 0 && (
   366→   366→                          <div
   367→   367→                            className="bg-amber-500 transition-all duration-500"
   368→   368→                            style={{ width: `${inProgressPct}%` }}
   369→   369→                            title={`In Progress: ${inProgress}`}
   370→   370→                          />
   371→   371→                        )}
   372→   372→                        {inReview > 0 && (
   373→   373→                          <div
   374→   374→                            className="bg-blue-500 transition-all duration-500"
   375→   375→                            style={{ width: `${inReviewPct}%` }}
   376→   376→                            title={`In Review: ${inReview}`}
   377→   377→                          />
   378→   378→                        )}
   379→   379→                        {completed > 0 && (
   380→   380→                          <div
   381→   381→                            className="bg-emerald-500 transition-all duration-500"
   382→   382→                            style={{ width: `${completedPct}%` }}
   383→   383→                            title={`Completed: ${completed}`}

   386→   386→                      </div>
   387→   387→                      {/* Legend */}

   389→   389→                        <span className="flex items-center gap-1.5">
   390→   390→                          <span className="w-2.5 h-2.5 rounded-sm bg-slate-400 dark:bg-slate-500" />
   391→   391→                          Not Started ({todo})
   392→   392→                        </span>
   393→   393→                        <span className="flex items-center gap-1.5">
   394→   394→                          <span className="w-2.5 h-2.5 rounded-sm bg-amber-500" />
   395→   395→                          In Progress ({inProgress})
   396→   396→                        </span>
   397→   397→                        <span className="flex items-center gap-1.5">
   398→   398→                          <span className="w-2.5 h-2.5 rounded-sm bg-blue-500" />
   399→   399→                          In Review ({inReview})
   400→   400→                        </span>
   401→   401→                        <span className="flex items-center gap-1.5">
   402→   402→                          <span className="w-2.5 h-2.5 rounded-sm bg-emerald-500" />
   403→   403→                          Completed ({completed})
   404→   404→                        </span>
   405→   405→                      </div>
   406→   406→                    </div>
   407→   407→                  )
   408→   408→                })}
   409→   409→              </div>
   410→   410→            </CardContent>
   411→   411→          </Card>
   412→   412→        </motion.div>
   413→   413→      )}
   414→   414→
   415→   415→      {projects.length === 0 ? (
   416→   416→        <EmptyState
   417→   417→          icon={FolderKanban}
   418→   418→          title="No active projects"
   419→   419→          description="You don't have any active projects yet."
   420→   420→        />
   421→   421→      ) : (

   570→   570→export function ClientProfile() {
   571→   571→  const { user, updateUser } = useAuthStore()
   572→   572→  const [loading, setLoading] = useState(true)
   573→   573→  const [saving, setSaving] = useState(false)
   574→   574→  const [savingPassword, setSavingPassword] = useState(false)
   575→   575→
   576→   576→  const [name, setName] = useState('')
   577→   577→  const [email, setEmail] = useState('')
   578→   578→  const [phone, setPhone] = useState('')
   579→   579→  const [company, setCompany] = useState('')
   580→   580→
   581→   581→  const [currentPassword, setCurrentPassword] = useState('')
   582→   582→  const [newPassword, setNewPassword] = useState('')
   583→   583→  const [confirmPassword, setConfirmPassword] = useState('')
   584→   584→
   585→   585→  useEffect(() => {
   586→   586→    if (!user?.id) return
   587→   587→    const fetchProfile = async () => {
   588→   588→      try {
   589→   589→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   590→   590→        const data = await res.json()
   591→   591→        if (data.user) {
   592→   592→          setName(data.user.name || '')
   593→   593→          setEmail(data.user.email || '')
   594→   594→          setPhone(data.user.phone || '')
   595→   595→          setCompany(data.user.company || '')
   596→   596→        }
   597→   597→      } catch {
   598→   598→        toast.error('Failed to load profile')
   599→   599→      } finally {
   600→   600→        setLoading(false)

   602→   602→    }

   605→   605→
   606→   606→  const saveProfile = async () => {
   607→   607→    if (!user?.id) return
   608→   608→    setSaving(true)
   609→   609→    try {
   610→   610→      const res = await fetch('/api/user/profile', {
   611→   611→        method: 'PATCH',
   612→   612→        headers: { 'Content-Type': 'application/json' },
   613→   613→        body: JSON.stringify({ userId: user.id, name, email, phone, company })
   614→   614→      })

   616→   616→        const data = await res.json()
   617→   617→        throw new Error(data.error || 'Failed')
   618→   618→      }
   619→   619→      const data = await res.json()

   622→   622→      if (data.user && user) {

   624→   624→      }
   625→   625→    } catch (e: unknown) {
   626→   626→      const msg = e instanceof Error ? e.message : 'Failed to update profile'
   627→   627→      toast.error(msg)
   628→   628→    } finally {
   629→   629→      setSaving(false)
   630→   630→    }
   631→   631→  }
   632→   632→
   633→   633→  const changePassword = async () => {
   634→   634→    if (!user?.id) return
   635→   635→    if (!currentPassword || !newPassword || !confirmPassword) {
   636→   636→      toast.error('Please fill in all password fields')
   637→   637→      return
   638→   638→    }
   639→   639→    if (newPassword !== confirmPassword) {
   640→   640→      toast.error('New passwords do not match')
   641→   641→      return
   642→   642→    }

   644→   644→      toast.error('Password must be at least 6 characters')
   645→   645→      return
   646→   646→    }

   648→   648→    try {
   649→   649→      const res = await fetch('/api/user/profile', {
   650→   650→        method: 'PATCH',
   651→   651→        headers: { 'Content-Type': 'application/json' },
   652→   652→        body: JSON.stringify({
   653→   653→          userId: user.id,
   654→   654→          currentPassword,
   655→   655→          newPassword
   656→   656→        })
   657→   657→      })
   658→   658→      if (!res.ok) {
   659→   659→        const data = await res.json()
   660→   660→        throw new Error(data.error || 'Failed')
   661→   661→      }
   662→   662→      toast.success('Password changed successfully')
   663→   663→      setCurrentPassword('')
   664→   664→      setNewPassword('')
   665→   665→      setConfirmPassword('')
   666→   666→    } catch (e: unknown) {
   667→   667→      const msg = e instanceof Error ? e.message : 'Failed to change password'
   668→   668→      toast.error(msg)
   669→   669→    } finally {
   670→   670→      setSavingPassword(false)
   671→   671→    }
   672→   672→  }

   674→   674→  if (loading) return <PageSkeleton />

   684→   684→          <p className="text-sm text-muted-foreground">Manage your account information</p>

   695→   695→                Client

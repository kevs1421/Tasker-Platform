    63→function ActivityIcon({ action }: { action: string }) {
    64→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    65→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    66→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    67→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    68→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    69→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    70→  return <CircleDot className="h-4 w-4 text-slate-400" />
    71→}

   284→export function AdminOnboarding() {
   285→  const { user } = useAuthStore()
   286→  const [loading, setLoading] = useState(true)
   287→  const [forms, setForms] = useState<OnboardingForm[]>([])
   288→  const [selectedForm, setSelectedForm] = useState<(OnboardingForm & { submissions: Onboarding[] }) | null>(null)
   289→  const [dialogOpen, setDialogOpen] = useState(false)
   290→  const [reviewing, setReviewing] = useState<string | null>(null)
   291→  const [reviewNotes, setReviewNotes] = useState<Record<string, string>>({})
   292→
   293→  // Create onboarding form state
   294→  const [createOpen, setCreateOpen] = useState(false)
   295→  const [creating, setCreating] = useState(false)
   296→  const [projects, setProjects] = useState<Project[]>([])
   297→  const [createForm, setCreateForm] = useState({
   298→    title: '',
   299→    description: '',
   300→    projectId: '',
   301→    selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'],
   302→  })
   303→  const [stepSearch, setStepSearch] = useState('')
   304→
   305→  const fetchForms = useCallback(async () => {
   306→    try {
   307→      const res = await fetch('/api/onboarding')
   308→      if (!res.ok) throw new Error('Failed')
   309→      const data = await res.json()
   310→      setForms(data.forms ?? [])
   311→    } catch {
   312→      toast.error('Failed to load onboarding forms')
   313→    } finally {
   314→      setLoading(false)
   315→    }
   316→  }, [])
   317→
   318→  const fetchProjects = useCallback(async () => {
   319→    try {
   320→      const res = await fetch('/api/projects')
   321→      if (res.ok) {
   322→        const data = await res.json()
   323→        setProjects(data.projects ?? [])
   324→      }
   325→    } catch { /* ignore */ }
   326→  }, [])
   327→
   328→  useEffect(() => {
   329→    fetchForms()
   330→    fetchProjects()
   331→  }, [fetchForms, fetchProjects])
   332→
   333→  async function openForm(form: OnboardingForm) {
   334→    try {
   335→      const res = await fetch(`/api/onboarding/${form.id}`)
   336→      if (!res.ok) throw new Error('Failed')
   337→      const data = await res.json()
   338→      setSelectedForm(data.form)
   339→      setDialogOpen(true)
   340→    } catch {
   341→      toast.error('Failed to load form details')
   342→    }
   343→  }
   344→
   345→  async function handleCreateOnboarding() {
   346→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   347→    setCreating(true)
   348→    try {
   349→      const stepsJson = JSON.stringify(
   350→        createForm.selectedSteps.map(stepId => {
   351→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   352→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   353→        })
   354→      )
   355→      const res = await fetch('/api/onboarding', {
   356→        method: 'POST',
   357→        headers: { 'Content-Type': 'application/json' },
   358→        body: JSON.stringify({
   359→          title: createForm.title.trim(),
   360→          description: createForm.description.trim() || null,
   361→          stepsJson,
   362→          projectId: createForm.projectId,
   363→          status: 'published',
   364→        }),
   365→      })
   366→      if (!res.ok) {
   367→        const d = await res.json()
   368→        throw new Error(d.error || 'Failed to create')
   369→      }
   370→      toast.success('Onboarding form created successfully')
   371→      setCreateOpen(false)
   372→      setCreateForm({ title: '', description: '', projectId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })
   373→      fetchForms()
   374→    } catch (e: unknown) {
   375→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding form')
   376→    } finally {
   377→      setCreating(false)
   378→    }
   379→  }
   380→
   381→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   382→    if (!user) return
   383→    setReviewing(submissionId)
   384→    try {
   385→      const notes = reviewNotes[submissionId] || ''
   386→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   387→        method: 'PATCH',
   388→        headers: { 'Content-Type': 'application/json' },
   389→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   390→      })
   391→      if (!patchRes.ok) {
   392→        const d = await patchRes.json()
   393→        throw new Error(d.error)
   394→      }
   395→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   396→      // Refresh
   397→      if (selectedForm) {
   398→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   399→        if (fresh.ok) {
   400→          const data = await fresh.json()
   401→          setSelectedForm(data.form)
   402→        }
   403→      }
   404→      fetchForms()
   405→    } catch (e: unknown) {
   406→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   407→    } finally {
   408→      setReviewing(null)
   409→    }
   410→  }
   411→

   414→      const res = await fetch(`/api/onboarding/${formId}`, {
   415→        method: 'DELETE',
   416→      })

   420→    } catch {
   421→      toast.error('Failed to delete onboarding form')
   422→    }
   423→  }
   424→
   425→  const toggleStep = (stepId: string) => {
   426→    setCreateForm(prev => ({
   427→      ...prev,
   428→      selectedSteps: prev.selectedSteps.includes(stepId)
   429→        ? prev.selectedSteps.filter(s => s !== stepId)
   430→        : [...prev.selectedSteps, stepId],
   431→    }))
   432→  }
   433→
   434→  const filteredSteps = ONBOARDING_STEPS_CONFIG.filter(s =>
   435→    !stepSearch || s.label.toLowerCase().includes(stepSearch.toLowerCase()) || s.description.toLowerCase().includes(stepSearch.toLowerCase())
   436→  )
   437→
   438→  if (loading) {
   439→    return (
   440→      <div className="space-y-4">
   441→        <Skeleton className="h-10 w-48" />
   442→        <Card><CardContent className="p-4"><Skeleton className="h-64 w-full" /></CardContent></Card>
   443→      </div>
   444→    )
   445→  }
   446→
   447→  return (
   448→    <motion.div {...fadeIn} className="space-y-6">
   449→      <div className="flex items-center justify-between">
   450→        <div>
   451→          <h2 className="text-xl font-semibold">Onboarding Management</h2>
   452→          <p className="text-sm text-muted-foreground">Create and manage client onboarding forms</p>
   453→        </div>
   454→        <Button onClick={() => setCreateOpen(true)} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   455→          <Plus className="h-4 w-4 mr-2" /> Create Onboarding
   456→        </Button>
   457→      </div>
   458→
   459→      {/* Summary cards */}
   460→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   461→        {[
   462→          { label: 'Total Forms', value: forms.length, color: 'text-slate-600', bg: 'bg-slate-100 dark:bg-slate-800' },

   839→const PRESET_COLORS = ['#f59e0b', '#10b981', '#ef4444', '#8b5cf6', '#ec4899', '#06b6d4', '#f97316', '#6366f1', '#14b8a6', '#e11d48']
   840→
   841→export function AdminProjects() {
   842→  const { user } = useAuthStore()
   843→  const [loading, setLoading] = useState(true)
   844→  const [projects, setProjects] = useState<Project[]>([])
   845→  const [createOpen, setCreateOpen] = useState(false)
   846→  const [creating, setCreating] = useState(false)
   847→  const [form, setForm] = useState({ name: '', description: '', priority: 'medium', dueDate: null as Date | null, color: PRESET_COLORS[0] })
   848→
   849→  const fetchProjects = useCallback(async () => {
   850→    try {
   851→      const res = await fetch('/api/projects')
   852→      if (!res.ok) throw new Error('Failed')
   853→      const data = await res.json()
   854→      setProjects(data.projects ?? [])
   855→    } catch {
   856→      toast.error('Failed to load projects')
   857→    } finally {
   858→      setLoading(false)
   859→    }
   860→  }, [])
   861→
   862→  useEffect(() => { fetchProjects() }, [fetchProjects])
   863→
   864→  async function handleCreate() {
   865→    if (!user || !form.name.trim()) return
   866→    setCreating(true)
   867→    try {
   868→      const createRes = await fetch('/api/projects', {
   869→        method: 'POST',
   870→        headers: { 'Content-Type': 'application/json' },
   871→        body: JSON.stringify({
   872→          name: form.name,
   873→          description: form.description || null,
   874→          priority: form.priority,
   875→          dueDate: form.dueDate ? form.dueDate.toISOString() : null,
   876→          color: form.color,
   877→          createdBy: user.id
   878→        })
   879→      })
   880→      if (!createRes.ok) {
   881→        const d = await createRes.json()
   882→        throw new Error(d.error)
   883→      }
   884→      toast.success('Project created successfully')
   885→      setCreateOpen(false)
   886→      setForm({ name: '', description: '', priority: 'medium', dueDate: null, color: PRESET_COLORS[0] })
   887→      fetchProjects()
   888→    } catch (e: unknown) {
   889→      toast.error(e instanceof Error ? e.message : 'Failed to create project')
   890→    } finally {
   891→      setCreating(false)
   892→    }
   893→  }
   894→
   895→  async function handleStatusChange(projectId: string, status: string) {
   896→    try {
   897→      const patchRes = await fetch(`/api/projects/${projectId}`, {
   898→        method: 'PATCH',
   899→        headers: { 'Content-Type': 'application/json' },
   900→        body: JSON.stringify({ status })
   901→      })
   902→      if (!patchRes.ok) throw new Error()
   903→      toast.success(`Project ${status}`)
   904→      fetchProjects()
   905→    } catch {
   906→      toast.error('Failed to update project status')
   907→    }
   908→  }
   909→
   910→  async function handleDelete(projectId: string) {
   911→    try {
   912→      const delRes = await fetch(`/api/projects/${projectId}`, { method: 'DELETE' })
   913→      if (!delRes.ok) throw new Error()
   914→      toast.success('Project deleted')
   915→      fetchProjects()
   916→    } catch {
   917→      toast.error('Failed to delete project')
   918→    }
   919→  }
   920→
   921→  function getProjectStatusBadge(status: string) {
   922→    switch (status) {
   923→      case 'active': return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   924→      case 'completed': return 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300'
   925→      case 'archived': return 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400'
   926→      default: return 'bg-slate-100 text-slate-700'
   927→    }
   928→  }
   929→
   930→  if (loading) {
   931→    return (
   932→      <div className="space-y-4">
   933→        <div className="flex justify-between items-center"><Skeleton className="h-10 w-48" /><Skeleton className="h-10 w-32" /></div>
   934→        <Card><CardContent className="p-4"><Skeleton className="h-64 w-full" /></CardContent></Card>
   935→      </div>
   936→    )
   937→  }
   938→
   939→  return (
   940→    <motion.div {...fadeIn} className="space-y-6">

    51→function UserAvatar({ user, size = 'md', className }: { user?: User | null; size?: 'sm' | 'md' | 'lg'; className?: string }) {
    52→  const sizeMap = { sm: 'h-8 w-8 text-[10px]', md: 'h-10 w-10 text-xs', lg: 'h-12 w-12 text-sm' }
    53→  if (!user) return <div className={cn(sizeMap[size], 'rounded-full bg-muted flex items-center justify-center', className)}>?</div>
    54→  return (
    55→    <Avatar className={cn(sizeMap[size], className)}>
    56→      {user.avatar && <AvatarImage src={user.avatar} />}
    57→      <AvatarFallback className="bg-slate-100 dark:bg-slate-800 font-medium">
    58→        {getInitials(user.name)}
    59→      </AvatarFallback>
    60→    </Avatar>
    61→  )
    62→}

    75→function CreateGroupDialog({
    76→  open, onOpenChange, onCreated, currentUser,
    77→}: {
    78→  open: boolean; onOpenChange: (open: boolean) => void;
    79→  onCreated: (group: MessageGroup) => void; currentUser: User;
    80→}) {
    81→  const [name, setName] = useState('')
    82→  const [users, setUsers] = useState<User[]>([])
    83→  const [selectedIds, setSelectedIds] = useState<string[]>([])
    84→  const [loading, setLoading] = useState(false)
    85→  const [creating, setCreating] = useState(false)
    86→  const [search, setSearch] = useState('')
    87→
    88→  useEffect(() => {
    89→    if (!open) return
    90→    setLoading(true)
    91→    fetch('/api/users')
    92→      .then((r) => r.json())
    93→      .then((d) => {
    94→        // Exclude current user from the list
    95→        setUsers((d.users ?? []).filter((u: User) => u.id !== currentUser.id))
    96→      })
    97→      .catch(() => {})
    98→      .finally(() => setLoading(false))
    99→    setName('')
   100→    setSelectedIds([])
   101→    setSearch('')
   102→  }, [open, currentUser.id])
   103→
   104→  const toggleUser = (userId: string) => {
   105→    setSelectedIds((prev) => prev.includes(userId) ? prev.filter((id) => id !== userId) : [...prev, userId])
   106→  }
   107→
   108→  const handleCreate = async () => {
   109→    if (!name.trim() || selectedIds.length === 0) return
   110→    setCreating(true)
   111→    try {
   112→      const res = await fetch('/api/message-groups', {
   113→        method: 'POST',
   114→        headers: { 'Content-Type': 'application/json' },
   115→        body: JSON.stringify({ name: name.trim(), createdBy: currentUser.id, memberIds: selectedIds }),
   116→      })
   117→      if (!res.ok) throw new Error()
   118→      const data = await res.json()
   119→      toast.success('Group created')
   120→      onCreated(data.group)
   121→      onOpenChange(false)
   122→    } catch {
   123→      toast.error('Failed to create group')
   124→    } finally {
   125→      setCreating(false)
   126→    }
   127→  }
   128→
   129→  const filteredUsers = users.filter(
   130→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   131→  )
   132→
   133→  return (
   134→    <Dialog open={open} onOpenChange={onOpenChange}>
   135→      <DialogContent className="sm:max-w-md">
   136→        <DialogHeader>
   137→          <DialogTitle className="flex items-center gap-2">
   138→            <Users className="h-5 w-5 text-violet-600" /> Create Group
   139→          </DialogTitle>
   140→          <DialogDescription>Start a group conversation with team members</DialogDescription>
   141→        </DialogHeader>
   142→
   143→        <div className="space-y-4 py-2">
   144→          <div>
   145→            <Input placeholder="Group name..." value={name} onChange={(e) => setName(e.target.value)} />
   146→          </div>
   147→
   148→          <div>
   149→            <div className="relative mb-2">
   150→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   151→              <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   152→            </div>
   153→            <ScrollArea className="max-h-48 rounded-md border">
   154→              {loading ? (
   155→                <div className="p-3 space-y-2">
   156→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   157→                </div>
   158→              ) : filteredUsers.length === 0 ? (
   159→                <p className="text-sm text-muted-foreground text-center py-4">No users found</p>
   160→              ) : (
   161→                <div className="p-1">
   162→                  {filteredUsers.map((u) => (
   163→                    <button
   164→                      key={u.id}
   165→                      className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   166→                      onClick={() => toggleUser(u.id)}
   167→                    >
   168→                      <Checkbox checked={selectedIds.includes(u.id)} />
   169→                      <UserAvatar user={u} size="sm" />
   170→                      <div className="flex-1 min-w-0">
   171→                        <p className="text-sm font-medium truncate">{u.name}</p>
   172→                        <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   173→                      </div>
   174→                      <Badge variant="secondary" className={cn('text-[10px]', getRoleBadgeClasses(u.role))}>{u.role}</Badge>
   175→                    </button>
   176→                  ))}
   177→                </div>
   178→              )}
   179→            </ScrollArea>
   180→            {selectedIds.length > 0 && (
   181→              <p className="text-xs text-muted-foreground mt-2">{selectedIds.length} member{selectedIds.length > 1 ? 's' : ''} selected</p>
   182→            )}
   183→          </div>
   184→        </div>
   185→
   186→        <DialogFooter>
   187→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   188→          <Button
   189→            onClick={handleCreate} disabled={!name.trim() || selectedIds.length === 0 || creating}
   190→            className="bg-violet-600 hover:bg-violet-700 text-white"
   191→          >
   192→            {creating && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   193→            Create Group
   194→          </Button>
   195→        </DialogFooter>
   196→      </DialogContent>
   197→    </Dialog>
   198→  )
   199→}

   203→function GroupInfoDialog({
   204→  open, onOpenChange, group, currentUserId,
   205→}: {
   206→  open: boolean; onOpenChange: (open: boolean) => void
   207→  group: { id: string; name: string; members?: GroupMember[]; _count?: { members: number } } | null
   208→  currentUserId: string
   209→}) {
   210→  const [members, setMembers] = useState<(GroupMember & { user: User })[]>([])
   211→  const [users, setUsers] = useState<User[]>([])
   212→  const [loading, setLoading] = useState(false)
   213→  const [showAddUser, setShowAddUser] = useState(false)
   214→  const [search, setSearch] = useState('')
   215→
   216→  useEffect(() => {
   217→    if (!open || !group) return
   218→    setLoading(true)
   219→    fetch(`/api/message-groups/${group.id}?userId=${currentUserId}`)
   220→      .then((r) => r.json())
   221→      .then((d) => {
   222→        setMembers((d.group?.members ?? []) as (GroupMember & { user: User })[])
   223→      })
   224→      .catch(() => {})
   225→      .finally(() => setLoading(false))
   226→    setSearch('')
   227→    setShowAddUser(false)
   228→  }, [open, group, currentUserId])
   229→
   230→  const handleAddMember = async (userId: string) => {
   231→    if (!group) return
   232→    try {
   233→      const res = await fetch(`/api/message-groups/${group.id}`, {
   234→        method: 'PATCH',
   235→        headers: { 'Content-Type': 'application/json' },
   236→        body: JSON.stringify({ addMemberIds: [userId] }),
   237→      })
   238→      if (!res.ok) throw new Error()
   239→      const data = await res.json()
   240→      setMembers(data.group.members ?? [])
   241→      toast.success('Member added')
   242→    } catch {
   243→      toast.error('Failed to add member')
   244→    }
   245→  }
   246→
   247→  const handleRemoveMember = async (userId: string) => {
   248→    if (!group) return
   249→    try {
   250→      const res = await fetch(`/api/message-groups/${group.id}`, {
   251→        method: 'PATCH',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({ removeMemberIds: [userId] }),
   254→      })
   255→      if (!res.ok) throw new Error()
   256→      const data = await res.json()
   257→      setMembers(data.group.members ?? [])
   258→      toast.success('Member removed')
   259→    } catch {
   260→      toast.error('Failed to remove member')
   261→    }
   262→  }
   263→
   264→  const handleLoadUsers = () => {
   265→    fetch('/api/users')
   266→      .then((r) => r.json())
   267→      .then((d) => {
   268→        const memberIds = members.map((m) => m.userId)
   269→        setUsers((d.users ?? []).filter((u: User) => !memberIds.includes(u.id)))
   270→      })
   271→      .catch(() => {})
   272→    setShowAddUser(true)
   273→  }
   274→
   275→  const filteredUsers = users.filter(
   276→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   277→  )
   278→
   279→  return (
   280→    <Dialog open={open} onOpenChange={onOpenChange}>
   281→      <DialogContent className="sm:max-w-md">
   282→        <DialogHeader>
   283→          <DialogTitle className="flex items-center gap-2">
   284→            <Users className="h-5 w-5 text-violet-600" /> Group Info
   285→          </DialogTitle>
   286→        </DialogHeader>
   287→
   288→        <div className="space-y-4 py-2">
   289→          {group && (
   290→            <div className="flex items-center gap-3">
   291→              <GroupAvatar name={group.name} size="lg" />
   292→              <div>
   293→                <p className="font-semibold">{group.name}</p>
   294→                <p className="text-sm text-muted-foreground">{group._count?.members ?? members.length} members</p>
   295→              </div>
   296→            </div>
   297→          )}
   298→
   299→          <Separator />
   300→
   301→          <div>
   302→            <div className="flex items-center justify-between mb-2">
   303→              <p className="text-sm font-medium">Members</p>
   304→              {!showAddUser && (
   305→                <Button variant="ghost" size="sm" className="h-7 text-xs gap-1" onClick={handleLoadUsers}>
   306→                  <UserPlus className="h-3.5 w-3.5" /> Add
   307→                </Button>
   308→              )}
   309→            </div>
   310→
   311→            {showAddUser && (
   312→              <div className="mb-3 space-y-2">
   313→                <div className="relative">
   314→                  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   315→                  <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   316→                </div>
   317→                <ScrollArea className="max-h-32 rounded-md border">
   318→                  <div className="p-1">
   319→                    {filteredUsers.slice(0, 8).map((u) => (
   320→                      <button
   321→                        key={u.id}
   322→                        className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   323→                        onClick={() => handleAddMember(u.id)}
   324→                      >
   325→                        <UserAvatar user={u} size="sm" />
   326→                        <div className="flex-1 min-w-0">
   327→                          <p className="text-sm font-medium truncate">{u.name}</p>
   328→                        </div>
   329→                        <UserPlus className="h-4 w-4 text-muted-foreground" />
   330→                      </button>
   331→                    ))}
   332→                  </div>
   333→                </ScrollArea>
   334→                <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setShowAddUser(false)}>Done</Button>
   335→              </div>
   336→            )}
   337→
   338→            <ScrollArea className="max-h-48 rounded-md border">
   339→              {loading ? (
   340→                <div className="p-3 space-y-2">
   341→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   342→                </div>
   343→              ) : (
   344→                <div className="p-1">
   345→                  {members.map((m) => (
   346→                    <div key={m.id} className="flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50">
   347→                      <UserAvatar user={m.user} size="sm" />
   348→                      <div className="flex-1 min-w-0">
   349→                        <p className="text-sm font-medium truncate">{m.user?.name ?? 'Unknown'}</p>
   350→                        <p className="text-xs text-muted-foreground">{m.role === 'admin' ? 'Admin' : 'Member'}</p>
   351→                      </div>
   352→                      {m.userId !== currentUserId && (
   353→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-muted-foreground hover:text-destructive" onClick={() => handleRemoveMember(m.userId)}>
   354→                          <X className="h-3.5 w-3.5" />
   355→                        </Button>
   356→                      )}
   357→                    </div>
   358→                  ))}
   359→                </div>
   360→              )}
   361→            </ScrollArea>
   362→          </div>
   363→        </div>
   364→      </DialogContent>
   365→    </Dialog>
   366→  )
   367→}

   369→// ─── Main MessagesPanel Component ───────────────────────────────────────────────

   371→export function MessagesPanel() {
   372→  const { user } = useAuthStore()
   373→  const [loading, setLoading] = useState(true)
   374→  const [conversations, setConversations] = useState<Conversation[]>([])
   375→  const [selectedConvo, setSelectedConvo] = useState<Conversation | null>(null)
   376→  const [messages, setMessages] = useState<Message[]>([])
   377→  const [msgText, setMsgText] = useState('')
   378→  const [sending, setSending] = useState(false)
   379→  const [msgLoading, setMsgLoading] = useState(false)
   380→  const [searchQuery, setSearchQuery] = useState('')
   381→  const [showCreateGroup, setShowCreateGroup] = useState(false)
   382→  const [showGroupInfo, setShowGroupInfo] = useState(false)
   383→  const messagesEndRef = useRef<HTMLDivElement>(null)
   384→
   385→  const fetchConversations = useCallback(async () => {
   386→    if (!user?.id) return
   387→    setLoading(true)
   388→    try {
   389→      const res = await fetch(`/api/messages?userId=${user.id}`)
   390→      if (!res.ok) throw new Error()
   391→      const data = await res.json()
   392→      setConversations(data.conversations ?? [])
   393→    } catch {
   394→      toast.error('Failed to load conversations')
   395→    } finally {
   396→      setLoading(false)
   397→    }
   398→  }, [user?.id])
   399→
   400→  useEffect(() => { fetchConversations() }, [fetchConversations])
   401→
   402→  useEffect(() => {
   403→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   404→  }, [messages])
   405→
   406→  // Determine conversation kind
   407→  const isGroupSelected = selectedConvo?.type === 'group'
   408→  const selectedGroup = isGroupSelected ? (selectedConvo as GroupConversation).group : null
   409→  const selectedDirectUser = !isGroupSelected ? (selectedConvo as DirectConversation | null)?.user : null
   410→
   411→  const selectConversation = async (convo: Conversation) => {
   412→    if (!user?.id) return
   413→    setSelectedConvo(convo)
   414→    setMsgText('')
   415→    setMsgLoading(true)
   416→
   417→    try {
   418→      let url = `/api/messages?userId=${user.id}`
   419→      if (convo.type === 'group') {
   420→        url += `&groupId=${convo.group.id}`
   421→      } else {
   422→        if (!convo.user) return
   423→        url += `&otherId=${convo.user.id}`
   424→      }
   425→      const res = await fetch(url)
   426→      if (!res.ok) throw new Error()
   427→      const data = await res.json()
   428→      setMessages(data.messages ?? [])
   429→    } catch {
   430→      toast.error('Failed to load messages')
   431→    } finally {
   432→      setMsgLoading(false)
   433→    }
   434→
   435→    // Clear unread locally
   436→    setConversations((prev) => prev.map((c) => {
   437→      if (c.type === 'direct' && convo.type === 'direct' && c.user?.id === convo.user?.id) return { ...c, unread: 0 }
   438→      if (c.type === 'group' && convo.type === 'group' && c.group.id === convo.group.id) return { ...c, unread: 0 }
   439→      return c
   440→    }))
   441→  }
   442→
   443→  const sendMessage = async () => {
   444→    if (!user?.id || !msgText.trim()) return
   445→    if (!selectedConvo) return
   446→
   447→    setSending(true)
   448→    const text = msgText.trim()
   449→    setMsgText('')
   450→
   451→    try {
   452→      const body: Record<string, string> = {
   453→        content: text,
   454→        senderId: user.id,
   455→      }
   456→
   457→      if (selectedConvo.type === 'group') {
   458→        body.groupId = selectedConvo.group.id
   459→      } else {
   460→        if (!selectedConvo.user) return
   461→        body.receiverId = selectedConvo.user.id
   462→      }
   463→
   464→      const res = await fetch('/api/messages', {
   465→        method: 'POST',
   466→        headers: { 'Content-Type': 'application/json' },
   467→        body: JSON.stringify(body),
   468→      })
   469→      if (!res.ok) throw new Error()
   470→      const data = await res.json()
   471→      setMessages((prev) => [...prev, data.message])
   472→      fetchConversations()
   473→    } catch {
   474→      toast.error('Failed to send message')
   475→      setMsgText(text)
   476→    } finally {
   477→      setSending(false)
   478→    }
   479→  }
   480→
   481→  const handleKeyDown = (e: React.KeyboardEvent) => {
   482→    if (e.key === 'Enter' && !e.shiftKey) {
   483→      e.preventDefault()
   484→      sendMessage()
   485→    }
   486→  }
   487→
   488→  // Filter conversations
   489→  const filteredConversations = conversations.filter((c) => {
   490→    if (!searchQuery.trim()) return true
   491→    const q = searchQuery.toLowerCase()
   492→    if (c.type === 'direct') return c.user?.name.toLowerCase().includes(q) || c.user?.email.toLowerCase().includes(q)
   493→    return c.group.name.toLowerCase().includes(q)
   494→  })
   495→
   496→  // Conversation display helpers
   497→  const getConvoTitle = (c: Conversation) => c.type === 'direct' ? (c.user?.name ?? 'Unknown') : c.group.name
   498→  const getConvoSubtitle = (c: Conversation) => {
   499→    if (c.type === 'group') {
   500→      return `${c.group._count.members} members`
   501→    }
   502→    return c.user?.company ?? c.user?.email ?? ''
   503→  }
   504→  const getConvoPreview = (c: Conversation) => {
   505→    if (!c.lastMessage) return 'No messages yet'
   506→    const senderName = c.lastMessage.senderId === user?.id ? 'You' : (c.lastMessage.sender?.name ?? 'Someone')
   507→    if (c.type === 'group') {
   508→      return `${senderName}: ${c.lastMessage.content}`
   509→    }
   510→    return `${c.lastMessage.senderId === user?.id ? 'You: ' : ''}${c.lastMessage.content}`
   511→  }
   512→
   513→  // ── Render ───────────────────────────────────────────────
   514→
   515→  const containerHeight = 'h-[calc(100vh-12rem)]'
   516→
   517→  if (loading) {
   518→    return (
   519→      <div className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}>
   520→        <div className="w-80 border-r p-4 space-y-3 flex-shrink-0">
   521→          <Skeleton className="h-10 w-full" />
   522→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-14 w-full" />)}
   523→        </div>
   524→        <div className="flex-1 flex items-center justify-center">
   525→          <p className="text-muted-foreground text-sm">Select a conversation</p>
   526→        </div>
   527→      </div>
   528→    )
   529→  }
   530→
   531→  return (
   532→    <motion.div
   533→      initial={{ opacity: 0, y: 8 }}
   534→      animate={{ opacity: 1, y: 0 }}
   535→      className={cn(containerHeight, 'flex border rounded-lg overflow-hidden')}
   536→    >
   537→      {/* ─── Conversation List ──────────────────────────── */}
   538→      <div className="w-80 border-r bg-background flex flex-col flex-shrink-0">
   539→        <div className="p-3 border-b space-y-2">
   540→          <div className="flex items-center justify-between">
   541→            <h2 className="text-lg font-semibold">Messages</h2>
   542→            <Button
   543→              variant="ghost" size="icon" className="h-8 w-8"
   544→              onClick={() => setShowCreateGroup(true)}
   545→              title="Create Group"
   546→            >
   547→              <Plus className="h-4 w-4" />
   548→            </Button>
   549→          </div>
   550→          <div className="relative">
   551→            <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   552→            <Input
   553→              placeholder="Search conversations..."
   554→              className="pl-9 h-9 text-sm"
   555→              value={searchQuery}
   556→              onChange={(e) => setSearchQuery(e.target.value)}
   557→            />
   558→          </div>
   559→        </div>
   560→        <ScrollArea className="flex-1">
   561→          <div className="p-1">
   562→            {filteredConversations.length === 0 && (
   563→              <p className="text-sm text-muted-foreground text-center py-8">
   564→                {searchQuery ? 'No matching conversations' : 'No conversations yet'}
   565→              </p>
   566→            )}
   567→            {filteredConversations.map((convo, idx) => {
   568→              const isSelected =
   569→                convo.type === 'direct'
   570→                  ? selectedDirectUser?.id === convo.user?.id
   571→                  : selectedGroup?.id === convo.group.id
   572→              return (
   573→                <button
   574→                  key={convo.type === 'direct' ? convo.user?.id ?? idx : convo.group.id}
   575→                  className={cn(
   576→                    'w-full text-left px-3 py-3 rounded-md flex items-start gap-3 transition-colors',
   577→                    isSelected ? 'bg-muted' : 'hover:bg-muted/50'
   578→                  )}
   579→                  onClick={() => selectConversation(convo)}
   580→                >
   581→                  {convo.type === 'direct' ? (
   582→                    <UserAvatar user={convo.user} size="md" className="mt-0.5 flex-shrink-0" />
   583→                  ) : (
   584→                    <GroupAvatar name={convo.group.name} size="md" className="mt-0.5 flex-shrink-0" />
   585→                  )}
   586→                  <div className="flex-1 min-w-0">
   587→                    <div className="flex items-center justify-between">
   588→                      <div className="flex items-center gap-1.5 min-w-0">
   589→                        <span className="text-sm font-medium truncate">{getConvoTitle(convo)}</span>
   590→                        {convo.type === 'group' && (
   591→                          <Users className="h-3 w-3 text-violet-500 flex-shrink-0" />
   592→                        )}
   593→                      </div>
   594→                      {convo.lastMessage && (
   595→                        <span className="text-[10px] text-muted-foreground flex-shrink-0 ml-2">
   596→                          {formatRelativeTime(convo.lastMessage.createdAt)}
   597→                        </span>
   598→                      )}
   599→                    </div>
   600→                    <p className="text-xs text-muted-foreground truncate mt-0.5">
   601→                      {getConvoPreview(convo)}
   602→                    </p>
   603→                  </div>
   604→                  {convo.unread > 0 && (
   605→                    <div className="h-5 w-5 rounded-full bg-emerald-500 text-white text-[10px] font-bold flex items-center justify-center flex-shrink-0 mt-1">
   606→                      {convo.unread}
   607→                    </div>
   608→                  )}
   609→                </button>
   610→              )
   611→            })}
   612→          </div>
   613→        </ScrollArea>
   614→      </div>
   615→
   616→      {/* ─── Chat Area ─────────────────────────────────── */}
   617→      <div className="flex-1 flex flex-col bg-background">
   618→        {selectedConvo ? (
   619→          <>
   620→            {/* Chat Header */}
   621→            <div className="p-3 border-b flex items-center gap-3">
   622→              {isGroupSelected ? (
   623→                <>
   624→                  <button onClick={() => setShowGroupInfo(true)} className="flex items-center gap-3 hover:bg-muted/50 rounded-md px-1 -ml-1 py-0.5 transition-colors">
   625→                    <GroupAvatar name={selectedGroup!.name} size="md" />
   626→                    <div>
   627→                      <p className="text-sm font-medium flex items-center gap-1.5">
   628→                        {selectedGroup!.name}
   629→                        <Users className="h-3.5 w-3.5 text-violet-500" />
   630→                      </p>
   631→                      <p className="text-xs text-muted-foreground">
   632→                        {selectedGroup!._count.members} members
   633→                      </p>
   634→                    </div>
   635→                  </button>
   636→                </>
   637→              ) : selectedDirectUser ? (
   638→                <>
   639→                  <UserAvatar user={selectedDirectUser} size="md" />
   640→                  <div>
   641→                    <p className="text-sm font-medium">{selectedDirectUser.name}</p>
   642→                    <p className="text-xs text-muted-foreground">
   643→                      {selectedDirectUser.company ?? selectedDirectUser.email}
   644→                    </p>
   645→                  </div>
   646→                  <Badge variant="secondary" className={cn('ml-auto text-xs', getRoleBadgeClasses(selectedDirectUser.role))}>
   647→                    {selectedDirectUser.role}
   648→                  </Badge>
   649→                </>
   650→              ) : null}
   651→            </div>
   652→
   653→            {/* Messages */}
   654→            <ScrollArea className="flex-1 p-4">
   655→              <div className="space-y-3 max-w-2xl mx-auto">
   656→                {msgLoading ? (
   657→                  <div className="space-y-3">
   658→                    {Array.from({ length: 4 }).map((_, i) => (
   659→                      <div key={i} className={cn('flex gap-2.5', i % 2 === 0 ? '' : 'justify-end')}>
   660→                        <Skeleton className="h-10 w-48 rounded-2xl" />
   661→                      </div>
   662→                    ))}
   663→                  </div>
   664→                ) : messages.length === 0 ? (
   665→                  <div className="flex flex-col items-center justify-center h-full text-center py-8">
   666→                    <MessageSquare className="h-10 w-10 text-muted-foreground/30 mb-3" />
   667→                    <p className="text-sm text-muted-foreground">
   668→                      {isGroupSelected ? 'No messages yet. Start the conversation!' : 'No messages yet. Say hello!'}
   669→                    </p>
   670→                  </div>
   671→                ) : (
   672→                  messages.map((msg, idx) => {
   673→                    const isSent = msg.senderId === user?.id
   674→                    return (
   675→                      <motion.div
   676→                        key={msg.id}
   677→                        initial={{ opacity: 0, y: 5 }}
   678→                        animate={{ opacity: 1, y: 0 }}
   679→                        transition={{ delay: Math.min(idx * 0.015, 0.2) }}
   680→                        className={cn('flex gap-2.5', isSent ? 'justify-end' : 'justify-start')}
   681→                      >
   682→                        {!isSent && (
   683→                          <UserAvatar user={msg.sender} size="sm" className="mt-auto" />
   684→                        )}
   685→                        <div className={cn('max-w-[70%]', isSent ? 'text-right' : '')}>
   686→                          {/* Show sender name in group chats */}
   687→                          {isGroupSelected && !isSent && (
   688→                            <p className="text-xs font-medium text-muted-foreground mb-1 px-1">
   689→                              {msg.sender?.name ?? 'Unknown'}
   690→                            </p>
   691→                          )}
   692→                          <div className={cn(
   693→                            'inline-block rounded-2xl px-4 py-2.5 text-sm',
   694→                            isSent
   695→                              ? 'bg-emerald-600 text-white rounded-br-md'
   696→                              : 'bg-muted rounded-bl-md',
   697→                          )}>
   698→                            <p className="whitespace-pre-wrap break-words">{msg.content}</p>
   699→                          </div>
   700→                          <div className={cn(
   701→                            'flex items-center gap-1 mt-1 text-[10px] text-muted-foreground',
   702→                            isSent ? 'justify-end' : 'justify-start'
   703→                          )}>
   704→                            <span>{formatRelativeTime(msg.createdAt)}</span>
   705→                            {isSent && msg.read && <CheckCheck className="w-3 h-3" />}
   706→                          </div>
   707→                        </div>
   708→                      </motion.div>
   709→                    )
   710→                  })
   711→                )}
   712→                <div ref={messagesEndRef} />
   713→              </div>
   714→            </ScrollArea>
   715→
   716→            {/* Message Input */}
   717→            <div className="p-3 border-t">
   718→              <div className="flex gap-2 max-w-2xl mx-auto">
   719→                <Input
   720→                  placeholder={isGroupSelected ? `Message ${selectedGroup!.name}...` : 'Type a message...'}
   721→                  value={msgText}
   722→                  onChange={(e) => setMsgText(e.target.value)}
   723→                  onKeyDown={handleKeyDown}
   724→                  className="flex-1"
   725→                />
   726→                <Button
   727→                  onClick={sendMessage}
   728→                  disabled={!msgText.trim() || sending}
   729→                  className="bg-emerald-600 hover:bg-emerald-700 text-white flex-shrink-0"
   730→                >
   731→                  {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   732→                </Button>
   733→              </div>
   734→            </div>
   735→          </>
   736→        ) : (
   737→          <div className="flex-1 flex items-center justify-center">
   738→            <div className="text-center space-y-3">
   739→              <div className="h-16 w-16 rounded-full bg-muted flex items-center justify-center mx-auto">
   740→                <MessageSquare className="h-8 w-8 text-muted-foreground" />
   741→              </div>
   742→              <h3 className="font-semibold text-lg">Select a conversation</h3>
   743→              <p className="text-sm text-muted-foreground max-w-xs">
   744→                Choose a conversation from the sidebar or create a new group.
   745→              </p>
   746→            </div>
   747→          </div>
   748→        )}
   749→      </div>
   750→
   751→      {/* Dialogs */}
   752→      <CreateGroupDialog
   753→        open={showCreateGroup}
   754→        onOpenChange={setShowCreateGroup}
   755→        onCreated={() => { fetchConversations(); setShowCreateGroup(false) }}
   756→        currentUser={user!}
   757→      />
   758→      <GroupInfoDialog
   759→        open={showGroupInfo}
   760→        onOpenChange={setShowGroupInfo}
   761→        group={selectedGroup ? { id: selectedGroup.id, name: selectedGroup.name, _count: selectedGroup._count } : null}
   762→        currentUserId={user?.id ?? ''}
   763→      />
   764→    </motion.div>
   765→  )

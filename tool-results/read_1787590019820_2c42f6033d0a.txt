    46→function formatElapsed(ms: number): string {
    47→  const totalSeconds = Math.floor(ms / 1000)
    48→  const h = Math.floor(totalSeconds / 3600)
    49→  const m = Math.floor((totalSeconds % 3600) / 60)
    50→  const s = totalSeconds % 60
    51→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    52→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    53→}
    54→
    55→function formatCountdown(ms: number): string {
    56→  if (ms <= 0) return '0:00'
    57→  const totalSeconds = Math.floor(ms / 1000)
    58→  const m = Math.floor(totalSeconds / 60)
    59→  const s = totalSeconds % 60
    60→  return `${m}:${String(s).padStart(2, '0')}`
    61→}

   894→function OfflineBanner() {
   895→  const [isOffline, setIsOffline] = useState(() => {
   896→    if (typeof navigator === 'undefined') return false
   897→    return !navigator.onLine
   898→  })
   899→
   900→  useEffect(() => {
   901→    const handleOnline = () => setIsOffline(false)
   902→    const handleOffline = () => setIsOffline(true)
   903→    window.addEventListener('online', handleOnline)
   904→    window.addEventListener('offline', handleOffline)
   905→    return () => {
   906→      window.removeEventListener('online', handleOnline)
   907→      window.removeEventListener('offline', handleOffline)
   908→    }
   909→  }, [])
   910→
   911→  if (!isOffline) return null
   912→
   913→  return (
   914→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   915→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   916→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   917→    </div>
   918→  )
   919→}

    64→// ─── File Upload Zone ───────────────────────────────────────────────────────────

    66→function FileUploadZone({
    67→  label,
    68→  required,
    69→  accept,
    70→  file,
    71→  onUpload,
    72→  onRemove,
    73→  icon: Icon = Upload,
    74→  description,
    75→}: {
    76→  label: string; required?: boolean; accept?: string
    77→  file: UploadedFile | null; onUpload: (file: UploadedFile) => void; onRemove: () => void
    78→  icon?: React.ElementType; description?: string
    79→}) {
    80→  const [uploading, setUploading] = useState(false)
    81→  const inputRef = useRef<HTMLInputElement>(null)
    82→
    83→  const handleFile = async (fileObj: File) => {
    84→    setUploading(true)
    85→    try {
    86→      const formData = new FormData()
    87→      formData.append('file', fileObj)
    88→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
    89→      if (!res.ok) {
    90→        const d = await res.json()
    91→        throw new Error(d.error || 'Upload failed')
    92→      }
    93→      const data = await res.json()
    94→      onUpload({
    95→        url: data.url,
    96→        originalName: data.originalName,
    97→        filename: data.filename,
    98→        size: data.size,
    99→        type: data.type,
   100→      })
   101→      toast.success('File uploaded')
   102→    } catch (e: unknown) {
   103→      toast.error(e instanceof Error ? e.message : 'Upload failed')
   104→    } finally {
   105→      setUploading(false)
   106→    }
   107→  }
   108→
   109→  const handleDrop = (e: React.DragEvent) => {
   110→    e.preventDefault()
   111→    const f = e.dataTransfer.files[0]
   112→    if (f) handleFile(f)
   113→  }
   114→
   115→  const formatSize = (bytes: number) => {
   116→    if (bytes < 1024) return `${bytes} B`
   117→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   118→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   119→  }
   120→
   121→  return (
   122→    <div className="space-y-2">
   123→      <Label className="text-sm font-medium">
   124→        {label} {required && <span className="text-red-500">*</span>}
   125→      </Label>
   126→      {file ? (
   127→        <div className="flex items-center gap-3 p-3 rounded-lg border bg-muted/30">
   128→          <div className="w-10 h-10 rounded-lg bg-primary/10 flex items-center justify-center flex-shrink-0">
   129→            <FileText className="w-5 h-5 text-primary" />
   130→          </div>
   131→          <div className="flex-1 min-w-0">
   132→            <p className="text-sm font-medium truncate">{file.originalName}</p>
   133→            <p className="text-xs text-muted-foreground">{formatSize(file.size)}</p>
   134→          </div>
   135→          <Button
   136→            type="button" variant="ghost" size="icon" className="h-8 w-8 text-muted-foreground hover:text-destructive"
   137→            onClick={onRemove}
   138→          >
   139→            <X className="h-4 w-4" />
   140→          </Button>
   141→        </div>
   142→      ) : (
   143→        <div
   144→          className="border-2 border-dashed rounded-lg p-6 flex flex-col items-center justify-center gap-2 cursor-pointer hover:border-primary/50 hover:bg-muted/20 transition-colors"
   145→          onClick={() => inputRef.current?.click()}
   146→          onDragOver={(e) => e.preventDefault()}
   147→          onDrop={handleDrop}
   148→        >
   149→          {uploading ? (
   150→            <Loader2 className="w-8 h-8 text-primary animate-spin" />
   151→          ) : (
   152→            <>
   153→              <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center">
   154→                <Icon className="w-6 h-6 text-muted-foreground" />
   155→              </div>
   156→              <div className="text-center">
   157→                <p className="text-sm font-medium text-muted-foreground">Click to upload</p>
   158→                {description && <p className="text-xs text-muted-foreground mt-1">{description}</p>}
   159→              </div>
   160→            </>
   161→          )}
   162→          <input
   163→            ref={inputRef}
   164→            type="file"
   165→            className="hidden"
   166→            accept={accept}
   167→            onChange={(e) => {
   168→              const f = e.target.files?.[0]
   169→              if (f) handleFile(f)
   170→              e.target.value = ''
   171→            }}
   172→          />
   173→        </div>
   174→      )}
   175→    </div>
   176→  )
   177→}

   179→// ─── Step Indicator Card ────────────────────────────────────────────────────────
   180→
   181→function StepCard({
   182→  step, index, currentStep, isCompleted, onClick,
   183→}: {
   184→  step: string; index: number; currentStep: number; isCompleted: boolean; onClick: () => void
   185→}) {
   186→  const Icon = STEP_ICONS[index]
   187→  const isCurrent = index === currentStep
   188→  const isClickable = isCompleted || index === currentStep
   189→
   190→  return (
   191→    <button
   192→      type="button"
   193→      onClick={isClickable ? onClick : undefined}
   194→      disabled={!isClickable}
   195→      className={cn(
   196→        'flex-1 min-w-0 flex flex-col items-center gap-1.5 py-3 px-2 rounded-xl transition-all duration-200 border',
   197→        isCompleted && 'bg-emerald-50 border-emerald-200 dark:bg-emerald-900/20 dark:border-emerald-800',
   198→        isCurrent && !isCompleted && 'bg-primary/10 border-primary/30',
   199→        !isCurrent && !isCompleted && 'bg-card border-transparent opacity-50',
   200→        isClickable && 'cursor-pointer hover:shadow-sm',
   201→        !isClickable && 'cursor-not-allowed',
   202→      )}
   203→    >
   204→      <div className={cn(
   205→        'w-10 h-10 rounded-full flex items-center justify-center transition-colors',
   206→        isCompleted && 'bg-emerald-500 text-white',
   207→        isCurrent && !isCompleted && 'bg-primary text-primary-foreground',
   208→        !isCurrent && !isCompleted && 'bg-muted text-muted-foreground',
   209→      )}>
   210→        {isCompleted ? <Check className="w-5 h-5" /> : <Icon className="w-5 h-5" />}
   211→      </div>
   212→      <span className={cn(
   213→        'text-[11px] md:text-xs font-medium text-center leading-tight',
   214→        isCompleted && 'text-emerald-700 dark:text-emerald-400',
   215→        isCurrent && !isCompleted && 'text-primary',
   216→        !isCurrent && !isCompleted && 'text-muted-foreground',
   217→      )}>
   218→        {step}
   219→      </span>
   220→    </button>
   221→  )
   222→}

   239→  const [submitting, setSubmitting] = useState(false)
   240→
   241→  const totalSteps = 4
   242→

   244→    if (!user?.id) return
   245→    try {
   246→      const res = await fetch(`/api/onboarding?clientId=${user.id}`)
   247→      const data = await res.json()
   248→      setForms(data.forms || [])
   249→    } catch {
   250→      toast.error('Failed to load onboarding forms')
   251→    } finally {
   252→      setLoading(false)
   253→    }
   254→  }, [user?.id])
   255→
   256→  useEffect(() => { fetchForms() }, [fetchForms])
   257→
   258→  const openForm = async (formId: string) => {
   259→    setFormLoading(true)
   260→    try {
   261→      const res = await fetch(`/api/onboarding/${formId}`)

   279→    if (mySubmission?.responsesJson) {
   280→      try {
   281→        const parsed = JSON.parse(mySubmission.responsesJson)
   282→        setResponses(parsed.responses || {})
   283→        setFiles(parsed.files || {})
   284→      } catch { /* ignore */ }
   285→    }
   286→  }, [mySubmission?.id])
   287→
   288→  const updateResponse = (name: string, value: string) => {
   289→    setResponses((prev) => ({ ...prev, [name]: value }))
   290→  }
   291→
   292→  const updateFile = (name: string, file: UploadedFile | null) => {
   293→    setFiles((prev) => ({ ...prev, [name]: file }))
   294→  }
   295→
   296→  // Validation
   297→  const isStepValid = (step: number) => {
   298→    switch (step) {
   299→      case 0:
   300→        return !!(responses.businessName?.trim() && responses.businessEmail?.trim())
   301→      case 1:
   302→        return !!files.companyLogo
   303→      case 2:

   306→        return !!(files.contractForm && files.paymentForm && files.workOrderAgreement)
   307→      default:
   308→        return true
   309→    }
   310→  }
   311→
   312→  const currentStepValid = isStepValid(currentStep)
   313→  const progress = ((currentStep + (currentStepValid ? 1 : 0.5)) / totalSteps) * 100
   314→
   315→  const goToStep = (step: number) => {
   316→    if (isLocked) return
   317→    if (step <= currentStep) setCurrentStep(step)
   318→  }
   319→
   320→  const handleNext = () => {
   321→    if (currentStep < totalSteps - 1 && currentStepValid) {
   322→      setCurrentStep(currentStep + 1)
   323→    }
   324→  }
   325→
   326→  const handlePrev = () => {
   327→    if (currentStep > 0) setCurrentStep(currentStep - 1)
   328→  }
   329→

   337→        method: 'PATCH',
   338→        headers: { 'Content-Type': 'application/json' },
   339→        body: JSON.stringify({
   340→          action: 'submit',
   341→          responsesJson: submissionData,
   342→          ...(submissionId ? {} : { formId: selectedForm.id, clientId: user.id }),
   343→        }),
   344→      })

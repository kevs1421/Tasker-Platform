    98→    98→export function ClientProjects() {
    99→    99→  const { user } = useAuthStore()
   100→   100→  const [projects, setProjects] = useState<ClientProject[]>([])
   101→   101→  const [loading, setLoading] = useState(true)
   102→   102→
   103→   103→  // Upload files state
   104→   104→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   105→   105→  const [uploading, setUploading] = useState(false)
   106→   106→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   107→   107→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   108→   108→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   109→   109→  const fileInputRef = useRef<HTMLInputElement>(null)
   110→   110→
   111→   111→  // Onboarding creation state
   112→   112→
   113→   113→
   114→   114→  const fetchProjects = useCallback(async () => {
   115→   115→    if (!user?.id) return
   116→   116→    try {
   117→   117→      // Use the client-only endpoint that queries through ClientProject table
   118→   118→      // This ensures only projects explicitly assigned to this client are returned
   119→   119→      const res = await fetch(`/api/client/projects?clientId=${user.id}`)
   120→   120→      const data = await res.json()
   121→   121→      setProjects(data.projects || [])
   122→   122→    } catch {
   123→   123→      toast.error('Failed to load projects')
   124→   124→    } finally {
   125→   125→      setLoading(false)
   126→   126→    }
   127→   127→  }, [user?.id])
   128→   128→
   129→   129→  // Fetch uploaded files from localStorage
   130→   130→  const fetchUploadedFiles = useCallback(() => {
   131→   131→    if (!user) return
   132→   132→    try {
   133→   133→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   134→   134→      if (stored) setUploadedFiles(JSON.parse(stored))
   135→   135→    } catch { /* ignore */ }
   136→   136→  }, [user])
   137→   137→
   138→   138→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   139→   139→    if (!user) return
   140→   140→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   141→   141→  }, [user])
   142→   142→
   143→   143→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   144→   144→
   145→   145→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   146→   146→  const handleFileUpload = async (files: FileList | null) => {
   147→   147→    if (!files || files.length === 0) return
   148→   148→    if (!uploadDialogProjectId) {
   149→   149→      toast.error('No project selected')
   150→   150→      return
   151→   151→    }
   152→   152→    setUploading(true)
   153→   153→    const newFiles: typeof uploadedFiles = []
   154→   154→    for (let i = 0; i < files.length; i++) {
   155→   155→      const file = files[i]
   156→   156→      try {
   157→   157→        const formData = new FormData()
   158→   158→        formData.append('file', file)
   159→   159→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   160→   160→        if (!res.ok) {
   161→   161→          const d = await res.json()
   162→   162→          throw new Error(d.error || 'Upload failed')
   163→   163→        }
   164→   164→        const data = await res.json()
   165→   165→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   166→   166→      } catch (e: unknown) {
   167→   167→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   168→   168→      }
   169→   169→    }
   170→   170→    if (newFiles.length > 0) {
   171→   171→      const updated = [...uploadedFiles, ...newFiles]
   172→   172→      setUploadedFiles(updated)
   173→   173→      saveUploadedFiles(updated)
   174→   174→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   175→   175→    }
   176→   176→    setUploading(false)
   177→   177→    if (fileInputRef.current) fileInputRef.current.value = ''
   178→   178→  }
   179→   179→
   180→   180→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   181→   181→    e.preventDefault()
   182→   182→    setUploadDragOver(null)
   183→   183→    setUploadDialogProjectId(projectId)
   184→   184→    handleFileUpload(e.dataTransfer.files)
   185→   185→  }
   186→   186→
   187→   187→  const openUploadDialog = (projectId: string) => {
   188→   188→    setUploadDialogProjectId(projectId)
   189→   189→    setUploadDialogOpen(true)
   190→   190→  }
   191→   191→
   192→   192→  const deleteUploadedFile = (idx: number) => {
   193→   193→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   194→   194→    setUploadedFiles(updated)
   195→   195→    saveUploadedFiles(updated)
   196→   196→    toast.success('File removed')
   197→   197→  }
   198→   198→
   199→   199→  const getFileIcon = (type: string) => {
   200→   200→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   201→   201→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   202→   202→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   203→   203→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   204→   204→    return <File className="w-4 h-4 text-muted-foreground" />
   205→   205→  }
   206→   206→
   207→   207→  const formatFileSize = (bytes: number) => {
   208→   208→    if (bytes < 1024) return `${bytes} B`
   209→   209→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   210→   210→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   211→   211→  }
   212→   212→
   213→   213→  if (loading) return <PageSkeleton />
   214→   214→
   215→   215→  return (
   216→   216→    <div className="p-4 md:p-6 space-y-6">
   217→   217→      {/* Page Header */}
   218→   218→      <div className="flex items-center gap-3">
   219→   219→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   220→   220→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   221→   221→        </div>
   222→   222→        <div>
   223→   223→          <h1 className="text-2xl font-bold">My Projects</h1>
   224→   224→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   225→   225→        </div>
   226→   226→      </div>
   227→   227→
   228→   228→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   229→   229→      <input
   230→   230→        ref={fileInputRef}
   231→   231→        type="file"
   232→   232→        multiple
   233→   233→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   234→   234→        className="hidden"
   235→   235→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   236→   236→      />
   237→   237→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   238→   238→        <DialogContent className="sm:max-w-md">
   239→   239→          <DialogHeader>
   240→   240→            <DialogTitle className="flex items-center gap-2">
   241→   241→              <Upload className="w-5 h-5 text-primary" />
   242→   242→              Upload Files
   243→   243→            </DialogTitle>
   244→   244→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   245→   245→          </DialogHeader>
   246→   246→          <div
   247→   247→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   248→   248→            onDragLeave={() => setUploadDragOver(null)}
   249→   249→            onDrop={(e) => {
   250→   250→              e.preventDefault()
   251→   251→              setUploadDragOver(null)
   252→   252→              // Set the project context before handling files
   253→   253→              const files = e.dataTransfer.files
   254→   254→              if (files && files.length > 0) {
   255→   255→                // Use a separate handler for dialog drops
   256→   256→                void (async () => {
   257→   257→                  setUploading(true)
   258→   258→                  const newFiles: typeof uploadedFiles = []
   259→   259→                  for (let i = 0; i < files.length; i++) {
   260→   260→                    const file = files[i]
   261→   261→                    try {
   262→   262→                      const formData = new FormData()
   263→   263→                      formData.append('file', file)
   264→   264→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   265→   265→                      if (!res.ok) {
   266→   266→                        const d = await res.json()
   267→   267→                        throw new Error(d.error || 'Upload failed')
   268→   268→                      }
   269→   269→                      const data = await res.json()
   270→   270→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   271→   271→                    } catch (err: unknown) {
   272→   272→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   273→   273→                    }
   274→   274→                  }
   275→   275→                  if (newFiles.length > 0) {
   276→   276→                    const updated = [...uploadedFiles, ...newFiles]
   277→   277→                    setUploadedFiles(updated)
   278→   278→                    saveUploadedFiles(updated)
   279→   279→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   280→   280→                  }
   281→   281→                  setUploading(false)
   282→   282→                })()
   283→   283→              }
   284→   284→            }}
   285→   285→            onClick={() => fileInputRef.current?.click()}
   286→   286→            className={cn(
   287→   287→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   288→   288→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   289→   289→            )}
   290→   290→          >
   291→   291→            {uploading ? (
   292→   292→              <div className="flex flex-col items-center gap-2">
   293→   293→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   294→   294→                <p className="text-sm text-muted-foreground">Uploading...</p>
   295→   295→              </div>
   296→   296→            ) : (
   297→   297→              <div className="flex flex-col items-center gap-2">
   298→   298→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   299→   299→                  <Upload className="w-5 h-5 text-muted-foreground" />
   300→   300→                </div>
   301→   301→                <div>
   302→   302→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   303→   303→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   304→   304→                </div>
   305→   305→              </div>
   306→   306→            )}
   307→   307→          </div>
   308→   308→          <DialogFooter>
   309→   309→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   310→   310→          </DialogFooter>
   311→   311→        </DialogContent>
   312→   312→      </Dialog>
   313→   313→
   314→   314→      {/* ─── Project Progress Overview ──────────────────────────── */}
   315→   315→      {projects.length > 0 && (
   316→   316→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   317→   317→          <Card>
   318→   318→            <CardHeader className="pb-3">
   319→   319→              <CardTitle className="text-base flex items-center gap-2">
   320→   320→                <BarChart3 className="w-4 h-4 text-primary" />
   321→   321→                Project Progress
   322→   322→              </CardTitle>
   323→   323→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   324→   324→            </CardHeader>
   325→   325→            <CardContent>
   326→   326→              <div className="space-y-5">
   327→   327→                {projects.map((project) => {
   328→   328→                  const todo = project.taskStats?.todo || 0
   329→   329→                  const inProgress = project.taskStats?.in_progress || 0
   330→   330→                  const inReview = project.taskStats?.in_review || 0
   331→   331→                  const completed = project.taskStats?.done || 0

   339→   339→
   340→   340→                  return (
   341→   341→                    <div key={project.id} className="space-y-2.5">
   342→   342→                      <div className="flex items-center justify-between">

   345→   345→                          <span className="text-sm font-medium">{project.name}</span>

   347→   347→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   348→   348→                      </div>
   349→   349→                      {/* Stacked progress bar */}
   350→   350→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   351→   351→                        {todo > 0 && (
   352→   352→                          <div
   353→   353→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   354→   354→                            style={{ width: `${todoPct}%` }}
   355→   355→                            title={`Not Started: ${todo}`}
   356→   356→                          />
   357→   357→                        )}
   358→   358→                        {inProgress > 0 && (
   359→   359→                          <div
   360→   360→                            className="bg-amber-500 transition-all duration-500"
   361→   361→                            style={{ width: `${inProgressPct}%` }}
   362→   362→                            title={`In Progress: ${inProgress}`}
   363→   363→                          />
   364→   364→                        )}
   365→   365→                        {inReview > 0 && (
   366→   366→                          <div
   367→   367→                            className="bg-blue-500 transition-all duration-500"
   368→   368→                            style={{ width: `${inReviewPct}%` }}
   369→   369→                            title={`In Review: ${inReview}`}
   370→   370→                          />
   371→   371→                        )}
   372→   372→                        {completed > 0 && (
   373→   373→                          <div
   374→   374→                            className="bg-emerald-500 transition-all duration-500"
   375→   375→                            style={{ width: `${completedPct}%` }}
   376→   376→                            title={`Completed: ${completed}`}
   377→   377→                          />
   378→   378→                        )}
   379→   379→                      </div>
   380→   380→                      {/* Legend */}
   381→   381→                      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   382→   382→                        <span className="flex items-center gap-1.5">
   383→   383→                          <span className="w-2.5 h-2.5 rounded-sm bg-slate-400 dark:bg-slate-500" />
   384→   384→                          Not Started ({todo})
   385→   385→                        </span>
   386→   386→                        <span className="flex items-center gap-1.5">
   387→   387→                          <span className="w-2.5 h-2.5 rounded-sm bg-amber-500" />
   388→   388→                          In Progress ({inProgress})
   389→   389→                        </span>
   390→   390→                        <span className="flex items-center gap-1.5">
   391→   391→                          <span className="w-2.5 h-2.5 rounded-sm bg-blue-500" />
   392→   392→                          In Review ({inReview})
   393→   393→                        </span>
   394→   394→                        <span className="flex items-center gap-1.5">
   395→   395→                          <span className="w-2.5 h-2.5 rounded-sm bg-emerald-500" />
   396→   396→                          Completed ({completed})
   397→   397→                        </span>
   398→   398→                      </div>
   399→   399→                    </div>
   400→   400→                  )
   401→   401→                })}
   402→   402→              </div>
   403→   403→            </CardContent>
   404→   404→          </Card>
   405→   405→        </motion.div>
   406→   406→      )}
   407→   407→
   408→   408→      {projects.length === 0 ? (
   409→   409→        <EmptyState
   410→   410→          icon={FolderKanban}
   411→   411→          title="No active projects"
   412→   412→          description="You don't have any active projects yet."
   413→   413→        />
   414→   414→      ) : (
   415→   415→        <>
   416→   416→          {/* ─── Project Cards ──────────────────────────────────── */}
   417→   417→          <div className="space-y-4">

   430→   430→              return (

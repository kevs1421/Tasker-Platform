    65→    65→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    66→    66→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    67→    67→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    68→    68→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    69→    69→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    70→    70→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    71→    71→  return <CircleDot className="h-4 w-4 text-slate-400" />
    72→    72→}

   416→   416→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   417→   417→      })

   419→   419→        const d = await patchRes.json()
   420→   420→        throw new Error(d.error)
   421→   421→      }
   422→   422→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)

   424→   424→      if (selectedForm) {
   425→   425→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)

   427→   427→          const data = await fresh.json()

   430→   430→      }
   431→   431→      fetchForms()
   432→   432→    } catch (e: unknown) {
   433→   433→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   434→   434→    } finally {
   435→   435→      setReviewing(null)
   436→   436→    }

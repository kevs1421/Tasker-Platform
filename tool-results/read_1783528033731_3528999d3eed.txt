   164→   164→    try {
   165→   165→      const res = await fetch('/api/dashboard/stats')

   167→   167→      const data = await res.json()

   178→   178→      const data = await res.json()

   180→   180→    } catch {
   181→   181→      toast.error('Failed to load projects')

   185→   185→  const fetchUsers = useCallback(async () => {
   186→   186→    try {
   187→   187→      const res = await fetch('/api/users')
   188→   188→      if (!res.ok) throw new Error('Failed to fetch users')
   189→   189→      const data = await res.json()

   191→   191→    } catch {
   192→   192→      toast.error('Failed to load users')
   193→   193→    }

   195→   195→
   196→   196→  const fetchActivities = useCallback(async () => {
   197→   197→    try {
   198→   198→      const res = await fetch('/api/dashboard/activity')
   199→   199→      if (!res.ok) throw new Error('Failed to fetch activities')

   202→   202→    } catch {

   204→   204→    }
   205→   205→  }, [])

   211→   211→      setLoading(false)

   229→   229→        headers: { 'Content-Type': 'application/json' },

   231→   231→          ...newProject,
   232→   232→          createdBy: currentUser.id,

   252→   252→        method: 'PATCH',

   292→   292→      toast.error('Failed to update project status')

   294→   294→  }, [fetchProjects, fetchStats])
   295→   295→
   296→   296→  // ── Chart Data ──

   316→   316→    if (lower.includes('updat') || lower.includes('edit')) return ACTIVITY_ICONS.updated
   317→   317→    return ACTIVITY_ICONS.created
   318→   318→  }
   319→   319→
   320→   320→  // ── Loading State ──
   321→   321→
   322→   322→  if (loading) {
   323→   323→    return (
   324→   324→      <div className="space-y-6 p-4 md:p-6">

   126→// ═══════════════════════════════════════════════════════════════════
   127→const ONBOARDING_STEPS = [
   128→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   129→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   130→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   131→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   132→]
   133→
   134→export function ClientProjects() {
   135→  const { user } = useAuthStore()
   136→  const [projects, setProjects] = useState<(Project & { tasks?: (Task & { _count: { comments: number; subtasks: number; completedSubtasks: number } })[] })[]>([])
   137→  const [loading, setLoading] = useState(true)
   138→  const [expandedProjectId, setExpandedProjectId] = useState<string | null>(null)
   139→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   140→  const [taskDetail, setTaskDetail] = useState<(Task & { _count: { comments: number; subtasks: number; completedSubtasks: number }; comments?: Comment[]; subtasks?: Subtask[] }) | null>(null)
   141→  const [taskDetailLoading, setTaskDetailLoading] = useState(false)
   142→
   143→  // Onboarding creation state
   144→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   145→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   146→  const [onboardingTitle, setOnboardingTitle] = useState('')
   147→  const [onboardingDesc, setOnboardingDesc] = useState('')
   148→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   149→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   150→
   151→  const fetchProjects = useCallback(async () => {
   152→    try {
   153→      const res = await fetch('/api/projects')
   154→      const data = await res.json()
   155→      // Filter to active projects and those the client is a member of
   156→      const clientProjects = (data.projects || [])
   157→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   158→      setProjects(clientProjects)
   159→    } catch {
   160→      toast.error('Failed to load projects')
   161→    } finally {
   162→      setLoading(false)
   163→    }
   164→  }, [user?.id])
   165→
   166→  useEffect(() => { fetchProjects() }, [fetchProjects])
   167→
   168→  const expandProject = async (projectId: string) => {
   169→    if (expandedProjectId === projectId) {
   170→      setExpandedProjectId(null)
   171→      setExpandedTasks([])
   172→      return
   173→    }
   174→    setExpandedProjectId(projectId)
   175→    try {
   176→      const res = await fetch(`/api/projects/${projectId}`)
   177→      const data = await res.json()
   178→      setExpandedTasks(data.project?.tasks || [])
   179→    } catch {
   180→      toast.error('Failed to load project tasks')
   181→    }
   182→  }
   183→
   184→  const openTaskDetail = async (taskId: string) => {
   185→    setTaskDetailLoading(true)
   186→    setTaskDetail(null)
   187→    try {
   188→      const res = await fetch(`/api/tasks/${taskId}`)
   189→      const data = await res.json()
   190→      setTaskDetail(data.task)
   191→    } catch {
   192→      toast.error('Failed to load task details')
   193→    } finally {
   194→      setTaskDetailLoading(false)
   195→    }
   196→  }
   197→
   198→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   199→    setOnboardingProjectId(projectId)
   200→    setOnboardingTitle(`Onboarding - ${projectName}`)
   201→    setOnboardingDesc('')
   202→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   203→    setOnboardingDialogOpen(true)
   204→  }
   205→
   206→  const handleCreateOnboarding = async () => {
   207→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   208→    setCreatingOnboarding(true)
   209→    try {
   210→      const stepsJson = JSON.stringify(
   211→        onboardingSteps.map(stepId => {
   212→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   213→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   214→        })
   215→      )
   216→      const res = await fetch('/api/onboarding', {
   217→        method: 'POST',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({
   220→          title: onboardingTitle.trim(),
   221→          description: onboardingDesc.trim() || null,
   222→          stepsJson,
   223→          projectId: onboardingProjectId,
   224→          clientId: user?.id || null,
   225→          status: 'published',
   226→        }),
   227→      })
   228→      if (!res.ok) {
   229→        const d = await res.json()
   230→        throw new Error(d.error || 'Failed to create')
   231→      }
   232→      toast.success('Onboarding form created successfully')
   233→      setOnboardingDialogOpen(false)
   234→    } catch (e: unknown) {
   235→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   236→    } finally {
   237→      setCreatingOnboarding(false)
   238→    }
   239→  }
   240→
   241→  const toggleOnboardingStep = (stepId: string) => {
   242→    setOnboardingSteps(prev =>
   243→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   244→    )
   245→  }
   246→
   247→  // Stats
   248→  const allTasks = projects.flatMap(p => p.tasks || [])
   249→  const totalTasks = projects.reduce((sum, p) => sum + (p._count?.tasks || 0), 0)
   250→  const completedTasks = allTasks.filter(t => t.status === 'done').length
   251→  const inProgressTasks = allTasks.filter(t => t.status === 'in_progress').length
   252→
   253→  if (loading) return <PageSkeleton />
   254→
   255→  return (
   256→    <div className="p-4 md:p-6 space-y-6">
   257→      <div className="flex items-center gap-3">
   258→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   259→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   260→        </div>
   261→        <div>
   262→          <h1 className="text-2xl font-bold">My Projects</h1>
   263→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   264→        </div>
   265→      </div>
   266→
   267→      {/* Stats */}
   268→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   269→        <StatCard icon={FolderKanban} label="My Projects" value={projects.length} color="bg-emerald-500" />
   270→        <StatCard icon={ListTodo} label="Total Tasks" value={totalTasks} color="bg-amber-500" />
   271→        <StatCard icon={CheckCircle2} label="Completed" value={completedTasks} color="bg-teal-500" />
   272→        <StatCard icon={Clock} label="In Progress" value={inProgressTasks} color="bg-orange-500" />
   273→      </div>
   274→
   275→      {/* Project Cards */}
   276→      {projects.length === 0 ? (
   277→        <EmptyState
   278→          icon={FolderKanban}
   279→          title="No active projects"
   280→          description="You don't have any active projects yet."
   281→        />
   282→      ) : (
   283→        <div className="space-y-4">
   284→          {projects.map((project, i) => {
   285→            const projectTasks = expandedProjectId === project.id ? expandedTasks : []
   286→            const doneTasks = projectTasks.filter(t => t.status === 'done').length
   287→            const totalT = project._count?.tasks || 0
   288→            const pct = totalT > 0 ? Math.round((doneTasks / totalT) * 100) : 0
   289→            const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   290→
   291→            return (
   292→              <motion.div
   293→                key={project.id}
   294→                initial={{ opacity: 0, y: 12 }}
   295→                animate={{ opacity: 1, y: 0 }}
   296→                transition={{ delay: i * 0.05 }}
   297→              >
   298→                <Card
   299→                  className="cursor-pointer hover:shadow-md transition-shadow border-l-4"
   300→                  style={{ borderLeftColor: project.color }}
   301→                  onClick={() => expandProject(project.id)}
   302→                >
   303→                  <CardContent className="p-4 md:p-5">
   304→                    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   305→                      <div className="min-w-0 flex-1">
   306→                        <div className="flex items-center gap-2 mb-1">
   307→                          <h3 className="font-semibold text-base truncate">{project.name}</h3>
   308→                          <PriorityBadge priority={project.priority} />
   309→                        </div>
   310→                        <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   311→                      </div>
   312→                      <div className="flex items-center gap-3 shrink-0 text-sm">
   313→                        {project.dueDate && (
   314→                          <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   315→                            <Calendar className="w-3.5 h-3.5" />
   316→                            {formatDate(project.dueDate)}
   317→                          </span>
   318→                        )}
   319→                        <span className="text-muted-foreground flex items-center gap-1">
   320→                          <Users className="w-3.5 h-3.5" />
   321→                          {project._count?.members || 0}
   322→                        </span>
   323→                        <Button
   324→                          size="sm"
   325→                          variant="outline"
   326→                          className="h-7 gap-1.5 text-xs"
   327→                          onClick={(e) => { e.stopPropagation(); openOnboardingDialog(project.id, project.name) }}
   328→                        >
   329→                          <ClipboardCheck className="w-3.5 h-3.5" />
   330→                          Onboarding
   331→                        </Button>
   332→                        <span className="text-muted-foreground cursor-pointer">
   333→                          {expandedProjectId === project.id ? '▲' : '▼'}
   334→                        </span>
   335→                      </div>
   336→                    </div>
   337→
   338→                    {/* Progress bar */}
   339→                    <div className="mt-3 space-y-1">
   340→                      <div className="flex items-center justify-between text-xs text-muted-foreground">
   341→                        <span>{doneTasks}/{totalT} tasks completed</span>
   342→                        <span>{pct}%</span>
   343→                      </div>
   344→                      <Progress value={pct} className="h-2" />
   345→                    </div>
   346→                  </CardContent>
   347→                </Card>
   348→
   349→                {/* Expanded tasks table */}
   350→                <AnimatePresence>
   351→                  {expandedProjectId === project.id && (
   352→                    <motion.div
   353→                      initial={{ opacity: 0, height: 0 }}
   354→                      animate={{ opacity: 1, height: 'auto' }}
   355→                      exit={{ opacity: 0, height: 0 }}
   356→                      transition={{ duration: 0.2 }}
   357→                      className="overflow-hidden"
   358→                    >
   359→                      <Card className="mt-2 rounded-t-none">
   360→                        <CardContent className="p-0">
   361→                          {projectTasks.length === 0 ? (
   362→                            <div className="py-8 text-center text-sm text-muted-foreground">
   363→                              No tasks in this project yet.
   364→                            </div>
   365→                          ) : (
   366→                            <div className="overflow-x-auto">
   367→                              <table className="w-full text-sm">
   368→                                <thead>
   369→                                  <tr className="border-b bg-muted/50">
   370→                                    <th className="text-left p-3 font-medium text-muted-foreground">Task</th>
   371→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden md:table-cell">Status</th>
   372→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden sm:table-cell">Priority</th>
   373→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Due Date</th>
   374→                                    <th className="text-left p-3 font-medium text-muted-foreground hidden lg:table-cell">Assignees</th>
   375→                                  </tr>
   376→                                </thead>
   377→                                <tbody>
   378→                                  {projectTasks.map(task => (
   379→                                    <tr
   380→                                      key={task.id}
   381→                                      className="border-b last:border-0 hover:bg-muted/30 cursor-pointer transition-colors"
   382→                                      onClick={(e) => { e.stopPropagation(); openTaskDetail(task.id) }}
   383→                                    >
   384→                                      <td className="p-3">
   385→                                        <div className="font-medium">{task.title}</div>
   386→                                        {task.description && (
   387→                                          <div className="text-xs text-muted-foreground line-clamp-1 mt-0.5">{task.description}</div>
   388→                                        )}
   389→                                      </td>
   390→                                      <td className="p-3 hidden md:table-cell"><StatusBadge status={task.status} /></td>
   391→                                      <td className="p-3 hidden sm:table-cell"><PriorityBadge priority={task.priority} /></td>
   392→                                      <td className="p-3 hidden lg:table-cell text-muted-foreground text-xs">
   393→                                        {task.dueDate ? formatDate(task.dueDate) : '—'}
   394→                                      </td>
   395→                                      <td className="p-3 hidden lg:table-cell">
   396→                                        <div className="flex -space-x-2">
   397→                                          {task.assignees?.slice(0, 3).map(a => (
   398→                                            <Avatar key={a.id} className="w-6 h-6 border-2 border-background">
   399→                                              <AvatarFallback className="text-[10px] bg-muted">
   400→                                                {getInitials(a.user?.name || '?')}
   401→                                              </AvatarFallback>
   402→                                            </Avatar>
   403→                                          ))}
   404→                                          {(task.assignees?.length || 0) > 3 && (
   405→                                            <span className="w-6 h-6 rounded-full bg-muted flex items-center justify-center text-[10px] border-2 border-background">
   406→                                              +{(task.assignees?.length || 0) - 3}
   407→                                            </span>
   408→                                          )}
   409→                                        </div>
   410→                                      </td>
   411→                                    </tr>
   412→                                  ))}
   413→                                </tbody>
   414→                              </table>
   415→                            </div>
   416→                          )}
   417→                        </CardContent>
   418→                      </Card>
   419→                    </motion.div>
   420→                  )}
   421→                </AnimatePresence>
   422→              </motion.div>
   423→            )
   424→          })}
   425→        </div>
   426→      )}
   427→
   428→      {/* Create Onboarding Dialog */}
   429→      <Dialog open={onboardingDialogOpen} onOpenChange={setOnboardingDialogOpen}>
   430→        <DialogContent className="sm:max-w-lg">
   431→          <DialogHeader>
   432→            <DialogTitle className="flex items-center gap-2">
   433→              <ClipboardCheck className="w-5 h-5 text-primary" />
   434→              Create Onboarding
   435→            </DialogTitle>
   436→            <DialogDescription>Create an onboarding form for this project</DialogDescription>
   437→          </DialogHeader>
   438→          <div className="space-y-4 py-2">
   439→            <div className="space-y-2">
   440→              <Label>Title</Label>
   441→              <Input
   442→                placeholder="Onboarding form title"
   443→                value={onboardingTitle}
   444→                onChange={(e) => setOnboardingTitle(e.target.value)}
   445→              />
   446→            </div>
   447→            <div className="space-y-2">
   448→              <Label>Description (optional)</Label>
   449→              <Textarea
   450→                placeholder="Brief description of the onboarding..."
   451→                value={onboardingDesc}
   452→                onChange={(e) => setOnboardingDesc(e.target.value)}
   453→                rows={2}
   454→              />
   455→            </div>
   456→            <div className="space-y-2">
   457→              <div className="flex items-center justify-between">
   458→                <Label>Steps</Label>
   459→                <span className="text-xs text-muted-foreground">{onboardingSteps.length} / {ONBOARDING_STEPS.length} selected</span>
   460→              </div>
   461→              <div className="space-y-2 max-h-48 overflow-y-auto">
   462→                {ONBOARDING_STEPS.map((step) => {
   463→                  const isSelected = onboardingSteps.includes(step.id)
   464→                  return (
   465→                    <button
   466→                      key={step.id}
   467→                      type="button"
   468→                      onClick={() => toggleOnboardingStep(step.id)}
   469→                      className={cn(
   470→                        'w-full flex items-start gap-3 p-3 rounded-lg border text-left transition-colors',
   471→                        isSelected ? 'border-primary bg-primary/5' : 'border-border hover:bg-muted/50'
   472→                      )}
   473→                    >
   474→                      <div className={cn(
   475→                        'mt-0.5 h-5 w-5 rounded border-2 flex items-center justify-center shrink-0 transition-colors',
   476→                        isSelected ? 'border-primary bg-primary text-primary-foreground' : 'border-muted-foreground/30'
   477→                      )}>
   478→                        {isSelected && <Check className="w-3 h-3" />}
   479→                      </div>
   480→                      <div className="flex-1 min-w-0">
   481→                        <p className="text-sm font-medium">{step.label}</p>
   482→                        <p className="text-xs text-muted-foreground mt-0.5">{step.description}</p>
   483→                      </div>
   484→                    </button>
   485→                  )
   486→                })}
   487→              </div>
   488→            </div>
   489→          </div>
   490→          <DialogFooter>
   491→            <Button variant="outline" onClick={() => setOnboardingDialogOpen(false)}>Cancel</Button>
   492→            <Button
   493→              onClick={handleCreateOnboarding}
   494→              disabled={creatingOnboarding || !onboardingTitle.trim() || onboardingSteps.length === 0}
   495→              className="gap-2"
   496→            >

   722→
   723→  const [currentPassword, setCurrentPassword] = useState('')

   726→
   727→  useEffect(() => {
   728→    if (!user?.id) return
   729→    const fetchProfile = async () => {
   730→      try {
   731→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   732→        const data = await res.json()
   733→        if (data.user) {
   734→          setName(data.user.name || '')
   735→          setEmail(data.user.email || '')
   736→          setPhone(data.user.phone || '')
   737→          setCompany(data.user.company || '')

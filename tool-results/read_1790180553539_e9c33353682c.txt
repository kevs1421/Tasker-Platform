   132→   132→   132→
   133→   133→   133→export function VideoCallPanel({
   134→   134→   134→  currentUser,
   135→   135→   135→  targetUser,
   136→   136→   136→  isInCall,
   137→   137→   137→  callType,
   138→   138→   138→  callStatus,
   139→   139→   139→  onEnd,
   140→   140→   140→  onToggleMic,
   141→   141→   141→  onToggleCamera,
   142→   142→   142→  onToggleScreen,
   143→   143→   143→  micOn,
   144→   144→   144→  cameraOn,
   145→   145→   145→  screenOn,
   146→   146→   146→  remoteScreenOn,
   147→   147→   147→  remoteMicOn,
   148→   148→   148→  remoteCameraOn,
   149→   149→   149→  screenShareLabel,
   150→   150→   150→}: VideoCallProps) {
   151→   151→   151→  const [isMinimized, setIsMinimized] = useState(false)
   152→   152→   152→  const [isFullscreen, setIsFullscreen] = useState(false)
   153→   153→   153→  const [callDuration, setCallDuration] = useState(0)
   154→   154→   154→  const [meetingLink, setMeetingLink] = useState('')
   155→   155→   155→  const [remoteVideoPresent, setRemoteVideoPresent] = useState(false)
   156→   156→   156→  const [remoteScreenPresent, setRemoteScreenPresent] = useState(false)
   157→   157→   157→  const timerRef = useRef<ReturnType<typeof setInterval>>
   158→   158→   158→  const panelRef = useRef<HTMLDivElement>(null)
   159→   159→   159→
   160→   160→   160→  // Timer for connected calls
   161→   161→   161→  useEffect(() => {
   162→   162→   162→    if (callStatus === 'connected') {
   163→   163→   163→      timerRef.current = setInterval(() => setCallDuration((p) => p + 1), 1000)
   164→   164→   164→    } else {
   165→   165→   165→      if (timerRef.current) clearInterval(timerRef.current)
   166→   166→   166→      setCallDuration(0)
   167→   167→   167→    }
   168→   168→   168→    return () => { if (timerRef.current) clearInterval(timerRef.current) }
   169→   169→   169→  }, [callStatus])
   170→   170→   170→
   171→   171→   171→  // Generate meeting link
   172→   172→   172→  useEffect(() => {
   173→   173→   173→    if (isInCall) {
   174→   174→   174→      const id = Math.random().toString(36).substring(2, 10)
   175→   175→   175→      setMeetingLink(`${window.location.origin}/meet/${id}`)
   176→   176→   176→    }
   177→   177→   177→  }, [isInCall])
   178→   178→   178→
   179→   179→   179→  // Monitor remote video tracks
   180→   180→   180→  useEffect(() => {
   181→   181→   181→    if (callStatus !== 'connected') {
   182→   182→   182→      setRemoteVideoPresent(false)
   183→   183→   183→      setRemoteScreenPresent(false)
   184→   184→   184→      return
   185→   185→   185→    }
   186→   186→   186→    const checkTracks = setInterval(() => {
   187→   187→   187→      const remoteVideo = document.querySelector('#remote-video') as HTMLVideoElement
   188→   188→   188→      const hasVideo = remoteVideo?.srcObject instanceof MediaStream &&
   189→   189→   189→        (remoteVideo.srcObject as MediaStream).getVideoTracks().some((t) => t.enabled && t.readyState === 'live')
   190→   190→   190→      setRemoteVideoPresent(!!hasVideo)
   191→   191→   191→
   192→   192→   192→      const remoteScreenVideo = document.querySelector('#remote-screen-video') as HTMLVideoElement
   193→   193→   193→      const hasScreen = remoteScreenVideo?.srcObject instanceof MediaStream &&
   194→   194→   194→        (remoteScreenVideo.srcObject as MediaStream).getVideoTracks().length > 0
   195→   195→   195→      setRemoteScreenPresent(!!hasScreen)
   196→   196→   196→    }, 1000)
   197→   197→   197→    return () => clearInterval(checkTracks)
   198→   198→   198→  }, [callStatus])
   199→   199→   199→
   200→   200→   200→  const copyLink = useCallback(() => {
   201→   201→   201→    if (meetingLink) {
   202→   202→   202→      navigator.clipboard.writeText(meetingLink)
   203→   203→   203→      toast.success('Meeting link copied!')
   204→   204→   204→    }
   205→   205→   205→  }, [meetingLink])
   206→   206→   206→
   207→   207→   207→  const formatTime = (seconds: number) => {
   208→   208→   208→    const h = Math.floor(seconds / 3600)
   209→   209→   209→    const m = Math.floor((seconds % 3600) / 60)
   210→   210→   210→    const s = seconds % 60
   211→   211→   211→    return h > 0
   212→   212→   212→      ? `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   213→   213→   213→      : `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   214→   214→   214→  }
   215→   215→   215→
   216→   216→   216→  const toggleFullscreen = useCallback(() => {
   217→   217→   217→    if (!panelRef.current) return
   218→   218→   218→    if (!document.fullscreenElement) {
   219→   219→   219→      panelRef.current.requestFullscreen().then(() => setIsFullscreen(true)).catch(() => {})
   220→   220→   220→    } else {
   221→   221→   221→      document.exitFullscreen().then(() => setIsFullscreen(false)).catch(() => {})
   222→   222→   222→    }
   223→   223→   223→  }, [])
   224→   224→   224→
   225→   225→   225→  if (!isInCall) return null
   226→   226→   226→
   227→   227→   227→  const isConnected = callStatus === 'connected'
   228→   228→   228→  const isRinging = callStatus === 'ringing'
   229→   229→   229→  const isConnecting = callStatus === 'connecting'
   230→   230→   230→  const isVideoCall = callType === 'video'
   231→   231→   231→  const isGroupCall = targetUser?.role === 'team' && targetUser?.id?.startsWith('group-')
   232→   232→   232→
   233→   233→   233→  // Whether the main view should show screen content
   234→   234→   234→  const showLocalScreen = isConnected && screenOn
   235→   235→   235→  const showRemoteScreen = isConnected && remoteScreenOn && remoteScreenPresent
   236→   236→   236→  const showScreenContent = showLocalScreen || showRemoteScreen
   237→   237→   237→
   238→   238→   238→  // ── Minimized view ──
   239→   239→   239→  if (isMinimized) {
   240→   240→   240→    return (
   241→   241→   241→      <AnimatePresence>
   242→   242→   242→        <motion.div
   243→   243→   243→          initial={{ opacity: 0, y: 40, scale: 0.9 }}
   244→   244→   244→          animate={{ opacity: 1, y: 0, scale: 1 }}
   245→   245→   245→          exit={{ opacity: 0, y: 40, scale: 0.9 }}
   246→   246→   246→          className="fixed bottom-6 right-6 z-50"
   247→   247→   247→        >
   248→   248→   248→          <div className="flex items-center gap-2 bg-card border rounded-2xl shadow-2xl px-3 py-2.5 pr-1.5">
   249→   249→   249→            {targetUser && (
   250→   250→   250→              <div className="flex items-center gap-2 min-w-0">
   251→   251→   251→                <div className={cn(
   252→   252→   252→                  'h-2.5 w-2.5 rounded-full flex-shrink-0',
   253→   253→   253→                  isConnected ? 'bg-emerald-500 animate-pulse' : 'bg-amber-500 animate-pulse'
   254→   254→   254→                )} />
   255→   255→   255→                {showScreenContent && <MonitorUp className="h-3.5 w-3.5 text-emerald-500 flex-shrink-0" />}
   256→   256→   256→                <span className="text-sm font-medium truncate max-w-[120px]">
   257→   257→   257→                  {targetUser.name}
   258→   258→   258→                </span>
   259→   259→   259→                <span className="text-xs text-muted-foreground font-mono">
   260→   260→   260→                  {isConnected ? formatTime(callDuration) : isRinging ? 'Ringing...' : 'Connecting...'}
   261→   261→   261→                </span>
   262→   262→   262→              </div>
   263→   263→   263→            )}
   264→   264→   264→            <Button variant="ghost" size="icon" className="h-8 w-8 flex-shrink-0" onClick={() => setIsMinimized(false)}>
   265→   265→   265→              <Maximize2 className="h-4 w-4" />
   266→   266→   266→            </Button>
   267→   267→   267→            <Button
   268→   268→   268→              variant="ghost" size="icon"
   269→   269→   269→              className="h-8 w-8 text-red-500 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-950/30 flex-shrink-0"
   270→   270→   270→              onClick={onEnd}
   271→   271→   271→            >
   272→   272→   272→              <PhoneOff className="h-4 w-4" />
   273→   273→   273→            </Button>
   274→   274→   274→          </div>
   275→   275→   275→        </motion.div>
   276→   276→   276→      </AnimatePresence>
   277→   277→   277→    )
   278→   278→   278→  }
   279→   279→   279→
   280→   280→   280→  // ── Full call panel ──
   281→   281→   281→  return (
   282→   282→   282→    <AnimatePresence>
   283→   283→   283→      <motion.div
   284→   284→   284→        ref={panelRef}
   285→   285→   285→        initial={{ opacity: 0, scale: 0.95, y: 20 }}
   286→   286→   286→        animate={{ opacity: 1, scale: 1, y: 0 }}
   287→   287→   287→        exit={{ opacity: 0, scale: 0.95, y: 20 }}
   288→   288→   288→        className={cn(
   289→   289→   289→          'fixed z-50 shadow-2xl border-2 border-slate-700 overflow-hidden',
   290→   290→   290→          showScreenContent
   291→   291→   291→            ? 'bottom-4 right-4 left-4 top-4'
   292→   292→   292→            : 'bottom-6 right-6',
   293→   293→   293→          showScreenContent
   294→   294→   294→            ? 'w-auto h-auto'
   295→   295→   295→            : isVideoCall ? 'w-[560px] h-[420px]' : 'w-[380px] h-[280px]',
   296→   296→   296→          'rounded-2xl bg-slate-950 text-white',
   297→   297→   297→          isFullscreen && 'fixed inset-0 !rounded-none !border-0'
   298→   298→   298→        )}
   299→   299→   299→      >
   300→   300→   300→        {/* ── Screen Share Banner ── */}
   301→   301→   301→        <AnimatePresence>
   302→   302→   302→          {showLocalScreen && (
   303→   303→   303→            <ScreenShareBanner
   304→   304→   304→              isLocal
   305→   305→   305→              label={screenShareLabel}
   306→   306→   306→              onStop={onToggleScreen}
   307→   307→   307→            />
   308→   308→   308→          )}
   309→   309→   309→          {showRemoteScreen && !showLocalScreen && (
   310→   310→   310→            <ScreenShareBanner
   311→   311→   311→              isLocal={false}
   312→   312→   312→              label={targetUser?.name ?? 'Participant'}
   313→   313→   313→              onStop={() => {}}
   314→   314→   314→            />
   315→   315→   315→          )}
   316→   316→   316→        </AnimatePresence>
   317→   317→   317→
   318→   318→   318→        {/* ── Call Header ── */}
   319→   319→   319→        <div className="flex items-center justify-between px-4 py-2.5 bg-slate-900/80 border-b border-white/10">
   320→   320→   320→          <div className="flex items-center gap-2.5 min-w-0">
   321→   321→   321→            <div className={cn(
   322→   322→   322→              'h-2.5 w-2.5 rounded-full flex-shrink-0',
   323→   323→   323→              isConnected ? 'bg-emerald-400' : 'bg-amber-400 animate-pulse'
   324→   324→   324→            )} />
   325→   325→   325→            <span className="text-sm font-medium text-white/90 truncate">
   326→   326→   326→              {targetUser ? targetUser.name : 'Meeting'}
   327→   327→   327→            </span>
   328→   328→   328→            {isGroupCall && <Users className="h-3.5 w-3.5 text-white/50" />}
   329→   329→   329→            <span className="text-xs text-white/50 font-mono">
   330→   330→   330→              {isConnected ? formatTime(callDuration) : isRinging ? 'Ringing...' : 'Connecting...'}
   331→   331→   331→            </span>
   332→   332→   332→          </div>
   333→   333→   333→          <div className="flex items-center gap-0.5">
   334→   334→   334→            <Button
   335→   335→   335→              variant="ghost" size="icon"
   336→   336→   336→              className="h-7 w-7 text-white/60 hover:text-white hover:bg-white/10"
   337→   337→   337→              onClick={copyLink}
   338→   338→   338→              title="Copy meeting link"
   339→   339→   339→            >
   340→   340→   340→              <Copy className="h-3.5 w-3.5" />
   341→   341→   341→            </Button>
   342→   342→   342→            <Button
   343→   343→   343→              variant="ghost" size="icon"
   344→   344→   344→              className="h-7 w-7 text-white/60 hover:text-white hover:bg-white/10"
   345→   345→   345→              onClick={toggleFullscreen}
   346→   346→   346→              title={isFullscreen ? 'Exit fullscreen' : 'Fullscreen'}
   347→   347→   347→            >
   348→   348→   348→              <Maximize2 className="h-3.5 w-3.5" />
   349→   349→   349→            </Button>
   350→   350→   350→            <Button
   351→   351→   351→              variant="ghost" size="icon"
   352→   352→   352→              className="h-7 w-7 text-white/60 hover:text-white hover:bg-white/10"
   353→   353→   353→              onClick={() => setIsMinimized(true)}
   354→   354→   354→            >
   355→   355→   355→              <Minimize2 className="h-3.5 w-3.5" />
   356→   356→   356→            </Button>
   357→   357→   357→          </div>
   358→   358→   358→        </div>
   359→   359→   359→
   360→   360→   360→        {/* ── Video Area ── */}
   361→   361→   361→        <div className="relative" style={{ height: 'calc(100% - 110px - 40px)' }}>
   362→   362→   362→          {/* ── SCREEN SHARE MAIN VIEW ── */}
   363→   363→   363→          {showScreenContent ? (
   364→   364→   364→            <div className="absolute inset-0 bg-slate-900">
   365→   365→   365→              {/* Local screen share (presenting) */}
   366→   366→   366→              {showLocalScreen && (
   367→   367→   367→                <div className="absolute inset-0 flex flex-col">
   368→   368→   368→                  <div className="flex-1 flex items-center justify-center p-4">
   369→   369→   369→                    <div className="relative w-full h-full bg-slate-800 rounded-lg overflow-hidden shadow-2xl border border-white/10">
   370→   370→   370→                      <video
   371→   371→   371→                        id="local-screen-video"
   372→   372→   372→                        autoPlay
   373→   373→   373→                        playsInline
   374→   374→   374→                        muted
   375→   375→   375→                        className="w-full h-full object-contain bg-black"
   376→   376→   376→                      />
   377→   377→   377→                      {/* Stop button overlay */}
   378→   378→   378→                      <div className="absolute bottom-4 left-1/2 -translate-x-1/2">
   379→   379→   379→                        <button
   380→   380→   380→                          onClick={onToggleScreen}
   381→   381→   381→                          className="flex items-center gap-2 px-4 py-2 bg-red-500 hover:bg-red-600 text-white rounded-full text-sm font-medium transition-colors shadow-lg"
   382→   382→   382→                        >
   383→   383→   383→                          <MonitorOff className="h-4 w-4" />
   384→   384→   384→                          Stop sharing

   372→   372→function TaskDetailDialog({
   373→   373→  task, userId, onUpdate
   374→   374→}: {
   375→   375→  task: Task & { _count: { comments: number; subtasks: number; completedSubtasks: number }; comments?: Comment[]; subtasks?: Subtask[] }
   376→   376→  userId: string
   377→   377→  onUpdate: () => void
   378→   378→}) {
   379→   379→  const [comments, setComments] = useState<Comment[]>(task.comments || [])
   380→   380→  const [newComment, setNewComment] = useState('')
   381→   381→  const [sendingComment, setSendingComment] = useState(false)
   382→   382→  const scrollRef = useRef<HTMLDivElement>(null)
   383→   383→
   384→   384→  useEffect(() => {
   385→   385→    if (scrollRef.current) {
   386→   386→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   387→   387→    }
   388→   388→  }, [comments])
   389→   389→
   390→   390→  const sendComment = async () => {
   391→   391→    if (!newComment.trim()) return
   392→   392→    setSendingComment(true)
   393→   393→    try {
   394→   394→      const res = await fetch('/api/comments', {
   395→   395→        method: 'POST',
   396→   396→        headers: { 'Content-Type': 'application/json' },
   397→   397→        body: JSON.stringify({ content: newComment.trim(), taskId: task.id, userId })
   398→   398→      })
   399→   399→      if (!res.ok) throw new Error()
   400→   400→      const data = await res.json()
   401→   401→      setComments(prev => [...prev, data.comment])
   402→   402→      setNewComment('')
   403→   403→      toast.success('Comment added')
   404→   404→    } catch {
   405→   405→      toast.error('Failed to add comment')
   406→   406→    } finally {
   407→   407→      setSendingComment(false)
   408→   408→    }
   409→   409→  }
   410→   410→
   411→   411→  const statusCfg = getStatusConfig(task.status)
   412→   412→  const priorityCfg = getPriorityConfig(task.priority)
   413→   413→
   414→   414→  return (
   415→   415→    <>
   416→   416→      <DialogHeader>
   417→   417→        <div className="flex items-center gap-2 flex-wrap">
   418→   418→          <DialogTitle className="text-xl">{task.title}</DialogTitle>
   419→   419→          <StatusBadge status={task.status} />
   420→   420→          <PriorityBadge priority={task.priority} />
   421→   421→        </div>
   422→   422→        {task.project && (

   560→   560→export function ClientProfile() {
   561→   561→  const { user, login } = useAuthStore()
   562→   562→  const [loading, setLoading] = useState(true)
   563→   563→  const [saving, setSaving] = useState(false)
   564→   564→  const [savingPassword, setSavingPassword] = useState(false)
   565→   565→
   566→   566→  const [name, setName] = useState('')
   567→   567→  const [email, setEmail] = useState('')
   568→   568→  const [phone, setPhone] = useState('')
   569→   569→  const [company, setCompany] = useState('')
   570→   570→
   571→   571→  const [currentPassword, setCurrentPassword] = useState('')

   573→   573→  const [confirmPassword, setConfirmPassword] = useState('')
   574→   574→
   575→   575→  useEffect(() => {
   576→   576→    if (!user?.id) return
   577→   577→    const fetchProfile = async () => {
   578→   578→      try {

   580→   580→        const data = await res.json()
   581→   581→        if (data.user) {
   582→   582→          setName(data.user.name || '')
   583→   583→          setEmail(data.user.email || '')
   584→   584→          setPhone(data.user.phone || '')

   588→   588→        toast.error('Failed to load profile')

   595→   595→
   596→   596→  const saveProfile = async () => {
   597→   597→    if (!user?.id) return
   598→   598→    setSaving(true)
   599→   599→    try {
   600→   600→      const res = await fetch('/api/user/profile', {
   601→   601→        method: 'PATCH',
   602→   602→        headers: { 'Content-Type': 'application/json' },

   606→   606→        const data = await res.json()
   607→   607→        throw new Error(data.error || 'Failed')
   608→   608→      }
   609→   609→      const data = await res.json()

   614→   614→      }
   615→   615→    } catch (e: unknown) {
   616→   616→      const msg = e instanceof Error ? e.message : 'Failed to update profile'
   617→   617→      toast.error(msg)
   618→   618→    } finally {
   619→   619→      setSaving(false)
   620→   620→    }
   621→   621→  }
   622→   622→
   623→   623→  const changePassword = async () => {
   624→   624→    if (!user?.id) return
   625→   625→    if (!currentPassword || !newPassword || !confirmPassword) {
   626→   626→      toast.error('Please fill in all password fields')
   627→   627→      return
   628→   628→    }
   629→   629→    if (newPassword !== confirmPassword) {
   630→   630→      toast.error('New passwords do not match')
   631→   631→      return
   632→   632→    }

   641→   641→        headers: { 'Content-Type': 'application/json' },
   642→   642→        body: JSON.stringify({
   643→   643→          userId: user.id,
   644→   644→          currentPassword,
   645→   645→          newPassword
   646→   646→        })
   647→   647→      })
   648→   648→      if (!res.ok) {
   649→   649→        const data = await res.json()
   650→   650→        throw new Error(data.error || 'Failed')
   651→   651→      }
   652→   652→      toast.success('Password changed successfully')

   655→   655→      setConfirmPassword('')
   656→   656→    } catch (e: unknown) {
   657→   657→      const msg = e instanceof Error ? e.message : 'Failed to change password'
   658→   658→      toast.error(msg)
   659→   659→    } finally {
   660→   660→      setSavingPassword(false)
   661→   661→    }
   662→   662→  }

   664→   664→  if (loading) return <PageSkeleton />

   690→   690→                Member since {formatDate(user.createdAt)}
   691→   691→              </p>
   692→   692→            )}

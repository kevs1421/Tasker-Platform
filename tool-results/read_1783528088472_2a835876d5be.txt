   431→interface TimelineViewProps {
   432→  tasks: Task[]
   433→  projects: Project[]
   434→  onTaskClick: (task: Task) => void
   435→  onCreateTask: () => void
   436→}
   437→
   438→function TimelineView({ tasks, projects, onTaskClick, onCreateTask }: TimelineViewProps) {
   439→  const grouped = useMemo(() => {
   440→    const map = new Map<string, { project: Project; tasks: Task[] }>()
   441→    for (const project of projects) {
   442→      map.set(project.id, { project, tasks: [] })
   443→    }
   444→    for (const task of tasks) {
   445→      const group = map.get(task.projectId)
   446→      if (group) group.tasks.push(task)
   447→    }
   448→    return Array.from(map.values()).filter((g) => g.tasks.length > 0)
   449→  }, [tasks, projects])
   450→
   451→  const overallCompleted = tasks.filter((t) => t.status === 'done').length
   452→  const overallTotal = tasks.length
   453→  const overallPercent = overallTotal > 0 ? Math.round((overallCompleted / overallTotal) * 100) : 0
   454→
   455→  return (
   456→    <motion.div {...viewTransition} className="space-y-6">
   457→      {/* Overall Summary */}
   458→      <Card>
   459→        <CardContent className="p-4">
   460→          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   461→            <div className="flex items-center gap-3">
   462→              <div className="w-9 h-9 rounded-lg bg-primary/10 flex items-center justify-center">
   463→                <FolderOpen className="w-4 h-4 text-primary" />
   464→              </div>
   465→              <div>
   466→                <h3 className="text-sm font-semibold">Overall Progress</h3>
   467→                <p className="text-xs text-muted-foreground">
   468→                  {overallCompleted} of {overallTotal} tasks completed
   469→                </p>
   470→              </div>
   471→            </div>
   472→            <div className="flex items-center gap-3 flex-1 max-w-md">
   473→              <Progress value={overallPercent} className="h-2.5 flex-1" />
   474→              <span className="text-sm font-semibold w-10 text-right">{overallPercent}%</span>
   475→            </div>
   476→          </div>
   477→        </CardContent>
   478→      </Card>
   479→
   480→      {/* Per-Project Groups */}
   481→      <div className="space-y-4">
   482→        {grouped.map(({ project, tasks: projectTasks }) => {
   483→          const completed = projectTasks.filter((t) => t.status === 'done').length
   484→          const total = projectTasks.length
   485→          const percent = total > 0 ? Math.round((completed / total) * 100) : 0
   486→          const inProgress = projectTasks.filter((t) => t.status === 'in_progress').length
   487→          const inReview = projectTasks.filter((t) => t.status === 'in_review').length
   488→          const todo = projectTasks.filter((t) => t.status === 'todo').length
   489→
   490→          return (
   491→            <Card key={project.id}>
   492→              <CardHeader className="pb-3">
   493→                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   494→                  <div className="flex items-center gap-3">
   495→                    <div
   496→                      className="w-8 h-8 rounded-lg flex items-center justify-center"
   497→                      style={{ backgroundColor: `${project.color}20` }}
   498→                    >
   499→                      <FolderOpen className="w-4 h-4" style={{ color: project.color }} />
   500→                    </div>
   501→                    <div>
   502→                      <h4 className="text-sm font-semibold">{project.name}</h4>
   503→                      <p className="text-xs text-muted-foreground">
   504→                        {project._count?.tasks ?? total} tasks
   505→                      </p>
   506→                    </div>
   507→                  </div>
   508→                  <div className="flex items-center gap-2">
   509→                    <Badge variant="outline" className="text-[10px] text-slate-500">
   510→                      {todo} To Do
   511→                    </Badge>
   512→                    <Badge variant="outline" className="text-[10px] text-amber-600">
   513→                      {inProgress} Active
   514→                    </Badge>
   515→                    <Badge variant="outline" className="text-[10px] text-purple-600">
   516→                      {inReview} Review
   517→                    </Badge>
   518→                    <Badge variant="outline" className="text-[10px] text-emerald-600">
   519→                      {completed} Done
   520→                    </Badge>
   521→                  </div>
   522→                </div>
   523→              </CardHeader>
   524→              <CardContent className="space-y-3">
   525→                {/* Progress Bar */}
   526→                <div className="flex items-center gap-3">
   527→                  <Progress value={percent} className="h-2 flex-1" />
   528→                  <span className="text-xs font-medium text-muted-foreground w-10 text-right">
   529→                    {percent}%
   530→                  </span>
   531→                </div>
   532→
   533→                {/* Task List */}
   534→                <div className="space-y-1">
   535→                  {projectTasks.map((task) => {
   536→                    const statusCfg = statusBadgeConfig[task.status]
   537→                    const isDone = task.status === 'done'
   538→                    return (
   539→                      <div
   540→                        key={task.id}
   541→                        className="flex items-center justify-between py-1.5 px-2 rounded-md cursor-pointer hover:bg-muted/60 transition-colors"
   542→                        onClick={() => onTaskClick(task)}
   543→                      >
   544→                        <div className="flex items-center gap-2.5 min-w-0">
   545→                          <div className={`w-1.5 h-1.5 rounded-full shrink-0 ${task.status === 'done' ? 'bg-emerald-500' : task.status === 'in_progress' ? 'bg-amber-500' : task.status === 'in_review' ? 'bg-purple-500' : 'bg-slate-400'}`} />
   546→                          <span className={`text-sm truncate ${isDone ? 'line-through text-muted-foreground' : ''}`}>
   547→                            {task.title}
   548→                          </span>
   549→                        </div>
   550→                        <Badge variant={statusCfg.variant} className={`text-[10px] px-2 py-0 shrink-0 ml-2 ${statusCfg.className}`}>
   551→                          {statusCfg.label}
   552→                        </Badge>
   553→                      </div>
   554→                    )
   555→                  })}
   556→                </div>
   557→              </CardContent>
   558→            </Card>
   559→          )
   560→        })}
   561→
   562→        {grouped.length === 0 && (
   563→          <Card>
   564→            <CardContent className="py-12 text-center">
   565→              <FolderOpen className="w-10 h-10 text-muted-foreground/40 mx-auto mb-3" />
   566→              <p className="text-sm text-muted-foreground">No tasks across projects yet</p>
   567→              <Button size="sm" className="mt-4 h-8 text-xs" onClick={onCreateTask}>
   568→                <Plus className="w-3.5 h-3.5 mr-1.5" />
   569→                Create Task
   570→              </Button>
   571→            </CardContent>
   572→          </Card>
   573→        )}
   574→      </div>
   575→    </motion.div>
   576→  )
   577→}

   579→// ─── Main TeamDashboard Component ────────────────────────────────────────────
   580→
   581→export default function TeamDashboard() {
   582→  const { user, teamView, selectedProjectId, setSelectedProjectId, setSelectedTaskId } = useAuthStore()
   583→
   584→  const [projects, setProjects] = useState<Project[]>([])
   585→  const [tasks, setTasks] = useState<Task[]>([])
   586→  const [loading, setLoading] = useState(true)
   587→  const [selectedTask, setSelectedTask] = useState<Task | null>(null)
   588→  const [taskDetailOpen, setTaskDetailOpen] = useState(false)
   589→  const [createModalOpen, setCreateModalOpen] = useState(false)
   590→  const [mobileColumn, setMobileColumn] = useState<TaskStatus>('todo')
   591→
   592→  // ── Data Fetching ──
   593→
   594→  const fetchProjects = useCallback(async () => {
   595→    try {
   596→      const res = await fetch('/api/projects')
   597→      if (!res.ok) throw new Error('Failed to fetch projects')
   598→      const data = await res.json()
   599→      let projectList: Project[] = data.projects || []
   600→
   601→      // If user is team member, filter to only their projects
   602→      if (user?.role === 'team') {
   603→        projectList = projectList.filter((p: Project) =>
   604→          p.members?.some((m) => m.userId === user.id)
   605→        )
   606→      }
   607→
   608→      setProjects(projectList)
   609→
   610→      // Auto-select first project if none selected
   611→      if (!selectedProjectId && projectList.length > 0) {
   612→        setSelectedProjectId(projectList[0].id)
   613→      }
   614→    } catch {
   615→      toast.error('Failed to load projects')
   616→    }
   617→  }, [user, selectedProjectId, setSelectedProjectId])
   618→
   619→  const fetchTasks = useCallback(async (projectId: string) => {
   620→    if (!projectId) return
   621→    setLoading(true)
   622→    try {
   623→      const res = await fetch(`/api/tasks?projectId=${projectId}`)
   624→      if (!res.ok) throw new Error('Failed to fetch tasks')
   625→      const data = await res.json()
   626→      setTasks(data.tasks || [])
   627→    } catch {
   628→      toast.error('Failed to load tasks')
   629→      setTasks([])
   630→    } finally {
   631→      setLoading(false)
   632→    }
   633→  }, [])
   634→
   635→  useEffect(() => {
   636→    fetchProjects()
   637→  }, [fetchProjects])
   638→
   639→  useEffect(() => {
   640→    if (selectedProjectId) {
   641→      fetchTasks(selectedProjectId)
   642→    }
   643→  }, [selectedProjectId, fetchTasks])
   644→
   645→  // ── Handlers ──
   646→
   647→  const handleTaskClick = useCallback((task: Task) => {
   648→    setSelectedTask(task)
   649→    setSelectedTaskId(task.id)
   650→    setTaskDetailOpen(true)
   651→  }, [setSelectedTaskId])
   652→
   653→  const handleTaskUpdate = useCallback(() => {
   654→    if (selectedProjectId) {
   655→      fetchTasks(selectedProjectId)
   656→    }
   657→    // Re-fetch the selected task for detail view
   658→    if (selectedTask?.id) {
   659→      fetch(`/api/tasks/${selectedTask.id}`)
   660→        .then((r) => r.json())
   661→        .then((d) => {
   662→          if (d.task) setSelectedTask(d.task)
   663→        })
   664→        .catch(() => {})
   665→    }
   666→  }, [selectedProjectId, selectedTask?.id])
   667→
   668→  const handleTaskCreated = useCallback(() => {
   669→    if (selectedProjectId) {
   670→      fetchTasks(selectedProjectId)
   671→    }
   672→  }, [selectedProjectId, fetchTasks])
   673→
   674→  const handleTaskDetailClose = useCallback(() => {
   675→    setTaskDetailOpen(false)
   676→    setSelectedTask(null)
   677→    setSelectedTaskId(null)
   678→  }, [setSelectedTaskId])
   679→
   680→  // ── Derived Data ──
   681→
   682→  const activeProject = useMemo(
   683→    () => projects.find((p) => p.id === selectedProjectId),
   684→    [projects, selectedProjectId]
   685→  )
   686→
   687→  const projectMembers = useMemo(() => {
   688→    if (!activeProject?.members) return []
   689→    return activeProject.members
   690→      .map((m) => m.user)
   691→      .filter((u): u is User => !!u)
   692→  }, [activeProject])
   693→
   694→  const tasksByStatus = useMemo(() => {
   695→    const map = new Map<TaskStatus, Task[]>()
   696→    for (const col of columns) {
   697→      map.set(col.key, tasks.filter((t) => t.status === col.key))
   698→    }
   699→    return map
   700→  }, [tasks])
   701→
   702→  // ── Render ──
   703→
   704→  if (projects.length === 0 && !loading) {
   705→    return (
   706→      <div className="flex flex-col items-center justify-center py-20 text-center">
   707→        <FolderOpen className="w-12 h-12 text-muted-foreground/40 mb-4" />
   708→        <h3 className="text-lg font-semibold mb-1">No Projects Yet</h3>
   709→        <p className="text-sm text-muted-foreground max-w-sm">
   710→          You don&apos;t have any projects assigned to you. Contact your admin to get started.
   711→        </p>
   712→      </div>
   713→    )
   714→  }
   715→
   716→  return (
   717→    <div className="space-y-4">
   718→      {/* ── Toolbar ── */}
   719→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   720→        <div className="flex items-center gap-3">
   721→          {/* Project Selector */}
   722→          <Select
   723→            value={selectedProjectId || ''}
   724→            onValueChange={(val) => setSelectedProjectId(val)}
   725→          >
   726→            <SelectTrigger className="w-[220px]">
   727→              <SelectValue placeholder="Select a project" />
   728→            </SelectTrigger>
   729→            <SelectContent>
   730→              {projects.map((project) => (
   731→                <SelectItem key={project.id} value={project.id}>
   732→                  <div className="flex items-center gap-2">
   733→                    <div
   734→                      className="w-2.5 h-2.5 rounded-full shrink-0"
   735→                      style={{ backgroundColor: project.color }}
   736→                    />
   737→                    {project.name}
   738→                  </div>
   739→                </SelectItem>
   740→              ))}
   741→            </SelectContent>
   742→          </Select>
   743→
   744→          {activeProject && (
   745→            <Badge variant="outline" className="text-[11px] hidden sm:flex">
   746→              <Users className="w-3 h-3 mr-1" />
   747→              {activeProject.members?.length ?? 0} members
   748→            </Badge>
   749→          )}
   750→        </div>
   751→
   752→        <Button
   753→          size="sm"
   754→          className="h-8 text-xs"
   755→          onClick={() => setCreateModalOpen(true)}
   756→          disabled={!selectedProjectId}
   757→        >
   758→          <Plus className="w-3.5 h-3.5 mr-1.5" />
   759→          Create Task
   760→        </Button>
   761→      </div>
   762→
   763→      {/* ── Mobile Column Tab Selector (Board view only) ── */}
   764→      {teamView === 'board' && (
   765→        <div className="md:hidden">
   766→          <Tabs
   767→            value={mobileColumn}
   768→            onValueChange={(val) => setMobileColumn(val as TaskStatus)}
   769→          >
   770→            <TabsList className="w-full grid grid-cols-4 h-9">
   771→              {columns.map((col) => (
   772→                <TabsTrigger
   773→                  key={col.key}
   774→                  value={col.key}
   775→                  className="text-xs gap-1.5 data-[state=active]:shadow-sm"
   776→                >
   777→                  <div className={`w-2 h-2 rounded-full ${col.dotColor}`} />
   778→                  <span className="hidden xs:inline">{col.label}</span>
   779→                  <span className="xs:hidden">{tasksByStatus.get(col.key)?.length ?? 0}</span>
   780→                </TabsTrigger>
   781→              ))}
   782→            </TabsList>
   783→          </Tabs>
   784→        </div>
   785→      )}
   786→
   787→      {/* ── View Content ── */}
   788→      <AnimatePresence mode="wait">
   789→        {teamView === 'board' && (
   790→          <motion.div
   791→            key="board-view"
   792→            {...viewTransition}
   793→          >
   794→            {loading ? (
   795→              <BoardSkeleton />
   796→            ) : (
   797→              <>
   798→                {/* Desktop: Full grid */}
   799→                <div className="hidden md:grid grid-cols-1 md:grid-cols-2 xl:grid-cols-4 gap-4">
   800→                  {columns.map((col) => (
   801→                    <BoardColumn
   802→                      key={col.key}
   803→                      config={col}
   804→                      tasks={tasksByStatus.get(col.key) || []}
   805→                      onTaskClick={handleTaskClick}
   806→                      onCreateTask={() => setCreateModalOpen(true)}
   807→                      showCreateButton={col.key === 'todo'}
   808→                    />
   809→                  ))}
   810→                </div>
   811→
   812→                {/* Mobile: Single column */}
   813→                <div className="md:hidden">
   814→                  {columns
   815→                    .filter((col) => col.key === mobileColumn)
   816→                    .map((col) => (
   817→                      <BoardColumn
   818→                        key={col.key}
   819→                        config={col}
   820→                        tasks={tasksByStatus.get(col.key) || []}
   821→                        onTaskClick={handleTaskClick}
   822→                        onCreateTask={() => setCreateModalOpen(true)}
   823→                        showCreateButton={col.key === 'todo'}
   824→                      />
   825→                    ))}
   826→                </div>
   827→              </>
   828→            )}
   829→          </motion.div>
   830→        )}
   831→
   832→        {teamView === 'list' && (
   833→          <motion.div key="list-view" {...viewTransition}>
   834→            {loading ? (
   835→              <ListSkeleton />
   836→            ) : (
   837→              <ListView
   838→                tasks={tasks}
   839→                projects={projects}
   840→                onTaskClick={handleTaskClick}
   841→                onCreateTask={() => setCreateModalOpen(true)}
   842→              />
   843→            )}
   844→          </motion.div>
   845→        )}
   846→
   847→        {teamView === 'timeline' && (
   848→          <motion.div key="timeline-view" {...viewTransition}>
   849→            {loading ? (
   850→              <TimelineSkeleton />
   851→            ) : (
   852→              <TimelineView
   853→                tasks={tasks}
   854→                projects={projects}
   855→                onTaskClick={handleTaskClick}
   856→                onCreateTask={() => setCreateModalOpen(true)}
   857→              />
   858→            )}
   859→          </motion.div>
   860→        )}
   861→      </AnimatePresence>
   862→
   863→      {/* ── Task Detail Dialog ── */}
   864→      <TaskDetail
   865→        task={selectedTask}
   866→        open={taskDetailOpen}
   867→        onClose={handleTaskDetailClose}
   868→        onUpdate={handleTaskUpdate}
   869→      />
   870→
   871→      {/* ── Create Task Modal ── */}
   872→      <CreateTaskModal
   873→        open={createModalOpen}
   874→        onClose={() => setCreateModalOpen(false)}
   875→        projectId={selectedProjectId || ''}
   876→        projectMembers={projectMembers}
   877→        onCreated={handleTaskCreated}
   878→      />
   879→    </div>
   880→  )

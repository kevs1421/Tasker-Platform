    66→function ActivityIcon({ action }: { action: string }) {
    67→  if (action.includes('created')) return <Plus className="h-4 w-4 text-emerald-500" />
    68→  if (action.includes('completed') || action.includes('done')) return <CheckCircle2 className="h-4 w-4 text-emerald-500" />
    69→  if (action.includes('moved') || action.includes('updated')) return <ArrowUpRight className="h-4 w-4 text-amber-500" />
    70→  if (action.includes('deleted') || action.includes('removed')) return <Trash2 className="h-4 w-4 text-rose-500" />
    71→  if (action.includes('comment')) return <MessageCircle className="h-4 w-4 text-slate-500" />
    72→  if (action.includes('onboarding')) return <FileCheck className="h-4 w-4 text-purple-500" />
    73→  return <CircleDot className="h-4 w-4 text-slate-400" />
    74→}

   363→  async function openForm(form: OnboardingForm) {
   364→    try {
   365→      const res = await fetch(`/api/onboarding/${form.id}`)
   366→      if (!res.ok) throw new Error('Failed')
   367→      const data = await res.json()
   368→      setSelectedForm(data.form)

   374→
   375→  async function handleCreateOnboarding() {
   376→    if (!user || !createForm.title.trim() || !createForm.projectId) return
   377→    setCreating(true)
   378→    try {
   379→      const stepsJson = JSON.stringify(
   380→        createForm.selectedSteps.map(stepId => {
   381→          const config = ONBOARDING_STEPS_CONFIG.find(s => s.id === stepId)
   382→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   383→        })
   384→      )
   385→      const res = await fetch('/api/onboarding', {
   386→        method: 'POST',
   387→        headers: { 'Content-Type': 'application/json' },
   388→        body: JSON.stringify({
   389→          title: createForm.title.trim(),
   390→          description: createForm.description.trim() || null,
   391→          stepsJson,
   392→          projectId: createForm.projectId,
   393→          clientId: createForm.clientId || null,
   394→          status: 'published',
   395→        }),
   396→      })
   397→      if (!res.ok) {
   398→        const d = await res.json()
   399→        throw new Error(d.error || 'Failed to create')
   400→      }
   401→      toast.success('Onboarding form created successfully')
   402→      setCreateOpen(false)
   403→      setCreateForm({ title: '', description: '', projectId: '', clientId: '', selectedSteps: ['business_info', 'brand_assets', 'project_kickoff', 'documents'] })
   404→      fetchForms()
   405→    } catch (e: unknown) {
   406→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding form')
   407→    } finally {
   408→      setCreating(false)
   409→    }
   410→  }

   412→  async function handleReview(submissionId: string, status: 'approved' | 'rejected') {
   413→    if (!user) return
   414→    setReviewing(submissionId)
   415→    try {
   416→      const notes = reviewNotes[submissionId] || ''
   417→      const patchRes = await fetch(`/api/onboarding/${submissionId}`, {
   418→        method: 'PATCH',
   419→        headers: { 'Content-Type': 'application/json' },
   420→        body: JSON.stringify({ action: 'review', reviewStatus: status, reviewNotes: notes, reviewedBy: user.id })
   421→      })
   422→      if (!patchRes.ok) {
   423→        const d = await patchRes.json()
   424→        throw new Error(d.error)
   425→      }
   426→      toast.success(`Submission ${status === 'approved' ? 'approved' : 'rejected'} successfully`)
   427→      // Refresh
   428→      if (selectedForm) {
   429→        const fresh = await fetch(`/api/onboarding/${selectedForm.id}`)
   430→        if (fresh.ok) {
   431→          const data = await fresh.json()
   432→          setSelectedForm(data.form)
   433→        }
   434→      }
   435→      fetchForms()
   436→    } catch (e: unknown) {
   437→      toast.error(e instanceof Error ? e.message : 'Failed to review')
   438→    } finally {
   439→      setReviewing(null)
   440→    }
   441→  }

   452→      toast.error('Failed to delete onboarding form')

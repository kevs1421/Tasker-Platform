    44→    44→function formatElapsed(ms: number): string {
    45→    45→  const totalSeconds = Math.floor(ms / 1000)
    46→    46→  const h = Math.floor(totalSeconds / 3600)
    47→    47→  const m = Math.floor((totalSeconds % 3600) / 60)
    48→    48→  const s = totalSeconds % 60
    49→    49→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    50→    50→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    51→    51→}
    52→    52→
    53→    53→function formatCountdown(ms: number): string {
    54→    54→  if (ms <= 0) return '0:00'
    55→    55→  const totalSeconds = Math.floor(ms / 1000)
    56→    56→  const m = Math.floor(totalSeconds / 60)
    57→    57→  const s = totalSeconds % 60
    58→    58→  return `${m}:${String(s).padStart(2, '0')}`
    59→    59→}

   741→   741→function OfflineBanner() {
   742→   742→  const [isOffline, setIsOffline] = useState(() => {
   743→   743→    if (typeof navigator === 'undefined') return false
   744→   744→    return !navigator.onLine
   745→   745→  })
   746→   746→
   747→   747→  useEffect(() => {
   748→   748→    const handleOnline = () => setIsOffline(false)
   749→   749→    const handleOffline = () => setIsOffline(true)
   750→   750→    window.addEventListener('online', handleOnline)
   751→   751→    window.addEventListener('offline', handleOffline)
   752→   752→    return () => {
   753→   753→      window.removeEventListener('online', handleOnline)
   754→   754→      window.removeEventListener('offline', handleOffline)
   755→   755→    }
   756→   756→  }, [])
   757→   757→
   758→   758→  if (!isOffline) return null
   759→   759→
   760→   760→  return (
   761→   761→    <div className="bg-amber-500 text-white text-center text-xs font-medium py-1.5 px-4 flex items-center justify-center gap-2">
   762→   762→      <WifiOff className="w-3.5 h-3.5 shrink-0" />
   763→   763→      <span>You are offline. Some features may be unavailable until connection is restored.</span>
   764→   764→    </div>
   765→   765→  )
   766→   766→}

   105→   105→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
   106→   106→  const visible = users.slice(0, max)
   107→   107→  const extra = users.length - max
   108→   108→  return (
   109→   109→    <div className="flex -space-x-2">
   110→   110→      {visible.map((u, i) =>
   111→   111→        u ? (
   112→   112→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   113→   113→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   114→   114→              {getInitials(u.name)}
   115→   115→            </AvatarFallback>
   116→   116→          </Avatar>
   117→   117→        ) : null
   118→   118→      )}
   119→   119→      {extra > 0 && (
   120→   120→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   121→   121→          +{extra}
   122→   122→        </div>
   123→   123→      )}
   124→   124→    </div>
   125→   125→  )

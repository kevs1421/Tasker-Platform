    97→function AvatarStack({ users, max = 3 }: { users: (User | undefined)[]; max?: number }) {
    98→  const visible = users.slice(0, max)
    99→  const extra = users.length - max
   100→  return (
   101→    <div className="flex -space-x-2">
   102→      {visible.map((u, i) =>
   103→        u ? (
   104→          <Avatar key={u.id} className="h-6 w-6 border-2 border-background ring-1 ring-border/50" style={{ zIndex: visible.length - i }}>
   105→            <AvatarFallback className="text-[10px] bg-slate-200 dark:bg-slate-700">
   106→              {getInitials(u.name)}
   107→            </AvatarFallback>
   108→          </Avatar>
   109→        ) : null
   110→      )}
   111→      {extra > 0 && (
   112→        <div className="h-6 w-6 rounded-full bg-slate-200 dark:bg-slate-700 border-2 border-background flex items-center justify-center text-[10px] font-medium text-slate-600 dark:text-slate-300" style={{ zIndex: 0 }}>
   113→          +{extra}
   114→        </div>
   115→      )}
   116→    </div>
   117→  )
   118→}

   120→function MemberAvatar({ user, size = 'md' }: { user: User | undefined; size?: 'sm' | 'md' }) {
   121→  const sizeClasses = size === 'sm' ? 'h-8 w-8 text-xs' : 'h-10 w-10 text-sm'
   122→  if (!user) return null
   123→  return (
   124→    <Avatar className={sizeClasses}>
   125→      <AvatarFallback className={cn('bg-slate-200 dark:bg-slate-700 font-medium')}>
   126→        {getInitials(user.name)}
   127→      </AvatarFallback>
   128→    </Avatar>
   129→  )
   130→}

   148→// ─── TeamProjects ───────────────────────────────────────────
   149→
   150→export function TeamProjects() {
   151→  const { user, setSelectedProjectId } = useAuthStore()
   152→  const isTraining = user?.role === 'training'
   153→  const [projects, setProjects] = useState<Project[]>([])
   154→  const [loading, setLoading] = useState(true)
   155→  const [expandedId, setExpandedId] = useState<string | null>(null)
   156→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   157→  const [expandedLoading, setExpandedLoading] = useState(false)
   158→  const [taskFilter, setTaskFilter] = useState<string>('all')
   159→  const [createTaskOpen, setCreateTaskOpen] = useState(false)
   160→  const [createTaskLoading, setCreateTaskLoading] = useState(false)
   161→  const [createTaskProjectId, setCreateTaskProjectId] = useState<string>('')
   162→  const [allTeamUsers, setAllTeamUsers] = useState<User[]>([])
   163→
   164→  // Create task form
   165→  const [taskTitle, setTaskTitle] = useState('')
   166→  const [taskDesc, setTaskDesc] = useState('')
   167→  const [taskStatus, setTaskStatus] = useState<'todo' | 'in_progress' | 'in_review' | 'done'>('todo')
   168→  const [taskPriority, setTaskPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium')
   169→  const [taskDueDate, setTaskDueDate] = useState<Date>()
   170→  const [taskTags, setTaskTags] = useState('')
   171→  const [taskAssignee, setTaskAssignee] = useState<string>('')
   172→  const [taskCalendarOpen, setTaskCalendarOpen] = useState(false)
   173→
   174→  const fetchProjects = useCallback(async () => {
   175→    try {
   176→      const res = await fetch('/api/projects')
   177→      const data = await res.json()
   178→      const filtered = (data.projects || []).filter((p: Project) =>
   179→        p.members?.some((m) => m.userId === user?.id)
   180→      )
   181→      setProjects(filtered)
   182→    } catch {
   183→      toast.error('Failed to load projects')
   184→    } finally {
   185→      setLoading(false)
   186→    }
   187→  }, [user])
   188→
   189→  useEffect(() => { fetchProjects() }, [fetchProjects])
   190→
   191→  // Fetch all active admin + team users for assignee picker
   192→  useEffect(() => {
   193→    fetch('/api/users')
   194→      .then(r => r.json())
   195→      .then(d => {
   196→        const members = (d.users || []).filter((u: User) => (u.role === 'team' || u.role === 'training' || u.role === 'admin') && u.status === 'active')
   197→        setAllTeamUsers(members)
   198→      })
   199→      .catch(() => {})
   200→  }, [])
   201→
   202→  const handleExpand = async (projectId: string) => {
   203→    if (expandedId === projectId) {
   204→      setExpandedId(null)
   205→      setExpandedTasks([])
   206→      return
   207→    }
   208→    setExpandedId(projectId)
   209→    setTaskFilter('all')
   210→    setExpandedLoading(true)
   211→    try {
   212→      const res = await fetch(`/api/projects/${projectId}`)
   213→      const data = await res.json()
   214→      setExpandedTasks(data.project?.tasks || [])
   215→    } catch {
   216→      toast.error('Failed to load tasks')
   217→    } finally {
   218→      setExpandedLoading(false)
   219→    }
   220→  }
   221→
   222→  const handleCreateTask = async () => {
   223→    if (!taskTitle.trim()) { toast.error('Task title is required'); return }
   224→    if (!user?.id) { toast.error('Not authenticated'); return }
   225→    setCreateTaskLoading(true)
   226→    try {
   227→      const res = await fetch('/api/tasks', {
   228→        method: 'POST',
   229→        headers: { 'Content-Type': 'application/json' },
   230→        body: JSON.stringify({
   231→          title: taskTitle, description: taskDesc || null, status: taskStatus,
   232→          priority: taskPriority, projectId: createTaskProjectId,
   233→          dueDate: taskDueDate ? taskDueDate.toISOString() : null,
   234→          tags: taskTags,
   235→          createdById: user.id,
   236→          assigneeIds: taskAssignee ? [taskAssignee] : [],
   237→        }),
   238→      })
   239→      if (!res.ok) {
   240→        const d = await res.json()
   241→        throw new Error(d.error || 'Failed to create task')
   242→      }
   243→      toast.success('Task created')
   244→      setCreateTaskOpen(false)
   245→      setTaskTitle(''); setTaskDesc(''); setTaskStatus('todo')
   246→      setTaskPriority('medium'); setTaskDueDate(undefined)
   247→      setTaskTags(''); setTaskAssignee('')
   248→      // Always refresh the expanded project view
   249→      handleExpand(createTaskProjectId)
   250→    } catch (e: unknown) {
   251→      toast.error(e instanceof Error ? e.message : 'Failed to create task')
   252→    } finally {
   253→      setCreateTaskLoading(false)
   254→    }
   255→  }
   256→
   257→  const openCreateTaskForProject = (projectId: string) => {
   258→    setCreateTaskProjectId(projectId)
   259→    setCreateTaskOpen(true)
   260→  }
   261→
   262→  const getProjectTaskStats = (projectId: string) => {
   263→    if (expandedId !== projectId) return null
   264→    const total = expandedTasks.length
   265→    const done = expandedTasks.filter((t) => t.status === 'done').length
   266→    return { total, done }
   267→  }
   268→
   269→  const filteredTasks = taskFilter === 'all'
   270→    ? expandedTasks
   271→    : expandedTasks.filter((t) => t.status === taskFilter)
   272→
   273→  if (loading) {
   274→    return (
   275→      <div>
   276→        <div className="flex items-center justify-between mb-6">
   277→          <Skeleton className="h-8 w-48" />
   278→          <Skeleton className="h-10 w-36" />
   279→        </div>
   280→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   281→          {Array.from({ length: 6 }).map((_, i) => <LoadingCard key={i} />)}
   282→        </div>
   283→      </div>
   284→    )
   285→  }
   286→
   287→  return (
   288→    <div className="space-y-6">
   289→      {/* Header */}
   290→      <div className="flex items-center justify-between">
   291→        <div>
   292→          <h1 className="text-2xl font-bold tracking-tight">Team Projects</h1>
   293→          <p className="text-sm text-muted-foreground mt-1">
   294→            {projects.length} project{projects.length !== 1 ? 's' : ''} you are assigned to
   295→          </p>
   296→        </div>
   297→      </div>
   298→
   299→      {/* Project Grid */}
   300→      {projects.length === 0 ? (
   301→        <EmptyState
   302→          icon={FolderKanban}
   303→          title="No projects yet"
   304→          description="You haven't been assigned to any projects yet. An admin will assign projects to you."
   305→        />
   306→      ) : (
   307→        <motion.div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   308→          <AnimatePresence>
   309→            {projects.map((project, idx) => {
   310→              const totalTasks = project._count?.tasks || 0
   311→              const members = project.members || []
   312→              const memberUsers = members.map((m) => m.user).filter(Boolean) as User[]
   313→              const pConfig = getPriorityConfig(project.priority)
   314→              const isExpanded = expandedId === project.id
   315→              const stats = getProjectTaskStats(project.id)
   316→
   317→              return (
   318→                <motion.div
   319→                  key={project.id}
   320→                  initial={{ opacity: 0, y: 20 }}
   321→                  animate={{ opacity: 1, y: 0 }}
   322→                  transition={{ delay: idx * 0.05 }}
   323→                >
   324→                  <Card
   325→                    className={cn(
   326→                      'overflow-hidden cursor-pointer transition-all hover:shadow-md',
   327→                      isExpanded && 'ring-1 ring-primary/20 shadow-md',
   328→                    )}
   329→                    onClick={() => handleExpand(project.id)}
   330→                  >
   331→                    <div className="flex">
   332→                      <div className="w-1.5 shrink-0" style={{ backgroundColor: project.color }} />
   333→                      <CardContent className="flex-1 p-4">
   334→                        <div className="flex items-start justify-between mb-2">
   335→                          <h3 className="font-semibold text-sm leading-tight line-clamp-1">{project.name}</h3>
   336→                          <Badge className={cn('text-[10px] px-1.5 py-0 ml-2 shrink-0', pConfig.color)}>
   337→                            {pConfig.label}
   338→                          </Badge>
   339→                        </div>
   340→                        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 min-h-[2rem]">
   341→                          {project.description || 'No description'}
   342→                        </p>
   343→                        <Progress
   344→                          value={stats ? (stats.total > 0 ? (stats.done / stats.total) * 100 : 0) : 0}
   345→                          className="h-1.5 mb-3"
   346→                        />
   347→                        <div className="flex items-center justify-between text-xs text-muted-foreground">
   348→                          <div className="flex items-center gap-3">
   349→                            <span className="flex items-center gap-1">
   350→                              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   351→                              {stats ? `${stats.done}/${stats.total}` : totalTasks}
   352→                            </span>
   353→                            <span className="flex items-center gap-1">
   354→                              <Users className="h-3 w-3" />
   355→                              {members.length}
   356→                            </span>
   357→                          </div>
   358→                          {project.dueDate && (
   359→                            <span className={cn(
   360→                              isPast(parseISO(project.dueDate)) && !isToday(parseISO(project.dueDate))
   361→                                ? 'text-red-500' : '',
   362→                            )}>
   363→                              {formatDate(project.dueDate)}
   364→                            </span>
   365→                          )}
   366→                        </div>
   367→                        {memberUsers.length > 0 && (
   368→                          <div className="mt-3">
   369→                            <AvatarStack users={memberUsers} max={4} />
   370→                          </div>
   371→                        )}
   372→                      </CardContent>
   373→                    </div>
   374→                  </Card>
   375→
   376→                  {/* Expanded Task List */}
   377→                  <AnimatePresence>
   378→                    {isExpanded && (
   379→                      <motion.div
   380→                        initial={{ opacity: 0, height: 0 }}
   381→                        animate={{ opacity: 1, height: 'auto' }}
   382→                        exit={{ opacity: 0, height: 0 }}
   383→                        transition={{ duration: 0.2 }}
   384→                        className="overflow-hidden"
   385→                        onClick={(e) => e.stopPropagation()}
   386→                      >
   387→                        <Card className="mt-3 p-4 space-y-4">
   388→                          <div className="flex items-center justify-between">
   389→                            <div className="flex items-center gap-2">
   390→                              <div className="flex gap-1">
   391→                                {['all', 'todo', 'in_progress', 'in_review', 'done'].map((s) => (
   392→                                  <Button
   393→                                    key={s}
   394→                                    variant={taskFilter === s ? 'default' : 'ghost'}
   395→                                    size="sm"
   396→                                    className="h-7 text-xs px-2"
   397→                                    onClick={() => setTaskFilter(s)}
   398→                                  >
   399→                                    {s === 'all' ? 'All' : getStatusConfig(s).label}
   400→                                  </Button>
   401→                                ))}
   402→                              </div>
   403→                            </div>
   404→                            {!isTraining && (
   405→                            <Button size="sm" className="h-8 text-xs gap-1" onClick={() => openCreateTaskForProject(project.id)}>
   406→                              <Plus className="h-3 w-3" /> Add Task
   407→                            </Button>
   408→                            )}
   409→                          </div>
   410→                          {expandedLoading ? (
   411→                            <div className="space-y-2">
   412→                              {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   413→                            </div>
   414→                          ) : filteredTasks.length === 0 ? (
   415→                            <p className="text-xs text-muted-foreground text-center py-4">No tasks found</p>
   416→                          ) : (
   417→                            <ScrollArea className="max-h-64">
   418→                              <div className="space-y-2">
   419→                                {filteredTasks.map((task) => {
   420→                                  const sConfig = getStatusConfig(task.status)
   421→                                  const pConf = getPriorityConfig(task.priority)
   422→                                  const assignees = (task.assignees || []).map((a) => a.user).filter(Boolean) as User[]
   423→                                  return (
   424→                                    <div
   425→                                      key={task.id}
   426→                                      className="flex items-center gap-3 p-2.5 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors group cursor-pointer"
   427→                                      onClick={() => setSelectedProjectId(project.id)}
   428→                                    >
   429→                                      <div className={cn('h-2 w-2 rounded-full shrink-0', sConfig.dotColor)} />
   430→                                      <div className="flex-1 min-w-0">
   431→                                        <p className="text-sm font-medium truncate">{task.title}</p>
   432→                                        <div className="flex items-center gap-2 mt-0.5">
   433→                                          <Badge className={cn('text-[10px] px-1.5 py-0 h-4', pConf.color)}>{pConf.label}</Badge>
   434→                                          {task.dueDate && (
   435→                                            <span className={cn('text-[10px] text-muted-foreground', isPast(parseISO(task.dueDate)) && !isToday(parseISO(task.dueDate)) ? 'text-red-500' : '')}>
   436→                                              {formatDate(task.dueDate)}
   437→                                            </span>
   438→                                          )}
   439→                                        </div>
   440→                                      </div>
   441→                                      <AvatarStack users={assignees} max={2} />
   442→                                    </div>
   443→                                  )
   444→                                })}
   445→                              </div>
   446→                            </ScrollArea>
   447→                          )}
   448→                        </Card>
   449→                      </motion.div>
   450→                    )}
   451→                  </AnimatePresence>
   452→                </motion.div>
   453→              )
   454→            })}
   455→          </AnimatePresence>
   456→        </motion.div>
   457→      )}
   458→
   459→      {/* Create Task Dialog */}
   460→      <Dialog open={createTaskOpen} onOpenChange={setCreateTaskOpen}>
   461→        <DialogContent className="sm:max-w-lg">
   462→          <DialogHeader>
   463→            <DialogTitle>Create New Task</DialogTitle>
   464→            <DialogDescription>Add a task to the project.</DialogDescription>
   465→          </DialogHeader>
   466→          <div className="space-y-4 py-2">
   467→            <div className="space-y-2">
   468→              <Label>Title</Label>
   469→              <Input placeholder="Task title" value={taskTitle} onChange={(e) => setTaskTitle(e.target.value)} />
   470→            </div>
   471→            <div className="space-y-2">
   472→              <Label>Description</Label>
   473→              <Textarea placeholder="Describe the task..." value={taskDesc} onChange={(e) => setTaskDesc(e.target.value)} rows={2} />
   474→            </div>
   475→            <div className="grid grid-cols-2 gap-4">
   476→              <div className="space-y-2">
   477→                <Label>Status</Label>
   478→                <Select value={taskStatus} onValueChange={(v) => setTaskStatus(v as typeof taskStatus)}>
   479→                  <SelectTrigger><SelectValue /></SelectTrigger>
   480→                  <SelectContent>
   481→                    <SelectItem value="todo">To Do</SelectItem>
   482→                    <SelectItem value="in_progress">In Progress</SelectItem>
   483→                    <SelectItem value="in_review">In Review</SelectItem>
   484→                    <SelectItem value="done">Done</SelectItem>
   485→                  </SelectContent>
   486→                </Select>
   487→              </div>
   488→              <div className="space-y-2">
   489→                <Label>Priority</Label>
   490→                <Select value={taskPriority} onValueChange={(v) => setTaskPriority(v as typeof taskPriority)}>
   491→                  <SelectTrigger><SelectValue /></SelectTrigger>
   492→                  <SelectContent>
   493→                    <SelectItem value="low">Low</SelectItem>
   494→                    <SelectItem value="medium">Medium</SelectItem>
   495→                    <SelectItem value="high">High</SelectItem>
   496→                    <SelectItem value="urgent">Urgent</SelectItem>
   497→                  </SelectContent>
   498→                </Select>
   499→              </div>
   500→            </div>
   501→            <div className="grid grid-cols-2 gap-4">
   502→              <div className="space-y-2">
   503→                <Label>Due Date</Label>
   504→                <Popover open={taskCalendarOpen} onOpenChange={setTaskCalendarOpen}>
   505→                  <PopoverTrigger asChild>
   506→                    <Button variant="outline" className="w-full justify-start text-left font-normal gap-2">

  1861→function formatDuration(minutes: number): string {
  1862→  const h = Math.floor(minutes / 60)
  1863→  const m = minutes % 60
  1864→  if (h === 0) return `${m}m`
  1865→  if (m === 0) return `${h}h`
  1866→  return `${h}h ${m}m`
  1867→}

  1869→function formatElapsed(startDate: string): string {
  1870→  const ms = Date.now() - new Date(startDate).getTime()
  1871→  const totalSec = Math.floor(ms / 1000)
  1872→  const h = Math.floor(totalSec / 3600)
  1873→  const m = Math.floor((totalSec % 3600) / 60)
  1874→  const s = totalSec % 60
  1875→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  1876→}

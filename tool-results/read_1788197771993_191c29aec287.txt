    59→function UserAvatar({ user, size = 'md', className }: { user?: User | null; size?: 'sm' | 'md' | 'lg'; className?: string }) {
    60→  const sizeMap = { sm: 'h-8 w-8 text-[10px]', md: 'h-10 w-10 text-xs', lg: 'h-12 w-12 text-sm' }
    61→  if (!user) return <div className={cn(sizeMap[size], 'rounded-full bg-muted flex items-center justify-center', className)}>?</div>
    62→  return (
    63→    <Avatar className={cn(sizeMap[size], className)}>
    64→      {user.avatar && <AvatarImage src={user.avatar} />}
    65→      <AvatarFallback className="bg-slate-100 dark:bg-slate-800 font-medium">
    66→        {getInitials(user.name)}
    67→      </AvatarFallback>
    68→    </Avatar>
    69→  )
    70→}

    83→function CreateGroupDialog({
    84→  open, onOpenChange, onCreated, currentUser,
    85→}: {
    86→  open: boolean; onOpenChange: (open: boolean) => void;
    87→  onCreated: (group: MessageGroup) => void; currentUser: User;
    88→}) {
    89→  const [name, setName] = useState('')
    90→  const [users, setUsers] = useState<User[]>([])
    91→  const [selectedIds, setSelectedIds] = useState<string[]>([])
    92→  const [creating, setCreating] = useState(false)
    93→  const [search, setSearch] = useState('')
    94→  const [loading, setLoading] = useState(false)
    95→
    96→  useEffect(() => {
    97→    if (!open) return
    98→    setLoading(true)
    99→    setName('')
   100→    setSelectedIds([])
   101→    setSearch('')
   102→    fetch('/api/users')
   103→      .then((r) => r.json())
   104→      .then((d) => {
   105→        setUsers((d.users ?? []).filter((u: User) => {
   106→          if (u.id === currentUser.id) return false
   107→          // Training users cannot message clients
   108→          if (currentUser.role === 'training' && u.role === 'client') return false
   109→          return true
   110→        }))
   111→      })
   112→      .catch(() => {})
   113→      .finally(() => setLoading(false))
   114→  }, [open, currentUser.id, currentUser.role])
   115→
   116→  const toggleUser = (userId: string) => {
   117→    setSelectedIds((prev) => prev.includes(userId) ? prev.filter((id) => id !== userId) : [...prev, userId])
   118→  }
   119→
   120→  const handleCreate = async () => {
   121→    if (!name.trim() || selectedIds.length === 0) return
   122→    setCreating(true)
   123→    try {
   124→      const res = await fetch('/api/message-groups', {
   125→        method: 'POST',
   126→        headers: { 'Content-Type': 'application/json' },
   127→        body: JSON.stringify({ name: name.trim(), createdBy: currentUser.id, memberIds: selectedIds }),
   128→      })
   129→      if (!res.ok) throw new Error()
   130→      const data = await res.json()
   131→      toast.success('Group created')
   132→      onCreated(data.group)
   133→      onOpenChange(false)
   134→    } catch {
   135→      toast.error('Failed to create group')
   136→    } finally {
   137→      setCreating(false)
   138→    }
   139→  }
   140→
   141→  const filteredUsers = users.filter(
   142→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   143→  )
   144→
   145→  return (
   146→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   147→      <DialogContent className="sm:max-w-md">
   148→        <DialogHeader>
   149→          <DialogTitle className="flex items-center gap-2">
   150→            <Users className="h-5 w-5 text-violet-600" /> Create Group
   151→          </DialogTitle>
   152→          <DialogDescription>Start a group conversation with team members</DialogDescription>
   153→        </DialogHeader>
   154→
   155→        <div className="space-y-4 py-2">
   156→          <div>
   157→            <Input placeholder="Group name..." value={name} onChange={(e) => setName(e.target.value)} autoFocus />
   158→          </div>
   159→
   160→          <div>
   161→            <div className="relative mb-2">
   162→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   163→              <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   164→            </div>
   165→            <ScrollArea className="max-h-48 rounded-md border">
   166→              {loading ? (
   167→                <div className="p-3 space-y-2">
   168→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   169→                </div>
   170→              ) : filteredUsers.length === 0 ? (
   171→                <p className="text-sm text-muted-foreground text-center py-4">No users found</p>
   172→              ) : (
   173→                <div className="p-1">
   174→                  {filteredUsers.map((u) => (
   175→                    <button
   176→                      key={u.id}
   177→                      type="button"
   178→                      className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   179→                      onClick={() => toggleUser(u.id)}
   180→                    >
   181→                      <Checkbox checked={selectedIds.includes(u.id)} />
   182→                      <UserAvatar user={u} size="sm" />
   183→                      <div className="flex-1 min-w-0">
   184→                        <p className="text-sm font-medium truncate">{u.name}</p>
   185→                        <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   186→                      </div>
   187→                      <Badge variant="secondary" className={cn('text-[10px]', getRoleBadgeClasses(u.role))}>{u.role}</Badge>
   188→                    </button>
   189→                  ))}
   190→                </div>
   191→              )}
   192→            </ScrollArea>
   193→            {selectedIds.length > 0 && (
   194→              <p className="text-xs text-muted-foreground mt-2">{selectedIds.length} member{selectedIds.length > 1 ? 's' : ''} selected</p>
   195→            )}
   196→          </div>
   197→        </div>
   198→
   199→        <DialogFooter>
   200→          <Button variant="outline" type="button" onClick={() => onOpenChange(false)}>Cancel</Button>
   201→          <Button
   202→            type="button"
   203→            onClick={handleCreate} disabled={!name.trim() || selectedIds.length === 0 || creating}
   204→            className="bg-violet-600 hover:bg-violet-700 text-white"
   205→          >
   206→            {creating && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   207→            Create Group
   208→          </Button>
   209→        </DialogFooter>
   210→      </DialogContent>
   211→    </Dialog>
   212→  )
   213→}

   300→function GroupInfoDialog({
   301→  open, onOpenChange, group, currentUserId, currentUserRole,
   302→}: {
   303→  open: boolean; onOpenChange: (open: boolean) => void
   304→  group: { id: string; name: string; members?: GroupMember[]; _count?: { members: number } } | null
   305→  currentUserId: string; currentUserRole: string
   306→}) {
   307→  const [members, setMembers] = useState<(GroupMember & { user: User })[]>([])
   308→  const [users, setUsers] = useState<User[]>([])
   309→  const [loading, setLoading] = useState(false)
   310→  const [showAddUser, setShowAddUser] = useState(false)
   311→  const [search, setSearch] = useState('')
   312→  useEffect(() => {
   313→    if (!open || !group) return
   314→    setLoading(true)
   315→    setSearch('')
   316→    setShowAddUser(false)
   317→    fetch(`/api/message-groups/${group.id}?userId=${currentUserId}`)
   318→      .then((r) => r.json())
   319→      .then((d) => {
   320→        setMembers((d.group?.members ?? []) as (GroupMember & { user: User })[])
   321→      })
   322→      .catch(() => {})
   323→      .finally(() => setLoading(false))
   324→  }, [open, group, currentUserId])
   325→
   326→  const handleAddMember = async (userId: string) => {
   327→    if (!group) return
   328→    try {
   329→      const res = await fetch(`/api/message-groups/${group.id}`, {
   330→        method: 'PATCH',
   331→        headers: { 'Content-Type': 'application/json' },
   332→        body: JSON.stringify({ addMemberIds: [userId] }),
   333→      })
   334→      if (!res.ok) throw new Error()
   335→      const data = await res.json()
   336→      setMembers(data.group.members ?? [])
   337→      toast.success('Member added')
   338→    } catch {
   339→      toast.error('Failed to add member')
   340→    }
   341→  }
   342→
   343→  const handleRemoveMember = async (userId: string) => {
   344→    if (!group) return
   345→    try {
   346→      const res = await fetch(`/api/message-groups/${group.id}`, {
   347→        method: 'PATCH',
   348→        headers: { 'Content-Type': 'application/json' },
   349→        body: JSON.stringify({ removeMemberIds: [userId] }),
   350→      })
   351→      if (!res.ok) throw new Error()
   352→      const data = await res.json()
   353→      setMembers(data.group.members ?? [])
   354→      toast.success('Member removed')
   355→    } catch {
   356→      toast.error('Failed to remove member')
   357→    }
   358→  }
   359→
   360→  const handleLoadUsers = () => {
   361→    fetch('/api/users')
   362→      .then((r) => r.json())
   363→      .then((d) => {
   364→        const memberIds = members.map((m) => m.userId)
   365→        setUsers((d.users ?? []).filter((u: User) => {
   366→          if (memberIds.includes(u.id)) return false
   367→          // Training users cannot add clients to groups
   368→          if (currentUserRole === 'training' && u.role === 'client') return false
   369→          return true
   370→        }))
   371→      })
   372→      .catch(() => {})
   373→    setShowAddUser(true)
   374→  }
   375→
   376→  const filteredUsers = users.filter(
   377→    (u) => u.name.toLowerCase().includes(search.toLowerCase()) || u.email.toLowerCase().includes(search.toLowerCase())
   378→  )
   379→
   380→  return (
   381→    <Dialog open={open} onOpenChange={onOpenChange} modal={true}>
   382→      <DialogContent className="sm:max-w-md">
   383→        <DialogHeader>
   384→          <DialogTitle className="flex items-center gap-2">
   385→            <Users className="h-5 w-5 text-violet-600" /> Group Info
   386→          </DialogTitle>
   387→        </DialogHeader>
   388→
   389→        <div className="space-y-4 py-2">
   390→          {group && (
   391→            <div className="flex items-center gap-3">
   392→              <GroupAvatar name={group.name} size="lg" />
   393→              <div>
   394→                <p className="font-semibold">{group.name}</p>
   395→                <p className="text-sm text-muted-foreground">{group._count?.members ?? members.length} members</p>
   396→              </div>
   397→            </div>
   398→          )}
   399→
   400→          <Separator />
   401→
   402→          <div>
   403→            <div className="flex items-center justify-between mb-2">
   404→              <p className="text-sm font-medium">Members</p>
   405→              {!showAddUser && (
   406→                <Button variant="ghost" size="sm" className="h-7 text-xs gap-1" type="button" onClick={handleLoadUsers}>
   407→                  <UserPlus className="h-3.5 w-3.5" /> Add
   408→                </Button>
   409→              )}
   410→            </div>
   411→
   412→            {showAddUser && (
   413→              <div className="mb-3 space-y-2">
   414→                <div className="relative">
   415→                  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   416→                  <Input placeholder="Search users..." className="pl-9 h-9 text-sm" value={search} onChange={(e) => setSearch(e.target.value)} />
   417→                </div>
   418→                <ScrollArea className="max-h-32 rounded-md border">
   419→                  <div className="p-1">
   420→                    {filteredUsers.slice(0, 8).map((u) => (
   421→                      <button
   422→                        key={u.id}
   423→                        type="button"
   424→                        className="w-full flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50 transition-colors text-left"
   425→                        onClick={() => handleAddMember(u.id)}
   426→                      >
   427→                        <UserAvatar user={u} size="sm" />
   428→                        <div className="flex-1 min-w-0">
   429→                          <p className="text-sm font-medium truncate">{u.name}</p>
   430→                        </div>
   431→                        <UserPlus className="h-4 w-4 text-muted-foreground" />
   432→                      </button>
   433→                    ))}
   434→                  </div>
   435→                </ScrollArea>
   436→                <Button variant="ghost" size="sm" className="h-7 text-xs" type="button" onClick={() => setShowAddUser(false)}>Done</Button>
   437→              </div>
   438→            )}
   439→
   440→            <ScrollArea className="max-h-48 rounded-md border">
   441→              {loading ? (
   442→                <div className="p-3 space-y-2">
   443→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   444→                </div>
   445→              ) : (
   446→                <div className="p-1">
   447→                  {members.map((m) => (
   448→                    <div key={m.id} className="flex items-center gap-3 px-3 py-2 rounded-md hover:bg-muted/50">
   449→                      <UserAvatar user={m.user} size="sm" />
   450→                      <div className="flex-1 min-w-0">
   451→                        <p className="text-sm font-medium truncate">{m.user?.name ?? 'Unknown'}</p>
   452→                        <p className="text-xs text-muted-foreground">{m.role === 'admin' ? 'Admin' : 'Member'}</p>
   453→                      </div>
   454→                      {m.userId !== currentUserId && (
   455→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-muted-foreground hover:text-destructive" type="button" onClick={() => handleRemoveMember(m.userId)}>
   456→                          <X className="h-3.5 w-3.5" />
   457→                        </Button>
   458→                      )}
   459→                    </div>
   460→                  ))}
   461→                </div>
   462→              )}
   463→            </ScrollArea>
   464→          </div>
   465→        </div>
   466→      </DialogContent>
   467→    </Dialog>
   468→  )
   469→}

   666→  if (loading) {

   829→                <Button

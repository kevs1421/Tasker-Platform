   170→   170→  const [onboardingDesc, setOnboardingDesc] = useState('')
   171→   171→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   172→   172→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   173→   173→
   174→   174→  const fetchProjects = useCallback(async () => {
   175→   175→    try {
   176→   176→      const res = await fetch('/api/projects')
   177→   177→      const data = await res.json()

   179→   179→      const clientProjects = (data.projects || [])
   180→   180→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   181→   181→      setProjects(clientProjects)
   182→   182→    } catch {
   183→   183→      toast.error('Failed to load projects')
   184→   184→    } finally {
   185→   185→      setLoading(false)
   186→   186→    }

   193→   193→      setExpandedProjectId(null)
   194→   194→      setExpandedTasks([])
   195→   195→      return
   196→   196→    }

   199→   199→      const res = await fetch(`/api/projects/${projectId}`)
   200→   200→      const data = await res.json()

   203→   203→      toast.error('Failed to load project tasks')

   213→   213→      setTaskDetail(data.task)
   214→   214→    } catch {
   215→   215→      toast.error('Failed to load task details')
   216→   216→    } finally {
   217→   217→      setTaskDetailLoading(false)
   218→   218→    }
   219→   219→  }
   220→   220→
   221→   221→  const openOnboardingDialog = (projectId: string, projectName: string) => {

   223→   223→    setOnboardingTitle(`Onboarding - ${projectName}`)

   233→   233→      const stepsJson = JSON.stringify(
   234→   234→        onboardingSteps.map(stepId => {
   235→   235→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   236→   236→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   237→   237→        })
   238→   238→      )
   239→   239→      const res = await fetch('/api/onboarding', {
   240→   240→        method: 'POST',
   241→   241→        headers: { 'Content-Type': 'application/json' },
   242→   242→        body: JSON.stringify({

   245→   245→          stepsJson,

   247→   247→          clientId: user?.id || null,

   249→   249→        }),

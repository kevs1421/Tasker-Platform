    98→const ONBOARDING_STEPS = [
    99→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   100→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },
   101→  { id: 'project_kickoff', label: 'Project Kickoff', description: 'Goals, audience, competitors, budget, timeline', required: true },
   102→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   103→]
   104→
   105→export function ClientProjects() {
   106→  const { user } = useAuthStore()
   107→  const [projects, setProjects] = useState<ClientProject[]>([])
   108→  const [loading, setLoading] = useState(true)
   109→
   110→  // Upload files state
   111→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ url: string; originalName: string; filename: string; size: number; type: string; uploadedAt: string; projectId: string }>>([])
   112→  const [uploading, setUploading] = useState(false)
   113→  const [uploadDragOver, setUploadDragOver] = useState<string | null>(null) // projectId
   114→  const [uploadDialogProjectId, setUploadDialogProjectId] = useState<string>('')
   115→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   116→  const fileInputRef = useRef<HTMLInputElement>(null)
   117→
   118→  // Onboarding creation state
   119→
   120→
   121→  const fetchProjects = useCallback(async () => {
   122→    if (!user?.id) return
   123→    try {
   124→      // Use the client-only endpoint that queries through ClientProject table
   125→      // This ensures only projects explicitly assigned to this client are returned
   126→      const res = await fetch(`/api/client/projects?clientId=${user.id}`)
   127→      const data = await res.json()
   128→      setProjects(data.projects || [])
   129→    } catch {
   130→      toast.error('Failed to load projects')
   131→    } finally {
   132→      setLoading(false)
   133→    }
   134→  }, [user?.id])
   135→
   136→  // Fetch uploaded files from localStorage
   137→  const fetchUploadedFiles = useCallback(() => {
   138→    if (!user) return
   139→    try {
   140→      const stored = localStorage.getItem(`client-uploads-${user.id}`)
   141→      if (stored) setUploadedFiles(JSON.parse(stored))
   142→    } catch { /* ignore */ }
   143→  }, [user])
   144→
   145→  const saveUploadedFiles = useCallback((files: typeof uploadedFiles) => {
   146→    if (!user) return
   147→    localStorage.setItem(`client-uploads-${user.id}`, JSON.stringify(files))
   148→  }, [user])
   149→
   150→  useEffect(() => { fetchProjects(); fetchUploadedFiles() }, [fetchProjects, fetchUploadedFiles])
   151→
   152→  // File upload handler — always uploads to the project set in uploadDialogProjectId
   153→  const handleFileUpload = async (files: FileList | null) => {
   154→    if (!files || files.length === 0) return
   155→    if (!uploadDialogProjectId) {
   156→      toast.error('No project selected')
   157→      return
   158→    }
   159→    setUploading(true)
   160→    const newFiles: typeof uploadedFiles = []
   161→    for (let i = 0; i < files.length; i++) {
   162→      const file = files[i]
   163→      try {
   164→        const formData = new FormData()
   165→        formData.append('file', file)
   166→        const res = await fetch('/api/upload', { method: 'POST', body: formData })
   167→        if (!res.ok) {
   168→          const d = await res.json()
   169→          throw new Error(d.error || 'Upload failed')
   170→        }
   171→        const data = await res.json()
   172→        newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   173→      } catch (e: unknown) {
   174→        toast.error(`${file.name}: ${e instanceof Error ? e.message : 'Upload failed'}`)
   175→      }
   176→    }
   177→    if (newFiles.length > 0) {
   178→      const updated = [...uploadedFiles, ...newFiles]
   179→      setUploadedFiles(updated)
   180→      saveUploadedFiles(updated)
   181→      toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   182→    }
   183→    setUploading(false)
   184→    if (fileInputRef.current) fileInputRef.current.value = ''
   185→  }
   186→
   187→  const handleDrop = (e: React.DragEvent, projectId: string) => {
   188→    e.preventDefault()
   189→    setUploadDragOver(null)
   190→    setUploadDialogProjectId(projectId)
   191→    handleFileUpload(e.dataTransfer.files)
   192→  }
   193→
   194→  const openUploadDialog = (projectId: string) => {
   195→    setUploadDialogProjectId(projectId)
   196→    setUploadDialogOpen(true)
   197→  }
   198→
   199→  const deleteUploadedFile = (idx: number) => {
   200→    const updated = uploadedFiles.filter((_, i) => i !== idx)
   201→    setUploadedFiles(updated)
   202→    saveUploadedFiles(updated)
   203→    toast.success('File removed')
   204→  }
   205→
   206→  const getFileIcon = (type: string) => {
   207→    if (type.startsWith('image/')) return <ImageIcon className="w-4 h-4 text-emerald-600" />
   208→    if (type.includes('pdf')) return <FileText className="w-4 h-4 text-red-500" />
   209→    if (type.includes('word') || type.includes('document')) return <FileText className="w-4 h-4 text-blue-500" />
   210→    if (type.includes('excel') || type.includes('sheet') || type.includes('spreadsheet')) return <FileSpreadsheet className="w-4 h-4 text-emerald-600" />
   211→    return <File className="w-4 h-4 text-muted-foreground" />
   212→  }
   213→
   214→  const formatFileSize = (bytes: number) => {
   215→    if (bytes < 1024) return `${bytes} B`
   216→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   217→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   218→  }
   219→
   220→  // Navigate to the onboarding tab
   221→
   222→  const handleCreateOnboarding = async () => {
   223→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   224→    setCreatingOnboarding(true)
   225→    try {
   226→      const stepsJson = JSON.stringify(
   227→        onboardingSteps.map(stepId => {
   228→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   229→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   230→        })
   231→      )
   232→      const res = await fetch('/api/onboarding', {
   233→        method: 'POST',
   234→        headers: { 'Content-Type': 'application/json' },
   235→        body: JSON.stringify({
   236→          title: onboardingTitle.trim(),
   237→          description: onboardingDesc.trim() || null,
   238→          stepsJson,
   239→          projectId: onboardingProjectId,
   240→          clientId: user?.id || null,
   241→          status: 'published',
   242→        }),
   243→      })
   244→      if (!res.ok) {
   245→        const d = await res.json()
   246→        throw new Error(d.error || 'Failed to create')
   247→      }
   248→      toast.success('Onboarding form created successfully')
   249→      setOnboardingDialogOpen(false)
   250→    } catch (e: unknown) {
   251→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   252→    } finally {
   253→      setCreatingOnboarding(false)
   254→    }
   255→  }
   256→
   257→  const toggleOnboardingStep = (stepId: string) => {
   258→    setOnboardingSteps(prev =>
   259→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   260→    )
   261→  }
   262→
   263→  if (loading) return <PageSkeleton />
   264→
   265→  return (
   266→    <div className="p-4 md:p-6 space-y-6">
   267→      {/* Page Header */}
   268→      <div className="flex items-center gap-3">
   269→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   270→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   271→        </div>
   272→        <div>
   273→          <h1 className="text-2xl font-bold">My Projects</h1>
   274→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   275→        </div>
   276→      </div>
   277→
   278→      {/* ─── Upload Files Section (hidden input + dialog) ─── */}
   279→      <input
   280→        ref={fileInputRef}
   281→        type="file"
   282→        multiple
   283→        accept=".jpg,.jpeg,.png,.gif,.webp,.svg,.pdf,.doc,.docx,.xls,.xlsx,.txt,.zip"
   284→        className="hidden"
   285→        onChange={(e) => { handleFileUpload(e.target.files); setUploadDialogOpen(false) }}
   286→      />
   287→      <Dialog open={uploadDialogOpen} onOpenChange={(open) => { if (!open) setUploadDialogOpen(false) }}>
   288→        <DialogContent className="sm:max-w-md">
   289→          <DialogHeader>
   290→            <DialogTitle className="flex items-center gap-2">
   291→              <Upload className="w-5 h-5 text-primary" />
   292→              Upload Files
   293→            </DialogTitle>
   294→            <DialogDescription>Choose files to upload for this project</DialogDescription>
   295→          </DialogHeader>
   296→          <div
   297→            onDragOver={(e) => { e.preventDefault(); setUploadDragOver('__dialog__') }}
   298→            onDragLeave={() => setUploadDragOver(null)}
   299→            onDrop={(e) => {
   300→              e.preventDefault()
   301→              setUploadDragOver(null)
   302→              // Set the project context before handling files
   303→              const files = e.dataTransfer.files
   304→              if (files && files.length > 0) {
   305→                // Use a separate handler for dialog drops
   306→                void (async () => {
   307→                  setUploading(true)
   308→                  const newFiles: typeof uploadedFiles = []
   309→                  for (let i = 0; i < files.length; i++) {
   310→                    const file = files[i]
   311→                    try {
   312→                      const formData = new FormData()
   313→                      formData.append('file', file)
   314→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   315→                      if (!res.ok) {
   316→                        const d = await res.json()
   317→                        throw new Error(d.error || 'Upload failed')
   318→                      }
   319→                      const data = await res.json()
   320→                      newFiles.push({ ...data, uploadedAt: new Date().toISOString(), projectId: uploadDialogProjectId })
   321→                    } catch (err: unknown) {
   322→                      toast.error(`${file.name}: ${err instanceof Error ? err.message : 'Upload failed'}`)
   323→                    }
   324→                  }
   325→                  if (newFiles.length > 0) {
   326→                    const updated = [...uploadedFiles, ...newFiles]
   327→                    setUploadedFiles(updated)
   328→                    saveUploadedFiles(updated)
   329→                    toast.success(`${newFiles.length} file${newFiles.length > 1 ? 's' : ''} uploaded successfully`)
   330→                  }
   331→                  setUploading(false)
   332→                })()
   333→              }
   334→            }}
   335→            onClick={() => fileInputRef.current?.click()}
   336→            className={cn(
   337→              'border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors',
   338→              uploadDragOver === '__dialog__' ? 'border-primary bg-primary/5' : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/30'
   339→            )}
   340→          >
   341→            {uploading ? (
   342→              <div className="flex flex-col items-center gap-2">
   343→                <Loader2 className="w-8 h-8 text-primary animate-spin" />
   344→                <p className="text-sm text-muted-foreground">Uploading...</p>
   345→              </div>
   346→            ) : (
   347→              <div className="flex flex-col items-center gap-2">
   348→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center">
   349→                  <Upload className="w-5 h-5 text-muted-foreground" />
   350→                </div>
   351→                <div>
   352→                  <p className="text-sm font-medium">Drop files here or click to browse</p>
   353→                  <p className="text-xs text-muted-foreground mt-1">JPG, PNG, PDF, DOC, DOCX, XLS, XLSX and more (max 10MB each)</p>
   354→                </div>
   355→              </div>
   356→            )}
   357→          </div>
   358→          <DialogFooter>
   359→            <Button variant="outline" onClick={() => setUploadDialogOpen(false)}>Close</Button>
   360→          </DialogFooter>
   361→        </DialogContent>
   362→      </Dialog>
   363→
   364→      {/* ─── Project Progress Overview ──────────────────────────── */}
   365→      {projects.length > 0 && (
   366→        <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   367→          <Card>
   368→            <CardHeader className="pb-3">
   369→              <CardTitle className="text-base flex items-center gap-2">
   370→                <BarChart3 className="w-4 h-4 text-primary" />
   371→                Project Progress
   372→              </CardTitle>
   373→              <p className="text-sm text-muted-foreground">Overview of task progress across your projects</p>
   374→            </CardHeader>
   375→            <CardContent>
   376→              <div className="space-y-5">
   377→                {projects.map((project) => {
   378→                  const todo = project.taskStats?.todo || 0
   379→                  const inProgress = project.taskStats?.in_progress || 0
   380→                  const inReview = project.taskStats?.in_review || 0
   381→                  const completed = project.taskStats?.done || 0
   382→                  const total = todo + inProgress + inReview + completed
   383→                  if (total === 0) return null
   384→
   385→                  const todoPct = Math.round((todo / total) * 100)
   386→                  const inProgressPct = Math.round((inProgress / total) * 100)
   387→                  const inReviewPct = Math.round((inReview / total) * 100)
   388→                  const completedPct = Math.round((completed / total) * 100)
   389→
   390→                  return (
   391→                    <div key={project.id} className="space-y-2.5">
   392→                      <div className="flex items-center justify-between">
   393→                        <div className="flex items-center gap-2">
   394→                          <span className="w-3 h-3 rounded-full" style={{ backgroundColor: project.color }} />
   395→                          <span className="text-sm font-medium">{project.name}</span>
   396→                        </div>
   397→                        <span className="text-sm font-semibold text-emerald-600 dark:text-emerald-400">{completedPct}%</span>
   398→                      </div>
   399→                      {/* Stacked progress bar */}
   400→                      <div className="flex h-3 w-full overflow-hidden rounded-full bg-muted">
   401→                        {todo > 0 && (
   402→                          <div
   403→                            className="bg-slate-400 dark:bg-slate-500 transition-all duration-500"
   404→                            style={{ width: `${todoPct}%` }}
   405→                            title={`Not Started: ${todo}`}
   406→                          />
   407→                        )}
   408→                        {inProgress > 0 && (
   409→                          <div
   410→                            className="bg-amber-500 transition-all duration-500"
   411→                            style={{ width: `${inProgressPct}%` }}
   412→                            title={`In Progress: ${inProgress}`}
   413→                          />
   414→                        )}
   415→                        {inReview > 0 && (
   416→                          <div
   417→                            className="bg-blue-500 transition-all duration-500"
   418→                            style={{ width: `${inReviewPct}%` }}
   419→                            title={`In Review: ${inReview}`}
   420→                          />
   421→                        )}
   422→                        {completed > 0 && (
   423→                          <div
   424→                            className="bg-emerald-500 transition-all duration-500"
   425→                            style={{ width: `${completedPct}%` }}
   426→                            title={`Completed: ${completed}`}
   427→                          />
   428→                        )}
   429→                      </div>
   430→                      {/* Legend */}
   431→                      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   432→                        <span className="flex items-center gap-1.5">
   433→                          <span className="w-2.5 h-2.5 rounded-sm bg-slate-400 dark:bg-slate-500" />
   434→                          Not Started ({todo})
   435→                        </span>
   436→                        <span className="flex items-center gap-1.5">
   437→                          <span className="w-2.5 h-2.5 rounded-sm bg-amber-500" />
   438→                          In Progress ({inProgress})
   439→                        </span>
   440→                        <span className="flex items-center gap-1.5">
   441→                          <span className="w-2.5 h-2.5 rounded-sm bg-blue-500" />
   442→                          In Review ({inReview})
   443→                        </span>
   444→                        <span className="flex items-center gap-1.5">
   445→                          <span className="w-2.5 h-2.5 rounded-sm bg-emerald-500" />
   446→                          Completed ({completed})
   447→                        </span>
   448→                      </div>
   449→                    </div>
   450→                  )
   451→                })}
   452→              </div>
   453→            </CardContent>
   454→          </Card>
   455→        </motion.div>
   456→      )}
   457→
   458→      {projects.length === 0 ? (
   459→        <EmptyState
   460→          icon={FolderKanban}
   461→          title="No active projects"
   462→          description="You don't have any active projects yet."
   463→        />
   464→      ) : (
   465→        <>
   466→          {/* ─── Project Cards ──────────────────────────────────── */}
   467→          <div className="space-y-4">
   468→            {projects.map((project, i) => {
   469→              const doneTasks = project.taskStats?.done || 0
   470→              const totalT = project._count?.tasks || 0
   471→              const pct = project.progress ?? 0
   472→              const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   473→
   474→              // Compute progress bar gradient colors
   475→              const barColor = pct >= 75 ? '#10b981' : pct >= 40 ? '#f59e0b' : '#ef4444'
   476→
   477→              // Files for this project only
   478→              const projectFiles = uploadedFiles.filter(f => f.projectId === project.id)
   479→
   480→              return (
   481→                <motion.div
   482→                  key={project.id}
   483→                  initial={{ opacity: 0, y: 12 }}
   484→                  animate={{ opacity: 1, y: 0 }}
   485→                  transition={{ delay: i * 0.05 }}
   486→                >
   487→                  <Card
   488→                    className="hover:shadow-md transition-all border-l-4"
   489→                    style={{ borderLeftColor: project.color }}
   490→                  >
   491→                    <CardContent className="p-4 md:p-5">
   492→                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   493→                        <div className="min-w-0 flex-1">
   494→                          <div className="flex items-center gap-2 mb-1">
   495→                            <h3 className="font-semibold text-base truncate">{project.name}</h3>
   496→                            <PriorityBadge priority={project.priority} />
   497→                          </div>
   498→                          <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   499→                        </div>
   500→                        <div className="flex items-center gap-2 shrink-0 text-sm flex-wrap">
   501→                          {project.dueDate && (
   502→                            <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   503→                              <Calendar className="w-3.5 h-3.5" />
   504→                              {formatDate(project.dueDate)}
   505→                            </span>
   506→                          )}
   507→                          <span className="text-muted-foreground flex items-center gap-1">
   508→                            <Users className="w-3.5 h-3.5" />
   509→                            {project._count?.members || 0}
   510→                          </span>
   511→                          <Button
   512→                            size="sm"

   688→// CLIENT PROFILE
   689→// ═══════════════════════════════════════════════════════════════════
   690→export function ClientProfile() {
   691→  const { user, updateUser } = useAuthStore()
   692→  const [loading, setLoading] = useState(true)
   693→  const [saving, setSaving] = useState(false)
   694→  const [savingPassword, setSavingPassword] = useState(false)
   695→
   696→  const [name, setName] = useState('')
   697→  const [email, setEmail] = useState('')
   698→  const [phone, setPhone] = useState('')
   699→  const [company, setCompany] = useState('')
   700→
   701→  const [currentPassword, setCurrentPassword] = useState('')
   702→  const [newPassword, setNewPassword] = useState('')
   703→  const [confirmPassword, setConfirmPassword] = useState('')
   704→
   705→  useEffect(() => {
   706→    if (!user?.id) return
   707→    const fetchProfile = async () => {
   708→      try {
   709→        const res = await fetch(`/api/user/profile?userId=${user.id}`)
   710→        const data = await res.json()
   711→        if (data.user) {
   712→          setName(data.user.name || '')
   713→          setEmail(data.user.email || '')
   714→          setPhone(data.user.phone || '')
   715→          setCompany(data.user.company || '')
   716→        }
   717→      } catch {
   718→        toast.error('Failed to load profile')
   719→      } finally {
   720→        setLoading(false)
   721→      }
   722→    }
   723→    fetchProfile()
   724→  }, [user?.id])
   725→
   726→  const saveProfile = async () => {
   727→    if (!user?.id) return
   728→    setSaving(true)
   729→    try {
   730→      const res = await fetch('/api/user/profile', {
   731→        method: 'PATCH',
   732→        headers: { 'Content-Type': 'application/json' },
   733→        body: JSON.stringify({ userId: user.id, name, email, phone, company })
   734→      })
   735→      if (!res.ok) {
   736→        const data = await res.json()
   737→        throw new Error(data.error || 'Failed')
   738→      }
   739→      const data = await res.json()
   740→      toast.success('Profile updated successfully')
   741→      // Update user in store
   742→      if (data.user && user) {
   743→        updateUser({ name: data.user.name, email: data.user.email, phone: data.user.phone, company: data.user.company })
   744→      }
   745→    } catch (e: unknown) {
   746→      const msg = e instanceof Error ? e.message : 'Failed to update profile'
   747→      toast.error(msg)
   748→    } finally {
   749→      setSaving(false)
   750→    }
   751→  }
   752→
   753→  const changePassword = async () => {
   754→    if (!user?.id) return
   755→    if (!currentPassword || !newPassword || !confirmPassword) {
   756→      toast.error('Please fill in all password fields')
   757→      return
   758→    }
   759→    if (newPassword !== confirmPassword) {
   760→      toast.error('New passwords do not match')
   761→      return
   762→    }
   763→    if (newPassword.length < 6) {
   764→      toast.error('Password must be at least 6 characters')
   765→      return
   766→    }
   767→    setSavingPassword(true)
   768→    try {
   769→      const res = await fetch('/api/user/profile', {
   770→        method: 'PATCH',
   771→        headers: { 'Content-Type': 'application/json' },
   772→        body: JSON.stringify({
   773→          userId: user.id,
   774→          currentPassword,
   775→          newPassword
   776→        })
   777→      })
   778→      if (!res.ok) {
   779→        const data = await res.json()
   780→        throw new Error(data.error || 'Failed')
   781→      }
   782→      toast.success('Password changed successfully')
   783→      setCurrentPassword('')
   784→      setNewPassword('')
   785→      setConfirmPassword('')
   786→    } catch (e: unknown) {
   787→      const msg = e instanceof Error ? e.message : 'Failed to change password'
   788→      toast.error(msg)
   789→    } finally {
   790→      setSavingPassword(false)
   791→    }
   792→  }
   793→
   794→  if (loading) return <PageSkeleton />
   795→
   796→  return (
   797→    <div className="p-4 md:p-6 space-y-6 max-w-2xl">
   798→      <div className="flex items-center gap-3">
   799→        <div className="w-10 h-10 rounded-lg bg-rose-100 dark:bg-rose-900/30 flex items-center justify-center">
   800→          <UserCircle className="w-5 h-5 text-rose-600 dark:text-rose-400" />
   801→        </div>
   802→        <div>
   803→          <h1 className="text-2xl font-bold">Profile Settings</h1>
   804→          <p className="text-sm text-muted-foreground">Manage your account information</p>
   805→        </div>
   806→      </div>
   807→
   808→      {/* Profile Card */}
   809→      <motion.div {...fadeSlide}>
   810→        <Card>
   811→          <CardHeader>
   812→            <div className="flex items-center justify-between">
   813→              <CardTitle className="text-lg">Personal Information</CardTitle>
   814→              <span className={`text-xs px-2.5 py-1 rounded-full font-medium ${getRoleBadgeClasses(user?.role || 'client')}`}>
   815→                Client
   816→              </span>
   817→            </div>
   818→            {user?.createdAt && (
   819→              <p className="text-sm text-muted-foreground">
   820→                Member since {formatDate(user.createdAt)}
   821→              </p>
   822→            )}
   823→          </CardHeader>
   824→          <CardContent className="space-y-4">
   825→            <div className="space-y-2">
   826→              <Label htmlFor="profile-name">Full Name</Label>
   827→              <Input
   828→                id="profile-name"
   829→                value={name}
   830→                onChange={e => setName(e.target.value)}
   831→                placeholder="Your full name"
   832→              />
   833→            </div>
   834→            <div className="space-y-2">
   835→              <Label htmlFor="profile-email">Email Address</Label>
   836→              <Input
   837→                id="profile-email"
   838→                type="email"
   839→                value={email}
   840→                onChange={e => setEmail(e.target.value)}
   841→                placeholder="your@email.com"
   842→              />
   843→            </div>
   844→            <div className="space-y-2">
   845→              <Label htmlFor="profile-phone">Phone Number</Label>
   846→              <Input
   847→                id="profile-phone"
   848→                type="tel"
   849→                value={phone}
   850→                onChange={e => setPhone(e.target.value)}
   851→                placeholder="+1 (555) 000-0000"
   852→              />
   853→            </div>
   854→            <div className="space-y-2">
   855→              <Label htmlFor="profile-company">Company</Label>
   856→              <Input
   857→                id="profile-company"
   858→                value={company}
   859→                onChange={e => setCompany(e.target.value)}
   860→                placeholder="Your company name"
   861→              />
   862→            </div>
   863→            <div className="pt-2">
   864→              <Button onClick={saveProfile} disabled={saving} className="w-full sm:w-auto">
   865→                {saving ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   866→                Save Changes
   867→              </Button>
   868→            </div>
   869→          </CardContent>
   870→        </Card>
   871→      </motion.div>
   872→
   873→      {/* Change Password */}
   874→      <motion.div {...fadeSlide} transition={{ delay: 0.1 }}>
   875→        <Card>
   876→          <CardHeader>
   877→            <CardTitle className="text-lg flex items-center gap-2">
   878→              <Shield className="w-5 h-5 text-muted-foreground" />
   879→              Change Password
   880→            </CardTitle>
   881→            <p className="text-sm text-muted-foreground">
   882→              Update your password to keep your account secure
   883→            </p>
   884→          </CardHeader>
   885→          <CardContent className="space-y-4">
   886→            <div className="space-y-2">
   887→              <Label htmlFor="current-password">Current Password</Label>
   888→              <div className="relative">
   889→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   890→                <Input
   891→                  id="current-password"
   892→                  type="password"
   893→                  value={currentPassword}
   894→                  onChange={e => setCurrentPassword(e.target.value)}
   895→                  placeholder="Enter current password"
   896→                  className="pl-9"
   897→                />
   898→              </div>
   899→            </div>
   900→            <div className="space-y-2">
   901→              <Label htmlFor="new-password">New Password</Label>
   902→              <div className="relative">
   903→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   904→                <Input
   905→                  id="new-password"
   906→                  type="password"
   907→                  value={newPassword}
   908→                  onChange={e => setNewPassword(e.target.value)}
   909→                  placeholder="Enter new password"

   918→                <Input

   925→                />
   926→              </div>
   927→            </div>
   928→            <div className="pt-2">

   930→                {savingPassword ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Shield className="w-4 h-4 mr-2" />}
   931→                Change Password

   127→   127→const ONBOARDING_STEPS = [
   128→   128→  { id: 'business_info', label: 'Business Information', description: 'Company details, contact info, industry', required: true },
   129→   129→  { id: 'brand_assets', label: 'Brand Assets', description: 'Logo, brand colors, typography, guidelines', required: true },

   131→   131→  { id: 'documents', label: 'Document Submission', description: 'Contract, payment form, work order agreement', required: true },
   132→   132→]
   133→   133→
   134→   134→export function ClientProjects() {
   135→   135→  const { user } = useAuthStore()
   136→   136→  const [projects, setProjects] = useState<(Project & { tasks?: (Task & { _count: { comments: number; subtasks: number; completedSubtasks: number } })[] })[]>([])
   137→   137→  const [loading, setLoading] = useState(true)
   138→   138→  const [expandedProjectId, setExpandedProjectId] = useState<string | null>(null)
   139→   139→  const [expandedTasks, setExpandedTasks] = useState<Task[]>([])
   140→   140→  const [taskDetail, setTaskDetail] = useState<(Task & { _count: { comments: number; subtasks: number; completedSubtasks: number }; comments?: Comment[]; subtasks?: Subtask[] }) | null>(null)
   141→   141→  const [taskDetailLoading, setTaskDetailLoading] = useState(false)
   142→   142→
   143→   143→  // Onboarding creation state
   144→   144→  const [onboardingDialogOpen, setOnboardingDialogOpen] = useState(false)
   145→   145→  const [onboardingProjectId, setOnboardingProjectId] = useState<string>('')
   146→   146→  const [onboardingTitle, setOnboardingTitle] = useState('')
   147→   147→  const [onboardingDesc, setOnboardingDesc] = useState('')
   148→   148→  const [onboardingSteps, setOnboardingSteps] = useState<string[]>(ONBOARDING_STEPS.map(s => s.id))
   149→   149→  const [creatingOnboarding, setCreatingOnboarding] = useState(false)
   150→   150→
   151→   151→  const fetchProjects = useCallback(async () => {
   152→   152→    try {
   153→   153→      const res = await fetch('/api/projects')
   154→   154→      const data = await res.json()
   155→   155→      // Filter to active projects and those the client is a member of
   156→   156→      const clientProjects = (data.projects || [])
   157→   157→        .filter((p: Project) => p.status === 'active' && p.members?.some(m => m.userId === user?.id))
   158→   158→      setProjects(clientProjects)
   159→   159→    } catch {
   160→   160→      toast.error('Failed to load projects')
   161→   161→    } finally {
   162→   162→      setLoading(false)
   163→   163→    }
   164→   164→  }, [user?.id])
   165→   165→
   166→   166→  useEffect(() => { fetchProjects() }, [fetchProjects])
   167→   167→
   168→   168→  const expandProject = async (projectId: string) => {
   169→   169→    if (expandedProjectId === projectId) {
   170→   170→      setExpandedProjectId(null)
   171→   171→      setExpandedTasks([])
   172→   172→      return
   173→   173→    }
   174→   174→    setExpandedProjectId(projectId)
   175→   175→    try {
   176→   176→      const res = await fetch(`/api/projects/${projectId}`)
   177→   177→      const data = await res.json()
   178→   178→      setExpandedTasks(data.project?.tasks || [])
   179→   179→    } catch {
   180→   180→      toast.error('Failed to load project tasks')
   181→   181→    }
   182→   182→  }
   183→   183→
   184→   184→  const openTaskDetail = async (taskId: string) => {
   185→   185→    setTaskDetailLoading(true)
   186→   186→    setTaskDetail(null)
   187→   187→    try {
   188→   188→      const res = await fetch(`/api/tasks/${taskId}`)
   189→   189→      const data = await res.json()
   190→   190→      setTaskDetail(data.task)
   191→   191→    } catch {
   192→   192→      toast.error('Failed to load task details')
   193→   193→    } finally {
   194→   194→      setTaskDetailLoading(false)
   195→   195→    }
   196→   196→  }
   197→   197→
   198→   198→  const openOnboardingDialog = (projectId: string, projectName: string) => {
   199→   199→    setOnboardingProjectId(projectId)
   200→   200→    setOnboardingTitle(`Onboarding - ${projectName}`)
   201→   201→    setOnboardingDesc('')
   202→   202→    setOnboardingSteps(ONBOARDING_STEPS.map(s => s.id))
   203→   203→    setOnboardingDialogOpen(true)
   204→   204→  }
   205→   205→
   206→   206→  const handleCreateOnboarding = async () => {
   207→   207→    if (!onboardingTitle.trim() || !onboardingProjectId) return
   208→   208→    setCreatingOnboarding(true)
   209→   209→    try {
   210→   210→      const stepsJson = JSON.stringify(
   211→   211→        onboardingSteps.map(stepId => {
   212→   212→          const config = ONBOARDING_STEPS.find(s => s.id === stepId)
   213→   213→          return { id: stepId, label: config?.label ?? stepId, description: config?.description ?? '', required: config?.required ?? false }
   214→   214→        })
   215→   215→      )
   216→   216→      const res = await fetch('/api/onboarding', {
   217→   217→        method: 'POST',
   218→   218→        headers: { 'Content-Type': 'application/json' },
   219→   219→        body: JSON.stringify({
   220→   220→          title: onboardingTitle.trim(),
   221→   221→          description: onboardingDesc.trim() || null,
   222→   222→          stepsJson,
   223→   223→          projectId: onboardingProjectId,
   224→   224→          clientId: user?.id || null,
   225→   225→          status: 'published',
   226→   226→        }),
   227→   227→      })
   228→   228→      if (!res.ok) {
   229→   229→        const d = await res.json()
   230→   230→        throw new Error(d.error || 'Failed to create')
   231→   231→      }
   232→   232→      toast.success('Onboarding form created successfully')
   233→   233→      setOnboardingDialogOpen(false)
   234→   234→    } catch (e: unknown) {
   235→   235→      toast.error(e instanceof Error ? e.message : 'Failed to create onboarding')
   236→   236→    } finally {
   237→   237→      setCreatingOnboarding(false)
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  const toggleOnboardingStep = (stepId: string) => {
   242→   242→    setOnboardingSteps(prev =>
   243→   243→      prev.includes(stepId) ? prev.filter(s => s !== stepId) : [...prev, stepId]
   244→   244→    )
   245→   245→  }
   246→   246→
   247→   247→  // Stats
   248→   248→  const allTasks = projects.flatMap(p => p.tasks || [])
   249→   249→  const totalTasks = projects.reduce((sum, p) => sum + (p._count?.tasks || 0), 0)
   250→   250→  const completedTasks = allTasks.filter(t => t.status === 'done').length
   251→   251→  const inProgressTasks = allTasks.filter(t => t.status === 'in_progress').length
   252→   252→
   253→   253→  if (loading) return <PageSkeleton />
   254→   254→
   255→   255→  return (
   256→   256→    <div className="p-4 md:p-6 space-y-6">
   257→   257→      <div className="flex items-center gap-3">
   258→   258→        <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   259→   259→          <FolderKanban className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   260→   260→        </div>
   261→   261→        <div>
   262→   262→          <h1 className="text-2xl font-bold">My Projects</h1>
   263→   263→          <p className="text-sm text-muted-foreground">Track progress on all your active projects</p>
   264→   264→        </div>
   265→   265→      </div>
   266→   266→
   267→   267→      {/* Stats */}
   268→   268→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   269→   269→        <StatCard icon={FolderKanban} label="My Projects" value={projects.length} color="bg-emerald-500" />
   270→   270→        <StatCard icon={ListTodo} label="Total Tasks" value={totalTasks} color="bg-amber-500" />
   271→   271→        <StatCard icon={CheckCircle2} label="Completed" value={completedTasks} color="bg-teal-500" />
   272→   272→        <StatCard icon={Clock} label="In Progress" value={inProgressTasks} color="bg-orange-500" />
   273→   273→      </div>
   274→   274→
   275→   275→      {/* Project Cards */}
   276→   276→      {projects.length === 0 ? (
   277→   277→        <EmptyState
   278→   278→          icon={FolderKanban}
   279→   279→          title="No active projects"
   280→   280→          description="You don't have any active projects yet."
   281→   281→        />
   282→   282→      ) : (
   283→   283→        <div className="space-y-4">
   284→   284→          {projects.map((project, i) => {
   285→   285→            const projectTasks = expandedProjectId === project.id ? expandedTasks : []
   286→   286→            const doneTasks = projectTasks.filter(t => t.status === 'done').length
   287→   287→            const totalT = project._count?.tasks || 0
   288→   288→            const pct = totalT > 0 ? Math.round((doneTasks / totalT) * 100) : 0
   289→   289→            const isOverdue = project.dueDate && isBefore(parseISO(project.dueDate), new Date())
   290→   290→
   291→   291→            return (
   292→   292→              <motion.div
   293→   293→                key={project.id}
   294→   294→                initial={{ opacity: 0, y: 12 }}
   295→   295→                animate={{ opacity: 1, y: 0 }}
   296→   296→                transition={{ delay: i * 0.05 }}
   297→   297→              >
   298→   298→                <Card
   299→   299→                  className="cursor-pointer hover:shadow-md transition-shadow border-l-4"
   300→   300→                  style={{ borderLeftColor: project.color }}
   301→   301→                  onClick={() => expandProject(project.id)}
   302→   302→                >
   303→   303→                  <CardContent className="p-4 md:p-5">
   304→   304→                    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   305→   305→                      <div className="min-w-0 flex-1">
   306→   306→                        <div className="flex items-center gap-2 mb-1">
   307→   307→                          <h3 className="font-semibold text-base truncate">{project.name}</h3>
   308→   308→                          <PriorityBadge priority={project.priority} />
   309→   309→                        </div>
   310→   310→                        <p className="text-sm text-muted-foreground line-clamp-1">{project.description || 'No description'}</p>
   311→   311→                      </div>
   312→   312→                      <div className="flex items-center gap-3 shrink-0 text-sm">
   313→   313→                        {project.dueDate && (
   314→   314→                          <span className={`flex items-center gap-1 ${isOverdue ? 'text-red-500 font-medium' : 'text-muted-foreground'}`}>
   315→   315→                            <Calendar className="w-3.5 h-3.5" />
   316→   316→                            {formatDate(project.dueDate)}
   317→   317→                          </span>
   318→   318→                        )}
   319→   319→                        <span className="text-muted-foreground flex items-center gap-1">
   320→   320→                          <Users className="w-3.5 h-3.5" />
   321→   321→                          {project._count?.members || 0}
   322→   322→                        </span>
   323→   323→                        <Button
   324→   324→                          size="sm"
   325→   325→                          variant="outline"
   326→   326→                          className="h-7 gap-1.5 text-xs"
   327→   327→                          onClick={(e) => { e.stopPropagation(); openOnboardingDialog(project.id, project.name) }}
   328→   328→                        >
   329→   329→                          <ClipboardCheck className="w-3.5 h-3.5" />
   330→   330→                          Onboarding

   332→   332→                        <span className="text-muted-foreground cursor-pointer">
   333→   333→                          {expandedProjectId === project.id ? '▲' : '▼'}
   334→   334→                        </span>
   335→   335→                      </div>
   336→   336→                    </div>
   337→   337→
   338→   338→                    {/* Progress bar */}
   339→   339→                    <div className="mt-3 space-y-1">
   340→   340→                      <div className="flex items-center justify-between text-xs text-muted-foreground">
   341→   341→                        <span>{doneTasks}/{totalT} tasks completed</span>
   342→   342→                        <span>{pct}%</span>
   343→   343→                      </div>
   344→   344→                      <Progress value={pct} className="h-2" />
   345→   345→                    </div>
   346→   346→                  </CardContent>
   347→   347→                </Card>
   348→   348→
   349→   349→                {/* Expanded tasks table */}
   350→   350→                <AnimatePresence>
   351→   351→                  {expandedProjectId === project.id && (
   352→   352→                    <motion.div
   353→   353→                      initial={{ opacity: 0, height: 0 }}
   354→   354→                      animate={{ opacity: 1, height: 'auto' }}
   355→   355→                      exit={{ opacity: 0, height: 0 }}
   356→   356→                      transition={{ duration: 0.2 }}
   357→   357→                      className="overflow-hidden"
   358→   358→                    >
   359→   359→                      <Card className="mt-2 rounded-t-none">
   360→   360→                        <CardContent className="p-0">
   361→   361→                          {projectTasks.length === 0 ? (
   362→   362→                            <div className="py-8 text-center text-sm text-muted-foreground">
   363→   363→                              No tasks in this project yet.
   364→   364→                            </div>
   365→   365→                          ) : (
   366→   366→                            <div className="overflow-x-auto">
   367→   367→                              <table className="w-full text-sm">

   375→   375→                                  </tr>
   376→   376→                                </thead>
   377→   377→                                <tbody>
   378→   378→                                  {projectTasks.map(task => (
   379→   379→                                    <tr
   380→   380→                                      key={task.id}
   381→   381→                                      className="border-b last:border-0 hover:bg-muted/30 cursor-pointer transition-colors"

   384→   384→                                      <td className="p-3">
   385→   385→                                        <div className="font-medium">{task.title}</div>

   389→   389→                                      </td>
   390→   390→                                      <td className="p-3 hidden md:table-cell"><StatusBadge status={task.status} /></td>
   391→   391→                                      <td className="p-3 hidden sm:table-cell"><PriorityBadge priority={task.priority} /></td>
   392→   392→                                      <td className="p-3 hidden lg:table-cell text-muted-foreground text-xs">
   393→   393→                                        {task.dueDate ? formatDate(task.dueDate) : '—'}
   394→   394→                                      </td>
   395→   395→                                      <td className="p-3 hidden lg:table-cell">
   396→   396→                                        <div className="flex -space-x-2">
   397→   397→                                          {task.assignees?.slice(0, 3).map(a => (

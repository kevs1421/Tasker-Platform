  3056→function formatDuration(minutes: number): string {
  3057→  const h = Math.floor(minutes / 60)
  3058→  const m = Math.round(minutes % 60)
  3059→  return `${h}h ${m}m`

  3062→function formatLiveTimer(startTime: string): string {
  3063→  const start = new Date(startTime).getTime()
  3064→  const now = Date.now()
  3065→  const diff = Math.max(0, Math.floor((now - start) / 1000))
  3066→  const h = Math.floor(diff / 3600)
  3067→  const m = Math.floor((diff % 3600) / 60)
  3068→  const s = diff % 60
  3069→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  3070→}

  3072→function getDateRange(preset: DatePreset, customFrom?: Date, customTo?: Date): { from?: string; to?: string } {
  3073→  const now = new Date()
  3074→  const todayStr = now.toISOString().split('T')[0]
  3075→
  3076→  if (preset === 'custom') {
  3077→    return {
  3078→      from: customFrom ? format(customFrom, 'yyyy-MM-dd') : undefined,
  3079→      to: customTo ? format(customTo, 'yyyy-MM-dd') : todayStr,
  3080→    }
  3081→  }
  3082→
  3083→  switch (preset) {
  3084→    case 'today':
  3085→      return { from: todayStr, to: todayStr }
  3086→    case 'week': {
  3087→      const dayOfWeek = now.getDay()
  3088→      const monday = new Date(now)
  3089→      monday.setDate(now.getDate() - ((dayOfWeek + 6) % 7))
  3090→      return { from: monday.toISOString().split('T')[0], to: todayStr }
  3091→    }
  3092→    case 'month': {
  3093→      const firstOfMonth = new Date(now.getFullYear(), now.getMonth(), 1)
  3094→      return { from: firstOfMonth.toISOString().split('T')[0], to: todayStr }
  3095→    }
  3096→    case 'all':
  3097→      return {}
  3098→  }
